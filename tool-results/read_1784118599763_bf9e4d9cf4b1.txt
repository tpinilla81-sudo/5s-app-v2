   419→ const handleCreateCompanyFull = async () => {
   420→  if (!newCompanyName.trim()) return
   421→  setIsCreatingCompany(true)
   422→  try {
   423→   // Step 1: Create company
   424→   const companyRes = await fetch('/api/companies', {
   425→    method: 'POST',
   426→    headers: { 'Content-Type': 'application/json' },
   427→    body: JSON.stringify({ name: newCompanyName, description: newCompanyDesc || undefined }),
   428→   })
   429→   const companyData = await companyRes.json()
   430→   if (!companyRes.ok) {
   431→    alert(companyData.error || 'Error al crear empresa')
   432→    return
   433→   }
   434→   const companyId = companyData.company?.id || companyData.id
   435→
   436→   // Step 2: Create or assign admin
   437→   let adminUserId: string | null = null
   438→   if (adminMode === 'create') {
   439→    if (!newAdminName.trim() || !newAdminEmail.trim() || !newAdminPassword.trim()) {
   440→     alert('Complete los datos del administrador')
   441→     return
   442→    }
   443→    // Create user with role 'admin' via /api/users
   444→    const userRes = await fetch('/api/users', {
   445→     method: 'POST',
   446→     headers: { 'Content-Type': 'application/json' },
   447→     body: JSON.stringify({
   448→      name: newAdminName,
   449→      email: newAdminEmail,
   450→      password: newAdminPassword,
   451→      role: 'admin',
   452→      active: true,
   453→     }),
   454→    })
   455→    const userData = await userRes.json()
   456→    if (!userRes.ok) {
   457→     alert(userData.error || 'Error al crear usuario admin')
   458→     return
   459→    }
   460→    adminUserId = userData.user?.id
   461→   } else {
   462→    // Assign existing user
   463→    if (!assignUserId) {
   464→     alert('Seleccione un usuario para asignar como administrador')
   465→     return
   466→    }
   467→    adminUserId = assignUserId
   468→    // Update user role to 'admin' if not already
   469→    const existingUser = users.find(u => u.id === assignUserId)
   470→    if (existingUser && existingUser.role !== 'admin') {
   471→     await fetch('/api/users', {
   472→      method: 'PUT',
   473→      headers: { 'Content-Type': 'application/json' },
   474→      body: JSON.stringify({ id: assignUserId, role: 'admin' }),
   475→     })
   476→    }
   477→   }
   478→
   479→   // Step 3: Add admin to company as admin_empresa
   480→   if (adminUserId) {
   481→    await fetch(`/api/companies/${companyId}/members`, {
   482→     method: 'POST',
   483→     headers: { 'Content-Type': 'application/json' },
   484→     body: JSON.stringify({ userId: adminUserId, role: 'admin_empresa' }),
   485→    })
   486→   }
   487→
   488→   // Step 4: Create subscription
   489→   await fetch('/api/subscriptions', {
   490→    method: 'POST',
   491→    headers: { 'Content-Type': 'application/json' },
   492→    body: JSON.stringify({
   493→     companyId,
   494→     plan: newSubPlan,
   495→     status: newSubStatus,
   496→     maxUsers: newSubMaxUsers,
   497→     maxProjects: newSubMaxProjects,
   498→     price: newSubPrice,
   499→     endDate: newSubEndDate || undefined,
   500→     trialEndsAt: newSubTrialEndsAt || undefined,
   501→     notes: newSubNotes || undefined,
   502→    }),
   503→   })
   504→
   505→   setShowNewCompany(false)
   506→   resetNewCompanyForm()
   507→   await loadStats()
   508→   await fetchCompanies()
   509→  } catch (error) {
   510→   console.error('Error creating company with admin:', error)
   511→   alert('Error al crear la empresa. Revise la consola para detalles.')
   512→  } finally {
   513→   setIsCreatingCompany(false)
   514→  }
   515→ }
