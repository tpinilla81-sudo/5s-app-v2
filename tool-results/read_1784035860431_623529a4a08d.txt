     1→'use client'
     2→
     3→import { useState } from 'react'
     4→import { use5SStore } from '@/lib/store'
     5→import { Button } from '@/components/ui/button'
     6→import { Input } from '@/components/ui/input'
     7→import { Label } from '@/components/ui/label'
     8→import { Textarea } from '@/components/ui/textarea'
     9→import { Card, CardContent, CardHeader, CardTitle, CardDescription } from '@/components/ui/card'
    10→import {
    11→  Select,
    12→  SelectContent,
    13→  SelectItem,
    14→  SelectTrigger,
    15→  SelectValue,
    16→} from '@/components/ui/select'
    17→import { Badge } from '@/components/ui/badge'
    18→import {
    19→  Table,
    20→  TableBody,
    21→  TableCell,
    22→  TableHead,
    23→  TableHeader,
    24→  TableRow,
    25→} from '@/components/ui/table'
    26→import {
    27→  Loader2,
    28→  Plus,
    29→  Trash2,
    30→  ChevronRight,
    31→  ChevronLeft,
    32→  Check,
    33→  Building2,
    34→  MapPin,
    35→  Users,
    36→  ClipboardCheck,
    37→} from 'lucide-react'
    38→import { Checkbox } from '@/components/ui/checkbox'
    39→
    40→const PRESET_COLORS = [
    41→  '#8B5CF6', '#EAB308', '#3B82F6', '#F43F5E',
    42→  '#F97316', '#22C55E', '#06B6D4', '#EC4899',
    43→]
    44→
    45→interface ZoneInput {
    46→  name: string
    47→  description: string
    48→  color: string
    49→}
    50→
    51→interface MemberInput {
    52→  name: string
    53→  email: string
    54→  role: string
    55→  zoneIds: string[]
    56→}
    57→
    58→export default function ProjectSetup() {
    59→  const { createProject, currentUser } = use5SStore()
    60→  const [step, setStep] = useState(1)
    61→  const [isCreating, setIsCreating] = useState(false)
    62→
    63→  // Step 1: Project Info
    64→  const [projectName, setProjectName] = useState('')
    65→  const [companyName, setCompanyName] = useState('')
    66→  const [description, setDescription] = useState('')
    67→
    68→  // Step 2: Zones
    69→  const [zones, setZones] = useState<ZoneInput[]>([
    70→    { name: '', description: '', color: PRESET_COLORS[0] },
    71→  ])
    72→
    73→  // Step 3: Team Members
    74→  const [members, setMembers] = useState<MemberInput[]>([])
    75→  const [newMember, setNewMember] = useState<MemberInput>({
    76→    name: '',
    77→    email: '',
    78→    role: 'empleado',
    79→    zoneIds: [],
    80→  })
    81→
    82→  const canGoNext = () => {
    83→    switch (step) {
    84→      case 1:
    85→        return projectName.trim() !== '' && companyName.trim() !== ''
    86→      case 2:
    87→        return zones.length > 0 && zones.every((z) => z.name.trim() !== '')
    88→      case 3:
    89→        return true // Members are optional (admin is auto-added)
    90→      case 4:
    91→        return true
    92→      default:
    93→        return false
    94→    }
    95→  }
    96→
    97→  const handleAddZone = () => {
    98→    const nextColor = PRESET_COLORS[zones.length % PRESET_COLORS.length]
    99→    setZones([...zones, { name: '', description: '', color: nextColor }])
   100→  }
   101→
   102→  const handleRemoveZone = (index: number) => {
   103→    if (zones.length > 1) {
   104→      setZones(zones.filter((_, i) => i !== index))
   105→    }
   106→  }
   107→
   108→  const handleZoneChange = (index: number, field: keyof ZoneInput, value: string) => {
   109→    const updated = [...zones]
   110→    updated[index] = { ...updated[index], [field]: value }
   111→    setZones(updated)
   112→  }
   113→
   114→  const handleAddMember = () => {
   115→    if (newMember.name.trim() && newMember.email.trim()) {
   116→      setMembers([...members, { ...newMember }])
   117→      setNewMember({ name: '', email: '', role: 'empleado', zoneIds: [] })
   118→    }
   119→  }
   120→
   121→  const handleRemoveMember = (index: number) => {
   122→    setMembers(members.filter((_, i) => i !== index))
   123→  }
   124→
   125→  const handleCreate = async () => {
   126→    setIsCreating(true)
   127→    try {
   128→      await createProject({
   129→        name: projectName,
   130→        description: description || undefined,
   131→        company: companyName,
   132→        zones: zones.map((z) => ({
   133→          name: z.name,
   134→          description: z.description || undefined,
   135→          color: z.color,
   136→        })),
   137→      })
   138→
   139→      // Add additional members (admin is auto-added in createProject)
   140→      const { currentProject } = use5SStore.getState()
   141→      if (currentProject) {
   142→        for (const member of members) {
   143→          // Map temporary zoneIds (e.g., "zone-0") to real zone IDs from created project
   144→          const realZoneIds: string[] = []
   145→          for (const zId of member.zoneIds) {
   146→            if (zId && zId.startsWith('zone-')) {
   147→              const zoneIndex = parseInt(zId.replace('zone-', ''), 10)
   148→              const realZoneId = currentProject.zones[zoneIndex]?.id
   149→              if (realZoneId) realZoneIds.push(realZoneId)
   150→            }
   151→          }
   152→
   153→          await fetch(`/api/projects/${currentProject.id}/members`, {
   154→            method: 'POST',
   155→            headers: { 'Content-Type': 'application/json' },
   156→            body: JSON.stringify({
   157→              email: member.email,
   158→              name: member.name,
   159→              role: member.role,
   160→              zoneIds: realZoneIds.length > 0 ? realZoneIds : undefined,
   161→            }),
   162→          })
   163→        }
   164→      }
   165→    } catch (error) {
   166→      console.error('Error creating project:', error)
   167→    } finally {
   168→      setIsCreating(false)
   169→    }
   170→  }
   171→
   172→  const getRoleLabel = (role: string) => {
   173→    const map: Record<string, string> = {
   174→      admin: 'Administrador',
   175→      gerente: 'Gerente',
   176→      responsable: 'Responsable',
   177→      empleado: 'Empleado',
   178→      auditor: 'Auditor',
   179→    }
   180→    return map[role] || role
   181→  }
   182→
   183→  const getRoleBadgeColor = (role: string) => {
   184→    const map: Record<string, string> = {
   185→      admin: 'bg-purple-100 text-purple-700 border-purple-200',
   186→      gerente: 'bg-indigo-100 text-indigo-700 border-indigo-200',
   187→      responsable: 'bg-blue-100 text-blue-700 border-blue-200',
   188→      empleado: 'bg-green-100 text-green-700 border-green-200',
   189→      auditor: 'bg-orange-100 text-orange-700 border-orange-200',
   190→    }
   191→    return map[role] || 'bg-gray-100 text-gray-700 border-gray-200'
   192→  }
   193→
   194→  const stepIcons = [Building2, MapPin, Users, ClipboardCheck]
   195→  const stepLabels = ['Proyecto', 'Zonas', 'Equipo', 'Confirmar']
   196→
   197→  return (
   198→    <div className="min-h-screen flex flex-col bg-gradient-to-br from-green-50 via-white to-emerald-50">
   199→      {/* Header */}
   200→      <header className="border-b bg-white/80 backdrop-blur-sm sticky top-0 z-10">
   201→        <div className="max-w-3xl mx-auto px-4 py-3 flex items-center gap-3">
   202→          <div className="w-10 h-10 rounded-full bg-gradient-to-br from-green-500 to-emerald-600 flex items-center justify-center text-white font-bold text-lg shadow-md">
   203→            5S
   204→          </div>
   205→          <div>
   206→            <h1 className="text-lg font-bold text-gray-900">Configurar Proyecto</h1>
   207→            <p className="text-xs text-muted-foreground">Crea tu primer proyecto 5S</p>
   208→          </div>
   209→        </div>
   210→      </header>
   211→
   212→      {/* Step Indicator */}
   213→      <div className="max-w-3xl mx-auto w-full px-4 pt-6">
   214→        <div className="flex items-center justify-between mb-8">
   215→          {stepLabels.map((label, i) => {
   216→            const Icon = stepIcons[i]
   217→            const stepNum = i + 1
   218→            const isActive = step === stepNum
   219→            const isCompleted = step > stepNum
   220→            return (
   221→              <div key={i} className="flex flex-col items-center flex-1">
   222→                <div className="flex items-center w-full">
   223→                  <div
   224→                    className={`w-10 h-10 rounded-full flex items-center justify-center text-sm font-semibold transition-all duration-300 ${
   225→                      isCompleted
   226→                        ? 'bg-green-500 text-white shadow-md shadow-green-200'
   227→                        : isActive
   228→                        ? 'bg-gradient-to-br from-green-500 to-emerald-600 text-white shadow-md shadow-green-200'
   229→                        : 'bg-gray-100 text-gray-400'
   230→                    }`}
   231→                  >
   232→                    {isCompleted ? <Check className="h-5 w-5" /> : <Icon className="h-4 w-4" />}
   233→                  </div>
   234→                  {i < 3 && (
   235→                    <div
   236→                      className={`flex-1 h-0.5 mx-2 transition-all duration-300 ${
   237→                        isCompleted ? 'bg-green-500' : 'bg-gray-200'
   238→                      }`}
   239→                    />
   240→                  )}
   241→                </div>
   242→                <span
   243→                  className={`text-xs mt-2 font-medium ${
   244→                    isActive ? 'text-green-600' : isCompleted ? 'text-green-500' : 'text-gray-400'
   245→                  }`}
   246→                >
   247→                  {label}
   248→                </span>
   249→              </div>
   250→            )
   251→          })}
   252→        </div>
   253→      </div>
   254→
   255→      {/* Main content */}
   256→      <main className="flex-1 max-w-3xl mx-auto w-full px-4 pb-8">
   257→          {/* Step 1: Project Info */}
   258→          {step === 1 && (
   259→            <div>
   260→              <Card className="border-0 shadow-lg shadow-green-100/50">
   261→                <CardHeader>
   262→                  <CardTitle className="flex items-center gap-2">
   263→                    <Building2 className="h-5 w-5 text-green-500" />
   264→                    Información del Proyecto
   265→                  </CardTitle>
   266→                  <CardDescription>
   267→                    Ingresa los datos básicos de tu proyecto 5S
   268→                  </CardDescription>
   269→                </CardHeader>
   270→                <CardContent className="space-y-4">
   271→                  <div className="space-y-2">
   272→                    <Label htmlFor="project-name">Nombre del Proyecto *</Label>
   273→                    <Input
   274→                      id="project-name"
   275→                      placeholder="Ej: Implementación 5S - Planta Principal"
   276→                      value={projectName}
   277→                      onChange={(e) => setProjectName(e.target.value)}
   278→                    />
   279→                  </div>
   280→                  <div className="space-y-2">
   281→                    <Label htmlFor="company-name">Empresa *</Label>
   282→                    <Input
   283→                      id="company-name"
   284→                      placeholder="Ej: Manufacturas ABC S.A."
   285→                      value={companyName}
   286→                      onChange={(e) => setCompanyName(e.target.value)}
   287→                    />
   288→                  </div>
   289→                  <div className="space-y-2">
   290→                    <Label htmlFor="description">Descripción (opcional)</Label>
   291→                    <Textarea
   292→                      id="description"
   293→                      placeholder="Describe brevemente el objetivo del proyecto..."
   294→                      value={description}
   295→                      onChange={(e) => setDescription(e.target.value)}
   296→                      rows={3}
   297→                    />
   298→                  </div>
   299→                </CardContent>
   300→              </Card>
   301→            </div>
   302→          )}
   303→
   304→          {/* Step 2: Zones */}
   305→          {step === 2 && (
   306→            <div>
   307→              <Card className="border-0 shadow-lg shadow-green-100/50">
   308→                <CardHeader>
   309→                  <CardTitle className="flex items-center gap-2">
   310→                    <MapPin className="h-5 w-5 text-green-500" />
   311→                    Zonas de Trabajo
   312→                  </CardTitle>
   313→                  <CardDescription>
   314→                    Define las zonas donde se implementará la metodología 5S
   315→                  </CardDescription>
   316→                </CardHeader>
   317→                <CardContent className="space-y-4">
   318→                  {zones.map((zone, index) => (
   319→                    <div
   320→                      key={index}
   321→                      className="p-4 rounded-lg border bg-white space-y-3"
   322→                    >
   323→                      <div className="flex items-center justify-between">
   324→                        <div className="flex items-center gap-2">
   325→                          <div
   326→                            className="w-4 h-4 rounded-full"
   327→                            style={{ backgroundColor: zone.color }}
   328→                          />
   329→                          <span className="text-sm font-medium">Zona {index + 1}</span>
   330→                        </div>
   331→                        {zones.length > 1 && (
   332→                          <Button
   333→                            variant="ghost"
   334→                            size="sm"
   335→                            onClick={() => handleRemoveZone(index)}
   336→                            className="text-red-500 hover:text-red-700 hover:bg-red-50 h-8 w-8 p-0"
   337→                          >
   338→                            <Trash2 className="h-4 w-4" />
   339→                          </Button>
   340→                        )}
   341→                      </div>
   342→
   343→                      <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   344→                        <div className="space-y-1">
   345→                          <Label className="text-xs">Nombre *</Label>
   346→                          <Input
   347→                            placeholder="Ej: Línea de ensamblaje"
   348→                            value={zone.name}
   349→                            onChange={(e) => handleZoneChange(index, 'name', e.target.value)}
   350→                          />
   351→                        </div>
   352→                        <div className="space-y-1">
   353→                          <Label className="text-xs">Descripción</Label>
   354→                          <Input
   355→                            placeholder="Ej: Área principal de producción"
   356→                            value={zone.description}
   357→                            onChange={(e) => handleZoneChange(index, 'description', e.target.value)}
   358→                          />
   359→                        </div>
   360→                      </div>
   361→
   362→                      <div className="space-y-1">
   363→                        <Label className="text-xs">Color</Label>
   364→                        <div className="flex gap-2 flex-wrap">
   365→                          {PRESET_COLORS.map((color) => (
   366→                            <button
   367→                              key={color}
   368→                              type="button"
   369→                              onClick={() => handleZoneChange(index, 'color', color)}
   370→                              className={`w-7 h-7 rounded-full border-2 transition-all ${
   371→                                zone.color === color
   372→                                  ? 'border-gray-800 scale-110 shadow-md'
   373→                                  : 'border-transparent hover:scale-105'
   374→                              }`}
   375→                              style={{ backgroundColor: color }}
   376→                            />
   377→                          ))}
   378→                        </div>
   379→                      </div>
   380→                    </div>
   381→                  ))}
   382→
   383→                  <Button
   384→                    variant="outline"
   385→                    onClick={handleAddZone}
   386→                    className="w-full border-dashed border-green-300 text-green-600 hover:bg-green-50 hover:text-green-700"
   387→                  >
   388→                    <Plus className="h-4 w-4 mr-2" />
   389→                    Agregar Zona
   390→                  </Button>
   391→                </CardContent>
   392→              </Card>
   393→            </div>
   394→          )}
   395→
   396→          {/* Step 3: Team Members */}
   397→          {step === 3 && (
   398→            <div>
   399→              <Card className="border-0 shadow-lg shadow-green-100/50">
   400→                <CardHeader>
   401→                  <CardTitle className="flex items-center gap-2">
   402→                    <Users className="h-5 w-5 text-green-500" />
   403→                    Equipo de Trabajo
   404→                  </CardTitle>
   405→                  <CardDescription>
   406→                    Agrega miembros al proyecto. Tú eres el administrador por defecto.
   407→                  </CardDescription>
   408→                </CardHeader>
   409→                <CardContent className="space-y-4">
   410→                  {/* Current user (admin) info */}
   411→                  {currentUser && (
   412→                    <div className="p-3 rounded-lg bg-green-50 border border-green-200">
   413→                      <div className="flex items-center gap-3">
   414→                        <div className="w-8 h-8 rounded-full bg-green-500 flex items-center justify-center text-white text-xs font-bold">
   415→                          {currentUser.name.charAt(0).toUpperCase()}
   416→                        </div>
   417→                        <div className="flex-1">
   418→                          <p className="text-sm font-medium text-green-900">
   419→                            {currentUser.name} (Tú)
   420→                          </p>
   421→                          <p className="text-xs text-green-600">{currentUser.email}</p>
   422→                        </div>
   423→                        <Badge className={`${getRoleBadgeColor('admin')} border`}>
   424→                          Administrador
   425→                        </Badge>
   426→                      </div>
   427→                    </div>
   428→                  )}
   429→
   430→                  {/* Add member form */}
   431→                  <div className="p-4 rounded-lg border bg-white space-y-3">
   432→                    <p className="text-sm font-medium">Agregar miembro</p>
   433→                    <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   434→                      <div className="space-y-1">
   435→                        <Label className="text-xs">Nombre</Label>
   436→                        <Input
   437→                          placeholder="Nombre del miembro"
   438→                          value={newMember.name}
   439→                          onChange={(e) =>
   440→                            setNewMember({ ...newMember, name: e.target.value })
   441→                          }
   442→                        />
   443→                      </div>
   444→                      <div className="space-y-1">
   445→                        <Label className="text-xs">Email</Label>
   446→                        <Input
   447→                          type="email"
   448→                          placeholder="email@ejemplo.com"
   449→                          value={newMember.email}
   450→                          onChange={(e) =>
   451→                            setNewMember({ ...newMember, email: e.target.value })
   452→                          }
   453→                        />
   454→                      </div>
   455→                      <div className="space-y-1">
   456→                        <Label className="text-xs">Rol</Label>
   457→                        <Select
   458→                          value={newMember.role}
   459→                          onValueChange={(value) =>
   460→                            setNewMember({ ...newMember, role: value })
   461→                          }
   462→                        >
   463→                          <SelectTrigger className="w-full">
   464→                            <SelectValue />
   465→                          </SelectTrigger>
   466→                          <SelectContent>
   467→                            <SelectItem value="gerente">Gerente</SelectItem>
   468→                            <SelectItem value="responsable">Responsable</SelectItem>
   469→                            <SelectItem value="empleado">Empleado</SelectItem>
   470→                            <SelectItem value="auditor">Auditor</SelectItem>
   471→                          </SelectContent>
   472→                        </Select>
   473→                      </div>
   474→                      <div className="space-y-1">
   475→                        <Label className="text-xs">Zonas asignadas</Label>
   476→                        <div className="space-y-0.5 max-h-32 overflow-y-auto border rounded-md p-2">
   477→                          {zones
   478→                            .filter((z) => z.name.trim())
   479→                            .map((zone, i) => (
   480→                              <label key={i} className="flex items-center gap-1.5 text-sm cursor-pointer hover:bg-gray-50 rounded px-1 py-0.5">
   481→                                <Checkbox
   482→                                  checked={newMember.zoneIds.includes(`zone-${i}`)}
   483→                                  onCheckedChange={(checked) => {
   484→                                    const zId = `zone-${i}`
   485→                                    if (checked) {
   486→                                      setNewMember({ ...newMember, zoneIds: [...newMember.zoneIds, zId] })
   487→                                    } else {
   488→                                      setNewMember({ ...newMember, zoneIds: newMember.zoneIds.filter(id => id !== zId) })
   489→                                    }
   490→                                  }}
   491→                                  className="h-4 w-4"
   492→                                />
   493→                                <div
   494→                                  className="w-2.5 h-2.5 rounded-full"
   495→                                  style={{ backgroundColor: zone.color }}
   496→                                />
   497→                                <span>{zone.name}</span>
   498→                              </label>
   499→                            ))}
   500→                        </div>
   501→                      </div>
   502→                    </div>
   503→                    <Button
   504→                      variant="outline"
   505→                      onClick={handleAddMember}
   506→                      disabled={!newMember.name.trim() || !newMember.email.trim()}
   507→                      className="w-full border-green-300 text-green-600 hover:bg-green-50"
   508→                    >
   509→                      <Plus className="h-4 w-4 mr-2" />
   510→                      Agregar Miembro
   511→                    </Button>
   512→                  </div>
   513→
   514→                  {/* Members table */}
   515→                  {members.length > 0 && (
   516→                    <div className="rounded-lg border overflow-hidden">
   517→                      <Table>
   518→                        <TableHeader>
   519→                          <TableRow>
   520→                            <TableHead>Nombre</TableHead>
   521→                            <TableHead>Email</TableHead>
   522→                            <TableHead>Rol</TableHead>
   523→                            <TableHead>Zona</TableHead>
   524→                            <TableHead className="w-10" />
   525→                          </TableRow>
   526→                        </TableHeader>
   527→                        <TableBody>
   528→                          {members.map((member, index) => (
   529→                            <TableRow key={index}>
   530→                              <TableCell className="font-medium text-sm">
   531→                                {member.name}
   532→                              </TableCell>
   533→                              <TableCell className="text-sm text-muted-foreground">
   534→                                {member.email}
   535→                              </TableCell>
   536→                              <TableCell>
   537→                                <Badge className={`${getRoleBadgeColor(member.role)} border`}>
   538→                                  {getRoleLabel(member.role)}
   539→                                </Badge>
   540→                              </TableCell>
   541→                              <TableCell className="text-sm">
   542→                                {member.zoneIds.length > 0
   543→                                  ? member.zoneIds.map(zId => {
   544→                                      const idx = parseInt(zId.replace('zone-', ''), 10)
   545→                                      return zones[idx]?.name
   546→                                    }).filter(Boolean).join(', ')
   547→                                  : '-'}
   548→                              </TableCell>
   549→                              <TableCell>
   550→                                <Button
   551→                                  variant="ghost"
   552→                                  size="sm"
   553→                                  onClick={() => handleRemoveMember(index)}
   554→                                  className="text-red-500 hover:text-red-700 h-8 w-8 p-0"
   555→                                >
   556→                                  <Trash2 className="h-4 w-4" />
   557→                                </Button>
   558→                              </TableCell>
   559→                            </TableRow>
   560→                          ))}
   561→                        </TableBody>
   562→                      </Table>
   563→                    </div>
   564→                  )}
   565→
   566→                  {members.length === 0 && (
   567→                    <p className="text-sm text-muted-foreground text-center py-4">
   568→                      Puedes agregar miembros más tarde desde la gestión del proyecto
   569→                    </p>
   570→                  )}
   571→                </CardContent>
   572→              </Card>
   573→            </div>
   574→          )}
   575→
   576→          {/* Step 4: Confirmation */}
