   266→   266→      if (status === 'ok') okCount++;
   267→   267→      else if (status === 'nok') nokCount++;
   268→   268→    }));
   269→   269→    const answeredCount = okCount + nokCount;
   270→   270→    const scorePercent = totalItems > 0 ? Math.min(Math.round((okCount / totalItems) * 100), 100) : 0;
   271→   271→    return { okCount, nokCount, answeredCount, scorePercent };
   272→   272→  }, [results, sections, totalItems]);
   273→   273→

   277→   277→

   313→   313→      ...prev,
   314→   314→      [itemId]: { ...prev[itemId], itemId, [field]: value },

   331→   331→        method: 'PUT',
   332→   332→        headers: { 'Content-Type': 'application/json' },
   333→   333→        body: JSON.stringify({
   334→   334→          completed: passed,
   335→   335→          score: scoring.scorePercent,
   336→   336→          notes: JSON.stringify({
   337→   337→            type: 'autoevaluacion',
   338→   338→            passed,
   339→   339→            notaMinima,
   340→   340→            results: effectiveResults,
   341→   341→            observaciones,
   342→   342→            fechaAutoevaluacion,
   343→   343→            horaAutoevaluacion,
   344→   344→            fechaProgramada: fechaProgramada || null,
   345→   345→            horaProgramada: horaProgramada || null,
   346→   346→          }),
   347→   347→          projectId: currentProject?.id,
   348→   348→          zoneId: currentZone?.id || null,
   349→   349→        }),
   350→   350→      });
   351→   351→      const json = await res.json();
   352→   352→      if (!res.ok || !json.success) {
   353→   353→        toast.error(`Error del servidor: ${json.error || res.statusText || 'Error desconocido'}`);
   354→   354→        return;
   355→   355→      }
   356→   356→      if (json.success) {
   357→   357→        // Feedback visible al usuario
   358→   358→        toast.success(`Autoevaluación guardada: ${effectiveResults.length} items (${scoring.okCount} OK, ${scoring.nokCount} NOK, score ${scoring.scorePercent}%)`);
   359→   359→        setIsCompleted(true);
   360→   360→        setFinalScore(scoring.scorePercent);
   361→   361→        await fetchProgress();
   362→   362→
   363→   363→        // ─── Create EmployeeProgress record for individual step 4 (autoevaluación) ───
   364→   364→        // Step 4 is individual for S1/S2/S3/S5 (done by employees) and for S4 (done by responsable)
   365→   365→        // We need to track individual completion so the gating system unlocks step 5

   367→   367→          try {
   368→   368→            await fetch('/api/employee-progress', {
   369→   369→              method: 'POST',
   370→   370→              headers: { 'Content-Type': 'application/json' },
   371→   371→              body: JSON.stringify({
   372→   372→                sStep,
   373→   373→                miniStep: 4,
   374→   374→                completed: true,
   375→   375→                score: scoring.scorePercent,
   376→   376→                projectId: currentProject.id,
   377→   377→                zoneId: currentZone.id,
   378→   378→                userId: currentUser.id,
   379→   379→              }),
   380→   380→            });
   381→   381→            // Also refresh employee progress so step 5 unlocks immediately
   382→   382→            const { fetchEmployeeProgress } = use5SStore.getState();
   383→   383→            await fetchEmployeeProgress(currentProject.id, currentZone.id);
   384→   384→          } catch (empErr) {
   385→   385→            console.error('Error creating employee progress for autoeval:', empErr);
   386→   386→          }
   387→   387→        }
   388→   388→
   389→   389→        // ─── Upload photos to library with traceability ───
   390→   390→        if (autoevalPhotos.length > 0) {
   391→   391→          setIsUploadingPhotos(true);
   392→   392→          for (let idx = 0; idx < autoevalPhotos.length; idx++) {
   393→   393→            const photo = autoevalPhotos[idx];
   394→   394→            try {
   395→   395→              // Step 1: Upload the file to get a server URL
   396→   396→              const formData = new FormData();
   397→   397→              formData.append('file', photo.file);
   398→   398→              formData.append('filename', `S${sStep}_autoeval_${currentZone?.name || 'zona'}_${idx + 1}_${Date.now()}.jpg`);
   399→   399→
   400→   400→              const uploadRes = await fetch('/api/upload', {
   401→   401→                method: 'POST',
   402→   402→                body: formData,
   403→   403→              });
   404→   404→              const uploadData = await uploadRes.json();
   405→   405→
   406→   406→              if (uploadData.success && uploadData.url) {
   407→   407→                // Step 2: Save to photo library with full traceability
   408→   408→                await fetch('/api/photo-library', {
   409→   409→                  method: 'POST',
   410→   410→                  headers: { 'Content-Type': 'application/json' },
   411→   411→                  body: JSON.stringify({
   412→   412→                    sStep,
   413→   413→                    miniStep: 4,
