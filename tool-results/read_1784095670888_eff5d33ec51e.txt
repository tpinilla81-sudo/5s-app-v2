   178→  return {
   179→    sections: checklist.map(section => ({

   181→      title: section.title,
   182→      items: section.items.map(item => ({
   183→        id: item.id,
   184→        description: item.description,
   185→        hasOther: item.hasOther || false,
   186→      }))
   187→    }))
   188→  }

   191→function getDefaultInventoryContent(sStep: number) {
   192→  const cfg = INVENTORY_CONFIGS[sStep]
   193→  if (!cfg) return { categories: [], extraFields: [] }
   194→  return {
   195→    categories: cfg.categories.map(c => ({ value: c.value, label: c.label, color: c.color })),
   196→    extraFields: cfg.extraFields.map(f => ({
   197→      key: f.key, label: f.label, type: f.type,
