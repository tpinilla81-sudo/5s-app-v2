   116→  // cuando se seleccionan varias fotos a la vez (era la causa del bloqueo).

   122→  const uploadPhoto = async (base64Data: string, index: number): Promise<string | null> => {
   123→    try {
   124→      const projectId = currentProject?.id || 'unknown';
   125→      const filename = generatePhotoFilename(projectId, sStep, miniStep, index);
   126→      const file = base64toFile(base64Data, filename);

   128→      const formData = new FormData();

   132→      const res = await fetch('/api/upload', {
   133→        method: 'POST',
   134→        body: formData,
   135→      });
   136→
   137→      const data = await res.json();
   138→      if (data.success && data.url) return data.url;
   139→      return null;
   140→    } catch (error) {
   141→      console.error('Upload error:', error);
   142→      return null;

   146→  // Procesa la cola de fotos UNA A UNA. Así el canvas + toDataURL + fetch
   147→  // no compiten por CPU/memoria, y la UI del móvil no se congela.
   148→  const processQueue = useCallback(async () => {
   149→    if (isProcessingRef.current) return;
   150→    if (uploadQueueRef.current.length === 0) return;
   151→    isProcessingRef.current = true;
   152→    console.log(`[FotosModal] processQueue: iniciando procesamiento de ${uploadQueueRef.current.length} foto(s)`);
   153→
   154→    try {
   155→      while (uploadQueueRef.current.length > 0) {
   156→        const item = uploadQueueRef.current.shift()!;
   157→        setQueueLength(uploadQueueRef.current.length);
   158→
   159→        const { rawBase64, photoType } = item;
   160→        const index = photoCounterRef.current++;
   161→        const photoId = `photo_${Date.now()}_${index}_${Math.random().toString(36).slice(2, 8)}`;
   162→        console.log(`[FotosModal] processQueue: procesando foto #${index + 1} (id=${photoId}), quedan ${uploadQueueRef.current.length} en cola`);
   163→
   164→        try {
   165→          // v2.58: si la compresión falla, usar el original sin comprimir
   166→          let compressed: string;
   167→          try {
   168→            compressed = await compressImage(rawBase64);
   169→          } catch (compressErr) {
   170→            console.warn('[FotosModal] Compression failed, using original:', compressErr);
   171→            compressed = rawBase64;
   172→          }
   173→          const estimatedSize = estimateBase64Size(compressed);
   174→
   175→          const newPhoto: PhotoItem = {
   176→            id: photoId,
   177→            preview: compressed,
   178→            serverUrl: '',
   179→            uploaded: false,
   180→            uploading: true,
   181→            estimatedSize,
   182→            title: generatePhotoTitle(index, photoType),
   183→            photoType,
   184→            savedToLibrary: false,
   185→          };
   186→
   187→          setPhotos(prev => [...prev, newPhoto]);
   188→
   189→          const url = await uploadPhoto(compressed, index);
   190→
   191→          // Actualiza por ID — robusto frente a reordenamientos/eliminaciones
   192→          setPhotos(prev =>
   193→            prev.map(p =>
   194→              p.id === photoId
   195→                ? { ...p, serverUrl: url || '', uploaded: !!url, uploading: false }
   196→                : p
   197→            )
   198→          );
   199→        } catch (err) {
   200→          console.error('Error processing queued photo:', err);
   201→          // Marcamos la foto como fallida (uploaded=false, uploading=false) para que el usuario la vea
   202→          setPhotos(prev =>
   203→            prev.map(p =>
   204→              p.id === photoId
   205→                ? { ...p, uploading: false, uploaded: false }
   206→                : p
   207→            )
   208→          );
   209→        }
   210→
   211→        // Cede al event loop para que el navegador pinte (evita congelar la UI)
   212→        await new Promise(r => setTimeout(r, 30));
   213→      }
   214→    } finally {
   215→      isProcessingRef.current = false;
   216→      setQueueLength(0);
   217→    }
   218→  }, [sStep, miniStep, currentProject?.id]);
   219→

   222→  const addPhoto = useCallback((rawBase64: string) => {
   223→    uploadQueueRef.current.push({ rawBase64, photoType: defaultPhotoType });
   224→    setQueueLength(uploadQueueRef.current.length);
   225→    processQueue();
   226→  }, [processQueue, defaultPhotoType]);
   227→
   228→  const stopStream = useCallback(() => {

   230→      stream.getTracks().forEach(track => track.stop());

   232→    }
   233→    setCameraMode('idle');
   234→  }, [stream]);
   235→

   240→        video: { facingMode, width: { ideal: 1920 }, height: { ideal: 1080 } },

   244→      setCameraMode('streaming');
   245→      if (videoRef.current) {
   246→        videoRef.current.srcObject = mediaStream;
   247→        await videoRef.current.play();

   250→      if (err.name === 'NotAllowedError') setCameraError('Permiso denegado. Permite el acceso a la cámara.');
   251→      else if (err.name === 'NotFoundError') setCameraError('No se encontró ninguna cámara.');
   252→      else setCameraError('Error al acceder a la cámara. Intenta subir una foto desde tu galería.');
   253→    }

   261→  const capturePhoto = useCallback(() => {

   296→
   297→  useEffect(() => {
   298→    if (open && currentProject?.id) {
   299→      setLoadingPhotos(true);
   300→      const zoneIdParam = currentZone?.id ? `&zoneId=${currentZone.id}` : ''
   301→      fetch(`/api/photo-library?projectId=${currentProject.id}&sStep=${sStep}&miniStep=2${zoneIdParam}`)
   302→        .then(res => res.json())
   303→        .then(data => {
   304→          if (data.success && data.data && data.data.length > 0) {
   305→            const existingPhotos: PhotoItem[] = data.data.map((p: any, idx: number) => ({

   310→              uploading: false,

   318→            setPhotos(existingPhotos);
   319→            // If step was already completed, mark it

   325→        })
   326→        .catch(err => {
   327→          console.error('Error loading existing photos:', err);
   328→          setLoadingPhotos(false);
   329→        });
   330→    }
   331→  }, [open, currentProject?.id, sStep, currentZone?.id]);
   332→
   333→  useEffect(() => {
   334→    if (!open) {
   335→      stopStream();
   336→      // Don't clear photos on close — they should persist
   337→      // Only clear transient UI state

   339→      setCameraMode('idle');

   341→      // v2.48: reset del flag "Añadir más fotos" al cerrar el modal
   342→      setShowAddMore(false);

   344→  }, [open, stopStream]);
   345→
   346→  // Lee todos los archivos y los encola. NO procesa aquí (eso saturationa el hilo UI).
   347→  // El procesamiento pesado (compresión + upload) lo hace processQueue secuencialmente.
   348→  const readFileAsDataURL = (file: File): Promise<string> =>
   349→    new Promise((resolve, reject) => {
   350→      const reader = new FileReader();
   351→      reader.onloadend = () => resolve(reader.result as string);
   352→      reader.onerror = () => reject(reader.error);
   353→      reader.readAsDataURL(file);
   354→    });
   355→
   356→  const handleFileSelect = async (e: React.ChangeEvent<HTMLInputElement>) => {
   357→    const files = e.target.files;
   358→    if (!files || files.length === 0) return;
   359→    // Captura el snapshot del FileList ANTES de limpiar el input.

   361→    const fileArr = Array.from(files);
   362→    if (fileInputRef.current) fileInputRef.current.value = '';
   363→    console.log(`[FotosModal] handleFileSelect: ${fileArr.length} archivo(s) seleccionado(s)`);
   364→    await Promise.all(fileArr.map(f => readFileAsDataURL(f).then(addPhoto).catch(err => console.error('Error leyendo archivo:', err))));
   365→  };
   366→
   367→  const handleCameraCapture = async (e: React.ChangeEvent<HTMLInputElement>) => {
   368→    const files = e.target.files;
   369→    if (!files || files.length === 0) return;
   370→    const fileArr = Array.from(files);
   371→    if (cameraInputRef.current) cameraInputRef.current.value = '';
   372→    console.log(`[FotosModal] handleCameraCapture: ${fileArr.length} archivo(s) capturado(s)`);
   373→    await Promise.all(fileArr.map(f => readFileAsDataURL(f).then(addPhoto).catch(err => console.error('Error leyendo archivo de cámara:', err))));
   374→  };
   375→
   376→  // v2.47: si la foto ya está guardada en la biblioteca (dbId presente),
   377→  // llamamos al backend DELETE para que el borrado persista. Si el backend
   378→  // devuelve 409 (Paso 2 ya completado), mostramos error y NO quitamos la
   379→  // foto del estado local — así el usuario entiende que no se puede borrar.
   380→  // Si la foto no está en la biblioteca (todavía no se hizo handleSubmit),
   381→  // simplemente la quitamos del estado local.
   382→  const removePhoto = async (index: number) => {
   383→    const photo = photos[index];
   384→    if (!photo) return;
   385→    if (photo.savedToLibrary && photo.dbId) {
   386→      try {
   387→        const res = await fetch(`/api/photo-library?id=${photo.dbId}`, { method: 'DELETE' });
   388→        if (!res.ok) {
   389→          const json = await res.json().catch(() => null);
   390→          toast.error(json?.error || 'No se puede eliminar esta foto.');
   391→          return; // NO la quitamos del estado local
   392→        }
   393→      } catch (e) {
   394→        console.error('Error deleting photo from backend:', e);
   395→        toast.error('Error de conexión al eliminar la foto.');
   396→        return;
   397→      }
   398→    }
   399→    setPhotos(prev => prev.filter((_, i) => i !== index));
   400→  };
   401→
   402→  const uploadingCount = photos.filter(p => p.uploading).length;
   403→  const isQueueBusy = queueLength > 0 || uploadingCount > 0;
   404→  const canSubmit = photos.length >= minPhotos && uploadingCount === 0 && queueLength === 0;
   405→  const totalSize = photos.reduce((sum, p) => sum + p.estimatedSize, 0);
   406→
   407→  const handleSubmit = async () => {
   408→    if (!canSubmit) return;
   409→    setIsSubmitting(true);
   410→    try {
   411→      const urls = photos.map(p => p.serverUrl || p.preview).join(',');
   412→
   413→      // v2.40: Cada foto del Paso 2 se vincula automáticamente a un elemento
   414→      // de inventario "borrador" que el usuario clasificará en el Paso 3.
   415→      // Así la foto está vinculada al registro DESDE EL MOMENTO en que se
   416→      // toma, no después. El flujo en InventarioModal cambia: el usuario ya
   417→      // no necesita vincular manualmente — solo rellenar los datos del
   418→      // elemento (nombre, categoría, decisión, etc.) y al hacerlo el
   419→      // borrador pasa a ser un item clasificado.
   420→      const sName = sStepData?.japaneseName || `S${sStep}`;
   421→      const zoneName = currentZone?.name || 'Zona';
   422→      const userLabel = currentUser?.name || 'Usuario';
   423→      let savedCount = 0;
   424→      let failedCount = 0;
   425→
   426→      // Procesamos SECUENCIALMENTE (no en paralelo) porque cada foto necesita
   427→      // el id del item creado en el paso anterior para vincularse.
   428→      // v2.48: trackea el dbId devuelto por el backend para cada foto recién
   429→      // guardada, y actualiza el estado local al final — así, si el usuario
   430→      // re-abre "Añadir más fotos", las fotos ya guardadas no se re-envían.
   431→      const newlySavedDbIds: { localId: string; dbId: string }[] = [];
   432→      for (let idx = 0; idx < photos.length; idx++) {
   433→        const p = photos[idx];
   434→        if (p.savedToLibrary) { savedCount++; continue; }
   435→        const photoUrlForDb = p.serverUrl || p.preview; // base64 fallback si el upload falló
   436→        const photoTitle = p.title || generatePhotoTitle(idx, p.photoType);
   437→
   438→        // 1) Guardar en PhotoLibrary
   439→        let photoId: string | null = null;
   440→        try {
   441→          const libRes = await fetch('/api/photo-library', {
   442→            method: 'POST',
   443→            headers: { 'Content-Type': 'application/json' },
   444→            body: JSON.stringify({
   445→              sStep,
   446→              miniStep: 2,
   447→              title: photoTitle,
   448→              description: `${sName} - ${zoneName} - Paso 2 Fotos - Subida por ${userLabel}`,
   449→              photoUrl: photoUrlForDb,
   450→              photoType: p.photoType || 'antes',
   451→              category: `paso2_s${sStep}`,
   452→              tags: JSON.stringify([`S${sStep}`, sName, zoneName, `paso2`, p.photoType]),
   453→              projectId: currentProject?.id,
   454→              zoneId: currentZone?.id || null,
   455→              uploadedBy: currentUser?.id || null,
   456→            }),
   457→          });
   458→          const libJson = await libRes.json();
   459→          if (libJson.success && libJson.data?.id) {
   460→            photoId = libJson.data.id;
   461→            newlySavedDbIds.push({ localId: p.id, dbId: photoId });
   462→            savedCount++;
   463→          } else {
   464→            failedCount++;
   465→            console.warn(`[FotosModal] Foto ${idx + 1} no se pudo guardar en la biblioteca`);
   466→            continue;
   467→          }
   468→        } catch (err) {
   469→          failedCount++;
   470→          console.error(`[FotosModal] Error guardando foto ${idx + 1} en biblioteca:`, err);
   471→          continue;
   472→        }
   473→
   474→        // 2) Crear elemento de inventario BORRADOR vinculado a esta foto.
