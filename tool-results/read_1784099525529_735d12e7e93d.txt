   123→// ─── Component ───────────────────────────────────────────────────────────────
   124→
   125→export default function GestorPanel() {
   126→  const { setCurrentView, fetchProjects, fetchCompanies, projects, setCurrentProject, currentProject } = use5SStore()
   127→  const [activeTab, setActiveTab] = useState<GestorTab>('dashboard')
   128→
   129→  // Platform stats
   130→  const [stats, setStats] = useState<PlatformStats | null>(null)
   131→  const [isLoadingStats, setIsLoadingStats] = useState(false)
   132→
   133→  // Companies management
   134→  const [showNewCompany, setShowNewCompany] = useState(false)
   135→  const [newCompanyName, setNewCompanyName] = useState('')
   136→  const [newCompanyDesc, setNewCompanyDesc] = useState('')
   137→  const [editingCompany, setEditingCompany] = useState<string | null>(null)
   138→  const [editCompanyData, setEditCompanyData] = useState({ name: '', description: '', active: true })
   139→
   140→  // Admin management
   141→  const [assigningAdminTo, setAssigningAdminTo] = useState<string | null>(null)
   142→  const [searchEmail, setSearchEmail] = useState('')
   143→  const [searchResults, setSearchResults] = useState<PlatformStats['users']>([])
   144→  const [isSearching, setIsSearching] = useState(false)
   145→  const [showCreateUser, setShowCreateUser] = useState(false)
   146→  const [newAdminData, setNewAdminData] = useState({ name: '', email: '', password: '' })
   147→  const [isAssigning, setIsAssigning] = useState(false)
   148→
   149→  // ─── Data loading ────────────────────────────────────────────────────────
   150→  const loadStats = useCallback(async () => {
   151→    setIsLoadingStats(true)
   152→    try {
   153→      const res = await fetch('/api/platform-stats')
   154→      const data = await res.json()
   155→      if (data.success) {
   156→        setStats(data.data)
   157→      }
   158→    } catch (error) {
   159→      console.error('Error loading platform stats:', error)
   160→    } finally {
   161→      setIsLoadingStats(false)
   162→    }
   163→  }, [])
   164→
   165→  useEffect(() => {
   166→    loadStats()
   167→  }, [loadStats])
   168→
   169→  // ─── Company actions ────────────────────────────────────────────────────
   170→  const handleCreateCompany = async () => {
   171→    if (!newCompanyName.trim()) return
   172→    try {
   173→      const res = await fetch('/api/companies', {
   174→        method: 'POST',
   175→        headers: { 'Content-Type': 'application/json' },
   176→        body: JSON.stringify({ name: newCompanyName, description: newCompanyDesc || undefined }),
   177→      })
   178→      if (res.ok) {
   179→        setShowNewCompany(false)
   180→        setNewCompanyName('')
   181→        setNewCompanyDesc('')
   182→        await loadStats()
   183→        await fetchCompanies()
   184→      } else {
   185→        const data = await res.json()
   186→        alert(data.error || 'Error al crear empresa')
   187→      }
   188→    } catch (error) {
   189→      console.error('Error creating company:', error)
   190→    }
   191→  }
   192→
   193→  const handleUpdateCompany = async (companyId: string) => {
   194→    try {
   195→      const res = await fetch(`/api/companies/${companyId}`, {
   196→        method: 'PUT',
   197→        headers: { 'Content-Type': 'application/json' },
   198→        body: JSON.stringify(editCompanyData),
   199→      })
   200→      if (res.ok) {
   201→        setEditingCompany(null)
   202→        await loadStats()
   203→        await fetchCompanies()
   204→      } else {
   205→        const data = await res.json()
   206→        alert(data.error || 'Error al actualizar empresa')
   207→      }
   208→    } catch (error) {
   209→      console.error('Error updating company:', error)
   210→    }
   211→  }
   212→
   213→  const handleToggleCompanyActive = async (companyId: string, currentActive: boolean) => {
   214→    try {
   215→      const res = await fetch(`/api/companies/${companyId}`, {
   216→        method: 'PUT',
   217→        headers: { 'Content-Type': 'application/json' },
   218→        body: JSON.stringify({ active: !currentActive }),
   219→      })
   220→      if (res.ok) {
   221→        await loadStats()
   222→        await fetchCompanies()
   223→      }
   224→    } catch (error) {
   225→      console.error('Error toggling company active:', error)
   226→    }
   227→  }
   228→
   229→  // ─── Admin assignment actions ───────────────────────────────────────────
   230→  const handleSearchUser = async () => {
   231→    if (!searchEmail.trim() || searchEmail.length < 3) {
   232→      setSearchResults([])
   233→      return
   234→    }
   235→    setIsSearching(true)
   236→    try {
   237→      const res = await fetch(`/api/users?search=${encodeURIComponent(searchEmail)}`)
   238→      const data = await res.json()
   239→      if (data.success) {
   240→        setSearchResults(data.users || [])
   241→      }
   242→    } catch (error) {
   243→      console.error('Error searching users:', error)
   244→    } finally {
   245→      setIsSearching(false)
   246→    }
   247→  }
   248→
   249→  useEffect(() => {
   250→    if (searchEmail.length >= 3) {
   251→      const t = setTimeout(() => handleSearchUser(), 300)
   252→      return () => clearTimeout(t)
   253→    } else {
   254→      setSearchResults([])
   255→    }
   256→  }, [searchEmail])
   257→
   258→  const handleAssignExistingUser = async (userId: string) => {
   259→    if (!assigningAdminTo) return
   260→    setIsAssigning(true)
   261→    try {
   262→      // First, ensure the user has 'admin' role
   263→      await fetch('/api/users', {
   264→        method: 'PUT',
   265→        headers: { 'Content-Type': 'application/json' },
   266→        body: JSON.stringify({ id: userId, role: 'admin' }),
   267→      })
   268→
   269→      // Then assign them to the company as admin
   270→      const res = await fetch(`/api/companies/${assigningAdminTo}/members`, {
   271→        method: 'POST',
   272→        headers: { 'Content-Type': 'application/json' },
   273→        body: JSON.stringify({ userId, role: 'admin_empresa' }),
   274→      })
   275→
   276→      if (res.ok) {
   277→        setAssigningAdminTo(null)
   278→        setSearchEmail('')
   279→        setSearchResults([])
   280→        await loadStats()
   281→      } else {
   282→        const data = await res.json()
   283→        alert(data.error || 'Error al asignar administrador')
   284→      }
   285→    } catch (error) {
   286→      console.error('Error assigning admin:', error)
   287→    } finally {
   288→      setIsAssigning(false)
   289→    }
   290→  }
   291→
   292→  const handleCreateAndAssignAdmin = async () => {
   293→    if (!assigningAdminTo) return
   294→    if (!newAdminData.name.trim() || !newAdminData.email.trim() || newAdminData.password.length < 6) {
   295→      alert('Completa todos los campos. La contraseña debe tener al menos 6 caracteres.')
   296→      return
   297→    }
   298→    setIsAssigning(true)
   299→    try {
   300→      // 1. Create user with role 'admin'
   301→      const createRes = await fetch('/api/users', {
   302→        method: 'POST',
   303→        headers: { 'Content-Type': 'application/json' },
   304→        body: JSON.stringify({
   305→          name: newAdminData.name,
   306→          email: newAdminData.email,
   307→          password: newAdminData.password,
   308→          role: 'admin',
   309→          active: true,
   310→        }),
   311→      })
   312→
   313→      if (!createRes.ok) {
   314→        const data = await createRes.json()
   315→        alert(data.error || 'Error al crear usuario')
   316→        setIsAssigning(false)
   317→        return
   318→      }
   319→
   320→      const createData = await createRes.json()
   321→      const newUserId = createData.user?.id
   322→
   323→      if (!newUserId) {
   324→        alert('Usuario creado pero no se pudo obtener el ID. Asigna el admin manualmente.')
   325→        setIsAssigning(false)
   326→        return
   327→      }
   328→
   329→      // 2. Assign user to company
   330→      const assignRes = await fetch(`/api/companies/${assigningAdminTo}/members`, {
   331→        method: 'POST',
   332→        headers: { 'Content-Type': 'application/json' },
   333→        body: JSON.stringify({ userId: newUserId, role: 'admin_empresa' }),
   334→      })
   335→
   336→      if (assignRes.ok) {
   337→        setAssigningAdminTo(null)
   338→        setShowCreateUser(false)
   339→        setNewAdminData({ name: '', email: '', password: '' })
   340→        await loadStats()
   341→      } else {
   342→        const data = await assignRes.json()
   343→        alert(data.error || 'Usuario creado pero error al asignar a la empresa')
   344→      }
   345→    } catch (error) {
   346→      console.error('Error creating admin:', error)
   347→    } finally {
   348→      setIsAssigning(false)
   349→    }
   350→  }
   351→
   352→  const handleRemoveAdmin = async (companyId: string, userId: string) => {
   353→    if (!confirm('¿Quitar a este administrador de la empresa?')) return
   354→    try {
   355→      const res = await fetch(`/api/companies/${companyId}/members`, {
   356→        method: 'DELETE',
   357→        headers: { 'Content-Type': 'application/json' },
   358→        body: JSON.stringify({ memberUserId: userId }),
   359→      })
   360→      if (res.ok) {
   361→        await loadStats()
   362→      } else {
   363→        const data = await res.json()
   364→        alert(data.error || 'Error al quitar administrador')
   365→      }
   366→    } catch (error) {
   367→      console.error('Error removing admin:', error)
   368→    }
   369→  }
   370→  // ─── Render ──────────────────────────────────────────────────────────────
   371→  const tabs: { key: GestorTab; label: string; icon: React.ReactNode }[] = [
   372→    { key: 'dashboard', label: 'KPIs', icon: <Activity className="h-4 w-4" /> },
   373→    { key: 'empresas', label: 'Empresas', icon: <Building2 className="h-4 w-4" /> },
   374→    { key: 'admins', label: 'Administradores', icon: <Shield className="h-4 w-4" /> },
   375→  ]
   376→
   377→  return (
   378→    <div className="min-h-screen bg-gradient-to-b from-slate-950 via-violet-950/20 to-slate-950 text-white">
   379→      {/* Header */}
   380→      <header className="border-b border-violet-800/30 bg-slate-950/80 backdrop-blur-sm sticky top-0 z-10">
   381→        <div className="max-w-7xl mx-auto px-4 py-3 flex items-center justify-between">
   382→          <div className="flex items-center gap-3">
   383→            <div className="w-10 h-10 rounded-xl bg-gradient-to-br from-violet-500 to-purple-700 flex items-center justify-center shadow-lg shadow-violet-500/30">
   384→              <Crown className="h-5 w-5 text-yellow-300" />
   385→            </div>
   386→            <div>
   387→              <h1 className="text-lg font-bold bg-gradient-to-r from-violet-300 to-purple-200 bg-clip-text text-transparent">
   388→                Gestor Panel
   389→              </h1>
   390→              <p className="text-xs text-violet-400">Dueño de la Plataforma</p>
   391→            </div>
   392→          </div>
   393→          <div className="flex items-center gap-2">
   394→            <Button
   395→              variant="outline"
   396→              size="sm"
   397→              onClick={() => {
   398→                if (!currentProject && projects.length > 0) {
   399→                  setCurrentProject(projects[0])
   400→                }
   401→                setCurrentView('board')
   402→              }}
   403→              className="gap-1.5 border-violet-700/50 text-violet-300 hover:bg-violet-900/30 hover:text-violet-200"
   404→            >
   405→              ← Vista App
   406→            </Button>
   407→            <Badge className="bg-gradient-to-r from-violet-600 to-purple-600 text-white border-0 px-3 py-1">
   408→              <Crown className="h-3 w-3 mr-1" /> GESTOR
   409→            </Badge>
   410→          </div>
   411→        </div>
   412→      </header>
   413→
   414→      {/* Tabs */}
   415→      <div className="border-b border-violet-800/20 bg-slate-950/60 backdrop-blur-sm">
   416→        <div className="max-w-7xl mx-auto px-4 flex gap-0 overflow-x-auto">
   417→          {tabs.map(tab => (
   418→            <button
   419→              key={tab.key}
   420→              onClick={() => setActiveTab(tab.key)}
   421→              className={`flex items-center gap-2 px-4 py-3 text-sm font-medium border-b-2 transition-all whitespace-nowrap ${
   422→                activeTab === tab.key
   423→                  ? 'border-violet-400 text-violet-300 bg-violet-900/20'
   424→                  : 'border-transparent text-violet-500 hover:text-violet-300 hover:border-violet-700/50'
   425→              }`}
   426→            >
   427→              {tab.icon}
   428→              {tab.label}
   429→            </button>
   430→          ))}
   431→        </div>
   432→      </div>
   433→
   434→      {/* Content */}
   435→      <main className="max-w-7xl mx-auto px-4 py-6">
   436→        <AnimatePresence mode="wait">
   437→          {/* ═══ DASHBOARD TAB (KPIs) ═══ */}
   438→          {activeTab === 'dashboard' && (
   439→            <motion.div key="dashboard" initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0 }} className="space-y-6">
   440→              {isLoadingStats ? (
   441→                <div className="flex justify-center py-20"><Loader2 className="h-10 w-10 text-violet-400 animate-spin" /></div>
   442→              ) : stats ? (
   443→                <>
   444→                  {/* KPI Cards */}
   445→                  <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   446→                    <Card className="bg-gradient-to-br from-violet-900/40 to-purple-900/20 border-violet-700/30 text-white">
   447→                      <CardContent className="p-4">
   448→                        <div className="flex items-center justify-between mb-2">
   449→                          <Building2 className="h-5 w-5 text-violet-400" />
   450→                          <Badge className="bg-violet-800/50 text-violet-300 border-0 text-[10px]">
   451→                            {stats.totals.activeCompanies}/{stats.totals.companies} activas
   452→                          </Badge>
   453→                        </div>
   454→                        <p className="text-2xl font-bold">{stats.totals.companies}</p>
   455→                        <p className="text-xs text-violet-400">Empresas</p>
   456→                      </CardContent>
   457→                    </Card>
   458→                    <Card className="bg-gradient-to-br from-blue-900/40 to-indigo-900/20 border-blue-700/30 text-white">
   459→                      <CardContent className="p-4">
   460→                        <div className="flex items-center justify-between mb-2">
   461→                          <Users className="h-5 w-5 text-blue-400" />
   462→                          <Badge className="bg-blue-800/50 text-blue-300 border-0 text-[10px]">
   463→                            {stats.totals.activeUsers}/{stats.totals.users} activos
   464→                          </Badge>
   465→                        </div>
   466→                        <p className="text-2xl font-bold">{stats.totals.users}</p>
   467→                        <p className="text-xs text-blue-400">Usuarios</p>
   468→                      </CardContent>
   469→                    </Card>
   470→                    <Card className="bg-gradient-to-br from-emerald-900/40 to-green-900/20 border-emerald-700/30 text-white">
   471→                      <CardContent className="p-4">
   472→                        <div className="flex items-center justify-between mb-2">
   473→                          <TrendingUp className="h-5 w-5 text-emerald-400" />
   474→                          <Badge className="bg-emerald-800/50 text-emerald-300 border-0 text-[10px]">
   475→                            {stats.totals.activeProjects}/{stats.totals.projects} activos
   476→                          </Badge>
   477→                        </div>
   478→                        <p className="text-2xl font-bold">{stats.totals.projects}</p>
   479→                        <p className="text-xs text-emerald-400">Proyectos</p>
   480→                      </CardContent>
   481→                    </Card>
   482→                    <Card className="bg-gradient-to-br from-orange-900/40 to-amber-900/20 border-orange-700/30 text-white">
   483→                      <CardContent className="p-4">
   484→                        <div className="flex items-center justify-between mb-2">
   485→                          <AlertTriangle className="h-5 w-5 text-orange-400" />
   486→                          <Badge className="bg-orange-800/50 text-orange-300 border-0 text-[10px]">
   487→                            de {stats.totals.actions} total
   488→                          </Badge>
   489→                        </div>
   490→                        <p className="text-2xl font-bold">{stats.totals.openActions}</p>
   491→                        <p className="text-xs text-orange-400">Acciones Abiertas</p>
   492→                      </CardContent>
   493→                    </Card>
   494→                  </div>
   495→
   496→                  {/* Role distribution + Platform health */}
   497→                  <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   498→                    <Card className="bg-slate-900/60 border-violet-700/20">
   499→                      <CardHeader className="pb-3">
   500→                        <CardTitle className="text-sm text-violet-300 flex items-center gap-2">
   501→                          <Users className="h-4 w-4" /> Distribución de Roles
   502→                        </CardTitle>
   503→                      </CardHeader>
   504→                      <CardContent>
   505→                        <div className="space-y-2">
   506→                          {Object.entries(stats.roleDistribution).map(([role, count]) => (
   507→                            <div key={role} className="flex items-center justify-between">
   508→                              <div className="flex items-center gap-2">
   509→                                <Badge className={`${ROLE_COLORS[role] || 'bg-gray-100 text-gray-700 border-gray-200'} border text-xs`}>
   510→                                  {ROLE_LABELS[role] || role}
   511→                                </Badge>
   512→                              </div>
   513→                              <div className="flex items-center gap-2">
   514→                                <div className="w-24 h-2 bg-slate-800 rounded-full overflow-hidden">
   515→                                  <div
   516→                                    className="h-full bg-gradient-to-r from-violet-500 to-purple-500 rounded-full"
   517→                                    style={{ width: `${Math.min((count / stats.totals.users) * 100, 100)}%` }}
   518→                                  />
   519→                                </div>
   520→                                <span className="text-sm font-semibold text-white w-8 text-right">{count}</span>
   521→                              </div>
   522→                            </div>
   523→                          ))}
   524→                        </div>
   525→                      </CardContent>
   526→                    </Card>
   527→
   528→                    <Card className="bg-slate-900/60 border-violet-700/20">
   529→                      <CardHeader className="pb-3">
   530→                        <CardTitle className="text-sm text-violet-300 flex items-center gap-2">
   531→                          <Activity className="h-4 w-4" /> Salud de Plataforma
   532→                        </CardTitle>
   533→                      </CardHeader>
   534→                      <CardContent>
   535→                        <div className="space-y-3">
   536→                          <div className="flex items-center justify-between">
   537→                            <span className="text-xs text-violet-400">Empresas activas</span>
   538→                            <span className="text-sm font-bold text-white">
   539→                              {stats.totals.companies > 0 ? Math.round((stats.totals.activeCompanies / stats.totals.companies) * 100) : 0}%
   540→                            </span>
