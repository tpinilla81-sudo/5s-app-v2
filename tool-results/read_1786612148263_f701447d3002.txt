    25→// ── Animated counter hook ──
    26→function useCounter(end: number, duration = 2000, startOnView = true) {
    27→  const [count, setCount] = useState(0)
    28→  const ref = useRef<HTMLDivElement>(null)
    29→  const started = useRef(false)
    30→
    31→  useEffect(() => {
    32→    if (!startOnView) return
    33→    const el = ref.current
    34→    if (!el) return
    35→    const obs = new IntersectionObserver(([entry]) => {
    36→      if (entry.isIntersecting && !started.current) {
    37→        started.current = true
    38→        const startTime = performance.now()
    39→        const step = () => {
    40→          const progress = Math.min((performance.now() - startTime) / duration, 1)
    41→          const eased = 1 - Math.pow(1 - progress, 3) // easeOutCubic
    42→          setCount(Math.round(eased * end))
    43→          if (progress < 1) requestAnimationFrame(step)
    44→        }
    45→        requestAnimationFrame(step)
    46→      }
    47→    }, { threshold: 0.3 })
    48→    obs.observe(el)
    49→    return () => obs.disconnect()
    50→  }, [end, duration, startOnView])
    51→
    52→  return { count, ref }
    53→}
