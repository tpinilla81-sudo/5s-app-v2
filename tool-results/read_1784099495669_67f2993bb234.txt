   286→  id: string
   287→  title: string

   291→function ChecklistEditor({ content, onChange }: { content: string; onChange: (v: string) => void }) {
   292→  let parsed: { sections: ChecklistSection[] }
   293→  try {
   294→    parsed = JSON.parse(content)
   295→  } catch {
   296→    return (
   297→      <div className="p-4 bg-red-50 border border-red-200 rounded-lg text-sm text-red-600">
   298→        El JSON no es válido. Corrígelo en modo JSON o carga el contenido por defecto.
   299→      </div>
   300→    )
   301→  }
   302→
   303→  const sections = parsed.sections || []
   304→  const sPrefix = String(content.match(/"id"\s*:\s*"(\d+)\./)?.[1] || '1')
   305→
   306→  const update = (newSections: ChecklistSection[]) => {
   307→    onChange(JSON.stringify({ ...parsed, sections: newSections }, null, 2))
   308→  }
   309→
   310→  const addSection = () => {
   311→    const newId = `${sPrefix}.${sections.length + 1}`
   312→    update([...sections, { id: newId, title: 'Nueva Sección', items: [] }])
   313→  }
   314→
   315→  const removeSection = (idx: number) => {
   316→    update(sections.filter((_, i) => i !== idx))
   317→  }
   318→
   319→  const updateSection = (idx: number, field: 'id' | 'title', value: string) => {
   320→    const updated = [...sections]
   321→    updated[idx] = { ...updated[idx], [field]: value }
   322→    update(updated)
   323→  }
   324→
   325→  const moveSection = (idx: number, dir: -1 | 1) => {
   326→    const target = idx + dir
   327→    if (target < 0 || target >= sections.length) return
   328→    const updated = [...sections]
   329→    ;[updated[idx], updated[target]] = [updated[target], updated[idx]]
   330→    update(updated)
   331→  }
   332→
   333→  const addItem = (sectionIdx: number) => {
   334→    const sec = sections[sectionIdx]
   335→    const newItemId = `${sec.id}.${sec.items.length + 1}`
   336→    const updated = [...sections]
   337→    updated[sectionIdx] = {
   338→      ...updated[sectionIdx],
   339→      items: [...updated[sectionIdx].items, { id: newItemId, description: '', hasOther: false }]
   340→    }
   341→    update(updated)
   342→  }
   343→
   344→  const removeItem = (sectionIdx: number, itemIdx: number) => {
   345→    const updated = [...sections]
   346→    updated[sectionIdx] = {
   347→      ...updated[sectionIdx],
   348→      items: updated[sectionIdx].items.filter((_, i) => i !== itemIdx)
   349→    }
   350→    update(updated)
   351→  }
   352→
   353→  const updateItem = (sectionIdx: number, itemIdx: number, field: string, value: string | boolean) => {
   354→    const updated = [...sections]
   355→    updated[sectionIdx] = {
   356→      ...updated[sectionIdx],
   357→      items: updated[sectionIdx].items.map((item, i) =>
   358→        i === itemIdx ? { ...item, [field]: value } : item
   359→      )
   360→    }
   361→    update(updated)
   362→  }
   363→
   364→  const moveItem = (sectionIdx: number, itemIdx: number, dir: -1 | 1) => {
   365→    const items = sections[sectionIdx].items
   366→    const target = itemIdx + dir
   367→    if (target < 0 || target >= items.length) return
   368→    const updated = [...sections]
   369→    const newItems = [...updated[sectionIdx].items]
   370→    ;[newItems[itemIdx], newItems[target]] = [newItems[target], newItems[itemIdx]]
   371→    updated[sectionIdx] = { ...updated[sectionIdx], items: newItems }
   372→    update(updated)
   373→  }
   374→
   375→  return (
   376→    <div className="space-y-4">
   377→      {sections.map((section, sIdx) => (
   378→        <div key={sIdx} className="border-2 rounded-lg overflow-hidden bg-white shadow-sm">
   379→          {/* Section header */}
   380→          <div className="flex items-center gap-3 px-4 py-3 bg-gray-50 border-b">
   381→            <div className="flex flex-col gap-0.5">
   382→              <button onClick={() => moveSection(sIdx, -1)} className="text-gray-400 hover:text-gray-600 leading-none" title="Subir sección">
   383→                <ChevronUp className="h-4 w-4" />
   384→              </button>
   385→              <button onClick={() => moveSection(sIdx, 1)} className="text-gray-400 hover:text-gray-600 leading-none" title="Bajar sección">
   386→                <ChevronDown className="h-4 w-4" />
   387→              </button>
   388→            </div>
   389→            <Input
   390→              value={section.id}
   391→              onChange={(e) => updateSection(sIdx, 'id', e.target.value)}
   392→              className="w-20 h-9 text-sm font-mono"
   393→              placeholder="ID"
   394→            />
   395→            <Input
   396→              value={section.title}
   397→              onChange={(e) => updateSection(sIdx, 'title', e.target.value)}
   398→              className="flex-1 h-9 text-base font-semibold"
   399→              placeholder="Título de sección"
   400→            />
   401→            <Button variant="ghost" size="sm" onClick={() => addItem(sIdx)}
   402→              className="h-9 w-9 p-0 text-green-600 hover:text-green-700 hover:bg-green-50" title="Añadir item">
   403→              <Plus className="h-5 w-5" />
   404→            </Button>
   405→            <Button variant="ghost" size="sm" onClick={() => removeSection(sIdx)}
   406→              className="h-9 w-9 p-0 text-red-500 hover:text-red-600 hover:bg-red-50" title="Eliminar sección">
   407→              <Trash2 className="h-4 w-4" />
   408→            </Button>
   409→          </div>
   410→
   411→          {/* Items */}
   412→          <div className="p-3 space-y-2">
   413→            {section.items.length === 0 && (
   414→              <p className="text-sm text-muted-foreground italic px-2 py-2">Sin items. Pulsa + para añadir.</p>
   415→            )}
   416→            {section.items.map((item, iIdx) => (
   417→              <div key={iIdx} className="flex items-center gap-3 group">
   418→                <div className="flex flex-col gap-0.5">
   419→                  <button onClick={() => moveItem(sIdx, iIdx, -1)} className="text-gray-300 hover:text-gray-500 leading-none" title="Subir">
   420→                    <ChevronUp className="h-3.5 w-3.5" />
   421→                  </button>
   422→                  <button onClick={() => moveItem(sIdx, iIdx, 1)} className="text-gray-300 hover:text-gray-500 leading-none" title="Bajar">
   423→                    <ChevronDown className="h-3.5 w-3.5" />
   424→                  </button>
   425→                </div>
   426→                <Input
   427→                  value={item.id}
   428→                  onChange={(e) => updateItem(sIdx, iIdx, 'id', e.target.value)}
   429→                  className="w-20 h-8 text-xs font-mono"
   430→                  placeholder="ID"
   431→                />
   432→                <Input
   433→                  value={item.description}
   434→                  onChange={(e) => updateItem(sIdx, iIdx, 'description', e.target.value)}
   435→                  className="flex-1 h-8 text-sm"
   436→                  placeholder="Descripción del item"
   437→                />
   438→                <label className="flex items-center gap-1.5 text-xs text-muted-foreground whitespace-nowrap cursor-pointer shrink-0">
   439→                  <input
   440→                    type="checkbox"
   441→                    checked={item.hasOther}
   442→                    onChange={(e) => updateItem(sIdx, iIdx, 'hasOther', e.target.checked)}
   443→                    className="rounded border-gray-300 h-4 w-4"
   444→                  />
   445→                  Otros
   446→                </label>
   447→                <Button variant="ghost" size="sm"
   448→                  onClick={() => removeItem(sIdx, iIdx)}
   449→                  className="h-8 w-8 p-0 text-red-400 hover:text-red-500 opacity-0 group-hover:opacity-100 transition-opacity" title="Eliminar">
   450→                  <Trash2 className="h-4 w-4" />
   451→                </Button>
   452→              </div>
   453→            ))}
   454→          </div>
   455→        </div>
   456→      ))}
   457→
   458→      <Button variant="outline" onClick={addSection}
   459→        className="w-full border-dashed border-2 text-green-600 hover:bg-green-50 hover:border-green-400 gap-1 h-10">
   460→        <Plus className="h-5 w-5" />
   461→        Añadir sección
   462→      </Button>
   463→
   464→      <div className="text-sm text-muted-foreground text-center">
   465→        {sections.length} sección(es) · {sections.reduce((s, sec) => s + sec.items.length, 0)} item(s) en total
   466→      </div>
   467→    </div>
   468→  )
   469→}

   547→              <button onClick={() => moveQuestion(qIdx, 1)} className="text-gray-400 hover:text-gray-600 leading-none">
