     5→     5→  Dialog,

     8→     8→  DialogTitle,
     9→     9→} from '@/components/ui/dialog';
    10→    10→import { Button } from '@/components/ui/button';
    11→    11→import { Card, CardContent } from '@/components/ui/card';
    12→    12→import { Badge } from '@/components/ui/badge';
    13→    13→import {

    37→    37→
    38→    38→interface FotosModalProps {
    39→    39→  open: boolean;
    40→    40→  onClose: () => void;
    41→    41→  sStep: number;
    42→    42→  miniStep: number;
    43→    43→}
    44→    44→
    45→    45→const BEFORE_PROMPT_BY_S: Record<number, string> = {
    46→    46→  1: 'Fotografía los elementos innecesarios que hay en la zona. Esto servirá como referencia "ANTES" de la clasificación.',
    47→    47→  2: 'Fotografía cómo está organizada la zona actualmente. Esto servirá como referencia "ANTES" de la reorganización.',
    48→    48→  3: 'Fotografía los puntos de suciedad de la zona. Esto servirá como referencia "ANTES" de la limpieza.',
    49→    49→  4: 'Fotografía el estado actual de la zona. Esto servirá como referencia "ANTES" de la estandarización.',
    50→    50→  5: 'Fotografía el nivel de cumplimiento de los estándares. Esto servirá como referencia "ANTES" de la disciplina.',
    51→    51→};
    52→    52→
    53→    53→interface PhotoItem {
    54→    54→  preview: string;
    55→    55→  serverUrl: string;
    56→    56→  uploaded: boolean;
    57→    57→  uploading: boolean;
    58→    58→  estimatedSize: number;
    59→    59→  title: string; // Título descriptivo de la foto
    60→    60→  photoType: string; // "antes", "despues", "referencia", "hallazgo"
    61→    61→  savedToLibrary: boolean; // Si ya se guardó en la biblioteca
    62→    62→}
    63→    63→
    64→    64→export default function FotosModal({ open, onClose, sStep, miniStep }: FotosModalProps) {
    65→    65→  const { fetchProgress, currentUser, adminFreeNavigation, currentProject, currentZone, canPerform, canView, hasPermission, openModal } = use5SStore();
    66→    66→  const sStepData = S_STEPS.find(s => s.id === sStep);
    67→    67→  const miniStepData = MINI_STEPS.find(m => m.id === miniStep);
    68→    68→
    69→    69→  const fileInputRef = useRef<HTMLInputElement>(null);
    70→    70→  const cameraInputRef = useRef<HTMLInputElement>(null);
    71→    71→  const videoRef = useRef<HTMLVideoElement>(null);
    72→    72→  const canvasRef = useRef<HTMLCanvasElement>(null);
    73→    73→
    74→    74→  const canSkipSteps = hasPermission('skip_steps');
    75→    75→  const canPerformStep = canPerform(sStep, miniStep);
    76→    76→  const canViewStep = canView(sStep, miniStep);
    77→    77→  // Permission-driven: read-only if no execute perm OR if candado closed for skip_steps users
    78→    78→  const isReadOnly = !canPerformStep || (canSkipSteps && !adminFreeNavigation);
    79→    79→
    80→    80→  const [isFullscreen, setIsFullscreen] = useState(true);
    81→    81→  const [photos, setPhotos] = useState<PhotoItem[]>([]);
    82→    82→  const [isSubmitting, setIsSubmitting] = useState(false);
    83→    83→  const [isCompleted, setIsCompleted] = useState(false);
    84→    84→  const [cameraMode, setCameraMode] = useState<'idle' | 'streaming' | 'capturing'>('idle');
    85→    85→  const [stream, setStream] = useState<MediaStream | null>(null);
    86→    86→  const [facingMode, setFacingMode] = useState<'environment' | 'user'>('environment');
    87→    87→  const [cameraError, setCameraError] = useState<string | null>(null);
    88→    88→  const [captureFlash, setCaptureFlash] = useState(false);
    89→    89→  const [activeTab, setActiveTab] = useState<'camera' | 'upload'>('camera');
    90→    90→  const [defaultPhotoType, setDefaultPhotoType] = useState<string>('antes');
    91→    91→
    92→    92→  const beforePrompt = BEFORE_PROMPT_BY_S[sStep] || 'Fotografía la zona para documentar su estado actual.';
    93→    93→
    94→    94→  // Generate auto title with traceability
    95→    95→  const generatePhotoTitle = (index: number, photoType: string): string => {
    96→    96→    const sName = sStepData?.japaneseName || `S${sStep}`
    97→    97→    const zoneName = currentZone?.name || 'Zona'
    98→    98→    const typeLabel = photoType === 'antes' ? 'ANTES' : photoType === 'despues' ? 'DESPUÉS' : photoType === 'referencia' ? 'Referencia' : 'Hallazgo'
    99→    99→    const date = new Date().toLocaleDateString('es-ES')
   100→   100→    return `S${sStep} ${sName} - ${zoneName} - ${typeLabel} ${index + 1} (${date})`
   101→   101→  }
   102→   102→
   103→   103→  const uploadPhoto = async (base64Data: string, index: number): Promise<string | null> => {
   104→   104→    try {
   105→   105→      const projectId = currentProject?.id || 'unknown';
   106→   106→      const filename = generatePhotoFilename(projectId, sStep, miniStep, index);
   107→   107→      const file = base64toFile(base64Data, filename);
   108→   108→
   109→   109→      const formData = new FormData();
   110→   110→      formData.append('file', file);
   111→   111→      formData.append('filename', filename);
   112→   112→
   113→   113→      const res = await fetch('/api/upload', {
   114→   114→        method: 'POST',
   115→   115→        body: formData,
   116→   116→      });
   117→   117→
   118→   118→      const data = await res.json();
   119→   119→      if (data.success && data.url) return data.url;
   120→   120→      return null;
   121→   121→    } catch (error) {
   122→   122→      console.error('Upload error:', error);
   123→   123→      return null;
   124→   124→    }
   125→   125→  };
   126→   126→
   127→   127→  const addPhoto = useCallback(async (rawBase64: string) => {
   128→   128→    try {
   129→   129→      const compressed = await compressImage(rawBase64);
   130→   130→      const estimatedSize = estimateBase64Size(compressed);
   131→   131→      const photoType = defaultPhotoType;
   132→   132→
   133→   133→      const newPhoto: PhotoItem = {
   134→   134→        preview: compressed,
   135→   135→        serverUrl: '',
   136→   136→        uploaded: false,
   137→   137→        uploading: true,
   138→   138→        estimatedSize,
   139→   139→        title: generatePhotoTitle(photos.length, photoType),
   140→   140→        photoType,
   141→   141→        savedToLibrary: false,
   142→   142→      };
   143→   143→
   144→   144→      setPhotos(prev => [...prev, newPhoto]);
   145→   145→
   146→   146→      const url = await uploadPhoto(compressed, photos.length);
   147→   147→
   148→   148→      setPhotos(prev =>
   149→   149→        prev.map((p, i) =>
   150→   150→          i === prev.length - 1 && p.uploading
   151→   151→            ? { ...p, serverUrl: url || '', uploaded: !!url, uploading: false }
   152→   152→            : p
   153→   153→        )
   154→   154→      );
   155→   155→    } catch (error) {
   156→   156→      console.error('Error processing photo:', error);
   157→   157→    }
   158→   158→  }, [sStep, miniStep, currentProject?.id, photos.length, defaultPhotoType]);
   159→   159→
   160→   160→  const stopStream = useCallback(() => {
   161→   161→    if (stream) {
   162→   162→      stream.getTracks().forEach(track => track.stop());
   163→   163→      setStream(null);
   164→   164→    }
   165→   165→    setCameraMode('idle');
   166→   166→  }, [stream]);
   167→   167→
   168→   168→  const startCamera = useCallback(async () => {
   169→   169→    setCameraError(null);
   170→   170→    try {
   171→   171→      const mediaStream = await navigator.mediaDevices.getUserMedia({
   172→   172→        video: { facingMode, width: { ideal: 1920 }, height: { ideal: 1080 } },
   173→   173→        audio: false,
   174→   174→      });
   175→   175→      setStream(mediaStream);
   176→   176→      setCameraMode('streaming');
   177→   177→      if (videoRef.current) {
   178→   178→        videoRef.current.srcObject = mediaStream;
   179→   179→        await videoRef.current.play();
   180→   180→      }
   181→   181→    } catch (err: any) {
   182→   182→      if (err.name === 'NotAllowedError') setCameraError('Permiso denegado. Permite el acceso a la cámara.');
   183→   183→      else if (err.name === 'NotFoundError') setCameraError('No se encontró ninguna cámara.');
   184→   184→      else setCameraError('Error al acceder a la cámara. Intenta subir una foto desde tu galería.');
   185→   185→    }
   186→   186→  }, [facingMode]);
   187→   187→
   188→   188→  const switchCamera = useCallback(async () => {
   189→   189→    stopStream();
   190→   190→    setFacingMode(prev => (prev === 'environment' ? 'user' : 'environment'));
   191→   191→  }, [stopStream]);
   192→   192→
   193→   193→  const capturePhoto = useCallback(() => {
   194→   194→    if (!videoRef.current || !canvasRef.current) return;
   195→   195→    const video = videoRef.current;
   196→   196→    const canvas = canvasRef.current;
   197→   197→    canvas.width = video.videoWidth;
   198→   198→    canvas.height = video.videoHeight;
   199→   199→    const ctx = canvas.getContext('2d');
   200→   200→    if (!ctx) return;
   201→   201→    if (facingMode === 'user') { ctx.translate(canvas.width, 0); ctx.scale(-1, 1); }
   202→   202→    ctx.drawImage(video, 0, 0);
   203→   203→    const base64 = canvas.toDataURL('image/jpeg', 0.9);
   204→   204→    setCaptureFlash(true);
   205→   205→    setTimeout(() => setCaptureFlash(false), 300);
   206→   206→    addPhoto(base64);
   207→   207→  }, [facingMode, addPhoto]);
   208→   208→
   209→   209→  // Load existing photos from DB when modal opens
   210→   210→  const [loadingPhotos, setLoadingPhotos] = useState(false);
   211→   211→
   212→   212→  useEffect(() => {
   213→   213→    if (open && currentProject?.id) {
   214→   214→      setLoadingPhotos(true);
   215→   215→      const zoneIdParam = currentZone?.id ? `&zoneId=${currentZone.id}` : ''
   216→   216→      fetch(`/api/photo-library?projectId=${currentProject.id}&sStep=${sStep}&miniStep=2${zoneIdParam}`)
   217→   217→        .then(res => res.json())
   218→   218→        .then(data => {
   219→   219→          if (data.success && data.data && data.data.length > 0) {
   220→   220→            const existingPhotos: PhotoItem[] = data.data.map((p: any) => ({
   221→   221→              preview: p.photoUrl, // Use the stored URL as preview
   222→   222→              serverUrl: p.photoUrl,
   223→   223→              uploaded: true,
   224→   224→              uploading: false,
   225→   225→              estimatedSize: 0,
   226→   226→              title: p.title || '',
   227→   227→              photoType: p.photoType || 'antes',
   228→   228→              savedToLibrary: true, // Already saved in DB
   229→   229→            }));
   230→   230→            setPhotos(existingPhotos);
   231→   231→            // If step was already completed, mark it
   232→   232→            const antesCount = existingPhotos.filter(p => p.photoType === 'antes').length;
   233→   233→            const despuesCount = existingPhotos.filter(p => p.photoType === 'despues').length;
   234→   234→            if (antesCount >= MIN_PHOTOS) setIsCompleted(true);
   235→   235→          }
   236→   236→          setLoadingPhotos(false);
   237→   237→        })
   238→   238→        .catch(err => {
   239→   239→          console.error('Error loading existing photos:', err);
   240→   240→          setLoadingPhotos(false);
   241→   241→        });
   242→   242→    }
   243→   243→  }, [open, currentProject?.id, sStep, currentZone?.id]);
   244→   244→
   245→   245→  useEffect(() => {
   246→   246→    if (!open) {
   247→   247→      stopStream();
   248→   248→      // Don't clear photos on close — they should persist
   249→   249→      // Only clear transient UI state
   250→   250→      setCameraError(null);
   251→   251→      setCameraMode('idle');
   252→   252→      setActiveTab('camera');
   253→   253→    }
   254→   254→  }, [open, stopStream]);
   255→   255→
   256→   256→  const handleFileSelect = async (e: React.ChangeEvent<HTMLInputElement>) => {
   257→   257→    const files = e.target.files;
   258→   258→    if (!files) return;
   259→   259→    for (const file of Array.from(files)) {
   260→   260→      const reader = new FileReader();
   261→   261→      reader.onloadend = async () => { await addPhoto(reader.result as string); };
   262→   262→      reader.readAsDataURL(file);
   263→   263→    }
   264→   264→    if (fileInputRef.current) fileInputRef.current.value = '';
   265→   265→  };
   266→   266→
   267→   267→  const handleCameraCapture = async (e: React.ChangeEvent<HTMLInputElement>) => {
   268→   268→    const files = e.target.files;
   269→   269→    if (!files) return;
   270→   270→    for (const file of Array.from(files)) {
   271→   271→      const reader = new FileReader();
   272→   272→      reader.onloadend = async () => { await addPhoto(reader.result as string); };
   273→   273→      reader.readAsDataURL(file);
   274→   274→    }
   275→   275→    if (cameraInputRef.current) cameraInputRef.current.value = '';
   276→   276→  };
   277→   277→
   278→   278→  const removePhoto = (index: number) => setPhotos(prev => prev.filter((_, i) => i !== index));
   279→   279→
   280→   280→  const uploadingCount = photos.filter(p => p.uploading).length;
   281→   281→  const canSubmit = photos.length >= MIN_PHOTOS && uploadingCount === 0;
   282→   282→  const totalSize = photos.reduce((sum, p) => sum + p.estimatedSize, 0);
   283→   283→
   284→   284→  const handleSubmit = async () => {
   285→   285→    if (!canSubmit) return;
   286→   286→    setIsSubmitting(true);
   287→   287→    try {
   288→   288→      const urls = photos.map(p => p.serverUrl || p.preview).join(',');
   289→   289→
   290→   290→      // Save each photo to the PhotoLibrary with full traceability
   291→   291→      // Use serverUrl if available, otherwise use preview (base64) as the photoUrl
   292→   292→      const libraryPromises = photos
   293→   293→        .filter(p => !p.savedToLibrary)
   294→   294→        .map((p, idx) => {
   295→   295→          const allUploadedBefore = photos.slice(0, idx).filter(pp => pp.serverUrl || pp.preview).length;
   296→   296→          const photoUrlForDb = p.serverUrl || p.preview; // Use base64 as fallback if upload failed
   297→   297→          return fetch('/api/photo-library', {
   298→   298→            method: 'POST',
   299→   299→            headers: { 'Content-Type': 'application/json' },
   300→   300→            body: JSON.stringify({
   301→   301→              sStep,
   302→   302→              miniStep: 2,
   303→   303→              title: p.title || generatePhotoTitle(allUploadedBefore, p.photoType),
   304→   304→              description: `${sStepData?.japaneseName || 'S' + sStep} - ${currentZone?.name || 'Zona'} - Paso 2 Fotos - Subida por ${currentUser?.name || 'Usuario'}`,
   305→   305→              photoUrl: photoUrlForDb,
   306→   306→              photoType: p.photoType || 'antes',
   307→   307→              category: `paso2_s${sStep}`,
   308→   308→              tags: JSON.stringify([`S${sStep}`, sStepData?.japaneseName || '', currentZone?.name || '', `paso2`, p.photoType]),
   309→   309→              projectId: currentProject?.id,
   310→   310→              zoneId: currentZone?.id || null,
   311→   311→              uploadedBy: currentUser?.id || null,
   312→   312→            }),
   313→   313→          });
   314→   314→        });
   315→   315→      
   316→   316→      // Wait for ALL photo library saves to complete before proceeding
   317→   317→      const libraryResults = await Promise.allSettled(libraryPromises);
   318→   318→      const savedCount = libraryResults.filter(r => r.status === 'fulfilled').length;
   319→   319→      const failedCount = libraryResults.filter(r => r.status === 'rejected').length;
   320→   320→      if (failedCount > 0) console.warn(`[FotosModal] ${failedCount} photos failed to save to library`);
   321→   321→      else if (savedCount > 0) console.log(`[FotosModal] ${savedCount} photos saved to library`);
   322→   322→      
   323→   323→      // Mark all photos as saved in local state
   324→   324→      setPhotos(prev => prev.map(p => ({ ...p, savedToLibrary: true })));
   325→   325→
   326→   326→      // Save progress
   327→   327→      const res = await fetch(`/api/progress/step?sStep=${sStep}&miniStep=${miniStep}`, {
   328→   328→        method: 'PUT',
   329→   329→        headers: { 'Content-Type': 'application/json' },
   330→   330→        body: JSON.stringify({ completed: true, photoUrls: urls, score: 100, projectId: currentProject?.id, zoneId: currentZone?.id || null }),
   331→   331→      });
   332→   332→      const json = await res.json();
   333→   333→      if (json.success) { setIsCompleted(true); stopStream(); await fetchProgress(); }
   334→   334→    } catch (error) {
   335→   335→      console.error('Error submitting photos:', error);
   336→   336→    } finally {
   337→   337→      setIsSubmitting(false);
   338→   338→    }
   339→   339→  };
   340→   340→
   341→   341→  const handleAdminSkip = async () => {
   342→   342→    try {
   343→   343→      const res = await fetch(`/api/progress/step?sStep=${sStep}&miniStep=${miniStep}`, {
   344→   344→        method: 'PUT',
   345→   345→        headers: { 'Content-Type': 'application/json' },
   346→   346→        body: JSON.stringify({ completed: true, score: 100, notes: 'Completado por administrador (skip)', skipMissingTemplate: true, projectId: currentProject?.id, zoneId: currentZone?.id || null }),
   347→   347→      });
   348→   348→      const json = await res.json();
   349→   349→      if (json.success) { await fetchProgress(); onClose(); }
   350→   350→    } catch (error) { console.error('Error admin skip:', error); }
   351→   351→  };
   352→   352→
   353→   353→  const isMobile = typeof navigator !== 'undefined' && /Android|webOS|iPhone|iPad|iPod|BlackBerry|IEMobile|Opera Mini/i.test(navigator.userAgent);
   354→   354→
   355→   355→  return (
   356→   356→    <Dialog open={open} onOpenChange={() => { stopStream(); onClose(); }}>
   357→   357→      <DialogContent size={isFullscreen ? "fullscreen" : "xl"} className="flex flex-col overflow-hidden p-0">
   358→   358→        <DialogHeader className="px-6 pt-6 pb-2 flex-shrink-0">
   359→   359→          <DialogTitle className="flex items-center gap-2">
   360→   360→            <Camera className="h-5 w-5" style={{ color: sStepData?.color }} />
   361→   361→            <span>Fotografías (Antes)</span>
   362→   362→            <Badge variant="outline" style={{ borderColor: sStepData?.color, color: sStepData?.color }}>
   363→   363→              {sStepData?.name}
   364→   364→            </Badge>
   365→   365→            <button
   366→   366→              onClick={() => setIsFullscreen(!isFullscreen)}
   367→   367→              className="ml-auto p-1 rounded hover:bg-muted transition-colors"
   368→   368→              title={isFullscreen ? "Reducir ventana" : "Pantalla completa"}
   369→   369→            >
   370→   370→              {isFullscreen ? <Minimize2 className="h-4 w-4 text-muted-foreground" /> : <Maximize2 className="h-4 w-4 text-muted-foreground" />}
   371→   371→            </button>
   372→   372→          </DialogTitle>
   373→   373→        </DialogHeader>
   374→   374→
   375→   375→        {canSkipSteps && !isCompleted && (
   376→   376→          <div className="flex items-center gap-2 p-2 mx-6 flex-shrink-0 bg-amber-50 border border-amber-200 rounded-lg">
   377→   377→            <span className="text-xs text-amber-700 font-medium">Modo Admin:</span>
   378→   378→            <Button variant="outline" size="sm" className="text-xs border-amber-300 text-amber-700 hover:bg-amber-100" onClick={handleAdminSkip}>
   379→   379→              Completar paso sin subir fotos
   380→   380→            </Button>
   381→   381→          </div>
   382→   382→        )}
   383→   383→
   384→   384→        {isReadOnly && (
   385→   385→          <div className="flex items-center gap-2 p-2 mx-6 flex-shrink-0 bg-blue-50 border border-blue-200 rounded-lg">
   386→   386→            <span className="text-xs text-blue-700 font-medium">Solo lectura: {canSkipSteps ? 'Activa el candado para poder realizar pasos.' : 'Puedes ver pero no modificar.'}</span>
   387→   387→          </div>
   388→   388→        )}
   389→   389→
   390→   390→        <div className="flex-1 overflow-y-auto px-6 pb-6 min-h-0">
   391→   391→        {isCompleted ? (
   392→   392→          <div className="text-center py-8">
   393→   393→            <CheckCircle className="h-16 w-16 text-green-500 mx-auto mb-3" />
   394→   394→            <h3 className="text-xl font-bold mb-2">Fotografías del ANTES Guardadas</h3>
   395→   395→            <p className="text-muted-foreground">Ha guardado {photos.length} fotos como evidencia del estado inicial.</p>
   396→   396→            <p className="text-xs text-muted-foreground mt-2">Tamaño total optimizado: {formatBytes(totalSize)}</p>
   397→   397→            <div className="mt-4 p-3 rounded-lg bg-blue-50 border border-blue-200 text-blue-700">
   398→   398→              <p className="text-sm font-semibold">→ Próximo paso: Inventario (Clasificar elementos)</p>
   399→   399→            </div>
   400→   400→            <div className="mt-4 flex justify-center">
   401→   401→              <Button
   402→   402→                onClick={() => { onClose(); openModal('inventario', 3); }}
   403→   403→                style={{ backgroundColor: sStepData?.color }}
   404→   404→                className="text-white"
   405→   405→              >
   406→   406→                Continuar al Inventario →
   407→   407→              </Button>
   408→   408→            </div>
   409→   409→          </div>
   410→   410→        ) : (
   411→   411→          <div className="space-y-4">
