   523→ const handleCreateCompanyFull = async () => {

   537→    return
   538→   }
   539→   companyId = companyData.company?.id || companyData.id
   540→
   541→   // Step 2: Create or assign admin
   542→   let adminUserId: string | null = null
   543→   if (adminMode === 'create') {

   549→     await loadStats()

   551→     return
   552→    }
   553→    // Create user with role 'admin' via /api/users
   554→    const userRes = await fetch('/api/users', {
   555→     method: 'POST',
   556→     headers: { 'Content-Type': 'application/json' },
   557→     body: JSON.stringify({
   558→      name: newAdminName,
   559→      email: newAdminEmail,
   560→      password: newAdminPassword,
   561→      role: 'admin',
   562→      active: true,
