   356→  canPerform: (sStep: number, miniStep: number): boolean => {

   365→
   366→  // ═══════════════════════════════════════════════════════
   367→  // getMiniStepStatus — Permission-Driven with Business Rules
   368→  // Permissions are the source of truth for WHO can do WHAT.
   369→  // Business rules define WHEN things are accessible:
   370→  //   - INTER-S progression: S2 requires S1 completed, S3 requires S2, etc.
   371→  //   - INTRA-S progression: step N requires step N-1 completed
   372→  //   - Step 5 (audit): requires steps 1-4 ALL completed within the same S
   373→  //   - a1 (perform) → can enter and act (if previous step done)
   374→  //   - a0 (view) only → can see the step exists, but CANNOT enter it
   375→  //   - No permission → locked (not even visible)
   376→  //   - IMPORTANT: Previous step completion is checked at ZONE level,
   377→  //     so an auditor can access step 5 if employees completed 1-4
   378→  //   - For step 1 (individual): also checks EmployeeProgress for the
   379→  //     current user, so the employee's own completion unlocks step 2
   380→  // ═══════════════════════════════════════════════════════
   381→  getMiniStepStatus: (sStep, miniStep) => {
   382→    const { progress, employeeProgress, currentUser, adminFreeNavigation, currentZone } = get()
   383→    if (!currentUser) return 'locked'
   384→
   385→    const canViewStep = get().canView(sStep, miniStep)
   386→    const canPerformStep = get().canPerform(sStep, miniStep)
   387→    // Permission-driven: skip_steps from DB replaces hardcoded isAdmin check
   388→    const canSkipSteps = get().hasPermission('skip_steps')
   389→    const skipLocks = canSkipSteps && adminFreeNavigation
   390→
   391→    // Check if a specific mini-step of a specific S is completed at zone level
   392→    const isStepCompletedAt = (s: number, ms: number): boolean => {
   393→      if (currentZone) {
   394→        const zoneStep = progress.find(p =>
   395→          p.sStep === s &&
   396→          p.miniStep === ms &&
   397→          (p.zoneId === currentZone.id || p.zoneId === null) &&
   398→          p.completed
   399→        )
   400→        return !!zoneStep
   401→      }
   402→      const anyStep = progress.find(p =>
   403→        p.sStep === s &&
   404→        p.miniStep === ms &&
   405→        p.completed
   406→      )
   407→      return !!anyStep
   408→    }
   409→
   410→    // Check if a specific S-step (all 5 mini-steps) is fully completed (quesito earned)
   411→    // Also checks employeeProgress for individual steps, consistent with areSteps1to4Completed
   412→    const isSCompleted = (s: number): boolean => {
   413→      for (let ms = 1; ms <= 5; ms++) {
   414→        // Check zone-level progress first
   415→        if (isStepCompletedAt(s, ms)) continue
   416→        // Also check employeeProgress for this step (e.g., individual formación step 1)
   417→        const anyEmpCompleted = currentZone
   418→          ? employeeProgress.some(ep =>
   419→              ep.sStep === s &&
   420→              ep.miniStep === ms &&
   421→              ep.zoneId === currentZone.id &&
   422→              ep.completed
   423→            )
   424→          : employeeProgress.some(ep =>
   425→              ep.sStep === s &&
   426→              ep.miniStep === ms &&
   427→              ep.completed
   428→            )
   429→        if (!anyEmpCompleted) return false
   430→      }
   431→      return true
   432→    }
   433→
   434→    // Check if the CURRENT USER has completed step 1 individually (EmployeeProgress)
   435→    // This is crucial for unlocking step 2 for an employee who passed the exam
   436→    // even if not ALL employees in the zone have completed it yet
   437→    const hasUserCompletedStep1 = (): boolean => {
   438→      if (!currentZone) return false
   439→      const userStep = employeeProgress.find(ep =>
   440→        ep.sStep === sStep &&
   441→        ep.miniStep === 1 &&
   442→        ep.zoneId === currentZone.id &&
   443→        ep.userId === currentUser.id &&
   444→        ep.completed
   445→      )
   446→      return !!userStep
   447→    }
   448→
   449→    // Check if already completed at zone level (or project level if no zone)
   450→    const isStepCompleted = (): boolean => isStepCompletedAt(sStep, miniStep)
   451→
   452→    // Check if the CURRENT USER has completed this individual step (EmployeeProgress)
   453→    // For step 1 (formación) and step 4 (autoevaluación), completion is tracked per employee
   454→    const hasUserCompletedIndividualStep = (): boolean => {
   455→      if (!currentZone) return false
   456→      if (miniStep !== 1 && miniStep !== 4) return false // Only individual steps
   457→      const userStep = employeeProgress.find(ep =>
   458→        ep.sStep === sStep &&
   459→        ep.miniStep === miniStep &&
   460→        ep.zoneId === currentZone.id &&
   461→        ep.userId === currentUser.id &&
   462→        ep.completed
   463→      )
   464→      return !!userStep
   465→    }
   466→
   467→    // Combined: step is "done" if either zone-level completed OR user completed individually
   468→    const isStepDoneForUser = (): boolean => isStepCompleted() || hasUserCompletedIndividualStep()
   469→
   470→    // Check if step is completed by ANYONE (zone or any employee)
   471→    // This is crucial: ALL users must see completed steps in green,
   472→    // even if they don't have permission to enter them (e.g., auditors viewing steps 1-4)
   473→    const isStepCompletedByAnyone = (): boolean => {
   474→      if (isStepCompleted()) return true
   475→      if (!currentZone) {
   476→        return employeeProgress.some(ep =>
   477→          ep.sStep === sStep &&
   478→          ep.miniStep === miniStep &&
   479→          ep.completed
   480→        )
   481→      }
   482→      return employeeProgress.some(ep =>
   483→        ep.sStep === sStep &&
   484→        ep.miniStep === miniStep &&
   485→        ep.zoneId === currentZone.id &&
   486→        ep.completed
   487→      )
   488→    }
   489→
   490→    // Helper: check if steps 1-4 are all completed for this S-step
   491→    // Uses BOTH zone-level progress AND any employee progress,
   492→    // so auditors can access step 5 when employees have completed 1-4
   493→    const areSteps1to4Completed = (): boolean => {
   494→      for (let ms = 1; ms <= 4; ms++) {
   495→        const zoneCompleted = isStepCompletedAt(sStep, ms)
   496→        if (zoneCompleted) continue
   497→        // Check if ANY employee has completed this step individually
   498→        const anyEmpCompleted = currentZone
   499→          ? employeeProgress.some(ep =>
   500→              ep.sStep === sStep &&
   501→              ep.miniStep === ms &&
   502→              ep.zoneId === currentZone.id &&
   503→              ep.completed
   504→            )
   505→          : employeeProgress.some(ep =>
   506→              ep.sStep === sStep &&
   507→              ep.miniStep === ms &&
   508→              ep.completed
   509→            )
   510→        if (!anyEmpCompleted) return false
   511→      }
   512→      return true
   513→    }
   514→
   515→    // ═══ INTER-S PROGRESSION CHECK ═══
   516→    // S1 is always available. S2 requires S1 completed. S3 requires S2. Etc.
   517→    // This enforces the 5S methodology sequence: Clasificar → Ordenar → Limpiar → Estandarizar → Mantener
   518→    const isPreviousSCompleted = (): boolean => {
   519→      if (sStep === 1) return true // S1 has no prerequisite
   520→      return isSCompleted(sStep - 1)
   521→    }
   522→
   523→    // ═══ INTRA-S PROGRESSION CHECK ═══
   524→    // Step 1: always available (if you have permission AND previous S is done)
   525→    // Steps 2-4: need the immediately previous step completed
   526→    // Step 5: needs ALL steps 1-4 completed
   527→    // IMPORTANT: Completion checks include BOTH zone-level AND any employee progress,
   528→    // so auditors/responsables can see steps as available when employees completed them
   529→    const isPreviousStepCompleted = (): boolean => {
   530→      if (miniStep === 1) return true
   531→      if (miniStep === 5) return areSteps1to4Completed()
   532→      // Steps 2, 3, 4: need previous step completed at zone level
   533→      if (isStepCompletedAt(sStep, miniStep - 1)) return true
   534→      // For step 2: current user completed step 1 individually (exam passed)
   535→      if (miniStep === 2 && hasUserCompletedStep1()) return true
   536→      // Also check: ANY employee completed previous step (for auditors/responsables)
   537→      const anyEmpCompletedPrev = currentZone
   538→        ? employeeProgress.some(ep =>
   539→            ep.sStep === sStep &&
   540→            ep.miniStep === miniStep - 1 &&
   541→            ep.zoneId === currentZone.id &&
   542→            ep.completed
   543→          )
   544→        : employeeProgress.some(ep =>
   545→            ep.sStep === sStep &&
   546→            ep.miniStep === miniStep - 1 &&
   547→            ep.completed
   548→          )
   549→      if (anyEmpCompletedPrev) return true
   550→      return false
   551→    }
   552→
   553→    // ── If already completed (zone-level OR user's individual), VERIFY COHERENCY ──
   554→    if (isStepDoneForUser()) {
   555→      const isChainCoherent = (): boolean => {
   556→        // First check inter-S: previous S must be completed
   557→        if (!isPreviousSCompleted()) return false
   558→        // Then check intra-S chain — reuse same logic as isPreviousStepCompleted
   559→        if (miniStep === 1) return true // Step 1 is always coherent if previous S is done
   560→        if (miniStep === 5) return areSteps1to4Completed()
   561→        if (isStepCompletedAt(sStep, miniStep - 1)) return true
   562→        if (miniStep === 2 && hasUserCompletedStep1()) return true
   563→        // Also check any employee completed previous step
   564→        const anyEmpPrev = currentZone
   565→          ? employeeProgress.some(ep =>
   566→              ep.sStep === sStep &&
   567→              ep.miniStep === miniStep - 1 &&
   568→              ep.zoneId === currentZone.id &&
   569→              ep.completed
   570→            )
   571→          : employeeProgress.some(ep =>
   572→              ep.sStep === sStep &&
   573→              ep.miniStep === miniStep - 1 &&
   574→              ep.completed
   575→            )
   576→        if (anyEmpPrev) return true
   577→        return false
   578→      }
   579→
   580→      if (isChainCoherent()) {
   581→        if (canPerformStep || canSkipSteps) return 'completed'
   582→        if (canViewStep) return 'completed_viewonly'
   583→        return 'locked'
   584→      }
   585→      // Chain is incoherent: treat as NOT completed — fall through
   586→    }
   587→
   588→    // ── Step completed by ANYONE (zone or employee) → show as green for ALL users ──
   589→    // Even users without permission must see completed steps in green
   590→    if (isStepCompletedByAnyone()) {
   591→      if (canPerformStep || canSkipSteps) return 'completed'
   592→      // Everyone else sees it as completed_viewonly (green, read-only)
   593→      return 'completed_viewonly'
   594→    }
   595→
   596→    // ── Admin with lock open: skip all checks ──
   597→    if (skipLocks) return 'available'
   598→
   599→    // ── No permission at all = locked ──
   600→    if (!canViewStep && !canPerformStep) return 'locked'
   601→
   602→    // ── Has perform permission (a1) → check progression ──
   603→    if (canPerformStep) {
   604→      // INTER-S: can't start this S until previous S is completed
   605→      if (!isPreviousSCompleted()) return 'locked'
   606→      // INTRA-S: can't enter step until previous step is completed
   607→      if (!isPreviousStepCompleted()) return 'locked'
   608→      return 'available'
   609→    }
   610→
   611→    // ── Has view permission only (a0) → can see but CANNOT enter ──
   612→    if (canViewStep) return 'locked'
   613→
   614→    return 'locked'
   615→  },
   616→
   617→  isZoneMiniStepComplete: (sStep, miniStep, zoneId) => {
   618→    const { progress, employeeProgress, currentProject } = get()
   619→    if (!currentProject) return false
   620→
   621→    if (ZONE_MINI_STEPS.includes(miniStep)) {
   622→      // Zone step: check if the zone's Progress record is completed
   623→      const zoneStep = progress.find(p =>
   624→        p.sStep === sStep &&
   625→        p.miniStep === miniStep &&
   626→        p.zoneId === zoneId
   627→      )
   628→      return zoneStep?.completed ?? false
   629→    }
   630→
   631→    if (INDIVIDUAL_MINI_STEPS.includes(miniStep)) {
   632→      // Individual step (1 only): ALL employees in the zone must have completed
   633→      // The zone-level Progress record for step 1 is only marked completed when ALL employees pass
   634→      const zoneStep = progress.find(p =>
   635→        p.sStep === sStep &&
   636→        p.miniStep === miniStep &&
   637→        (p.zoneId === zoneId || p.zoneId === null) &&
   638→        p.completed
   639→      )
   640→      return !!zoneStep
   641→    }
   642→
   643→    return false
   644→  },
   645→
   646→  isQuesitoEarned: (sStep) => {
   647→    const { progress, employeeProgress, currentZone } = get()
   648→    const zoneId = currentZone?.id || null
   649→
   650→    // Check ALL 5 mini-steps are completed — each one individually
   651→    // For zone-level steps (2,3,5): check progress table
   652→    // For individual steps (1,4): also check employeeProgress as fallback
   653→    for (let miniStep = 1; miniStep <= 5; miniStep++) {
   654→      // Check zone-level progress first
   655→      const zoneStep = progress.find(p =>
   656→        p.sStep === sStep &&
   657→        p.miniStep === miniStep &&
   658→        (p.zoneId === zoneId || p.zoneId === null) &&
   659→        p.completed
   660→      )
   661→      if (zoneStep) continue
   662→
   663→      // Also check employeeProgress for this step (e.g., individual formación step 1)
   664→      // This ensures the quesito is earned even when not all employees have completed step 1,
   665→      // but the auditor was able to pass step 5 (audit) because areSteps1to4Completed returned true
   666→      const anyEmpCompleted = currentZone
   667→        ? employeeProgress.some(ep =>
   668→            ep.sStep === sStep &&
   669→            ep.miniStep === miniStep &&
   670→            ep.zoneId === currentZone.id &&
   671→            ep.completed
   672→          )
   673→        : employeeProgress.some(ep =>
   674→            ep.sStep === sStep &&
   675→            ep.miniStep === miniStep &&
   676→            ep.completed
   677→          )
   678→      if (anyEmpCompleted) continue
   679→
   680→      return false
   681→    }
   682→    return true
   683→  },
   684→
   685→  is5SCompleted: () => {
   686→    for (let i = 1; i <= 5; i++) {
   687→      if (!get().isQuesitoEarned(i)) return false
   688→    }
   689→    return true
   690→  },
   691→
   692→  getCompletedCount: () => {
   693→    const { progress, currentZone } = get()
   694→
   695→    if (currentZone) {
   696→      const completedMiniSteps = progress.filter(p =>
   697→        (p.zoneId === currentZone.id || p.zoneId === null) &&
   698→        p.completed
   699→      ).length
   700→      let completedSSteps = 0

   703→      }
   704→      return { sSteps: completedSSteps, miniSteps: completedMiniSteps, total: 25 }
   705→    }
   706→
   707→    const completedMiniSteps = progress.filter(p => p.completed).length
   708→    let completedSSteps = 0
   709→    for (let i = 1; i <= 5; i++) {
   710→      if (get().isQuesitoEarned(i)) completedSSteps++
   711→    }
   712→    return { sSteps: completedSSteps, miniSteps: completedMiniSteps, total: 25 }
   713→  },

   716→  login: async (email: string, password: string) => {
   717→    try {
   718→      set({ isLoginLoading: true, authError: null })
   719→      const res = await fetch('/api/auth', {
   720→        method: 'POST',
   721→        headers: { 'Content-Type': 'application/json' },
   722→        body: JSON.stringify({ email, password }),
   723→      })
   724→      const data = await res.json()
   725→
   726→      if (!res.ok) {
   727→        const errorMsg = data.error || 'Email o contraseña incorrectos'
   728→        set({ isLoginLoading: false, authError: errorMsg })
   729→        return false
   730→      }
   731→
   732→      if (!data.user) {
   733→        set({ isLoginLoading: false, authError: 'Error inesperado al iniciar sesión' })
   734→        return false
   735→      }
   736→
   737→      set({ currentUser: data.user, isLoginLoading: false, authError: null })
   738→
   739→      // Load permissions after login
   740→      try {
   741→        await get().fetchPermissions()
   742→      } catch (e) {
   743→        console.error('Error loading permissions after login:', e)
   744→      }
   745→
   746→      // Gestor (dueño de la app) goes directly to management panel
   747→      if (data.user.role === 'gestor') {
   748→        set({ authView: 'board', activeTab: 'gestion' })
   749→        try {
   750→          await get().fetchCompanies()
   751→        } catch (e) {
   752→          console.error('Error fetching companies after login:', e)
   753→        }
   754→        return true
   755→      }
   756→
   757→      // Check for projects after login
   758→      try {
   759→        await get().fetchProjects()
   760→      } catch (e) {
   761→        console.error('Error fetching projects after login:', e)
   762→      }
   763→      const { projects } = get()
   764→
   765→      if (projects.length > 0) {
   766→        // No auto-selecciona: el usuario elige proyecto+zona en la pantalla de selección
   767→        set({ currentProject: null, currentZone: null, authView: 'project_selector' })
   768→        // Fetch user's assigned zones after login (para filtrar zonas accesibles)
   769→        try {
   770→          await get().fetchUserZones()
   771→        } catch (e) {
   772→          console.error('Error fetching user zones after login:', e)
   773→        }
   774→      } else {
   775→        // Only admin can create projects via setup wizard
   776→        // Non-admin users see a waiting screen
   777→        if (data.user.role === 'admin') {
   778→          set({ authView: 'setup' })
   779→        } else {
   780→          set({ authView: 'no_projects' })
   781→        }
   782→      }
   783→
   784→      return true
   785→    } catch (error) {
   786→      console.error('Login error:', error)
   787→      set({ isLoginLoading: false, authError: 'Error de conexión. Inténtalo de nuevo.' })
   788→      return false
   789→    }
   790→  },
