    33→interface AutoevaluacionModalProps {
    34→  open: boolean;

    40→
    41→
    42→export default function AutoevaluacionModal({ open, onClose, sStep, miniStep }: AutoevaluacionModalProps) {
    43→  const { fetchProgress, currentUser, adminFreeNavigation, currentProject, currentZone, canPerform, canView, hasPermission, openModal } = use5SStore();
    44→  const sStepData = S_STEPS.find(s => s.id === sStep);
    45→  const canSkipSteps = hasPermission('skip_steps');
    46→  const canPerformStep = canPerform(sStep, miniStep);
    47→  const canViewStep = canView(sStep, miniStep);
    48→  // Permission-driven: read-only if no execute perm OR if candado closed for skip_steps users
    49→  const isReadOnly = !canPerformStep || (canSkipSteps && !adminFreeNavigation);
    50→  const canPerformAutoeval = canPerformStep;
    51→
    52→  const [isFullscreen, setIsFullscreen] = useState(true);
    53→  const [results, setResults] = useState<Record<string, AuditItemResult>>({});
    54→  const [expandedSections, setExpandedSections] = useState<Record<string, boolean>>({});
    55→  const [observaciones, setObservaciones] = useState('');
    56→  const [isSubmitting, setIsSubmitting] = useState(false);
    57→  const [isCompleted, setIsCompleted] = useState(false);
    58→  const [finalScore, setFinalScore] = useState(0);
    59→  // Track which (open, sStep, sections-signature) we've already initialized for,
    60→  // so we don't reset user edits if `sections` ref changes mid-session.
    61→  const initializedFor = useRef<string>('');
    62→  const [notaMinima, setNotaMinima] = useState(70);
    63→  const [autoevalPhotos, setAutoevalPhotos] = useState<{ file: File; preview: string; uploading?: boolean; serverUrl?: string }[]>([]);
    64→  const [isUploadingPhotos, setIsUploadingPhotos] = useState(false);
    65→  const photoInputRef = useRef<HTMLInputElement>(null);
    66→
    67→  // Project members for responsable selector on NOK items
    68→  const [projectMembers, setProjectMembers] = useState<Array<{ id: string; userId: string; role: string; user: { id: string; name: string; email: string; role: string; active: boolean } }>>([]);
    69→
    70→  // Date/Time for scheduling and recording the evaluation
    71→  const [fechaAutoevaluacion, setFechaAutoevaluacion] = useState('');
    72→  const [horaAutoevaluacion, setHoraAutoevaluacion] = useState('');
    73→  const [fechaProgramada, setFechaProgramada] = useState('');
    74→  const [horaProgramada, setHoraProgramada] = useState('');
    75→  // Track auditoría schedule (miniStep 5) — employee sees this as read-only (auditor sets it)
    76→  const [auditFechaProgramada, setAuditFechaProgramada] = useState('');
    77→  const [auditHoraProgramada, setAuditHoraProgramada] = useState('');
    78→
    79→  // Load template from API (uses board config if zone has one)
    80→  const { sections, isLoading: isLoadingTemplate, notaMinima: templateNotaMinima } = useChecklistTemplate('autoevaluacion', sStep, open, currentZone?.boardConfigId);
    81→
    82→  // Apply template notaMinima when loaded
    83→  useEffect(() => {
    84→    if (templateNotaMinima !== null) setNotaMinima(templateNotaMinima);
    85→  }, [templateNotaMinima]);
    86→
    87→  // Load project members (for responsable selector on NOK items)
    88→  useEffect(() => {
    89→    if (open && currentProject?.id) {
    90→      fetch(`/api/projects/${currentProject.id}/members`)
    91→        .then(r => r.json())
    92→        .then(data => setProjectMembers(data?.members || []))
    93→        .catch(err => console.error('Error loading project members:', err));
    94→    } else {
    95→      setProjectMembers([]);
    96→    }
    97→  }, [open, currentProject?.id]);
    98→
    99→
   100→
   101→  // Fetch dynamic threshold (overrides template if present)
   102→  useEffect(() => {
   103→    if (open && currentProject?.id) {
   104→      const fetchThreshold = async () => {
   105→        try {
   106→          const params = new URLSearchParams({ projectId: currentProject.id });
   107→          if (currentZone?.id) params.set('zoneId', currentZone.id);
   108→          const res = await fetch(`/api/audit-targets?${params}`);
   109→          const json = await res.json();
   110→          if (json.success && json.data) {
   111→            const zoneTarget = json.data.find((t: any) => t.sStep === sStep && t.miniStep === 4 && t.zoneId === currentZone?.id);
   112→            const projectTarget = json.data.find((t: any) => t.sStep === sStep && t.miniStep === 4 && t.zoneId === null);
   113→            const target = zoneTarget || projectTarget;
   114→            if (target?.notaMinima) setNotaMinima(target.notaMinima);
   115→          }
   116→        } catch (e) { console.error('Error fetching threshold:', e); }
   117→      };
   118→      fetchThreshold();
   119→    }
   120→  }, [open, sStep, currentProject?.id, currentZone?.id]);
   121→
   122→  // ─────────────────────────────────────────────────────────────
   123→  // ROBUST PRE-TICK: Merge missing items as 'ok' whenever sections change.
   124→  // This replaces the fragile one-shot pre-tick. Instead of setting results
   125→  // once and praying the effect fires after sections loads, we continuously
   126→  // MERGE missing items as 'ok' into `results` without overwriting user
   127→  // edits. This guarantees that every visible item always has an 'ok' entry
   128→  // unless the user explicitly marks it NOK/N/A.
   129→  // ─────────────────────────────────────────────────────────────
   130→  useEffect(() => {
   131→    if (!open || sections.length === 0) return;
   132→    setResults(prev => {
   133→      const next = { ...prev };
   134→      let changed = false;
   135→      sections.forEach(s => s.items.forEach(item => {
   136→        if (!next[item.id]) {
   137→          next[item.id] = { itemId: item.id, status: 'ok' };
   138→          changed = true;
   139→        }
   140→      }));
   141→      return changed ? next : prev;
   142→    });
   143→  }, [open, sections]);
   144→
   145→  // Initialize expanded sections + reset form state once per (sStep + sections signature)
   146→  useEffect(() => {
   147→    if (!open || sections.length === 0) return;
   148→    const sig = `${sStep}:${sections.map(s => s.items.map(i => i.id).join(',')).join('|')}`;
   149→    if (initializedFor.current === sig) return;
   150→    initializedFor.current = sig;
   151→
   152→    const expanded: Record<string, boolean> = {};
   153→    sections.forEach(s => { expanded[s.id] = true; });
   154→    setExpandedSections(expanded);
   155→    setObservaciones('');
   156→    setIsCompleted(false);
   157→    setFinalScore(0);
   158→    setAutoevalPhotos([]);
   159→    // Auto-fill current date/time for the evaluation
   160→    const now = new Date();
   161→    setFechaAutoevaluacion(now.toISOString().split('T')[0]);
   162→    setHoraAutoevaluacion(now.toTimeString().slice(0, 5));
   163→    // Load scheduled date if available
   164→    loadScheduledDate();
   165→    // Load previously saved results (so user can see what was saved before)
   166→    loadSavedResults();
   167→  }, [open, sStep, sections]);
   168→
   169→  // Load previously saved audit results from the backend so the user can see
   170→  // what was actually persisted. This makes the save behavior verifiable.
   171→  const loadSavedResults = async () => {
   172→    if (!currentProject?.id) return;
   173→    try {
   174→      const res = await fetch(`/api/progress/step?sStep=${sStep}&miniStep=${miniStep}&projectId=${currentProject.id}${currentZone?.id ? `&zoneId=${currentZone.id}` : ''}`);
   175→      const json = await res.json();
   176→      if (json.success && json.data?.notes) {
   177→        try {
   178→          const notes = typeof json.data.notes === 'string' ? JSON.parse(json.data.notes) : json.data.notes;
   179→          if (notes?.results && Array.isArray(notes.results)) {
   180→            const loaded: Record<string, AuditItemResult> = {};
   181→            notes.results.forEach((r: any) => {
   182→              if (r?.itemId) loaded[r.itemId] = r;
   183→            });
   184→            // Only set if there's at least one real entry (don't overwrite with empty)
   185→            if (Object.keys(loaded).length > 0) {
   186→              setResults(loaded);
   187→              if (notes.observaciones) setObservaciones(notes.observaciones);
   188→              if (notes.fechaAutoevaluacion) setFechaAutoevaluacion(notes.fechaAutoevaluacion);
   189→              if (notes.horaAutoevaluacion) setHoraAutoevaluacion(notes.horaAutoevaluacion);
   190→            }
   191→          }
   192→        } catch (e) {
   193→          console.error('Error parsing saved notes:', e);
   194→        }
   195→      }
   196→    } catch (e) {
   197→      console.error('Error loading saved results:', e);
   198→    }
   199→  };
   200→
   201→  // Reset the init guard when modal closes, so next open re-initializes
   202→  useEffect(() => {
   203→    if (!open) {
   204→      initializedFor.current = '';
   205→      // Also clear results on close so next open starts fresh (then loads saved)
   206→      setResults({});
   207→    }
   208→  }, [open]);
   209→
   210→  // Load scheduled date/time for this autoevaluación AND the auditoría schedule
   211→  const loadScheduledDate = async () => {
   212→    if (!currentProject?.id || !currentZone?.id) return;
   213→    try {
   214→      // Load autoevaluación schedule (miniStep 4)
   215→      const res = await fetch(`/api/evaluation-schedule?sStep=${sStep}&miniStep=4&projectId=${currentProject.id}&zoneId=${currentZone.id}`);
   216→      const json = await res.json();
   217→      if (json.success && json.data) {
   218→        setFechaProgramada(json.data.fechaProgramada || '');
   219→        setHoraProgramada(json.data.horaProgramada || '');
   220→      }
   221→      // Also load auditoría schedule (miniStep 5) — employee sees if auditor has set a date
   222→      const auditRes = await fetch(`/api/evaluation-schedule?sStep=${sStep}&miniStep=5&projectId=${currentProject.id}&zoneId=${currentZone.id}`);
   223→      const auditJson = await auditRes.json();
   224→      if (auditJson.success && auditJson.data) {
   225→        setAuditFechaProgramada(auditJson.data.fechaProgramada || '');
   226→        setAuditHoraProgramada(auditJson.data.horaProgramada || '');
   227→      }
   228→    } catch (e) {
   229→      console.error('Error loading scheduled date:', e);
   230→    }
   231→  };
   232→
   233→  // Save scheduled date/time
   234→  const handleSaveSchedule = async () => {
   235→    if (!currentProject?.id || !currentZone?.id) return;
   236→    try {
   237→      await fetch('/api/evaluation-schedule', {
   238→        method: 'POST',
   239→        headers: { 'Content-Type': 'application/json' },
   240→        body: JSON.stringify({
   241→          sStep,
   242→          miniStep: 4,
   243→          projectId: currentProject.id,
   244→          zoneId: currentZone.id,
   245→          fechaProgramada,
   246→          horaProgramada,
   247→        }),
   248→      });
   249→      toast.success('Fecha programada guardada');
   250→    } catch (e) {
   251→      console.error('Error saving schedule:', e);
   252→      toast.error('Error al guardar la fecha programada');
   253→    }
   254→  };
   255→
   256→  const totalItems = sections.reduce((sum, s) => sum + s.items.length, 0) || 26;
   257→
   258→  // SCORING: Treat missing results as 'ok' (default).
   259→  // This is bulletproof — does NOT depend on useEffect timing.
   260→  // Every visible item that hasn't been explicitly marked NOK/N/A counts as OK.
   261→  const scoring = useMemo(() => {
   262→    let okCount = 0;
   263→    let nokCount = 0;
   264→    sections.forEach(s => s.items.forEach(item => {
   265→      const status = results[item.id]?.status ?? 'ok';
   266→      if (status === 'ok') okCount++;
   267→      else if (status === 'nok') nokCount++;
   268→    }));
   269→    const answeredCount = okCount + nokCount;
   270→    const scorePercent = totalItems > 0 ? Math.min(Math.round((okCount / totalItems) * 100), 100) : 0;
   271→    return { okCount, nokCount, answeredCount, scorePercent };
   272→  }, [results, sections, totalItems]);
   273→
   274→  // Check that all NOK items have hallazgo, mejora and responsable filled
   275→  const nokItems = Object.values(results).filter(r => r.status === 'nok');
   276→  const allNokCompleted = nokItems.length === 0 || nokItems.every(r => (r.hallazgo || '').trim() !== '' && (r.mejora || '').trim() !== '' && (r.responsable || '').trim() !== '');
   277→
   278→  const passed = scoring.scorePercent >= notaMinima;
   279→  const canSubmit = canPerformAutoeval && scoring.answeredCount > 0 && allNokCompleted;
   280→
   281→  const handlePhotoSelect = async (e: React.ChangeEvent<HTMLInputElement>) => {
   282→    const files = e.target.files;
   283→    if (!files) return;
   284→    for (const file of Array.from(files)) {
   285→      const preview = URL.createObjectURL(file);
   286→      setAutoevalPhotos(prev => [...prev, { file, preview }]);
   287→    }
   288→    // Reset input
   289→    if (photoInputRef.current) photoInputRef.current.value = '';
   290→  };
   291→
   292→  const removeAutoevalPhoto = (index: number) => {
   293→    setAutoevalPhotos(prev => {
   294→      const photo = prev[index];
   295→      if (photo.preview) URL.revokeObjectURL(photo.preview);
   296→      return prev.filter((_, i) => i !== index);
   297→    });
   298→  };
   299→
   300→  const toggleSection = (sectionId: string) => {
   301→    setExpandedSections(prev => ({ ...prev, [sectionId]: !prev[sectionId] }));
   302→  };
   303→
   304→  const setItemStatus = (itemId: string, status: 'ok' | 'nok' | 'na') => {
   305→    setResults(prev => ({
   306→      ...prev,
   307→      [itemId]: { ...prev[itemId], itemId, status },
   308→    }));
   309→  };
   310→
   311→  const setItemField = (itemId: string, field: 'hallazgo' | 'mejora' | 'otherText' | 'responsable', value: string) => {
   312→    setResults(prev => ({
   313→      ...prev,
   314→      [itemId]: { ...prev[itemId], itemId, [field]: value },
   315→    }));
   316→  };
   317→
   318→  const handleSubmit = async () => {
   319→    if (!canSubmit) return;
   320→    if (!canPerformAutoeval) return; // Only responsable/admin for S4, or any employee for S1/S2/S3/S5
   321→    setIsSubmitting(true);
   322→    try {
   323→      // Build effective results: merge missing items as 'ok' (default) before sending to backend
   324→      const effectiveResults = sections.flatMap(s => s.items).map(item => {
   325→        const r = results[item.id];
   326→        return r ?? { itemId: item.id, status: 'ok' as const };
   327→      });
   328→      // Only mark as completed if score meets notaMinima threshold
   329→      // If not passed, still save results but step stays available for retry
   330→      const res = await fetch(`/api/progress/step?sStep=${sStep}&miniStep=${miniStep}`, {
   331→        method: 'PUT',
   332→        headers: { 'Content-Type': 'application/json' },
   333→        body: JSON.stringify({
   334→          completed: passed,
   335→          score: scoring.scorePercent,
   336→          notes: JSON.stringify({
   337→            type: 'autoevaluacion',
   338→            passed,
   339→            notaMinima,
   340→            results: effectiveResults,
   341→            observaciones,
   342→            fechaAutoevaluacion,
   343→            horaAutoevaluacion,
   344→            fechaProgramada: fechaProgramada || null,
   345→            horaProgramada: horaProgramada || null,
   346→          }),
   347→          projectId: currentProject?.id,
   348→          zoneId: currentZone?.id || null,
   349→        }),
   350→      });
   351→      const json = await res.json();
   352→      if (!res.ok || !json.success) {
   353→        toast.error(`Error del servidor: ${json.error || res.statusText || 'Error desconocido'}`);
   354→        return;
   355→      }
   356→      if (json.success) {
   357→        // Feedback visible al usuario
   358→        toast.success(`Autoevaluación guardada: ${effectiveResults.length} items (${scoring.okCount} OK, ${scoring.nokCount} NOK, score ${scoring.scorePercent}%)`);
   359→        setIsCompleted(true);
   360→        setFinalScore(scoring.scorePercent);
   361→        await fetchProgress();
   362→
   363→        // ─── Create EmployeeProgress record for individual step 4 (autoevaluación) ───
   364→        // Step 4 is individual for S1/S2/S3/S5 (done by employees) and for S4 (done by responsable)
   365→        // We need to track individual completion so the gating system unlocks step 5
   366→        if (passed && currentUser?.id && currentProject?.id && currentZone?.id) {
   367→          try {
   368→            await fetch('/api/employee-progress', {
   369→              method: 'POST',
   370→              headers: { 'Content-Type': 'application/json' },
   371→              body: JSON.stringify({
   372→                sStep,
   373→                miniStep: 4,
   374→                completed: true,
   375→                score: scoring.scorePercent,
   376→                projectId: currentProject.id,
   377→                zoneId: currentZone.id,
   378→                userId: currentUser.id,
   379→              }),
   380→            });
   381→            // Also refresh employee progress so step 5 unlocks immediately
   382→            const { fetchEmployeeProgress } = use5SStore.getState();
   383→            await fetchEmployeeProgress(currentProject.id, currentZone.id);
   384→          } catch (empErr) {
   385→            console.error('Error creating employee progress for autoeval:', empErr);
   386→          }
   387→        }
   388→
   389→        // ─── Upload photos to library with traceability ───
   390→        if (autoevalPhotos.length > 0) {
   391→          setIsUploadingPhotos(true);
   392→          for (let idx = 0; idx < autoevalPhotos.length; idx++) {
   393→            const photo = autoevalPhotos[idx];
   394→            try {
   395→              // Step 1: Upload the file to get a server URL
   396→              const formData = new FormData();
   397→              formData.append('file', photo.file);
   398→              formData.append('filename', `S${sStep}_autoeval_${currentZone?.name || 'zona'}_${idx + 1}_${Date.now()}.jpg`);
   399→
   400→              const uploadRes = await fetch('/api/upload', {
   401→                method: 'POST',
   402→                body: formData,
   403→              });
   404→              const uploadData = await uploadRes.json();
   405→
   406→              if (uploadData.success && uploadData.url) {
   407→                // Step 2: Save to photo library with full traceability
   408→                await fetch('/api/photo-library', {
   409→                  method: 'POST',
   410→                  headers: { 'Content-Type': 'application/json' },
   411→                  body: JSON.stringify({
   412→                    sStep,
   413→                    miniStep: 4,
   414→                    title: `Autoeval S${sStep} - Foto ${idx + 1}`,
   415→                    description: `${sStepData?.japaneseName || 'S' + sStep} - ${currentZone?.name || 'Zona'} - Paso 4 Autoevaluación - Subida por ${currentUser?.name || 'Usuario'}`,
   416→                    photoUrl: uploadData.url,
   417→                    photoType: 'hallazgo',
   418→                    category: `paso4_s${sStep}`,
   419→                    tags: JSON.stringify([`S${sStep}`, sStepData?.japaneseName || '', currentZone?.name || '', 'paso4', 'autoevaluacion', 'hallazgo']),
   420→                    projectId: currentProject?.id,
   421→                    zoneId: currentZone?.id || null,
   422→                    uploadedBy: currentUser?.id || null,
   423→                  }),
   424→                });
   425→              }
   426→            } catch (photoErr) {
   427→              console.error('Error uploading autoeval photo:', photoErr);
   428→            }
   429→          }
   430→          setIsUploadingPhotos(false);
   431→        }
   432→
   433→        // ─── Create Action Items for NOK (disfunciones) ───
   434→        const nokResults = Object.values(results).filter((r: any) => r.status === 'nok');
   435→        for (const nok of nokResults) {
   436→          if (!nok.hallazgo && !nok.mejora) continue; // Skip items without description
   437→          try {
   438→            await fetch('/api/actions', {
   439→              method: 'POST',
   440→              headers: { 'Content-Type': 'application/json' },
   441→              body: JSON.stringify({
   442→                sStep,
   443→                miniStep: 4,
   444→                itemId: nok.itemId,
   445→                itemDescription: `Disfunción detectada en autoevaluación: ${nok.itemId}`,
   446→                hallazgo: nok.hallazgo || nok.itemId,
   447→                mejora: nok.mejora || '',
   448→                responsable: nok.responsable || null,
   449→                prioridad: 'media',
   450→                estado: 'abierta',
   451→                source: 'autoevaluacion',
   452→                auditor: null,
   453→                projectId: currentProject?.id,
   454→                zoneId: currentZone?.id || null,
   455→              }),
   456→            });
   457→          } catch (actionError) {
   458→            console.error('Error creating action item from autoevaluación:', actionError);
   459→          }
   460→        }
   461→
   462→        // ─── Notify responsables of NOK disfunciones ───
   463→        if (nokResults.length > 0 && currentProject?.id && currentZone?.id) {
   464→          try {
   465→            const membersRes = await fetch(`/api/projects/${currentProject.id}/members`);
   466→            const membersData = await membersRes.json();
   467→            const allMembers = membersData?.members || [];
   468→
   469→            const sStepData = S_STEPS.find(s => s.id === sStep);
   470→            const disfuncionMessage = `Se han detectado ${nokResults.length} disfunción(es) en la autoevaluación de S${sStep} (${sStepData?.japaneseName || ''}) en la zona "${currentZone.name}". Revisa el Plan de Acción.`;
   471→
   472→            // Notify zone responsable
   473→            const responsableIds = new Set<string>();
   474→            if (currentZone.responsableId) responsableIds.add(currentZone.responsableId);
   475→            const responsables = allMembers.filter((m: any) => m.role === 'responsable');
   476→            for (const resp of responsables) responsableIds.add(resp.userId);
   477→
   478→            for (const respId of responsableIds) {
   479→              await fetch('/api/notifications', {
   480→                method: 'POST',
   481→                headers: { 'Content-Type': 'application/json' },
   482→                body: JSON.stringify({
   483→                  userId: respId,
   484→                  type: 'disfuncion',
   485→                  title: `Disfunciones detectadas: S${sStep} — ${sStepData?.japaneseName || ''}`,
   486→                  message: disfuncionMessage,
   487→                  sStep,
   488→                  zoneId: currentZone.id,
   489→                  projectId: currentProject.id,
   490→                }),
   491→              });
