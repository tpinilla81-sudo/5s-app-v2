   106→  // Generate auto title with traceability

   115→  // Refs para cola de subida secuencial — evita saturar CPU/memoria en móvil
   116→  // cuando se seleccionan varias fotos a la vez (era la causa del bloqueo).
   117→  const uploadQueueRef = useRef<{ rawBase64: string; photoType: string }[]>([]);

   119→  const photoCounterRef = useRef(0);
   120→  const [queueLength, setQueueLength] = useState(0);
   121→
   122→  const uploadPhoto = async (base64Data: string, index: number): Promise<string | null> => {
   123→    try {
   124→      const projectId = currentProject?.id || 'unknown';
   125→      const filename = generatePhotoFilename(projectId, sStep, miniStep, index);
   126→      const file = base64toFile(base64Data, filename);
   127→
   128→      const formData = new FormData();
   129→      formData.append('file', file);
   130→      formData.append('filename', filename);
   131→
   132→      const res = await fetch('/api/upload', {
   133→        method: 'POST',
   134→        body: formData,
   135→      });
   136→
   137→      const data = await res.json();
   138→      if (data.success && data.url) return data.url;
   139→      return null;
   140→    } catch (error) {
   141→      console.error('Upload error:', error);
   142→      return null;
   143→    }
   144→  };
   145→
   146→  // Procesa la cola de fotos UNA A UNA. Así el canvas + toDataURL + fetch
   147→  // no compiten por CPU/memoria, y la UI del móvil no se congela.
   148→  const processQueue = useCallback(async () => {
   149→    if (isProcessingRef.current) return;
   150→    if (uploadQueueRef.current.length === 0) return;
   151→    isProcessingRef.current = true;
   152→    console.log(`[FotosModal] processQueue: iniciando procesamiento de ${uploadQueueRef.current.length} foto(s)`);
   153→
   154→    try {
   155→      while (uploadQueueRef.current.length > 0) {
   156→        const item = uploadQueueRef.current.shift()!;
   157→        setQueueLength(uploadQueueRef.current.length);
   158→
   159→        const { rawBase64, photoType } = item;
   160→        const index = photoCounterRef.current++;
   161→        const photoId = `photo_${Date.now()}_${index}_${Math.random().toString(36).slice(2, 8)}`;
   162→        console.log(`[FotosModal] processQueue: procesando foto #${index + 1} (id=${photoId}), quedan ${uploadQueueRef.current.length} en cola`);
   163→
   164→        try {
   165→          // v2.58: si la compresión falla, usar el original sin comprimir
   166→          let compressed: string;
   167→          try {
   168→            compressed = await compressImage(rawBase64);
   169→          } catch (compressErr) {
   170→            console.warn('[FotosModal] Compression failed, using original:', compressErr);
   171→            compressed = rawBase64;
   172→          }
   173→          const estimatedSize = estimateBase64Size(compressed);
   174→
   175→          const newPhoto: PhotoItem = {
   176→            id: photoId,
   177→            preview: compressed,
   178→            serverUrl: '',
   179→            uploaded: false,
   180→            uploading: true,
   181→            estimatedSize,
   182→            title: generatePhotoTitle(index, photoType),
   183→            photoType,
   184→            savedToLibrary: false,
   185→          };
   186→
   187→          setPhotos(prev => [...prev, newPhoto]);
   188→
   189→          const url = await uploadPhoto(compressed, index);
   190→
   191→          // Actualiza por ID — robusto frente a reordenamientos/eliminaciones
   192→          setPhotos(prev =>
   193→            prev.map(p =>
   194→              p.id === photoId
   195→                ? { ...p, serverUrl: url || '', uploaded: !!url, uploading: false }
   196→                : p
   197→            )
   198→          );
   199→        } catch (err) {
   200→          console.error('Error processing queued photo:', err);
   201→          // Marcamos la foto como fallida (uploaded=false, uploading=false) para que el usuario la vea
   202→          setPhotos(prev =>
   203→            prev.map(p =>
   204→              p.id === photoId
   205→                ? { ...p, uploading: false, uploaded: false }
   206→                : p
   207→            )
   208→          );
   209→        }
   210→
   211→        // Cede al event loop para que el navegador pinte (evita congelar la UI)
   212→        await new Promise(r => setTimeout(r, 30));
   213→      }
   214→    } finally {
   215→      isProcessingRef.current = false;
   216→      setQueueLength(0);
   217→    }
   218→  }, [sStep, miniStep, currentProject?.id]);
   219→
   220→  // Encolar (no procesa directamente). Múltiples llamadas paralelas solo llenan la cola,
   221→  // luego processQueue las consume de una en una.
   222→  const addPhoto = useCallback((rawBase64: string) => {
   223→    uploadQueueRef.current.push({ rawBase64, photoType: defaultPhotoType });
   224→    setQueueLength(uploadQueueRef.current.length);
   225→    processQueue();
   226→  }, [processQueue, defaultPhotoType]);
   227→
   228→  const stopStream = useCallback(() => {
   229→    if (stream) {

   232→    }
   233→    setCameraMode('idle');
   234→  }, [stream]);
   235→
   236→  const startCamera = useCallback(async () => {
   237→    setCameraError(null);
   238→    try {
   239→      const mediaStream = await navigator.mediaDevices.getUserMedia({
   240→        video: { facingMode, width: { ideal: 1920 }, height: { ideal: 1080 } },

   244→      setCameraMode('streaming');

   246→        videoRef.current.srcObject = mediaStream;
   247→        await videoRef.current.play();

   250→      if (err.name === 'NotAllowedError') setCameraError('Permiso denegado. Permite el acceso a la cámara.');
   251→      else if (err.name === 'NotFoundError') setCameraError('No se encontró ninguna cámara.');
   252→      else setCameraError('Error al acceder a la cámara. Intenta subir una foto desde tu galería.');
   253→    }
   254→  }, [facingMode]);
   255→
   256→  const switchCamera = useCallback(async () => {

   260→
   261→  const capturePhoto = useCallback(() => {
   262→    if (!videoRef.current || !canvasRef.current) return;
   263→    const video = videoRef.current;
   264→    const canvas = canvasRef.current;
   265→    // v2.60: validar que el video tiene un frame listo (videoWidth > 0).
   266→    // Si el stream acaba de iniciar o está en transición, videoWidth puede
   267→    // ser 0 y el canvas resultante sería 0x0 → imagen negra al comprimir.
   268→    if (!video.videoWidth || !video.videoHeight || video.videoWidth < 2 || video.videoHeight < 2) {
   269→      console.warn('[FotosModal] Video not ready (dimensions:', video.videoWidth, 'x', video.videoHeight, '). Skipping capture.');
   270→      toast.error('La cámara aún no está lista. Espera un instante e inténtalo de nuevo.');
   271→      return;
   272→    }
   273→    canvas.width = video.videoWidth;
   274→    canvas.height = video.videoHeight;
   275→    const ctx = canvas.getContext('2d');
   276→    if (!ctx) return;
   277→    // v2.60: llenar el canvas con blanco antes de dibujar (igual que compressImage)
   278→    // para evitar píxeles negros si la imagen tuviera transparencia.
   279→    ctx.fillStyle = '#ffffff';
   280→    ctx.fillRect(0, 0, canvas.width, canvas.height);
   281→    if (facingMode === 'user') { ctx.translate(canvas.width, 0); ctx.scale(-1, 1); }
   282→    ctx.drawImage(video, 0, 0);
   283→    const base64 = canvas.toDataURL('image/jpeg', 0.9);
   284→    // v2.60: sanity check — si la captura es sospechosamente pequeña,
   285→    // no la añadimos a la cola (sería una foto negra).
   286→    const base64Len = base64.split(',')[1]?.length || 0;
   287→    const estimatedSize = (base64Len * 3) / 4;
   288→    if (estimatedSize < 1024) {
   289→      console.error('[FotosModal] Capture produced empty image (' + estimatedSize + ' bytes). Skipping.');
   290→      toast.error('La captura salió vacía. Inténtalo de nuevo.');
   291→      return;
   292→    }
   293→    setCaptureFlash(true);
   294→    setTimeout(() => setCaptureFlash(false), 300);
   295→    addPhoto(base64);
   296→  }, [facingMode, addPhoto]);
   297→
   298→  // Load existing photos from DB when modal opens
   299→  const [loadingPhotos, setLoadingPhotos] = useState(false);
   300→
   301→  // v2.35: cargar límite dinámico de fotos al abrir el modal.

   314→      })
   315→      .catch(err => console.error('[FotosModal] Error al cargar photo-limits:', err));
   316→  }, [open, currentProject?.id, currentZone?.id, sStep, miniStep]);
   317→
   318→  useEffect(() => {
   319→    if (open && currentProject?.id) {
   320→      setLoadingPhotos(true);
   321→      const zoneIdParam = currentZone?.id ? `&zoneId=${currentZone.id}` : ''
   322→      fetch(`/api/photo-library?projectId=${currentProject.id}&sStep=${sStep}&miniStep=2${zoneIdParam}`)
   323→        .then(res => res.json())
   324→        .then(data => {
   325→          if (data.success && data.data && data.data.length > 0) {
   326→            const existingPhotos: PhotoItem[] = data.data.map((p: any, idx: number) => ({
   327→              id: `existing_${p.id || idx}_${Math.random().toString(36).slice(2, 8)}`,
   328→              preview: p.photoUrl, // Use the stored URL as preview
   329→              serverUrl: p.photoUrl,
   330→              uploaded: true,
   331→              uploading: false,
   332→              estimatedSize: 0,
   333→              title: p.title || '',
   334→              photoType: p.photoType || 'antes',
   335→              savedToLibrary: true, // Already saved in DB
   336→              dbId: p.id, // v2.47: id real en PhotoLibrary
   337→            }));
   338→            photoCounterRef.current = existingPhotos.length;
   339→            setPhotos(existingPhotos);
   340→            // If step was already completed, mark it

   342→            const despuesCount = existingPhotos.filter(p => p.photoType === 'despues').length;
   343→            if (antesCount >= minPhotos) setIsCompleted(true);
   344→          }
   345→          setLoadingPhotos(false);
   346→        })
   347→        .catch(err => {
   348→          console.error('Error loading existing photos:', err);
   349→          setLoadingPhotos(false);
   350→        });
   351→    }
   352→  }, [open, currentProject?.id, sStep, currentZone?.id]);
   353→
   354→  useEffect(() => {
   355→    if (!open) {
   356→      stopStream();
   357→      // Don't clear photos on close — they should persist
   358→      // Only clear transient UI state
   359→      setCameraError(null);
   360→      setCameraMode('idle');
   361→      setActiveTab('camera');
   362→      // v2.48: reset del flag "Añadir más fotos" al cerrar el modal
   363→      setShowAddMore(false);
   364→    }
   365→  }, [open, stopStream]);
   366→
   367→  // Lee todos los archivos y los encola. NO procesa aquí (eso saturationa el hilo UI).
   368→  // El procesamiento pesado (compresión + upload) lo hace processQueue secuencialmente.
   369→  const readFileAsDataURL = (file: File): Promise<string> =>
   370→    new Promise((resolve, reject) => {
   371→      const reader = new FileReader();
   372→      reader.onloadend = () => resolve(reader.result as string);
   373→      reader.onerror = () => reject(reader.error);
   374→      reader.readAsDataURL(file);
   375→    });
   376→
   377→  const handleFileSelect = async (e: React.ChangeEvent<HTMLInputElement>) => {
   378→    const files = e.target.files;
   379→    if (!files || files.length === 0) return;
   380→    // Captura el snapshot del FileList ANTES de limpiar el input.
   381→    // Si limpiamos antes, el navegador vacía el FileList y Array.from(files) da [].
   382→    const fileArr = Array.from(files);
   383→    if (fileInputRef.current) fileInputRef.current.value = '';
   384→    console.log(`[FotosModal] handleFileSelect: ${fileArr.length} archivo(s) seleccionado(s)`);
   385→    await Promise.all(fileArr.map(f => readFileAsDataURL(f).then(addPhoto).catch(err => console.error('Error leyendo archivo:', err))));
   386→  };
   387→
   388→  const handleCameraCapture = async (e: React.ChangeEvent<HTMLInputElement>) => {
   389→    const files = e.target.files;
   390→    if (!files || files.length === 0) return;
   391→    const fileArr = Array.from(files);
   392→    if (cameraInputRef.current) cameraInputRef.current.value = '';
   393→    console.log(`[FotosModal] handleCameraCapture: ${fileArr.length} archivo(s) capturado(s)`);
   394→    await Promise.all(fileArr.map(f => readFileAsDataURL(f).then(addPhoto).catch(err => console.error('Error leyendo archivo de cámara:', err))));
   395→  };
   396→
   397→  // v2.47: si la foto ya está guardada en la biblioteca (dbId presente),
   398→  // llamamos al backend DELETE para que el borrado persista. Si el backend
   399→  // devuelve 409 (Paso 2 ya completado), mostramos error y NO quitamos la
   400→  // foto del estado local — así el usuario entiende que no se puede borrar.
   401→  // Si la foto no está en la biblioteca (todavía no se hizo handleSubmit),
   402→  // simplemente la quitamos del estado local.
   403→  const removePhoto = async (index: number) => {
   404→    const photo = photos[index];
   405→    if (!photo) return;
   406→    if (photo.savedToLibrary && photo.dbId) {
   407→      try {
   408→        const res = await fetch(`/api/photo-library?id=${photo.dbId}`, { method: 'DELETE' });
   409→        if (!res.ok) {
   410→          const json = await res.json().catch(() => null);
   411→          toast.error(json?.error || 'No se puede eliminar esta foto.');
   412→          return; // NO la quitamos del estado local
   413→        }
   414→      } catch (e) {
   415→        console.error('Error deleting photo from backend:', e);
   416→        toast.error('Error de conexión al eliminar la foto.');
   417→        return;
   418→      }
   419→    }
   420→    setPhotos(prev => prev.filter((_, i) => i !== index));
   421→  };
   422→
   423→  const uploadingCount = photos.filter(p => p.uploading).length;
   424→  const isQueueBusy = queueLength > 0 || uploadingCount > 0;
   425→  const canSubmit = photos.length >= minPhotos && uploadingCount === 0 && queueLength === 0;
   426→  const totalSize = photos.reduce((sum, p) => sum + p.estimatedSize, 0);
   427→
   428→  const handleSubmit = async () => {
   429→    if (!canSubmit) return;
   430→    setIsSubmitting(true);
   431→    try {
   432→      const urls = photos.map(p => p.serverUrl || p.preview).join(',');
   433→
   434→      // v2.40: Cada foto del Paso 2 se vincula automáticamente a un elemento
   435→      // de inventario "borrador" que el usuario clasificará en el Paso 3.
   436→      // Así la foto está vinculada al registro DESDE EL MOMENTO en que se
   437→      // toma, no después. El flujo en InventarioModal cambia: el usuario ya
   438→      // no necesita vincular manualmente — solo rellenar los datos del
   439→      // elemento (nombre, categoría, decisión, etc.) y al hacerlo el
   440→      // borrador pasa a ser un item clasificado.
   441→      const sName = sStepData?.japaneseName || `S${sStep}`;
   442→      const zoneName = currentZone?.name || 'Zona';
   443→      const userLabel = currentUser?.name || 'Usuario';
   444→      let savedCount = 0;
   445→      let failedCount = 0;
   446→
   447→      // Procesamos SECUENCIALMENTE (no en paralelo) porque cada foto necesita
   448→      // el id del item creado en el paso anterior para vincularse.
   449→      // v2.48: trackea el dbId devuelto por el backend para cada foto recién
   450→      // guardada, y actualiza el estado local al final — así, si el usuario
   451→      // re-abre "Añadir más fotos", las fotos ya guardadas no se re-envían.
   452→      const newlySavedDbIds: { localId: string; dbId: string }[] = [];
   453→      for (let idx = 0; idx < photos.length; idx++) {
   454→        const p = photos[idx];
   455→        if (p.savedToLibrary) { savedCount++; continue; }
   456→        const photoUrlForDb = p.serverUrl || p.preview; // base64 fallback si el upload falló
   457→        const photoTitle = p.title || generatePhotoTitle(idx, p.photoType);
   458→
   459→        // 1) Guardar en PhotoLibrary
   460→        let photoId: string | null = null;
   461→        try {
   462→          const libRes = await fetch('/api/photo-library', {
   463→            method: 'POST',
   464→            headers: { 'Content-Type': 'application/json' },
   465→            body: JSON.stringify({
