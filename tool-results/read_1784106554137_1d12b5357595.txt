   300→// VISUAL EDITOR: ChecklistEditor (autoevaluacion / auditoria)
   301→// ═══════════════════════════════════════════════════════
   302→interface ChecklistSection {
   303→  id: string
   304→  title: string
   305→  items: { id: string; description: string; hasOther: boolean }[]
   306→}
   307→
   308→function ChecklistEditor({ content, onChange }: { content: string; onChange: (v: string) => void }) {
   309→  let parsed: { sections: ChecklistSection[] }
   310→  try {
   311→    parsed = JSON.parse(content)
   312→  } catch {
   313→    return (
   314→      <div className="p-4 bg-red-50 border border-red-200 rounded-lg text-sm text-red-600">
   315→        El JSON no es válido. Corrígelo en modo JSON o carga el contenido por defecto.
   316→      </div>
   317→    )
   318→  }
   319→
   320→  const sections = parsed.sections || []
   321→  const sPrefix = String(content.match(/"id"\s*:\s*"(\d+)\./)?.[1] || '1')
   322→
   323→  const update = (newSections: ChecklistSection[]) => {
   324→    onChange(JSON.stringify({ ...parsed, sections: newSections }, null, 2))
   325→  }
   326→
   327→  const addSection = () => {
   328→    const newId = `${sPrefix}.${sections.length + 1}`
   329→    update([...sections, { id: newId, title: 'Nueva Sección', items: [] }])
   330→  }
   331→
   332→  const removeSection = (idx: number) => {
   333→    update(sections.filter((_, i) => i !== idx))
   334→  }
   335→
   336→  const updateSection = (idx: number, field: 'id' | 'title', value: string) => {
   337→    const updated = [...sections]
   338→    updated[idx] = { ...updated[idx], [field]: value }
   339→    update(updated)
   340→  }
   341→
   342→  const moveSection = (idx: number, dir: -1 | 1) => {
   343→    const target = idx + dir
   344→    if (target < 0 || target >= sections.length) return
   345→    const updated = [...sections]
   346→    ;[updated[idx], updated[target]] = [updated[target], updated[idx]]
   347→    update(updated)
   348→  }
   349→
   350→  const addItem = (sectionIdx: number) => {
   351→    const sec = sections[sectionIdx]
   352→    const newItemId = `${sec.id}.${sec.items.length + 1}`
   353→    const updated = [...sections]
   354→    updated[sectionIdx] = {
   355→      ...updated[sectionIdx],
   356→      items: [...updated[sectionIdx].items, { id: newItemId, description: '', hasOther: false }]
   357→    }
   358→    update(updated)
   359→  }
   360→
   361→  const removeItem = (sectionIdx: number, itemIdx: number) => {
   362→    const updated = [...sections]
   363→    updated[sectionIdx] = {
   364→      ...updated[sectionIdx],
   365→      items: updated[sectionIdx].items.filter((_, i) => i !== itemIdx)
   366→    }
   367→    update(updated)
   368→  }
   369→
   370→  const updateItem = (sectionIdx: number, itemIdx: number, field: string, value: string | boolean) => {
   371→    const updated = [...sections]
   372→    updated[sectionIdx] = {
   373→      ...updated[sectionIdx],
   374→      items: updated[sectionIdx].items.map((item, i) =>
   375→        i === itemIdx ? { ...item, [field]: value } : item
   376→      )
   377→    }
   378→    update(updated)
   379→  }
   380→
   381→  const moveItem = (sectionIdx: number, itemIdx: number, dir: -1 | 1) => {
   382→    const items = sections[sectionIdx].items
   383→    const target = itemIdx + dir
   384→    if (target < 0 || target >= items.length) return
   385→    const updated = [...sections]
   386→    const newItems = [...updated[sectionIdx].items]
   387→    ;[newItems[itemIdx], newItems[target]] = [newItems[target], newItems[itemIdx]]
   388→    updated[sectionIdx] = { ...updated[sectionIdx], items: newItems }
   389→    update(updated)
   390→  }
   391→
   392→  return (
   393→    <div className="space-y-4">
   394→      {sections.map((section, sIdx) => (
   395→        <div key={sIdx} className="border-2 rounded-lg overflow-hidden bg-white shadow-sm">
   396→          {/* Section header */}
   397→          <div className="flex items-center gap-3 px-4 py-3 bg-gray-50 border-b">
   398→            <div className="flex flex-col gap-0.5">
   399→              <button onClick={() => moveSection(sIdx, -1)} className="text-gray-400 hover:text-gray-600 leading-none" title="Subir sección">
   400→                <ChevronUp className="h-4 w-4" />
   401→              </button>
   402→              <button onClick={() => moveSection(sIdx, 1)} className="text-gray-400 hover:text-gray-600 leading-none" title="Bajar sección">
   403→                <ChevronDown className="h-4 w-4" />
   404→              </button>
   405→            </div>
   406→            <Input
   407→              value={section.id}
   408→              onChange={(e) => updateSection(sIdx, 'id', e.target.value)}
   409→              className="w-20 h-9 text-sm font-mono"
   410→              placeholder="ID"
   411→            />
   412→            <Input
   413→              value={section.title}
   414→              onChange={(e) => updateSection(sIdx, 'title', e.target.value)}
   415→              className="flex-1 h-9 text-base font-semibold"
   416→              placeholder="Título de sección"
   417→            />
   418→            <Button variant="ghost" size="sm" onClick={() => addItem(sIdx)}
   419→              className="h-9 w-9 p-0 text-green-600 hover:text-green-700 hover:bg-green-50" title="Añadir item">
   420→              <Plus className="h-5 w-5" />
   421→            </Button>
   422→            <Button variant="ghost" size="sm" onClick={() => removeSection(sIdx)}
   423→              className="h-9 w-9 p-0 text-red-500 hover:text-red-600 hover:bg-red-50" title="Eliminar sección">
   424→              <Trash2 className="h-4 w-4" />
   425→            </Button>
   426→          </div>
   427→
   428→          {/* Items */}
   429→          <div className="p-3 space-y-2">
   430→            {section.items.length === 0 && (
   431→              <p className="text-sm text-muted-foreground italic px-2 py-2">Sin items. Pulsa + para añadir.</p>
   432→            )}
   433→            {section.items.map((item, iIdx) => (
   434→              <div key={iIdx} className="flex items-center gap-3 group">
   435→                <div className="flex flex-col gap-0.5">
   436→                  <button onClick={() => moveItem(sIdx, iIdx, -1)} className="text-gray-300 hover:text-gray-500 leading-none" title="Subir">
   437→                    <ChevronUp className="h-3.5 w-3.5" />
   438→                  </button>
   439→                  <button onClick={() => moveItem(sIdx, iIdx, 1)} className="text-gray-300 hover:text-gray-500 leading-none" title="Bajar">
   440→                    <ChevronDown className="h-3.5 w-3.5" />
   441→                  </button>
   442→                </div>
   443→                <Input
   444→                  value={item.id}
   445→                  onChange={(e) => updateItem(sIdx, iIdx, 'id', e.target.value)}
   446→                  className="w-20 h-8 text-xs font-mono"
   447→                  placeholder="ID"
   448→                />
   449→                <Input
   450→                  value={item.description}
   451→                  onChange={(e) => updateItem(sIdx, iIdx, 'description', e.target.value)}
   452→                  className="flex-1 h-8 text-sm"
   453→                  placeholder="Descripción del item"
   454→                />
   455→                <label className="flex items-center gap-1.5 text-xs text-muted-foreground whitespace-nowrap cursor-pointer shrink-0">
   456→                  <input
   457→                    type="checkbox"
   458→                    checked={item.hasOther}
   459→                    onChange={(e) => updateItem(sIdx, iIdx, 'hasOther', e.target.checked)}
   460→                    className="rounded border-gray-300 h-4 w-4"
   461→                  />
   462→                  Otros
   463→                </label>
   464→                <Button variant="ghost" size="sm"
   465→                  onClick={() => removeItem(sIdx, iIdx)}
   466→                  className="h-8 w-8 p-0 text-red-400 hover:text-red-500 opacity-0 group-hover:opacity-100 transition-opacity" title="Eliminar">
   467→                  <Trash2 className="h-4 w-4" />
   468→                </Button>
   469→              </div>
   470→            ))}
   471→          </div>
   472→        </div>
   473→      ))}
   474→
   475→      <Button variant="outline" onClick={addSection}
   476→        className="w-full border-dashed border-2 text-green-600 hover:bg-green-50 hover:border-green-400 gap-1 h-10">
   477→        <Plus className="h-5 w-5" />
   478→        Añadir sección
   479→      </Button>
   480→
   481→      <div className="text-sm text-muted-foreground text-center">
   482→        {sections.length} sección(es) · {sections.reduce((s, sec) => s + sec.items.length, 0)} item(s) en total
   483→      </div>
   484→    </div>
   485→  )
   486→}

   501→  }
   502→
   503→  const questions = parsed.questions || []

   518→    const updated = [...questions]
   519→    updated[idx] = { ...updated[idx], [field]: value }
   520→    update(updated)
   521→  }
   522→

   524→    const updated = [...questions]

   526→    update(updated)

   531→    const letters = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ'

   533→    updated[qIdx] = { ...updated[qIdx], options: [...updated[qIdx].options, `Opción ${letters[newIdx] || newIdx + 1}`] }
   534→    update(updated)

   538→    const updated = [...questions]
   539→    const newOptions = updated[qIdx].options.filter((_, i) => i !== oIdx)
   540→    let newCorrect = updated[qIdx].correctIndex
   541→    if (newCorrect >= newOptions.length) newCorrect = 0
   542→    else if (newCorrect > oIdx) newCorrect--
   543→    updated[qIdx] = { ...updated[qIdx], options: newOptions, correctIndex: newCorrect }
   544→    update(updated)
   545→  }
   546→
   547→  const moveQuestion = (idx: number, dir: -1 | 1) => {
   548→    const target = idx + dir
   549→    if (target < 0 || target >= questions.length) return
   550→    const updated = [...questions]

   561→              <button onClick={() => moveQuestion(qIdx, -1)} className="text-gray-400 hover:text-gray-600 leading-none">
   562→                <ChevronUp className="h-4 w-4" />
   563→              </button>
   564→              <button onClick={() => moveQuestion(qIdx, 1)} className="text-gray-400 hover:text-gray-600 leading-none">
   565→                <ChevronDown className="h-4 w-4" />
   566→              </button>
   567→            </div>
   568→            <Badge className="bg-amber-200 text-amber-800 shrink-0 text-sm px-2 py-0.5">P{qIdx + 1}</Badge>
   569→            <Input
   570→              value={q.question}
   571→              onChange={(e) => updateQuestion(qIdx, 'question', e.target.value)}
   572→              className="flex-1 h-9 text-base"
   573→              placeholder="Pregunta"
   574→            />
   575→            <Button variant="ghost" size="sm" onClick={() => removeQuestion(qIdx)}
   576→              className="h-9 w-9 p-0 text-red-500 hover:text-red-600 hover:bg-red-50" title="Eliminar pregunta">
   577→              <Trash2 className="h-4 w-4" />
   578→            </Button>
   579→          </div>

   581→          <div className="p-4 space-y-2">
