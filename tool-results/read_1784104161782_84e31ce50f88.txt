   282→   282→function ChecklistEditor({ content, onChange }: { content: string; onChange: (v: string) => void }) {
   283→   283→  let parsed: { sections: ChecklistSection[] }
   284→   284→  try {
   285→   285→    parsed = JSON.parse(content)
   286→   286→  } catch {
   287→   287→    return (
   288→   288→      <div className="p-4 bg-red-50 border border-red-200 rounded-lg text-sm text-red-600">
   289→   289→        El JSON no es válido. Corrígelo en modo JSON o carga el contenido por defecto.
   290→   290→      </div>
   291→   291→    )
   292→   292→  }
   293→   293→
   294→   294→  const sections = parsed.sections || []
   295→   295→  const sPrefix = String(content.match(/"id"\s*:\s*"(\d+)\./)?.[1] || '1')
   296→   296→
   297→   297→  const update = (newSections: ChecklistSection[]) => {
   298→   298→    onChange(JSON.stringify({ ...parsed, sections: newSections }, null, 2))
   299→   299→  }
   300→   300→
   301→   301→  const addSection = () => {
   302→   302→    const newId = `${sPrefix}.${sections.length + 1}`
   303→   303→    update([...sections, { id: newId, title: 'Nueva Sección', items: [] }])
   304→   304→  }
   305→   305→
   306→   306→  const removeSection = (idx: number) => {
   307→   307→    update(sections.filter((_, i) => i !== idx))
   308→   308→  }
   309→   309→
   310→   310→  const updateSection = (idx: number, field: 'id' | 'title', value: string) => {
   311→   311→    const updated = [...sections]
   312→   312→    updated[idx] = { ...updated[idx], [field]: value }
   313→   313→    update(updated)
   314→   314→  }
   315→   315→
   316→   316→  const moveSection = (idx: number, dir: -1 | 1) => {
   317→   317→    const target = idx + dir
   318→   318→    if (target < 0 || target >= sections.length) return
   319→   319→    const updated = [...sections]
   320→   320→    ;[updated[idx], updated[target]] = [updated[target], updated[idx]]
   321→   321→    update(updated)
   322→   322→  }
   323→   323→
   324→   324→  const addItem = (sectionIdx: number) => {
   325→   325→    const sec = sections[sectionIdx]
   326→   326→    const newItemId = `${sec.id}.${sec.items.length + 1}`
   327→   327→    const updated = [...sections]
   328→   328→    updated[sectionIdx] = {
   329→   329→      ...updated[sectionIdx],
   330→   330→      items: [...updated[sectionIdx].items, { id: newItemId, description: '', hasOther: false }]
   331→   331→    }
   332→   332→    update(updated)
   333→   333→  }
   334→   334→
   335→   335→  const removeItem = (sectionIdx: number, itemIdx: number) => {
   336→   336→    const updated = [...sections]
   337→   337→    updated[sectionIdx] = {
   338→   338→      ...updated[sectionIdx],
   339→   339→      items: updated[sectionIdx].items.filter((_, i) => i !== itemIdx)
   340→   340→    }
   341→   341→    update(updated)
   342→   342→  }
   343→   343→
   344→   344→  const updateItem = (sectionIdx: number, itemIdx: number, field: string, value: string | boolean) => {
   345→   345→    const updated = [...sections]
   346→   346→    updated[sectionIdx] = {
   347→   347→      ...updated[sectionIdx],
   348→   348→      items: updated[sectionIdx].items.map((item, i) =>
   349→   349→        i === itemIdx ? { ...item, [field]: value } : item
   350→   350→      )
   351→   351→    }
   352→   352→    update(updated)
   353→   353→  }
   354→   354→
   355→   355→  const moveItem = (sectionIdx: number, itemIdx: number, dir: -1 | 1) => {
   356→   356→    const items = sections[sectionIdx].items
   357→   357→    const target = itemIdx + dir
   358→   358→    if (target < 0 || target >= items.length) return
   359→   359→    const updated = [...sections]
   360→   360→    const newItems = [...updated[sectionIdx].items]
   361→   361→    ;[newItems[itemIdx], newItems[target]] = [newItems[target], newItems[itemIdx]]
   362→   362→    updated[sectionIdx] = { ...updated[sectionIdx], items: newItems }
   363→   363→    update(updated)
   364→   364→  }
   365→   365→
   366→   366→  return (
   367→   367→    <div className="space-y-4">
   368→   368→      {sections.map((section, sIdx) => (

   370→   370→          {/* Section header */}

   372→   372→            <div className="flex flex-col gap-0.5">
   373→   373→              <button onClick={() => moveSection(sIdx, -1)} className="text-gray-400 hover:text-gray-600 leading-none" title="Subir sección">
   374→   374→                <ChevronUp className="h-4 w-4" />
   375→   375→              </button>
   376→   376→              <button onClick={() => moveSection(sIdx, 1)} className="text-gray-400 hover:text-gray-600 leading-none" title="Bajar sección">
   377→   377→                <ChevronDown className="h-4 w-4" />
   378→   378→              </button>
   379→   379→            </div>
   380→   380→            <Input

   382→   382→              onChange={(e) => updateSection(sIdx, 'id', e.target.value)}

   384→   384→              placeholder="ID"
   385→   385→            />
   386→   386→            <Input

   390→   390→              placeholder="Título de sección"
   391→   391→            />

   396→   396→            <Button variant="ghost" size="sm" onClick={() => removeSection(sIdx)}
   397→   397→              className="h-9 w-9 p-0 text-red-500 hover:text-red-600 hover:bg-red-50" title="Eliminar sección">
   398→   398→              <Trash2 className="h-4 w-4" />

   400→   400→          </div>
   401→   401→
   402→   402→          {/* Items */}
   403→   403→          <div className="p-3 space-y-2">
   404→   404→            {section.items.length === 0 && (
   405→   405→              <p className="text-sm text-muted-foreground italic px-2 py-2">Sin items. Pulsa + para añadir.</p>
   406→   406→            )}
   407→   407→            {section.items.map((item, iIdx) => (
   408→   408→              <div key={iIdx} className="flex items-center gap-3 group">

   411→   411→                    <ChevronUp className="h-3.5 w-3.5" />
   412→   412→                  </button>

   422→   422→                />

   958→   958→interface StdField { key: string; label: string; type: 'photo' | 'text' | 'number' | 'select'; options?: string[]; required?: boolean }
   959→   959→
   960→   960→function StandardTemplateEditor({ content, onChange }: { content: string; onChange: (v: string) => void }) {
   961→   961→  let parsed: { fields: StdField[] }
   962→   962→  try {
   963→   963→    parsed = JSON.parse(content)
   964→   964→  } catch {
   965→   965→    return (
   966→   966→      <div className="p-4 bg-red-50 border border-red-200 rounded-lg text-sm text-red-600">
   967→   967→        El JSON no es válido. Corrígelo en modo JSON o carga el contenido por defecto.
   968→   968→      </div>
   969→   969→    )
   970→   970→  }
   971→   971→
   972→   972→  const fields = parsed.fields || []
   973→   973→
   974→   974→  const update = (newFields: StdField[]) => {
   975→   975→    onChange(JSON.stringify({ ...parsed, fields: newFields }, null, 2))
   976→   976→  }
   977→   977→
   978→   978→  const addField = () => {
   979→   979→    update([...fields, { key: '', label: '', type: 'text', required: false }])
   980→   980→  }
   981→   981→
   982→   982→  const removeField = (idx: number) => {
   983→   983→    update(fields.filter((_, i) => i !== idx))
   984→   984→  }
   985→   985→
   986→   986→  const updateField = (idx: number, field: string, value: string | boolean) => {
   987→   987→    const updated = [...fields]
   988→   988→    updated[idx] = { ...updated[idx], [field]: value }
   989→   989→    // Remove options if type is not select
   990→   990→    if (field === 'type' && value !== 'select') {
   991→   991→      const { options, ...rest } = updated[idx]
   992→   992→      updated[idx] = rest as StdField
   993→   993→    }
   994→   994→    update(updated)
   995→   995→  }
   996→   996→
   997→   997→  const moveField = (idx: number, dir: -1 | 1) => {
   998→   998→    const target = idx + dir
   999→   999→    if (target < 0 || target >= fields.length) return
  1000→  1000→    const updated = [...fields]
  1001→  1001→    ;[updated[idx], updated[target]] = [updated[target], updated[idx]]
  1002→  1002→    update(updated)
  1003→  1003→  }
  1004→  1004→
  1005→  1005→  const addOption = (fIdx: number) => {
  1006→  1006→    const updated = [...fields]
  1007→  1007→    const opts = [...(updated[fIdx].options || []), '']

  1009→  1009→    update(updated)
  1010→  1010→  }

  1013→  1013→    const updated = [...fields]
  1014→  1014→    const opts = (updated[fIdx].options || []).filter((_, i) => i !== oIdx)
  1015→  1015→    updated[fIdx] = { ...updated[fIdx], options: opts }
  1016→  1016→    update(updated)
  1017→  1017→  }

  1019→  1019→  const updateOption = (fIdx: number, oIdx: number, value: string) => {
  1020→  1020→    const updated = [...fields]

  1023→  1023→    updated[fIdx] = { ...updated[fIdx], options: opts }
  1024→  1024→    update(updated)
  1025→  1025→  }

  1087→  1087→      <Button variant="outline" onClick={addField} size="sm"
