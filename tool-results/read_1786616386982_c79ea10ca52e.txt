   134→// ─── Field helper for Datos Empresa ─────────────────────────────────────────
   135→// Muestra el valor en modo lectura o un Input editable según `editing`.
   136→function Field({
   137→  label, k, value, edit, setEdit, editing,
   138→}: {
   139→  label: string
   140→  k: string
   141→  value: string | null | undefined
   142→  edit: Record<string, string>
   143→  setEdit: React.Dispatch<React.SetStateAction<Record<string, string>>>
   144→  editing: boolean
   145→}) {
   146→  return (
   147→    <div className="space-y-1">
   148→      <Label className="text-xs text-muted-foreground">{label}</Label>
   149→      {editing ? (
   150→        <Input
   151→          value={edit[k] ?? ''}
   152→          onChange={e => setEdit(prev => ({ ...prev, [k]: e.target.value }))}
   153→          className="h-8 text-sm"
   154→          placeholder={label}
   155→        />
   156→      ) : (
   157→        <div className={`h-8 px-2 flex items-center text-sm rounded-md border border-transparent bg-gray-50 ${
   158→          (value ?? '').trim() === '' ? 'text-muted-foreground italic' : 'text-gray-900'
   159→        }`}>
   160→          {(value ?? '').trim() === '' ? '— sin rellenar —' : value}
   161→        </div>
   162→      )}
   163→    </div>
   164→  )
   165→}
