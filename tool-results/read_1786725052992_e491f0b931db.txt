    51→
    52→  const [isFullscreen, setIsFullscreen] = useState(true);

    54→  const [expandedSections, setExpandedSections] = useState<Record<string, boolean>>({});
    55→  const [observaciones, setObservaciones] = useState('');
    56→  const [isSubmitting, setIsSubmitting] = useState(false);
    57→  const [isCompleted, setIsCompleted] = useState(false);
    58→  const [finalScore, setFinalScore] = useState(0);

    60→  // so we don't reset user edits if `sections` ref changes mid-session.
    61→  const initializedFor = useRef<string>('');
    62→  const [notaMinima, setNotaMinima] = useState(70);
    63→  const [autoevalPhotos, setAutoevalPhotos] = useState<{ file: File; preview: string; uploading?: boolean; serverUrl?: string }[]>([]);
    64→  const [isUploadingPhotos, setIsUploadingPhotos] = useState(false);
    65→  const photoInputRef = useRef<HTMLInputElement>(null);
    66→  // v2.63: Per-NOK-finding photos — each NOK item can have its own photos
    67→  // that will be linked to the corresponding ActionItem via photoRefs.
    68→  const [nokPhotos, setNokPhotos] = useState<Record<string, { file: File; preview: string }[]>>({});
    69→  const nokPhotoInputRef = useRef<Record<string, HTMLInputElement | null>>({});
    70→  const [activeNokItemId, setActiveNokItemId] = useState<string | null>(null);
    71→
    72→  // Project members for responsable selector on NOK items
    73→  const [projectMembers, setProjectMembers] = useState<Array<{ id: string; userId: string; role: string; user: { id: string; name: string; email: string; role: string; active: boolean } }>>([]);
    74→
    75→  // Date/Time for scheduling and recording the evaluation
    76→  const [fechaAutoevaluacion, setFechaAutoevaluacion] = useState('');
    77→  const [horaAutoevaluacion, setHoraAutoevaluacion] = useState('');
    78→  const [fechaProgramada, setFechaProgramada] = useState('');
    79→  const [horaProgramada, setHoraProgramada] = useState('');
    80→  // Track auditoría schedule (miniStep 5) — employee sees this as read-only (auditor sets it)
    81→  const [auditFechaProgramada, setAuditFechaProgramada] = useState('');
    82→  const [auditHoraProgramada, setAuditHoraProgramada] = useState('');
    83→
    84→  // Load template from API (uses board config if zone has one)
    85→  const { sections, isLoading: isLoadingTemplate, notaMinima: templateNotaMinima } = useChecklistTemplate('autoevaluacion', sStep, open, currentZone?.boardConfigId);
    86→
    87→  // Apply template notaMinima when loaded
    88→  useEffect(() => {
    89→    if (templateNotaMinima !== null) setNotaMinima(templateNotaMinima);
    90→  }, [templateNotaMinima]);
    91→
    92→  // Load project members (for responsable selector on NOK items)
    93→  useEffect(() => {
    94→    if (open && currentProject?.id) {
    95→      fetch(`/api/projects/${currentProject.id}/members`)
    96→        .then(r => r.json())
    97→        .then(data => setProjectMembers(data?.members || []))
    98→        .catch(err => console.error('Error loading project members:', err));
    99→    } else {
   100→      setProjectMembers([]);
   101→    }
   102→  }, [open, currentProject?.id]);
   103→
   104→
   105→
   106→  // Fetch dynamic threshold (overrides template if present)
   107→  useEffect(() => {
   108→    if (open && currentProject?.id) {
   109→      const fetchThreshold = async () => {
   110→        try {
   111→          const params = new URLSearchParams({ projectId: currentProject.id });
   112→          if (currentZone?.id) params.set('zoneId', currentZone.id);
   113→          const res = await fetch(`/api/audit-targets?${params}`);
   114→          const json = await res.json();
   115→          if (json.success && json.data) {
   116→            const zoneTarget = json.data.find((t: any) => t.sStep === sStep && t.miniStep === 4 && t.zoneId === currentZone?.id);
   117→            const projectTarget = json.data.find((t: any) => t.sStep === sStep && t.miniStep === 4 && t.zoneId === null);
   118→            const target = zoneTarget || projectTarget;
   119→            if (target?.notaMinima) setNotaMinima(target.notaMinima);
   120→          }
   121→        } catch (e) { console.error('Error fetching threshold:', e); }
   122→      };
   123→      fetchThreshold();
   124→    }
   125→  }, [open, sStep, currentProject?.id, currentZone?.id]);
   126→
   127→  // ─────────────────────────────────────────────────────────────
   128→  // ROBUST PRE-TICK: Merge missing items as 'ok' whenever sections change.
   129→  // This replaces the fragile one-shot pre-tick. Instead of setting results
   130→  // once and praying the effect fires after sections loads, we continuously
   131→  // MERGE missing items as 'ok' into `results` without overwriting user
   132→  // edits. This guarantees that every visible item always has an 'ok' entry
   133→  // unless the user explicitly marks it NOK/N/A.
   134→  // ─────────────────────────────────────────────────────────────
   135→  useEffect(() => {
   136→    if (!open || sections.length === 0) return;
   137→    setResults(prev => {
   138→      const next = { ...prev };
   139→      let changed = false;
   140→      sections.forEach(s => s.items.forEach(item => {
   141→        if (!next[item.id]) {
   142→          next[item.id] = { itemId: item.id, status: 'ok' };
   143→          changed = true;
   144→        }
   145→      }));
   146→      return changed ? next : prev;
   147→    });
   148→  }, [open, sections]);
   149→
   150→  // Initialize expanded sections + reset form state once per (sStep + sections signature)
   151→  useEffect(() => {
   152→    if (!open || sections.length === 0) return;
   153→    const sig = `${sStep}:${sections.map(s => s.items.map(i => i.id).join(',')).join('|')}`;
   154→    if (initializedFor.current === sig) return;
   155→    initializedFor.current = sig;
   156→
   157→    const expanded: Record<string, boolean> = {};
   158→    sections.forEach(s => { expanded[s.id] = true; });
   159→    setExpandedSections(expanded);
   160→    setObservaciones('');
   161→    setIsCompleted(false);
   162→    setFinalScore(0);
   163→    setAutoevalPhotos([]);
   164→    setNokPhotos({});
   165→    // Auto-fill current date/time for the evaluation
   166→    const now = new Date();
   167→    setFechaAutoevaluacion(now.toISOString().split('T')[0]);
   168→    setHoraAutoevaluacion(now.toTimeString().slice(0, 5));
   169→    // Load scheduled date if available
   170→    loadScheduledDate();
   171→    // Load previously saved results (so user can see what was saved before)
   172→    loadSavedResults();
   173→  }, [open, sStep, sections]);
   174→
   175→  // Load previously saved audit results from the backend so the user can see
   176→  // what was actually persisted. This makes the save behavior verifiable.
   177→  const loadSavedResults = async () => {
   178→    if (!currentProject?.id) return;
   179→    try {
   180→      const res = await fetch(`/api/progress/step?sStep=${sStep}&miniStep=${miniStep}&projectId=${currentProject.id}${currentZone?.id ? `&zoneId=${currentZone.id}` : ''}`);
   181→      const json = await res.json();
   182→      if (json.success && json.data?.notes) {
   183→        try {
   184→          const notes = typeof json.data.notes === 'string' ? JSON.parse(json.data.notes) : json.data.notes;
   185→          if (notes?.results && Array.isArray(notes.results)) {
   186→            const loaded: Record<string, AuditItemResult> = {};
   187→            notes.results.forEach((r: any) => {
   188→              if (r?.itemId) loaded[r.itemId] = r;
   189→            });
   190→            // Only set if there's at least one real entry (don't overwrite with empty)
   191→            if (Object.keys(loaded).length > 0) {
   192→              setResults(loaded);
   193→              if (notes.observaciones) setObservaciones(notes.observaciones);
   194→              if (notes.fechaAutoevaluacion) setFechaAutoevaluacion(notes.fechaAutoevaluacion);
   195→              if (notes.horaAutoevaluacion) setHoraAutoevaluacion(notes.horaAutoevaluacion);
   196→            }
   197→          }
   198→        } catch (e) {
   199→          console.error('Error parsing saved notes:', e);
   200→        }
   201→      }
   202→    } catch (e) {
   203→      console.error('Error loading saved results:', e);
   204→    }
   205→  };
   206→
   207→  // Reset the init guard when modal closes, so next open re-initializes
   208→  useEffect(() => {
   209→    if (!open) {
   210→      initializedFor.current = '';
   211→      // Also clear results on close so next open starts fresh (then loads saved)
   212→      setResults({});
   213→    }
   214→  }, [open]);
   215→
   216→  // Load scheduled date/time for this autoevaluación AND the auditoría schedule
   217→  const loadScheduledDate = async () => {
   218→    if (!currentProject?.id || !currentZone?.id) return;
   219→    try {
   220→      // Load autoevaluación schedule (miniStep 4)
   221→      const res = await fetch(`/api/evaluation-schedule?sStep=${sStep}&miniStep=4&projectId=${currentProject.id}&zoneId=${currentZone.id}`);
   222→      const json = await res.json();
   223→      if (json.success && json.data) {
   224→        setFechaProgramada(json.data.fechaProgramada || '');
   225→        setHoraProgramada(json.data.horaProgramada || '');
   226→      }
   227→      // Also load auditoría schedule (miniStep 5) — employee sees if auditor has set a date
   228→      const auditRes = await fetch(`/api/evaluation-schedule?sStep=${sStep}&miniStep=5&projectId=${currentProject.id}&zoneId=${currentZone.id}`);
   229→      const auditJson = await auditRes.json();
   230→      if (auditJson.success && auditJson.data) {
   231→        setAuditFechaProgramada(auditJson.data.fechaProgramada || '');
   232→        setAuditHoraProgramada(auditJson.data.horaProgramada || '');
   233→      }
   234→    } catch (e) {
   235→      console.error('Error loading scheduled date:', e);
   236→    }
   237→  };
   238→
   239→  // Save scheduled date/time
   240→  const handleSaveSchedule = async () => {
   241→    if (!currentProject?.id || !currentZone?.id) return;
   242→    try {
   243→      await fetch('/api/evaluation-schedule', {
   244→        method: 'POST',
   245→        headers: { 'Content-Type': 'application/json' },
   246→        body: JSON.stringify({
   247→          sStep,
   248→          miniStep: 4,
   249→          projectId: currentProject.id,
   250→          zoneId: currentZone.id,
   251→          fechaProgramada,
   252→          horaProgramada,
   253→        }),
   254→      });
   255→      toast.success('Fecha programada guardada');
   256→    } catch (e) {
   257→      console.error('Error saving schedule:', e);
   258→      toast.error('Error al guardar la fecha programada');
   259→    }
   260→  };
   261→
   262→  const totalItems = sections.reduce((sum, s) => sum + s.items.length, 0) || 26;
   263→
   264→  // SCORING: Treat missing results as 'ok' (default).
   265→  // This is bulletproof — does NOT depend on useEffect timing.
   266→  // Every visible item that hasn't been explicitly marked NOK/N/A counts as OK.
   267→  const scoring = useMemo(() => {
   268→    let okCount = 0;
   269→    let nokCount = 0;
   270→    sections.forEach(s => s.items.forEach(item => {
   271→      const status = results[item.id]?.status ?? 'ok';
   272→      if (status === 'ok') okCount++;
   273→      else if (status === 'nok') nokCount++;
   274→    }));
   275→    const answeredCount = okCount + nokCount;
   276→    const scorePercent = totalItems > 0 ? Math.min(Math.round((okCount / totalItems) * 100), 100) : 0;
   277→    return { okCount, nokCount, answeredCount, scorePercent };
   278→  }, [results, sections, totalItems]);
   279→
   280→  // Check that all NOK items have hallazgo, mejora and responsable filled
   281→  const nokItems = Object.values(results).filter(r => r.status === 'nok');
   282→  const allNokCompleted = nokItems.length === 0 || nokItems.every(r => (r.hallazgo || '').trim() !== '' && (r.mejora || '').trim() !== '' && (r.responsable || '').trim() !== '');
   283→
   284→  const passed = scoring.scorePercent >= notaMinima;
   285→  const canSubmit = canPerformAutoeval && scoring.answeredCount > 0 && allNokCompleted;
   286→
   287→  const handlePhotoSelect = async (e: React.ChangeEvent<HTMLInputElement>) => {
   288→    const files = e.target.files;
   289→    if (!files) return;
   290→    for (const file of Array.from(files)) {
   291→      const preview = URL.createObjectURL(file);
   292→      setAutoevalPhotos(prev => [...prev, { file, preview }]);
   293→    }
   294→    // Reset input
   295→    if (photoInputRef.current) photoInputRef.current.value = '';
   296→  };
   297→
   298→  const removeAutoevalPhoto = (index: number) => {
   299→    setAutoevalPhotos(prev => {
   300→      const photo = prev[index];
   301→      if (photo.preview) URL.revokeObjectURL(photo.preview);
   302→      return prev.filter((_, i) => i !== index);
   303→    });
   304→  };
   305→
   306→  // v2.63: Per-NOK photo handlers — photos attached to a specific finding
   307→  // (hallazgo) flow: paso 2 (capture) → paso 3 (action plan with photoRefs).
   308→  // v2.64: Al añadir una foto, se autocompletan:
   309→  //   1. Descripción del hallazgo con IA (VLM)
   310→  //   2. Punto de mejora sugerido con IA
   311→  //   3. Responsable: el responsable de la zona (si existe) o el currentUser
   312→  const [nokPhotoAnalyzing, setNokPhotoAnalyzing] = useState<Record<string, boolean>>({});
   313→  const handleNokPhotoSelect = async (itemId: string, e: React.ChangeEvent<HTMLInputElement>) => {
   314→    const files = e.target.files;
   315→    if (!files) return;
   316→    const newPhotos: { file: File; preview: string }[] = [];
   317→    for (const file of Array.from(files)) {
   318→      const preview = URL.createObjectURL(file);
   319→      newPhotos.push({ file, preview });
   320→    }
   321→    setNokPhotos(prev => ({
   322→      ...prev,
   323→      [itemId]: [...(prev[itemId] || []), ...newPhotos],
   324→    }));
   325→    if (nokPhotoInputRef.current[itemId]) nokPhotoInputRef.current[itemId]!.value = '';
   326→
   327→    // v2.64: Autocompletar campos del NOK con IA + responsable de zona
   328→    // Solo autocompletar si el hallazgo está vacío (no sobreescribir)
   329→    const currentResult = results[itemId];
   330→    const shouldAutoDescribe = !(currentResult?.hallazgo || '').trim();
   331→    if (shouldAutoDescribe && newPhotos.length > 0) {
   332→      setNokPhotoAnalyzing(prev => ({ ...prev, [itemId]: true }));
   333→      try {
   334→        // Convertir primera foto a base64 data URL
   335→        const file = newPhotos[0].file;
   336→        const dataUrl = await new Promise<string>((resolve, reject) => {
   337→          const reader = new FileReader();
   338→          reader.onload = () => resolve(reader.result as string);
   339→          reader.onerror = reject;
   340→          reader.readAsDataURL(file);
   341→        });
   342→
   343→        // Llamar a la API de VLM
   344→        const descRes = await fetch('/api/photo-describe', {
   345→          method: 'POST',
   346→          headers: { 'Content-Type': 'application/json' },
   347→          body: JSON.stringify({ imageUrl: dataUrl, sStep }),
   348→        });
   349→        const descJson = await descRes.json();
   350→        if (descJson.success && descJson.description) {
   351→          // Autocompletar hallazgo con la descripción IA
   352→          setItemField(itemId, 'hallazgo', descJson.description);
   353→          // Autocompletar mejora sugerida (plantilla basada en la descripción)
   354→          const mejoraSuggested = `Aplicar 5S en ${sStepData?.japaneseName || 'S' + sStep}: corregir desviación detectada — ${descJson.description.slice(0, 100)}`;
   355→          setItemField(itemId, 'mejora', mejoraSuggested);
   356→          toast.success(`Descripción generada por IA para ${itemId}`);
   357→        }
   358→      } catch (err) {
   359→        console.error('[autoeval] Error describiendo foto con IA:', err);
   360→        // No mostrar error al usuario — la foto se sigue guardando
   361→      } finally {
   362→        setNokPhotoAnalyzing(prev => ({ ...prev, [itemId]: false }));
   363→      }
   364→    }
   365→
   366→    // Autocompletar responsable con el responsable de la zona (si está vacío)
   367→    const shouldAutoResponsable = !(currentResult?.responsable || '').trim();
   368→    if (shouldAutoResponsable) {
   369→      let responsableName: string | null = null;
   370→      // 1. Si la zona tiene responsableId, buscar su nombre en projectMembers
   371→      if (currentZone?.responsableId) {
   372→        const zoneResp = projectMembers.find(m => m.userId === currentZone.responsableId);
   373→        if (zoneResp?.user?.name) responsableName = zoneResp.user.name;
   374→      }
   375→      // 2. Si no, buscar el primer miembro con rol 'responsable' en la zona actual
   376→      if (!responsableName) {
   377→        const respMember = projectMembers.find(m =>
   378→          m.role === 'responsable' &&
   379→          // Si el miembro tiene zonas asignadas, que la actual esté entre ellas
   380→          (m as any).zones?.some?.((z: any) => z.id === currentZone?.id) !== false
   381→        );
   382→        if (respMember?.user?.name) responsableName = respMember.user.name;
   383→      }
   384→      // 3. Si no hay responsable, usar el usuario actual (empleado que hace la autoeval)
   385→      if (!responsableName && currentUser?.name) {
   386→        responsableName = currentUser.name;
   387→      }
   388→      if (responsableName) {
   389→        setItemField(itemId, 'responsable', responsableName);
   390→      }
   391→    }
   392→  };
   393→
   394→  const removeNokPhoto = (itemId: string, index: number) => {
   395→    setNokPhotos(prev => {
   396→      const photos = prev[itemId] || [];
   397→      const photo = photos[index];
   398→      if (photo?.preview) URL.revokeObjectURL(photo.preview);
   399→      const updated = photos.filter((_, i) => i !== index);
   400→      const next = { ...prev };
   401→      if (updated.length === 0) delete next[itemId];
   402→      else next[itemId] = updated;
   403→      return next;
   404→    });
   405→  };
   406→
   407→  // v2.64: Auto-completar responsable para TODOS los NOKs a la vez
   408→  // (cuando se marca un item como NOK, autocompletar responsable si está vacío)
   409→  const autoFillResponsableForNok = (itemId: string) => {
   410→    const currentResult = results[itemId];
   411→    if ((currentResult?.responsable || '').trim()) return; // ya tiene
   412→    let responsableName: string | null = null;
   413→    if (currentZone?.responsableId) {
   414→      const zoneResp = projectMembers.find(m => m.userId === currentZone.responsableId);
   415→      if (zoneResp?.user?.name) responsableName = zoneResp.user.name;
   416→    }
   417→    if (!responsableName) {
   418→      const respMember = projectMembers.find(m => m.role === 'responsable');
   419→      if (respMember?.user?.name) responsableName = respMember.user.name;
   420→    }
   421→    if (!responsableName && currentUser?.name) {
   422→      responsableName = currentUser.name;
   423→    }
   424→    if (responsableName) {
   425→      setItemField(itemId, 'responsable', responsableName);
   426→    }
   427→  };
   428→
   429→  const toggleSection = (sectionId: string) => {
   430→    setExpandedSections(prev => ({ ...prev, [sectionId]: !prev[sectionId] }));
   431→  };
   432→
   433→  const setItemStatus = (itemId: string, status: 'ok' | 'nok' | 'na') => {
   434→    setResults(prev => ({
   435→      ...prev,
   436→      [itemId]: { ...prev[itemId], itemId, status },
   437→    }));
   438→    // v2.64: Si se marca como NOK, autocompletar el responsable automáticamente
   439→    if (status === 'nok') {
   440→      // Pequeño defer para que el state se actualice antes de leerlo
   441→      setTimeout(() => autoFillResponsableForNok(itemId), 0);
   442→    }
   443→  };
   444→
   445→  const setItemField = (itemId: string, field: 'hallazgo' | 'mejora' | 'otherText' | 'responsable', value: string) => {
   446→    setResults(prev => ({
   447→      ...prev,
   448→      [itemId]: { ...prev[itemId], itemId, [field]: value },
   449→    }));
   450→  };
   451→
   452→  const handleSubmit = async () => {
   453→    if (!canSubmit) return;
   454→    if (!canPerformAutoeval) return; // Only responsable/admin for S4, or any employee for S1/S2/S3/S5
   455→    setIsSubmitting(true);
   456→    try {
   457→      // Build effective results: merge missing items as 'ok' (default) before sending to backend
   458→      const effectiveResults = sections.flatMap(s => s.items).map(item => {
   459→        const r = results[item.id];
   460→        return r ?? { itemId: item.id, status: 'ok' as const };
   461→      });
   462→      // Only mark as completed if score meets notaMinima threshold
   463→      // If not passed, still save results but step stays available for retry
   464→      const res = await fetch(`/api/progress/step?sStep=${sStep}&miniStep=${miniStep}`, {
   465→        method: 'PUT',
   466→        headers: { 'Content-Type': 'application/json' },
   467→        body: JSON.stringify({
   468→          completed: passed,
   469→          score: scoring.scorePercent,
