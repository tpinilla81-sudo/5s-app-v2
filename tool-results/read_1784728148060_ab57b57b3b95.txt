   184→   184→// NORMALIZE HELPER
   185→   185→// ═══════════════════════════════════════════════════════
   186→   186→const normalize = (obj: PermMap): string => {
   187→   187→  const sorted: PermMap = {}
   188→   188→  Object.keys(obj).sort().forEach(role => {
   189→   189→    sorted[role] = {}
   190→   190→    Object.keys(obj[role]).sort().forEach(perm => {
   191→   191→      sorted[role][perm] = obj[role][perm]
   192→   192→    })
   193→   193→  })
   194→   194→  return JSON.stringify(sorted)
   195→   195→}
