    95→    95→
    96→    96→export default function ActionPlanModal({ open, onClose, sStep, miniStep }: ActionPlanModalProps) {
    97→    97→  const { fetchProgress, currentUser, adminFreeNavigation, currentProject, currentZone, canPerform, canView, hasPermission } = use5SStore();
    98→    98→  const sStepData = S_STEPS.find(s => s.id === sStep);
    99→    99→  const canSkipSteps = hasPermission('skip_steps');
   100→   100→  const canPerformStep = canPerform(sStep, miniStep);
   101→   101→  const canViewStep = canView(sStep, miniStep);
   102→   102→  // Permission-driven: read-only if no execute perm OR if candado closed for skip_steps users
   103→   103→  const isReadOnly = !canPerformStep || (canSkipSteps && !adminFreeNavigation);
   104→   104→
   105→   105→  const [actions, setActions] = useState<ActionItemData[]>([]);
   106→   106→  const [isCompleted, setIsCompleted] = useState(false);
   107→   107→  const [zones, setZones] = useState<ZoneData[]>([]);
   108→   108→  const [expandedRow, setExpandedRow] = useState<string | null>(null);
   109→   109→  const [isFullscreen, setIsFullscreen] = useState(true);
   110→   110→
   111→   111→  const loadZones = async () => {
   112→   112→    if (!currentProject) return;
   113→   113→    try {
   114→   114→      const res = await fetch(`/api/projects/${currentProject.id}/zones`);
   115→   115→      if (res.ok) {
   116→   116→        const json = await res.json();
   117→   117→        setZones(json.zones || json.data || []);
   118→   118→      }
   119→   119→    } catch (error) {
   120→   120→      console.error('Error loading zones:', error);
   121→   121→    }
   122→   122→  };
   123→   123→
   124→   124→  const loadActions = async () => {
   125→   125→    try {
   126→   126→      const params = new URLSearchParams();
   127→   127→      // When opened globally (miniStep=0), load ALL actions; otherwise filter by sStep
   128→   128→      const isGlobal = miniStep === 0;
   129→   129→      if (!isGlobal) {
   130→   130→        params.set('sStep', String(sStep));
   131→   131→      }
   132→   132→      if (currentProject?.id) params.set('projectId', currentProject.id);
   133→   133→      if (currentUser?.id) params.set('userId', currentUser.id);
   134→   134→      if (currentUser?.role) params.set('userRole', currentUser.role);
   135→   135→      const res = await fetch(`/api/actions?${params.toString()}`);
   136→   136→      const json = await res.json();
   137→   137→      if (json.success && json.data) {
   138→   138→        setActions(json.data.map((a: any) => ({
   139→   139→          id: a.id,
   140→   140→          numeroEntrada: a.numeroEntrada || 0,
   141→   141→          fechaEntrada: a.fechaEntrada ? new Date(a.fechaEntrada).toISOString().split('T')[0] : (a.createdAt ? new Date(a.createdAt).toISOString().split('T')[0] : ''),
   142→   142→          comunicadoPor: a.comunicadoPor || '',
   143→   143→          semana: a.semana || '',
   144→   144→          seccionDemandante: a.seccionDemandante || '',
   145→   145→          clienteZona: a.clienteZona || a.zone?.name || '',
   146→   146→          personaDemandada: a.personaDemandada || '',
   147→   147→          seccionDemandada: a.seccionDemandada || '',
   148→   148→          hallazgo: a.hallazgo || a.itemDescription || '',
   149→   149→          impactoObjetivo: a.impactoObjetivo || '',
   150→   150→          enviado: a.enviado || '',
   151→   151→          accionCorrectiva: a.accionCorrectiva || a.mejora || '',
   152→   152→          accionesPreventivas: a.accionesPreventivas || '',
   153→   153→          semanaPrevista: a.semanaPrevista || '',
   154→   154→          responsable: a.responsable || '',
   155→   155→          porcentaje: a.porcentaje || 0,
   156→   156→          estado: a.estado === 'abierta' ? 'abierta' : a.estado === 'en_proceso' ? 'en_proceso' : a.estado === 'resuelta' || a.estado === 'cerrada' ? 'resuelta' : 'abierta',
   157→   157→          semanaReal: a.semanaReal || '',
   158→   158→          // Legacy
   159→   159→          descripcion: a.itemDescription || a.hallazgo || '',
   160→   160→          fechaCompromiso: a.fechaCompromiso ? new Date(a.fechaCompromiso).toISOString().split('T')[0] : '',
   161→   161→          fechaLimite: a.fechaLimite ? new Date(a.fechaLimite).toISOString().split('T')[0] : '',
   162→   162→          fechaReal: a.fechaReal ? new Date(a.fechaReal).toISOString().split('T')[0] : '',
   163→   163→          prioridad: a.prioridad || 'media',
   164→   164→          zoneId: a.zoneId || '',
   165→   165→          zoneName: a.zone?.name || '',
   166→   166→          verificadoPor: a.verificadoPor || '',
   167→   167→          sStep: a.sStep || sStep,
   168→   168→          miniStep: a.miniStep || 3,
   169→   169→        })));
   170→   170→      }
   171→   171→    } catch (error) {
   172→   172→      console.error('Error loading actions:', error);
   173→   173→      toast.error('Error al cargar las acciones');
   174→   174→    }
   175→   175→  };
   176→   176→
   177→   177→  useEffect(() => {
   178→   178→    if (open) {
   179→   179→      loadActions();
   180→   180→      loadZones();
   181→   181→    }
   182→   182→  }, [open, sStep]);
   183→   183→
   184→   184→  const getNextNumero = () => {
   185→   185→    if (actions.length === 0) return 1;
   186→   186→    return Math.max(...actions.map(a => a.numeroEntrada || 0)) + 1;
   187→   187→  };
   188→   188→
   189→   189→  const getCurrentWeek = () => {
   190→   190→    const now = new Date();
   191→   191→    const start = new Date(now.getFullYear(), 0, 1);
   192→   192→    const diff = now.getTime() - start.getTime();
   193→   193→    const oneWeek = 1000 * 60 * 60 * 24 * 7;
   194→   194→    return `W${Math.ceil((diff / oneWeek) + start.getDay() / 7)}`;
   195→   195→  };
   196→   196→
   197→   197→  const handleAddAction = async () => {
   198→   198→    if (!currentProject?.id) {
   199→   199→      toast.error('No hay proyecto seleccionado.');
   200→   200→      return;
   201→   201→    }
   202→   202→
   203→   203→    const nextNumero = getNextNumero();
   204→   204→    const today = new Date().toISOString().split('T')[0];
   205→   205→    const currentWeek = getCurrentWeek();
   206→   206→
   207→   207→    try {
   208→   208→      const res = await fetch('/api/actions', {
   209→   209→        method: 'POST',
   210→   210→        headers: { 'Content-Type': 'application/json' },
   211→   211→        body: JSON.stringify({
   212→   212→          sStep,
   213→   213→          miniStep: 3,
   214→   214→          itemId: `PA-${sStep}-${Date.now()}`,
   215→   215→          itemDescription: '',
   216→   216→          hallazgo: '',
   217→   217→          source: 'actionplan',
   218→   218→          projectId: currentProject.id,
   219→   219→          zoneId: currentZone?.id || null,
   220→   220→          // New Plan de Acción fields
   221→   221→          numeroEntrada: nextNumero,
   222→   222→          fechaEntrada: today,
   223→   223→          semana: currentWeek,
   224→   224→          estado: 'abierta',
   225→   225→          enviado: 'Pendiente',
   226→   226→          porcentaje: 0,
   227→   227→        }),
   228→   228→      });
   229→   229→
   230→   230→      const json = await res.json();
   231→   231→      if (json.success) {
   232→   232→        toast.success('Entrada agregada correctamente');
   233→   233→        await loadActions();
   234→   234→      } else {
   235→   235→        toast.error(`Error: ${json.error || 'Error desconocido'}`);
   236→   236→      }
   237→   237→    } catch (error) {
   238→   238→      console.error('Error adding action:', error);
   239→   239→      toast.error('Error de conexión al agregar la entrada');
   240→   240→    }
   241→   241→  };
   242→   242→
   243→   243→  const handleUpdateField = async (actionId: string, field: string, value: any) => {
   244→   244→    // Optimistic update
   245→   245→    setActions(prev => prev.map(a =>
   246→   246→      a.id === actionId ? { ...a, [field]: value } : a
   247→   247→    ));
   248→   248→
   249→   249→    try {
   250→   250→      const res = await fetch(`/api/actions?id=${actionId}`, {
   251→   251→        method: 'PUT',
   252→   252→        headers: { 'Content-Type': 'application/json' },
   253→   253→        body: JSON.stringify({ [field]: value }),
   254→   254→      });
   255→   255→      const json = await res.json();
   256→   256→      if (!json.success) {
   257→   257→        toast.error(`Error al actualizar: ${json.error || 'Error desconocido'}`);
   258→   258→        await loadActions(); // Revert
   259→   259→      }
   260→   260→    } catch (error) {
   261→   261→      console.error('Error updating action:', error);
   262→   262→      await loadActions(); // Revert
   263→   263→    }
   264→   264→  };
   265→   265→
   266→   266→  const handleDeleteAction = async (id: string) => {
   267→   267→    if (!confirm('¿Eliminar esta entrada?')) return;
   268→   268→    try {
   269→   269→      const res = await fetch(`/api/actions?id=${id}`, { method: 'DELETE' });
   270→   270→      const json = await res.json();
   271→   271→      if (json.success) {
   272→   272→        setActions(prev => prev.filter(a => a.id !== id));
   273→   273→        toast.success('Entrada eliminada');
   274→   274→      } else {
   275→   275→        toast.error(`Error al eliminar: ${json.error || 'Error desconocido'}`);
   276→   276→      }
   277→   277→    } catch (error) {
   278→   278→      console.error('Error deleting action:', error);
   279→   279→      toast.error('Error de conexión al eliminar');
   280→   280→    }
   281→   281→  };
   282→   282→
   283→   283→  // Count completed actions
   284→   284→  const completedCount = actions.filter(a => a.estado === 'resuelta' || a.estado === 'cerrada').length;
   285→   285→  const totalActions = actions.length;
   286→   286→  const canComplete = totalActions >= ACTION_PLAN_MIN_ITEMS;
   287→   287→  const progressPercent = totalActions > 0 ? Math.round((completedCount / totalActions) * 100) : 0;
   288→   288→
   289→   289→  const handleComplete = async () => {
   290→   290→    if (!canComplete) return;
   291→   291→    try {
   292→   292→      const res = await fetch(`/api/progress/step?sStep=${sStep}&miniStep=${miniStep}`, {
   293→   293→        method: 'PUT',
   294→   294→        headers: { 'Content-Type': 'application/json' },
   295→   295→        body: JSON.stringify({
   296→   296→          completed: true,
   297→   297→          score: progressPercent,
   298→   298→          projectId: currentProject?.id,
   299→   299→          zoneId: currentZone?.id || null,
   300→   300→        }),
   301→   301→      });
   302→   302→      const json = await res.json();
   303→   303→      if (json.success) {
   304→   304→        setIsCompleted(true);
   305→   305→        await fetchProgress();
   306→   306→      }
   307→   307→    } catch (error) {
   308→   308→      console.error('Error completing action plan:', error);
   309→   309→    }
   310→   310→  };
   311→   311→
   312→   312→  const handleAdminSkip = async () => {
   313→   313→    try {
   314→   314→      const res = await fetch(`/api/progress/step?sStep=${sStep}&miniStep=${miniStep}`, {
   315→   315→        method: 'PUT',
   316→   316→        headers: { 'Content-Type': 'application/json' },
   317→   317→        body: JSON.stringify({ completed: true, score: 100, notes: 'Completado por administrador (skip)', skipMissingTemplate: true, projectId: currentProject?.id, zoneId: currentZone?.id || null }),
   318→   318→      });
   319→   319→      const json = await res.json();
   320→   320→      if (json.success) {
   321→   321→        await fetchProgress();
   322→   322→        onClose();
   323→   323→      }
   324→   324→    } catch (error) {
   325→   325→      console.error('Error admin skip:', error);
   326→   326→    }
   327→   327→  };
   328→   328→
   329→   329→  const getEstadoBadge = (estado: string) => {
   330→   330→    const opt = ESTADO_OPTIONS.find(e => e.value === estado);
   331→   331→    return opt || ESTADO_OPTIONS[0];
   332→   332→  };
   333→   333→
   334→   334→  return (
   335→   335→    <Dialog open={open} onOpenChange={() => onClose()}>
   336→   336→      <DialogContent size={isFullscreen ? "fullscreen" : "xl"} className="flex flex-col overflow-hidden p-0">
   337→   337→        <DialogHeader className="px-6 pt-6 pb-2 flex-shrink-0">
   338→   338→          <DialogTitle className="flex items-center gap-2">
   339→   339→            <ListChecks className="h-5 w-5" style={{ color: sStepData?.color }} />
   340→   340→            <span>Plan de Acción{miniStep === 0 ? ' — Global' : ` — ${sStepData?.name}`}</span>
   341→   341→            {sStepData && miniStep !== 0 && (
   342→   342→              <Badge variant="outline" style={{ borderColor: sStepData?.color, color: sStepData?.color }}>
   343→   343→                {sStepData?.japaneseName}
   344→   344→              </Badge>
   345→   345→            )}
   346→   346→            <button
   347→   347→              onClick={() => setIsFullscreen(!isFullscreen)}
   348→   348→              className="ml-auto p-1 rounded hover:bg-muted transition-colors"
   349→   349→              title={isFullscreen ? "Reducir ventana" : "Pantalla completa"}
   350→   350→            >
   351→   351→              {isFullscreen ? <Minimize2 className="h-4 w-4 text-muted-foreground" /> : <Maximize2 className="h-4 w-4 text-muted-foreground" />}
   352→   352→            </button>
   353→   353→          </DialogTitle>
   354→   354→        </DialogHeader>
   355→   355→
   356→   356→        {canSkipSteps && !isCompleted && (
   357→   357→          <div className="mx-6 flex items-center gap-2 p-2 bg-amber-50 border border-amber-200 rounded-lg flex-shrink-0">
   358→   358→            <span className="text-xs text-amber-700 font-medium">Modo Admin:</span>
   359→   359→            <Button
   360→   360→              variant="outline"
   361→   361→              size="sm"
   362→   362→              className="text-xs border-amber-300 text-amber-700 hover:bg-amber-100"
   363→   363→              onClick={handleAdminSkip}
   364→   364→            >
   365→   365→              Completar paso sin plan
   366→   366→            </Button>
   367→   367→          </div>
   368→   368→        )}
   369→   369→
   370→   370→        {isReadOnly && (
   371→   371→          <div className="flex items-center gap-2 p-2 mx-6 flex-shrink-0 bg-blue-50 border border-blue-200 rounded-lg">
   372→   372→            <span className="text-xs text-blue-700 font-medium">Solo lectura: {canSkipSteps ? 'Activa el candado para poder realizar pasos.' : 'Puedes ver pero no modificar.'}</span>
   373→   373→          </div>
   374→   374→        )}
   375→   375→
   376→   376→        <div className="flex-1 overflow-y-auto px-6 pb-6 min-h-0">
   377→   377→        {isCompleted ? (
   378→   378→          <div className="text-center py-8">
   379→   379→            <div className="text-4xl mb-3">✅</div>
   380→   380→            <h3 className="text-xl font-bold mb-2">¡Plan de Acción Completado!</h3>
   381→   381→            <p className="text-muted-foreground">
   382→   382→              Se han definido {totalActions} acciones, {completedCount} completadas ({progressPercent}%).
   383→   383→            </p>
   384→   384→          </div>
   385→   385→        ) : (
   386→   386→          <div className="flex-1 flex flex-col min-h-0 space-y-3">
   387→   387→            {/* Stats bar */}
   388→   388→            <div className="flex items-center justify-between p-2 bg-muted rounded-lg text-sm">
   389→   389→              <div className="flex items-center gap-3">
   390→   390→                <span className="font-medium">{totalActions} entradas</span>
   391→   391→                <span className="text-muted-foreground">({completedCount} resueltas — {progressPercent}%)</span>
   392→   392→              </div>
   393→   393→              <div className="flex items-center gap-2">
   394→   394→                <Button
   395→   395→                  onClick={handleAddAction}
   396→   396→                  size="sm"
   397→   397→                  style={{ backgroundColor: sStepData?.color }}
   398→   398→                  className="text-white"
   399→   399→                >
   400→   400→                  <Plus className="h-4 w-4 mr-1" /> Nueva Entrada
   401→   401→                </Button>
   402→   402→              </div>
   403→   403→            </div>
   404→   404→
   405→   405→            {/* Scrollable table */}
   406→   406→            <div className="flex-1 overflow-auto border rounded-lg">
   407→   407→              <table className="w-full text-xs border-collapse min-w-[800px] md:min-w-[1200px]">
   408→   408→                <thead className="sticky top-0 z-10">
   409→   409→                  <tr>
   410→   410→                    {/* Yellow section: Demanda */}
   411→   411→                    <th colSpan={9} className={`${HEADER_COLORS.demandante} px-2 py-1.5 text-center text-xs font-bold border border-amber-500`}>
   412→   412→                      DEMANDA
   413→   413→                    </th>
   414→   414→                    {/* Blue section: Acción */}
   415→   415→                    <th colSpan={4} className={`${HEADER_COLORS.accion} px-2 py-1.5 text-center text-xs font-bold border border-sky-500`}>
   416→   416→                      ACCIÓN
   417→   417→                    </th>
   418→   418→                    {/* Orange section: Seguimiento */}
   419→   419→                    <th colSpan={5} className={`${HEADER_COLORS.seguimiento} px-2 py-1.5 text-center text-xs font-bold border border-orange-500`}>
   420→   420→                      SEGUIMIENTO
   421→   421→                    </th>
   422→   422→                    <th className="bg-gray-400 text-white px-1 py-1.5 text-center text-xs font-bold border border-gray-500 w-8">
   423→   423→                      🗑
   424→   424→                    </th>
   425→   425→                  </tr>
   426→   426→                  <tr>
