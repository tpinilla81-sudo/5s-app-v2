     1→'use client'
     2→
     3→import { useState, useEffect, useCallback, useRef, useLayoutEffect } from 'react'
     4→import { Card, CardContent } from '@/components/ui/card'
     5→import { Badge } from '@/components/ui/badge'
     6→import { Button } from '@/components/ui/button'
     7→import { Input } from '@/components/ui/input'
     8→import { Label } from '@/components/ui/label'
     9→import { Checkbox } from '@/components/ui/checkbox'
    10→import {
    11→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    12→} from '@/components/ui/select'
    13→import {
    14→  Dialog, DialogContent, DialogHeader, DialogTitle,
    15→} from '@/components/ui/dialog'
    16→import {
    17→  Loader2, X, FileText, Award, Settings2, Plus, Trash2, Edit3, Save,
    18→  ChevronDown, ChevronUp, Star, Copy, LayoutGrid,
    19→} from 'lucide-react'
    20→import { S_STEPS } from '@/lib/5s-constants'
    21→
    22→// ═══════════════════════════════════════════════════════
    23→// TYPES
    24→// ═══════════════════════════════════════════════════════
    25→
    26→interface TemplateOption {
    27→  id: string
    28→  type: string
    29→  title: string
    30→  sStep: number
    31→  miniStep: number
    32→}
    33→
    34→interface StandardOption {
    35→  id: string
    36→  title: string
    37→  sStep: number
    38→  category: string
    39→}
    40→
    41→interface SlotTemplate {
    42→  id: string
    43→  slotId: string
    44→  templateId: string
    45→  sortOrder: number
    46→  template: TemplateOption
    47→}
    48→
    49→interface SlotStandard {
    50→  id: string
    51→  slotId: string
    52→  standardId: string
    53→  sortOrder: number
    54→  standard: StandardOption
    55→}
    56→
    57→interface BoardSlotData {
    58→  id: string
    59→  boardConfigId: string
    60→  sStep: number
    61→  miniStep: number
    62→  templates: SlotTemplate[]
    63→  standards: SlotStandard[]
    64→}
    65→
    66→interface BoardConfigData {
    67→  id: string
    68→  name: string
    69→  description: string | null
    70→  isDefault: boolean
    71→  _count?: { slots: number; zones: number }
    72→}
    73→

    75→// CONSTANTS
    76→// ═══════════════════════════════════════════════════════
    77→
    78→const S_COLORS: Record<number, string> = {
    79→  1: '#8B5CF6', 2: '#EAB308', 3: '#3B82F6', 4: '#F43F5E', 5: '#F97316',
    80→}
    81→
    82→const S_BG_COLORS: Record<number, string> = {
    83→  1: 'bg-violet-50 border-violet-200',
    84→  2: 'bg-yellow-50 border-yellow-200',
    85→  3: 'bg-blue-50 border-blue-200',
    86→  4: 'bg-rose-50 border-rose-200',
    87→  5: 'bg-orange-50 border-orange-200',
    88→}
    89→
    90→const S_BORDER_COLORS: Record<number, string> = {
    91→  1: 'border-l-violet-500',
    92→  2: 'border-l-yellow-500',
    93→  3: 'border-l-blue-500',
    94→  4: 'border-l-rose-500',
    95→  5: 'border-l-orange-500',
    96→}
    97→
    98→const PASO_LABELS: Record<number, { label: string; types: string[] }> = {
    99→  1: { label: 'Formación y Exámenes', types: ['formacion', 'examen'] },
   100→  2: { label: 'Fotografías', types: ['fotos'] },
   101→  3: { label: 'Inventario / Estándar', types: ['inventario', 'estandar'] },
   102→  4: { label: 'Autoevaluación', types: ['autoevaluacion'] },
   103→  5: { label: 'Auditoría', types: ['auditoria'] },
   104→}
   105→
   106→const TYPE_LABELS: Record<string, string> = {
   107→  formacion: 'Formación', examen: 'Examen', fotos: 'Fotos',
   108→  inventario: 'Inventario', estandar: 'Estándar',
   109→  autoevaluacion: 'Autoevaluación', auditoria: 'Auditoría',
   110→}
   111→
   112→const TYPE_COLORS: Record<string, string> = {
   113→  formacion: 'bg-blue-100 text-blue-700', examen: 'bg-amber-100 text-amber-700',
   114→  fotos: 'bg-pink-100 text-pink-700', inventario: 'bg-green-100 text-green-700',
   115→  estandar: 'bg-purple-100 text-purple-700', autoevaluacion: 'bg-cyan-100 text-cyan-700',
   116→  auditoria: 'bg-red-100 text-red-700',
   117→}
   118→
   119→const REFERENCE_WIDTH = 1100
   120→
   121→// ═══════════════════════════════════════════════════════
   122→// COMPONENT
   123→// ═══════════════════════════════════════════════════════
   124→
   125→export default function Tablero5S() {
   126→  const [configs, setConfigs] = useState<BoardConfigData[]>([])
   127→  const [selectedConfigId, setSelectedConfigId] = useState<string | null>(null)
   128→  const [slots, setSlots] = useState<BoardSlotData[]>([])
   129→  const [templates, setTemplates] = useState<TemplateOption[]>([])
   130→  const [standards, setStandards] = useState<StandardOption[]>([])
   131→  const [isLoading, setIsLoading] = useState(false)
   132→  const [savingSlot, setSavingSlot] = useState<string | null>(null)
   133→  const [expandedCell, setExpandedCell] = useState<string | null>(null)
   134→
   135→  // ─── Responsive scaling ───────────────────────────────────────────────
   136→  const boardContainerRef = useRef<HTMLDivElement>(null)
   137→  const [zoom, setZoom] = useState(1)
   138→
   139→  const boardVisible = !!(selectedConfigId && configs.length > 0 && !isLoading)
   140→
   141→  useLayoutEffect(() => {
   142→    const container = boardContainerRef.current
   143→    if (!container) return
   144→
   145→    const update = () => {
   146→      const containerWidth = container.clientWidth
   147→      if (containerWidth === 0) return
   148→      const newZoom = containerWidth / REFERENCE_WIDTH
   149→      setZoom(newZoom)
   150→    }
   151→
   152→    update()
   153→
   154→    const ro = new ResizeObserver(update)
   155→    ro.observe(container)
   156→
   157→    return () => ro.disconnect()
   158→  }, [boardVisible])
   159→
   160→  // Config edit dialog
   161→  const [showConfigDialog, setShowConfigDialog] = useState(false)
   162→  const [editingConfig, setEditingConfig] = useState<BoardConfigData | null>(null)
   163→  const [configName, setConfigName] = useState('')
   164→  const [configDescription, setConfigDescription] = useState('')
   165→  const [configIsDefault, setConfigIsDefault] = useState(false)
   166→  const [isSavingConfig, setIsSavingConfig] = useState(false)
   167→
   168→  // ─── Data loading ────────────────────────────────────────────────────────
   169→  const loadConfigs = useCallback(async () => {
   170→    try {
   171→      const res = await fetch('/api/board-configs')
   172→      const data = await res.json()
   173→      if (data.success) {
   174→        const configsData = data.data || []
   175→        setConfigs(configsData)
   176→        // Auto-select default or first
   177→        if (!selectedConfigId && configsData.length > 0) {
   178→          const defaultConfig = configsData.find((c: BoardConfigData) => c.isDefault)
   179→          setSelectedConfigId((defaultConfig || configsData[0]).id)
   180→        }
   181→      }
   182→    } catch (error) {
   183→      console.error('Error loading board configs:', error)
   184→    }
   185→  }, [selectedConfigId])
   186→
   187→  const loadSlots = useCallback(async () => {
   188→    if (!selectedConfigId) return
   189→    setIsLoading(true)
   190→    try {
   191→      const res = await fetch(`/api/board-slots?boardConfigId=${selectedConfigId}`)
   192→      const data = await res.json()
   193→      if (data.success) {
   194→        setSlots(data.data || [])
   195→      }
   196→    } catch (error) {
   197→      console.error('Error loading board slots:', error)
   198→    } finally {
   199→      setIsLoading(false)
   200→    }
   201→  }, [selectedConfigId])
   202→
   203→  const loadTemplates = useCallback(async () => {
   204→    try {
   205→      const res = await fetch('/api/templates?includeInactive=false')
   206→      const data = await res.json()
   207→      if (data.success) setTemplates(data.data || [])
   208→    } catch (error) {
   209→      console.error('Error loading templates:', error)
   210→    }
   211→  }, [])
   212→
   213→  const loadStandards = useCallback(async () => {
   214→    try {
   215→      const res = await fetch('/api/standards')
   216→      const data = await res.json()
   217→      const list = data.standards || (Array.isArray(data) ? data : [])
   218→      setStandards(list.map((s: any) => ({
   219→        id: s.id, title: s.title, sStep: s.sStep, category: s.category,
   220→      })))
   221→    } catch (error) {
   222→      console.error('Error loading standards:', error)
   223→    }
   224→  }, [])
   225→
   226→  useEffect(() => { loadConfigs() }, [loadConfigs])
   227→  useEffect(() => { if (selectedConfigId) loadSlots() }, [loadSlots])
   228→  useEffect(() => { loadTemplates(); loadStandards() }, [loadTemplates, loadStandards])
   229→
   230→  // ─── Slot helpers ────────────────────────────────────────────────────────
   231→  const getSlot = (sStep: number, miniStep: number): BoardSlotData | undefined =>
   232→    slots.find(s => s.sStep === sStep && s.miniStep === miniStep)
   233→
   234→  const getTemplatesForPaso = (sStep: number, miniStep: number): TemplateOption[] => {
   235→    const pasoConfig = PASO_LABELS[miniStep]
   236→    if (!pasoConfig) return []
   237→    return templates.filter(t => pasoConfig.types.includes(t.type) && t.sStep === sStep)
   238→  }
   239→
   240→  const getStandardsForS = (sStep: number): StandardOption[] =>
   241→    standards.filter(s => s.sStep === sStep)
   242→
   243→  // ─── Save slot ──────────────────────────────────────────────────────────
   244→  const handleSaveSlot = async (sStep: number, miniStep: number, templateIds: string[], standardIds: string[]) => {
   245→    if (!selectedConfigId) return
   246→    const key = `${sStep}-${miniStep}`
   247→    setSavingSlot(key)
   248→    try {
   249→      const res = await fetch('/api/board-slots', {
   250→        method: 'POST',
   251→        headers: { 'Content-Type': 'application/json' },
   252→        body: JSON.stringify({ boardConfigId: selectedConfigId, sStep, miniStep, templateIds, standardIds }),
   253→      })
   254→      if (res.ok) {
   255→        const data = await res.json()
   256→        if (data.success) {
   257→          setSlots(prev => {
   258→            const idx = prev.findIndex(s => s.sStep === sStep && s.miniStep === miniStep)
   259→            if (idx >= 0) { const u = [...prev]; u[idx] = data.data; return u }
   260→            return [...prev, data.data]
   261→          })
   262→        }
   263→      }
   264→    } catch (error) { console.error('Error saving board slot:', error) }
   265→    finally { setSavingSlot(null) }
   266→  }
   267→
   268→  const handleToggleTemplate = async (sStep: number, miniStep: number, templateId: string) => {
   269→    const slot = getSlot(sStep, miniStep)
   270→    const currentIds = slot?.templates.map(t => t.templateId) || []
   271→    const newIds = currentIds.includes(templateId) ? currentIds.filter(id => id !== templateId) : [...currentIds, templateId]
   272→    await handleSaveSlot(sStep, miniStep, newIds, slot?.standards.map(s => s.standardId) || [])
   273→  }
   274→
   275→  const handleToggleStandard = async (sStep: number, miniStep: number, standardId: string) => {
   276→    const slot = getSlot(sStep, miniStep)
   277→    const templateIds = slot?.templates.map(t => t.templateId) || []
   278→    const currentIds = slot?.standards.map(s => s.standardId) || []
   279→    const newIds = currentIds.includes(standardId) ? currentIds.filter(id => id !== standardId) : [...currentIds, standardId]
   280→    await handleSaveSlot(sStep, miniStep, templateIds, newIds)
   281→  }
   282→
   283→  const handleClearSlot = async (sStep: number, miniStep: number) => {
   284→    await handleSaveSlot(sStep, miniStep, [], [])
   285→    setExpandedCell(null)
   286→  }
   287→
   288→  // ─── Config CRUD ─────────────────────────────────────────────────────────
   289→  const openNewConfig = () => {
   290→    setEditingConfig(null)
   291→    setConfigName('')
   292→    setConfigDescription('')
   293→    setConfigIsDefault(configs.length === 0)
   294→    setShowConfigDialog(true)
   295→  }
   296→
   297→  const openEditConfig = (config: BoardConfigData) => {
   298→    setEditingConfig(config)
   299→    setConfigName(config.name)
   300→    setConfigDescription(config.description || '')
   301→    setConfigIsDefault(config.isDefault)
   302→    setShowConfigDialog(true)
   303→  }
   304→
   305→  const handleSaveConfig = async () => {
   306→    if (!configName) return
   307→    setIsSavingConfig(true)
   308→    try {
   309→      if (editingConfig) {
   310→        await fetch('/api/board-configs', {
   311→          method: 'PUT',
   312→          headers: { 'Content-Type': 'application/json' },
   313→          body: JSON.stringify({ id: editingConfig.id, name: configName, description: configDescription, isDefault: configIsDefault }),
   314→        })
   315→      } else {
   316→        await fetch('/api/board-configs', {
   317→          method: 'POST',
   318→          headers: { 'Content-Type': 'application/json' },
   319→          body: JSON.stringify({ name: configName, description: configDescription, isDefault: configIsDefault }),
   320→        })
   321→      }
   322→      setShowConfigDialog(false)
   323→      await loadConfigs()
   324→    } catch (error) { console.error('Error saving config:', error) }
   325→    finally { setIsSavingConfig(false) }
   326→  }
   327→
   328→  const handleDeleteConfig = async (id: string) => {
   329→    if (!confirm('¿Eliminar este tablero? Las zonas que lo usen quedarán sin tablero asignado.')) return
   330→    try {
   331→      await fetch(`/api/board-configs?id=${id}`, { method: 'DELETE' })
   332→      if (selectedConfigId === id) setSelectedConfigId(null)
   333→      await loadConfigs()
   334→    } catch (error) { console.error('Error deleting config:', error) }
   335→  }
   336→
   337→  const handleDuplicateConfig = async (config: BoardConfigData) => {
   338→    const newName = `${config.name} (copia)`
   339→    try {
   340→      const res = await fetch('/api/board-configs', {
   341→        method: 'POST',
   342→        headers: { 'Content-Type': 'application/json' },
   343→        body: JSON.stringify({ name: newName, description: config.description }),
   344→      })
   345→      const data = await res.json()
   346→      if (data.success && data.data) {
   347→        // Now copy all slots from the original config
   348→        const sourceSlotsRes = await fetch(`/api/board-slots?boardConfigId=${config.id}`)
   349→        const sourceSlotsData = await sourceSlotsRes.json()
   350→        if (sourceSlotsData.success && sourceSlotsData.data) {
   351→          for (const slot of sourceSlotsData.data) {
   352→            const templateIds = slot.templates.map((t: SlotTemplate) => t.templateId)
   353→            const standardIds = slot.standards.map((s: SlotStandard) => s.standardId)
   354→            await fetch('/api/board-slots', {
   355→              method: 'POST',
   356→              headers: { 'Content-Type': 'application/json' },
   357→              body: JSON.stringify({
   358→                boardConfigId: data.data.id,
   359→                sStep: slot.sStep,
   360→                miniStep: slot.miniStep,
   361→                templateIds,
   362→                standardIds,
   363→              }),
   364→            })
   365→          }
   366→        }
   367→        await loadConfigs()
   368→        setSelectedConfigId(data.data.id)
   369→      }
   370→    } catch (error) { console.error('Error duplicating config:', error) }
   371→  }
   372→
   373→  const handleSetDefault = async (config: BoardConfigData) => {
   374→    try {
   375→      await fetch('/api/board-configs', {
   376→        method: 'PUT',
   377→        headers: { 'Content-Type': 'application/json' },
   378→        body: JSON.stringify({ id: config.id, isDefault: true }),
   379→      })
   380→      await loadConfigs()
   381→    } catch (error) { console.error('Error setting default:', error) }
   382→  }
   383→
   384→  // ─── Render ──────────────────────────────────────────────────────────
   385→  const selectedConfig = configs.find(c => c.id === selectedConfigId)
   386→
   387→  return (
   388→    <div className="space-y-6">
   389→      {/* Header */}
   390→      <div className="flex items-center gap-3">
   391→        <div className="w-10 h-10 rounded-lg bg-gradient-to-br from-purple-500 to-indigo-600 flex items-center justify-center">
   392→          <Settings2 className="h-5 w-5 text-white" />
   393→        </div>
   394→        <div>
   395→          <h2 className="text-lg font-bold text-gray-900">Configuración de Tableros</h2>
   396→          <p className="text-sm text-muted-foreground">
   397→            Crea tableros con plantillas y estándares. Luego asígnalos a las zonas de cada proyecto.
   398→          </p>
   399→        </div>
   400→      </div>
   401→
   402→      {/* Board configs selector + CRUD */}
   403→      <div className="border rounded-xl p-4 bg-gradient-to-r from-gray-50 to-white space-y-3">
   404→        <div className="flex items-center gap-3">
   405→          <Label className="text-sm font-semibold whitespace-nowrap">Tablero:</Label>
   406→          <Select value={selectedConfigId || ''} onValueChange={setSelectedConfigId}>
   407→            <SelectTrigger className="h-9 flex-1 max-w-md">
   408→              <SelectValue placeholder="Selecciona un tablero" />
   409→            </SelectTrigger>
   410→            <SelectContent>
   411→              {configs.map(c => (
   412→                <SelectItem key={c.id} value={c.id}>
   413→                  <div className="flex items-center gap-2">
   414→                    {c.isDefault && <Star className="h-3 w-3 text-amber-500 fill-amber-500" />}
   415→                    <span>{c.name}</span>
   416→                    {c._count && (
   417→                      <span className="text-xs text-gray-400">
   418→                        ({c._count.zones} zona{c._count.zones !== 1 ? 's' : ''})
   419→                      </span>
   420→                    )}
   421→                  </div>
   422→                </SelectItem>
   423→              ))}
   424→            </SelectContent>
   425→          </Select>
   426→          <Button variant="outline" size="sm" onClick={openNewConfig} className="gap-1">
   427→            <Plus className="h-4 w-4" /> Nuevo
   428→          </Button>
   429→          {selectedConfig && (
   430→            <>
   431→              <Button variant="outline" size="sm" onClick={() => openEditConfig(selectedConfig)} className="gap-1">
   432→                <Edit3 className="h-3.5 w-3.5" />
   433→              </Button>
   434→              <Button variant="outline" size="sm" onClick={() => handleDuplicateConfig(selectedConfig)} className="gap-1">
   435→                <Copy className="h-3.5 w-3.5" />
   436→              </Button>
   437→              {!selectedConfig.isDefault && (
   438→                <Button variant="outline" size="sm" onClick={() => handleSetDefault(selectedConfig)}
   439→                  className="gap-1 text-amber-600 hover:text-amber-700" title="Establecer como predeterminado">
   440→                  <Star className="h-3.5 w-3.5" />
   441→                </Button>
   442→              )}
   443→              {configs.length > 1 && (
   444→                <Button variant="outline" size="sm" onClick={() => handleDeleteConfig(selectedConfig.id)}
   445→                  className="gap-1 text-red-500 hover:text-red-600">
   446→                  <Trash2 className="h-3.5 w-3.5" />
   447→                </Button>
   448→              )}
   449→            </>
   450→          )}
   451→        </div>
   452→        {selectedConfig && (
   453→          <div className="flex items-center gap-2 text-xs text-muted-foreground">
   454→            <LayoutGrid className="h-3 w-3" />
   455→            <span>Tablero: <strong>{selectedConfig.name}</strong></span>
   456→            {selectedConfig.isDefault && (
   457→              <Badge className="text-[9px] bg-amber-100 text-amber-700 border-0 px-1.5 py-0">
   458→                <Star className="h-2.5 w-2.5 mr-0.5 fill-amber-500 text-amber-500" />Predeterminado
   459→              </Badge>
   460→            )}
   461→            {selectedConfig.description && <span>· {selectedConfig.description}</span>}
   462→          </div>
   463→        )}
   464→      </div>
   465→
   466→      {/* No configs message */}
   467→      {configs.length === 0 && (
   468→        <div className="text-center py-16 space-y-3">
   469→          <LayoutGrid className="h-12 w-12 text-gray-300 mx-auto" />
   470→          <p className="text-muted-foreground">No hay tableros configurados.</p>
   471→          <Button onClick={openNewConfig} className="gap-2">
   472→            <Plus className="h-4 w-4" /> Crear el primer tablero
   473→          </Button>
   474→        </div>
   475→      )}
   476→
   477→      {/* Board grid */}
   478→      {selectedConfigId && configs.length > 0 && (
   479→        isLoading ? (
   480→          <div className="flex items-center justify-center py-12">
   481→            <Loader2 className="h-8 w-8 text-purple-500 animate-spin" />
   482→          </div>
   483→        ) : (
   484→          <div
   485→            ref={boardContainerRef}
   486→            className="w-full overflow-hidden"
   487→          >
   488→            <div
   489→              style={{
   490→                zoom: zoom,
   491→                width: REFERENCE_WIDTH,
   492→                transformOrigin: 'top left',
   493→              }}
   494→            >
   495→              {/* Column headers */}
   496→              <div className="grid grid-cols-[180px_repeat(5,1fr)] gap-2 mb-2">
   497→                <div />
   498→                {[1, 2, 3, 4, 5].map(paso => (
   499→                  <div key={paso} className="text-center">
   500→                    <div className="bg-gray-100 rounded-t-lg px-3 py-2">
   501→                      <p className="text-xs font-bold text-gray-600">PASO {paso}</p>
   502→                      <p className="text-[10px] text-gray-500 leading-tight">{PASO_LABELS[paso]?.label}</p>
   503→                    </div>
   504→                  </div>
   505→                ))}
   506→              </div>
   507→
   508→              {/* S rows */}
   509→              {S_STEPS.map(s => (
   510→                <div key={s.id} className="grid grid-cols-[180px_repeat(5,1fr)] gap-2 mb-2">
   511→                  <div className={`rounded-lg border-l-4 ${S_BORDER_COLORS[s.id]} ${S_BG_COLORS[s.id]} flex flex-col items-center justify-center p-3`}>
   512→                    <span className="text-2xl font-black" style={{ color: S_COLORS[s.id] }}>S{s.id}</span>
   513→                    <span className="text-xs font-semibold text-gray-700 mt-0.5">{s.japaneseName}</span>
   514→                    <span className="text-[10px] text-gray-500">{s.spanishName}</span>
   515→                  </div>
   516→
   517→                  {[1, 2, 3, 4, 5].map(paso => {
   518→                    const slot = getSlot(s.id, paso)
