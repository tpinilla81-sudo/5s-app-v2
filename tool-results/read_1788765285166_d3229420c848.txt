   144→// ─── Field helper for Datos Empresa ─────────────────────────────────────────
   145→// Muestra el valor en modo lectura o un Input editable según `editing`.
   146→function Field({
   147→  label, k, value, edit, setEdit, editing,
   148→}: {
   149→  label: string
   150→  k: string
   151→  value: string | null | undefined
   152→  edit: Record<string, string>
   153→  setEdit: React.Dispatch<React.SetStateAction<Record<string, string>>>
   154→  editing: boolean
   155→}) {
   156→  return (
   157→    <div className="space-y-1">
   158→      <Label className="text-xs text-muted-foreground">{label}</Label>
   159→      {editing ? (
   160→        <Input
   161→          value={edit[k] ?? ''}
   162→          onChange={e => setEdit(prev => ({ ...prev, [k]: e.target.value }))}
   163→          className="h-8 text-sm"
   164→          placeholder={label}
   165→        />
   166→      ) : (
   167→        <div className={`h-8 px-2 flex items-center text-sm rounded-md border border-transparent bg-gray-50 ${
   168→          (value ?? '').trim() === '' ? 'text-muted-foreground italic' : 'text-gray-900'
   169→        }`}>
   170→          {(value ?? '').trim() === '' ? '— sin rellenar —' : value}
   171→        </div>
   172→      )}
   173→    </div>
   174→  )
   175→}

   476→          const assignProjectRes = await fetch(`/api/projects/${firstProject.id}/members`, {
   477→            method: 'POST',
   478→            headers: { 'Content-Type': 'application/json' },
   479→            body: JSON.stringify({ userId, role: newCompanyUserRole }),
   480→          })

   482→            console.log(`[handleCreateCompanyUser] Usuario ${email} asignado al proyecto ${firstProject.name}`)
   483→          } else {
   484→            console.warn(`[handleCreateCompanyUser] No se pudo asignar al proyecto:`, await assignProjectRes.json())
   485→          }
   486→        } catch (e) {
   487→          console.warn(`[handleCreateCompanyUser] Error asignando al proyecto:`, e)
   488→        }
   489→      }
   490→
   491→      // 4. Recargar users y limpiar formulario
   492→      await loadUsers()

   494→      setNewCompanyUserEmail('')
   495→      setNewCompanyUserPassword('')
