   363→// ═══════════════════════════════════════════════════════
   364→// VISUAL EDITOR: ChecklistEditor (autoevaluacion / auditoria)
   365→// ═══════════════════════════════════════════════════════
   366→interface ChecklistItem {
   367→  id: string
   368→  description: string
   369→  hasOther: boolean
   370→  active?: boolean // true (default) → aparece en el tablero (apartados 4/5); false → oculta
   371→}
   372→
   373→interface ChecklistSection {
   374→  id: string
   375→  title: string
   376→  items: ChecklistItem[]
   377→}
   378→
   379→function ChecklistEditor({ content, onChange }: { content: string; onChange: (v: string) => void }) {
   380→  let parsed: { sections: ChecklistSection[] }
   381→  try {
   382→    parsed = JSON.parse(content)
   383→  } catch {
   384→    return (
   385→      <div className="p-4 bg-red-50 border border-red-200 rounded-lg text-sm text-red-600">
   386→        El JSON no es válido. Corrígelo en modo JSON o carga el contenido por defecto.
   387→      </div>
   388→    )
   389→  }
   390→
   391→  const sections = parsed.sections || []
   392→  const sPrefix = String(content.match(/"id"\s*:\s*"(\d+)\./)?.[1] || '1')
   393→
   394→  const update = (newSections: ChecklistSection[]) => {
   395→    onChange(JSON.stringify({ ...parsed, sections: newSections }, null, 2))
   396→  }
   397→
   398→  const addSection = () => {
   399→    const newId = `${sPrefix}.${sections.length + 1}`
   400→    update([...sections, { id: newId, title: 'Nueva Sección', items: [] }])
   401→  }
   402→
   403→  const removeSection = (idx: number) => {
   404→    update(sections.filter((_, i) => i !== idx))
   405→  }
   406→
   407→  const updateSection = (idx: number, field: 'id' | 'title', value: string) => {
   408→    const updated = [...sections]
   409→    updated[idx] = { ...updated[idx], [field]: value }
   410→    update(updated)
   411→  }
   412→
   413→  const moveSection = (idx: number, dir: -1 | 1) => {
   414→    const target = idx + dir
   415→    if (target < 0 || target >= sections.length) return
   416→    const updated = [...sections]
   417→    ;[updated[idx], updated[target]] = [updated[target], updated[idx]]
   418→    update(updated)
   419→  }
   420→
   421→  const addItem = (sectionIdx: number) => {
   422→    const sec = sections[sectionIdx]
   423→    const newItemId = `${sec.id}.${sec.items.length + 1}`
   424→    const updated = [...sections]
   425→    updated[sectionIdx] = {
   426→      ...updated[sectionIdx],
   427→      items: [...updated[sectionIdx].items, { id: newItemId, description: '', hasOther: false, active: true }]
   428→    }
   429→    update(updated)
   430→  }
   431→
   432→  const removeItem = (sectionIdx: number, itemIdx: number) => {
   433→    const updated = [...sections]
   434→    updated[sectionIdx] = {
   435→      ...updated[sectionIdx],
   436→      items: updated[sectionIdx].items.filter((_, i) => i !== itemIdx)
   437→    }
   438→    update(updated)
   439→  }
   440→
   441→  const updateItem = (sectionIdx: number, itemIdx: number, field: string, value: string | boolean) => {
   442→    const updated = [...sections]
   443→    updated[sectionIdx] = {
   444→      ...updated[sectionIdx],
   445→      items: updated[sectionIdx].items.map((item, i) =>
   446→        i === itemIdx ? { ...item, [field]: value } : item
   447→      )
   448→    }
   449→    update(updated)
   450→  }
   451→
   452→  const moveItem = (sectionIdx: number, itemIdx: number, dir: -1 | 1) => {
   453→    const items = sections[sectionIdx].items
   454→    const target = itemIdx + dir
   455→    if (target < 0 || target >= items.length) return
   456→    const updated = [...sections]
   457→    const newItems = [...updated[sectionIdx].items]
   458→    ;[newItems[itemIdx], newItems[target]] = [newItems[target], newItems[itemIdx]]
   459→    updated[sectionIdx] = { ...updated[sectionIdx], items: newItems }
   460→    update(updated)
   461→  }
   462→
   463→  return (
   464→    <div className="space-y-4">
   465→      {/* Leyenda rápida */}
   466→      <div className="flex items-center gap-3 px-3 py-2 bg-indigo-50 border border-indigo-200 rounded-md text-xs text-indigo-800">
   467→        <Eye className="h-3.5 w-3.5 shrink-0" />
   468→        <span><b>En tablero</b> = la pregunta aparece en los apartados 4/5 de las S del tablero.</span>
   469→        <span className="text-indigo-400">·</span>
   470→        <EyeOff className="h-3.5 w-3.5 shrink-0" />
   471→        <span><b>Oculta</b> = no aparece en el tablero. Pulsa el badge para alternar.</span>
   472→      </div>
   473→      {sections.map((section, sIdx) => {
   474→        const activeCount = section.items.filter(it => it.active !== false).length
   475→        const totalCount = section.items.length
   476→        return (
   477→        <div key={sIdx} className="border-2 rounded-lg overflow-hidden bg-white shadow-sm">
   478→          {/* Section header */}
   479→          <div className="flex items-center gap-3 px-4 py-3 bg-gray-50 border-b">
   480→            <div className="flex flex-col gap-0.5">
   481→              <button onClick={() => moveSection(sIdx, -1)} className="text-gray-400 hover:text-gray-600 leading-none" title="Subir sección">
   482→                <ChevronUp className="h-4 w-4" />
   483→              </button>
   484→              <button onClick={() => moveSection(sIdx, 1)} className="text-gray-400 hover:text-gray-600 leading-none" title="Bajar sección">
   485→                <ChevronDown className="h-4 w-4" />
   486→              </button>
   487→            </div>
   488→            <Input
   489→              value={section.id}
   490→              onChange={(e) => updateSection(sIdx, 'id', e.target.value)}
   491→              className="w-20 h-9 text-sm font-mono"
   492→              placeholder="ID"
   493→            />
   494→            <Input
   495→              value={section.title}
   496→              onChange={(e) => updateSection(sIdx, 'title', e.target.value)}
   497→              className="flex-1 h-9 text-base font-semibold"
   498→              placeholder="Título de sección"
   499→            />
   500→            <span
   501→              className={`text-xs font-medium px-2 py-1 rounded-md whitespace-nowrap shrink-0 ${
   502→                activeCount === totalCount
   503→                  ? 'bg-green-100 text-green-700'
   504→                  : activeCount === 0
   505→                  ? 'bg-gray-200 text-gray-600'
   506→                  : 'bg-amber-100 text-amber-700'
   507→              }`}
   508→              title={`${activeCount} de ${totalCount} preguntas aparecen en el tablero`}
   509→            >
   510→              {activeCount}/{totalCount} en tablero
   511→            </span>
   512→            <Button variant="ghost" size="sm" onClick={() => addItem(sIdx)}
   513→              className="h-9 w-9 p-0 text-green-600 hover:text-green-700 hover:bg-green-50" title="Añadir item">
   514→              <Plus className="h-5 w-5" />
   515→            </Button>
   516→            <Button variant="ghost" size="sm" onClick={() => removeSection(sIdx)}
   517→              className="h-9 w-9 p-0 text-red-500 hover:text-red-600 hover:bg-red-50" title="Eliminar sección">
   518→              <Trash2 className="h-4 w-4" />
   519→            </Button>
   520→          </div>
   521→
   522→          {/* Items */}
   523→          <div className="p-3 space-y-2">
   524→            {section.items.length === 0 && (
   525→              <p className="text-sm text-muted-foreground italic px-2 py-2">Sin items. Pulsa + para añadir.</p>
   526→            )}
   527→            {section.items.map((item, iIdx) => {
   528→              // Un item se considera "en tablero" a menos que explícitamente esté active:false.
   529→              // Esto mantiene compatibilidad con plantillas antiguas que no tienen el campo `active`.
   530→              const isOnBoard = item.active !== false
   531→              return (
   532→                <div
   533→                  key={iIdx}
   534→                  className={`flex items-center gap-3 group rounded-md px-1 py-0.5 transition-colors ${
   535→                    isOnBoard ? '' : 'bg-gray-50'
   536→                  }`}
   537→                >
   538→                  <div className="flex flex-col gap-0.5">
   539→                    <button onClick={() => moveItem(sIdx, iIdx, -1)} className="text-gray-300 hover:text-gray-500 leading-none" title="Subir">
   540→                      <ChevronUp className="h-3.5 w-3.5" />
   541→                    </button>
   542→                    <button onClick={() => moveItem(sIdx, iIdx, 1)} className="text-gray-300 hover:text-gray-500 leading-none" title="Bajar">
   543→                      <ChevronDown className="h-3.5 w-3.5" />
   544→                    </button>
   545→                  </div>
   546→                  <Input
   547→                    value={item.id}
   548→                    onChange={(e) => updateItem(sIdx, iIdx, 'id', e.target.value)}
   549→                    className="w-20 h-8 text-xs font-mono"
   550→                    placeholder="ID"
   551→                  />
   552→                  <Input
   553→                    value={item.description}
   554→                    onChange={(e) => updateItem(sIdx, iIdx, 'description', e.target.value)}
   555→                    className={`flex-1 h-8 text-sm ${
   556→                      isOnBoard ? '' : 'text-gray-400 line-through decoration-gray-300'
   557→                    }`}
   558→                    placeholder="Descripción del item"
   559→                  />
   560→                  {/* Toggle "En tablero" — Controla si esta pregunta aparece en los apartados 4/5 de las S del tablero */}
   561→                  <button
   562→                    type="button"
   563→                    onClick={() => updateItem(sIdx, iIdx, 'active', !isOnBoard)}
   564→                    className={`flex items-center gap-1 px-2 h-8 rounded-md text-xs font-medium border transition-colors whitespace-nowrap shrink-0 ${
   565→                      isOnBoard
   566→                        ? 'bg-green-50 text-green-700 border-green-200 hover:bg-green-100'
   567→                        : 'bg-gray-100 text-gray-500 border-gray-200 hover:bg-gray-200'
   568→                    }`}
   569→                    title={
   570→                      isOnBoard
   571→                        ? 'Esta pregunta aparece en el tablero. Clic para ocultarla.'
   572→                        : 'Esta pregunta NO aparece en el tablero. Clic para mostrarla.'
