    62→export default function AuditoriaModal({ open, onClose, sStep, miniStep }: AuditoriaModalProps) {
    63→  const { fetchProgress, currentUser, adminFreeNavigation, currentProject, currentZone, canPerform, hasPermission } = use5SStore();
    64→  const sStepData = S_STEPS.find(s => s.id === sStep);
    65→  const canSkipSteps = hasPermission('skip_steps');
    66→  const canAudit = canPerform(sStep, 5);
    67→
    68→  const [auditorName, setAuditorName] = useState('');
    69→  const [results, setResults] = useState<Record<string, AuditItemResult>>({});
    70→  const [expandedSections, setExpandedSections] = useState<Record<string, boolean>>({});
    71→  const [observaciones, setObservaciones] = useState('');
    72→  const [isSubmitting, setIsSubmitting] = useState(false);
    73→  const [isCompleted, setIsCompleted] = useState(false);
    74→  const [finalScore, setFinalScore] = useState(0);
    75→  // Track which (open, sStep, sections-signature) we've already initialized for,
    76→  // so we don't reset user edits if `sections` ref changes mid-session.
    77→  const initializedFor = useRef<string>('');
    78→  const [notaMinima, setNotaMinima] = useState(75);
    79→  const [fechaAuditoria, setFechaAuditoria] = useState('');
    80→  const [horaAuditoria, setHoraAuditoria] = useState('');
    81→  const [auditPhotos, setAuditPhotos] = useState<{ file: File; preview: string; uploading?: boolean; serverUrl?: string }[]>([]);
    82→  const [isUploadingPhotos, setIsUploadingPhotos] = useState(false);
    83→  const photoInputRef = useRef<HTMLInputElement>(null);
    84→
    85→  // Project members for responsable selector on NOK items
    86→  const [projectMembers, setProjectMembers] = useState<Array<{ id: string; userId: string; role: string; user: { id: string; name: string; email: string; role: string; active: boolean } }>>([]);
    87→
    88→  // Scheduling for audit
    89→  const [fechaProgramada, setFechaProgramada] = useState('');
    90→  const [horaProgramada, setHoraProgramada] = useState('');
    91→
    92→  // Load template from API (uses board config if zone has one)
    93→  const { sections, isLoading: isLoadingTemplate, notaMinima: templateNotaMinima } = useChecklistTemplate('auditoria', sStep, open, currentZone?.boardConfigId);
    94→
    95→  // Apply template notaMinima when loaded
    96→  useEffect(() => {
    97→    if (templateNotaMinima !== null) setNotaMinima(templateNotaMinima);
    98→  }, [templateNotaMinima]);
    99→
   100→  // Load project members (for responsable selector on NOK items)
   101→  useEffect(() => {
   102→    if (open && currentProject?.id) {
   103→      fetch(`/api/projects/${currentProject.id}/members`)
   104→        .then(r => r.json())
   105→        .then(data => setProjectMembers(data?.members || []))
   106→        .catch(err => console.error('Error loading project members:', err));
   107→    } else {
   108→      setProjectMembers([]);
   109→    }
   110→  }, [open, currentProject?.id]);
   111→
   112→  // Mejoras realizadas
   113→  const [isFullscreen, setIsFullscreen] = useState(true);
   114→  const [haMejoras, setHaMejoras] = useState<boolean | null>(null);
   115→  const [mejoras, setMejoras] = useState<Array<{ id: string; descripcion: string; responsable: string; fecha: string }>>([]);
   116→
   117→
   118→
   119→  // Fetch dynamic threshold
   120→  useEffect(() => {
   121→    if (open && currentProject?.id) {
   122→      const fetchThreshold = async () => {
   123→        try {
   124→          const params = new URLSearchParams({ projectId: currentProject.id });
   125→          if (currentZone?.id) params.set('zoneId', currentZone.id);
   126→          const res = await fetch(`/api/audit-targets?${params}`);
   127→          const json = await res.json();
   128→          if (json.success && json.data) {
   129→            const zoneTarget = json.data.find((t: any) => t.sStep === sStep && t.miniStep === 5 && t.zoneId === currentZone?.id);
   130→            const projectTarget = json.data.find((t: any) => t.sStep === sStep && t.miniStep === 5 && t.zoneId === null);
   131→            const target = zoneTarget || projectTarget;
   132→            if (target?.notaMinima) setNotaMinima(target.notaMinima);
   133→          }
   134→        } catch (e) { console.error('Error fetching threshold:', e); }
   135→      };
   136→      fetchThreshold();
   137→    }
   138→  }, [open, sStep, currentProject?.id, currentZone?.id]);
   139→
   140→  // ─────────────────────────────────────────────────────────────
   141→  // ROBUST PRE-TICK: Merge missing items as 'ok' whenever sections change.
   142→  // This replaces the fragile one-shot pre-tick. Instead of setting results
   143→  // once and praying the effect fires after sections loads, we continuously
   144→  // MERGE missing items as 'ok' into `results` without overwriting user
   145→  // edits. This guarantees that every visible item always has an 'ok' entry
   146→  // unless the user explicitly marks it NOK/N/A.
   147→  // ─────────────────────────────────────────────────────────────
   148→  useEffect(() => {
   149→    if (!open || sections.length === 0) return;
   150→    setResults(prev => {
   151→      const next = { ...prev };
   152→      let changed = false;
   153→      sections.forEach(s => s.items.forEach(item => {
   154→        if (!next[item.id]) {
   155→          next[item.id] = { itemId: item.id, status: 'ok' };
   156→          changed = true;
   157→        }
   158→      }));
   159→      return changed ? next : prev;
   160→    });
   161→  }, [open, sections]);
   162→
   163→  // Initialize expanded sections + reset form state once per (sStep + sections signature)
   164→  useEffect(() => {
   165→    if (!open || sections.length === 0) return;
   166→    const sig = `${sStep}:${sections.map(s => s.items.map(i => i.id).join(',')).join('|')}`;
   167→    if (initializedFor.current === sig) return;
   168→    initializedFor.current = sig;
   169→
   170→    const expanded: Record<string, boolean> = {};
   171→    sections.forEach(s => { expanded[s.id] = true; });
   172→    setExpandedSections(expanded);
   173→    setAuditorName(currentUser?.name || '');
   174→    setObservaciones('');
   175→    setIsCompleted(false);
   176→    setFinalScore(0);
   177→    setHaMejoras(null);
   178→    setAuditPhotos([]);
   179→    setMejoras([]);
   180→    // Auto-fill date and time of the audit
   181→    const now = new Date();
   182→    setFechaAuditoria(now.toISOString().slice(0, 10));
   183→    setHoraAuditoria(now.toTimeString().slice(0, 5));
   184→    // Load scheduled date if available
   185→    loadScheduledDate();
   186→    // Load previously saved results (so user can see what was saved before)
   187→    loadSavedResults();
   188→  }, [open, sStep, sections]);
   189→
   190→  // Load previously saved audit results from the backend
   191→  const loadSavedResults = async () => {
   192→    if (!currentProject?.id) return;
   193→    try {
   194→      const res = await fetch(`/api/progress/step?sStep=${sStep}&miniStep=${miniStep}&projectId=${currentProject.id}${currentZone?.id ? `&zoneId=${currentZone.id}` : ''}`);
   195→      const json = await res.json();
   196→      if (json.success && json.data?.notes) {
   197→        try {
   198→          const notes = typeof json.data.notes === 'string' ? JSON.parse(json.data.notes) : json.data.notes;
   199→          if (notes?.results && Array.isArray(notes.results)) {
   200→            const loaded: Record<string, AuditItemResult> = {};
   201→            notes.results.forEach((r: any) => {
   202→              if (r?.itemId) loaded[r.itemId] = r;
   203→            });
   204→            if (Object.keys(loaded).length > 0) {
   205→              setResults(loaded);
   206→              if (notes.observaciones) setObservaciones(notes.observaciones);
   207→              if (notes.fechaAuditoria) setFechaAuditoria(notes.fechaAuditoria);
   208→              if (notes.horaAuditoria) setHoraAuditoria(notes.horaAuditoria);
   209→              if (notes.auditorName) setAuditorName(notes.auditorName);
   210→            }
   211→          }
   212→        } catch (e) {
   213→          console.error('Error parsing saved notes:', e);
   214→        }
   215→      }
   216→    } catch (e) {
   217→      console.error('Error loading saved results:', e);
   218→    }
   219→  };
   220→
   221→  // Reset the init guard when modal closes, so next open re-initializes
   222→  useEffect(() => {
   223→    if (!open) {
   224→      initializedFor.current = '';
   225→      setResults({});
   226→    }
   227→  }, [open]);
   228→
   229→  // Load scheduled date/time for this auditoría
   230→  const loadScheduledDate = async () => {
   231→    if (!currentProject?.id || !currentZone?.id) return;
   232→    try {
   233→      const res = await fetch(`/api/evaluation-schedule?sStep=${sStep}&miniStep=5&projectId=${currentProject.id}&zoneId=${currentZone.id}`);
   234→      const json = await res.json();
   235→      if (json.success && json.data) {
   236→        setFechaProgramada(json.data.fechaProgramada || '');
   237→        setHoraProgramada(json.data.horaProgramada || '');
   238→      }
   239→    } catch (e) {
   240→      console.error('Error loading scheduled date:', e);
   241→    }
   242→  };
   243→
   244→  // Save scheduled date/time
   245→  const handleSaveSchedule = async () => {
   246→    if (!currentProject?.id || !currentZone?.id) return;
   247→    try {
   248→      await fetch('/api/evaluation-schedule', {
   249→        method: 'POST',
   250→        headers: { 'Content-Type': 'application/json' },
   251→        body: JSON.stringify({
   252→          sStep,
   253→          miniStep: 5,
   254→          projectId: currentProject.id,
   255→          zoneId: currentZone.id,
   256→          fechaProgramada,
   257→          horaProgramada,
   258→        }),
   259→      });
   260→      toast.success('Fecha programada guardada');
   261→    } catch (e) {
   262→      console.error('Error saving schedule:', e);
   263→      toast.error('Error al guardar la fecha programada');
   264→    }
   265→  };
   266→
   267→  const totalItems = sections.reduce((sum, s) => sum + s.items.length, 0) || 26;
   268→
   269→  const scoring = useMemo(() => {
   270→    // SCORING: Treat missing results as 'ok' (default).
   271→    // Bulletproof — does NOT depend on useEffect timing.
   272→    let okCount = 0;
   273→    let nokCount = 0;
   274→    sections.forEach(s => s.items.forEach(item => {
   275→      const status = results[item.id]?.status ?? 'ok';
   276→      if (status === 'ok') okCount++;
   277→      else if (status === 'nok') nokCount++;
   278→    }));
   279→    const answeredCount = okCount + nokCount;
   280→    // Checklist maxes at 90%
   281→    const checklistScore = totalItems > 0 ? Math.round((okCount / totalItems) * 90) : 0;
   282→    // Each mejora with description adds 5%, max 2 mejoras = +10%
   283→    const validMejorasCount = haMejoras ? mejoras.filter(m => m.descripcion.trim()).length : 0;
   284→    const mejorasScore = Math.min(validMejorasCount, 2) * 5;
   285→    const scorePercent = Math.min(checklistScore + mejorasScore, 100); // HARD CAP: never exceeds 100%
   286→    return { okCount, nokCount, answeredCount, checklistScore, mejorasScore, validMejorasCount, scorePercent };
   287→  }, [results, sections, totalItems, haMejoras, mejoras]);
   288→
   289→  const canSubmit = canAudit && auditorName.trim() !== '' && scoring.answeredCount > 0;
   290→
   291→  // Check that all NOK items have hallazgo and mejora filled
   292→  const nokItems = Object.values(results).filter(r => r.status === 'nok');
   293→  const allNokCompleted = nokItems.length === 0 || nokItems.every(r => (r.hallazgo || '').trim() !== '' && (r.mejora || '').trim() !== '' && (r.responsable || '').trim() !== '');
   294→  const canSubmitFinal = canSubmit && allNokCompleted;
   295→
   296→  const toggleSection = (sectionId: string) => {
   297→    setExpandedSections(prev => ({ ...prev, [sectionId]: !prev[sectionId] }));
   298→  };
   299→
   300→  const setItemStatus = (itemId: string, status: 'ok' | 'nok' | 'na') => {
   301→    setResults(prev => ({
   302→      ...prev,
   303→      [itemId]: { ...prev[itemId], itemId, status },
   304→    }));
   305→  };
   306→
   307→  // Marcar TODOS los items como OK de una sola vez.
   308→  // Respeta los items ya marcados como NOK/N/A (no los sobreescribe).
   309→  const markAllOk = () => {
   310→    setResults(prev => {
   311→      const next = { ...prev };
   312→      sections.forEach(s => s.items.forEach(item => {
   313→        const existing = next[item.id];
   314→        if (!existing || existing.status === 'ok') {
   315→          next[item.id] = { ...existing, itemId: item.id, status: 'ok' };
   316→        }
   317→      }));
   318→      return next;
   319→    });
   320→  };
   321→
   322→  // Forzar TODOS los items como OK (sobreescribe incluso NOK/N/A). Con confirmación.
   323→  const forceAllOk = () => {
   324→    const nokCount = sections.reduce((acc, s) => acc + s.items.filter(i => results[i.id]?.status === 'nok').length, 0);
   325→    if (nokCount > 0) {
   326→      if (!confirm(`¿Sobreescribir ${nokCount} hallazgo(s) NOK y marcar TODO como OK?`)) return;
   327→    }
   328→    setResults(prev => {
   329→      const next = { ...prev };
   330→      sections.forEach(s => s.items.forEach(item => {
   331→        next[item.id] = { ...next[item.id], itemId: item.id, status: 'ok' };
   332→      }));
   333→      return next;
   334→    });
   335→  };
   336→
   337→  const setItemField = (itemId: string, field: 'hallazgo' | 'mejora' | 'otherText' | 'responsable', value: string) => {
   338→    setResults(prev => ({
   339→      ...prev,
   340→      [itemId]: { ...prev[itemId], itemId, [field]: value },
   341→    }));
   342→  };
   343→
   344→  const handleAuditPhotoSelect = async (e: React.ChangeEvent<HTMLInputElement>) => {
   345→    const files = e.target.files;
   346→    if (!files) return;
   347→    for (const file of Array.from(files)) {
   348→      const preview = URL.createObjectURL(file);
   349→      setAuditPhotos(prev => [...prev, { file, preview }]);
   350→    }
   351→    if (photoInputRef.current) photoInputRef.current.value = '';
   352→  };
   353→
   354→  const removeAuditPhoto = (index: number) => {
   355→    setAuditPhotos(prev => {
   356→      const photo = prev[index];
   357→      if (photo.preview) URL.revokeObjectURL(photo.preview);
   358→      return prev.filter((_, i) => i !== index);
   359→    });
   360→  };
   361→
   362→  const handleSubmit = async () => {
   363→    if (!canSubmit) return;
   364→    setIsSubmitting(true);
   365→    const isApto = scoring.scorePercent >= notaMinima;
   366→    try {
   367→      // Save audit result (always save, even if not apto)
   368→      const auditRes = await fetch('/api/audit', {
   369→        method: 'POST',
   370→        headers: { 'Content-Type': 'application/json' },
   371→        body: JSON.stringify({
   372→          sStep,
   373→          auditorName,
   374→          result: isApto ? 'apto' : 'no_apto',
   375→          score: Math.min(scoring.scorePercent, 100),
   376→          observations: observaciones || null,
   377→          fechaAuditoria,
   378→          horaAuditoria,
   379→          projectId: currentProject?.id,
   380→          zoneId: currentZone?.id || null,
   381→          checklistData: JSON.stringify(Object.values(results)),
   382→          mejorasData: haMejoras ? JSON.stringify(mejoras.filter(m => m.descripcion.trim())) : null,
   383→        }),
   384→      });
   385→
   386→      const auditJson = await auditRes.json();
   387→
   388→      // Handle server-side role rejection
   389→      if (auditRes.status === 403) {
   390→        setIsSubmitting(false);
   391→        alert(auditJson.error || 'No tienes permisos para realizar auditorías externas');
   392→        return;
   393→      }
   394→
   395→      if (!auditJson.success) {
   396→        setIsSubmitting(false);
   397→        alert(auditJson.error || 'Error al guardar la auditoría');
   398→        return;
   399→      }
   400→      if (auditJson.success) {
   401→        // Feedback visible
   402→        toast.success(`Auditoría guardada: ${Object.keys(results).length} items (${scoring.okCount} OK, ${scoring.nokCount} NOK, score ${scoring.scorePercent}%)`);
   403→        // Also save results in progress.notes so loadSavedResults can find them on reopen
   404→        try {
   405→          await fetch(`/api/progress/step?sStep=${sStep}&miniStep=${miniStep}`, {
   406→            method: 'PUT',
   407→            headers: { 'Content-Type': 'application/json' },
   408→            body: JSON.stringify({
   409→              completed: isApto,
   410→              score: scoring.scorePercent,
   411→              notes: JSON.stringify({
   412→                type: 'auditoria',
   413→                passed: isApto,
   414→                notaMinima,
   415→                results: Object.values(results),
   416→                observaciones,
   417→                auditorName,
   418→                fechaAuditoria,
   419→                horaAuditoria,
   420→              }),
   421→              projectId: currentProject?.id,
   422→              zoneId: currentZone?.id || null,
   423→            }),
   424→          });
   425→        } catch (progressErr) {
   426→          console.error('Error saving audit results to progress.notes:', progressErr);
   427→        }
   428→
   429→        // Create action items for each NOK to transmit disfunciones to the operator
   430→        const nokResults = Object.values(results).filter(r => r.status === 'nok');
   431→        for (const nok of nokResults) {
   432→          await fetch('/api/actions', {
   433→            method: 'POST',
   434→            headers: { 'Content-Type': 'application/json' },
   435→            body: JSON.stringify({
   436→              sStep,
   437→              miniStep: 5,
   438→              itemId: nok.itemId,
   439→              itemDescription: `Disfunción detectada en auditoría: ${nok.itemId}`,
   440→              hallazgo: nok.hallazgo || nok.itemId,
   441→              mejora: nok.mejora || '',
   442→              responsable: nok.responsable || null,
   443→              prioridad: 'alta',
   444→              estado: 'abierta',
   445→              source: 'auditoria',
   446→              auditor: auditorName,
   447→              projectId: currentProject?.id,
   448→              zoneId: currentZone?.id || null,
   449→            }),
   450→          });
   451→        }
   452→
   453→        // ─── Notify responsables of NOK disfunciones from audit ───
   454→        if (nokResults.length > 0 && currentProject?.id && currentZone?.id) {
   455→          try {
   456→            const membersRes = await fetch(`/api/projects/${currentProject.id}/members`);
   457→            const membersData = await membersRes.json();
   458→            const allMembers = membersData?.members || [];
   459→
   460→            const sStepData = S_STEPS.find(s => s.id === sStep);
   461→            const disfuncionMessage = `El auditor ha detectado ${nokResults.length} disfunción(es) en la auditoría de S${sStep} (${sStepData?.japaneseName || ''}) en la zona "${currentZone.name}". Revisa el Plan de Acción.`;
   462→
   463→            // Notify zone responsable
   464→            const responsableIds = new Set<string>();
   465→            if (currentZone.responsableId) responsableIds.add(currentZone.responsableId);
   466→            const responsables = allMembers.filter((m: any) => m.role === 'responsable');
   467→            for (const resp of responsables) responsableIds.add(resp.userId);
   468→
   469→            for (const respId of responsableIds) {
   470→              await fetch('/api/notifications', {
   471→                method: 'POST',
   472→                headers: { 'Content-Type': 'application/json' },
   473→                body: JSON.stringify({
   474→                  userId: respId,
   475→                  type: 'disfuncion',
   476→                  title: `Disfunciones en auditoría: S${sStep} — ${sStepData?.japaneseName || ''}`,
   477→                  message: disfuncionMessage,
   478→                  sStep,
   479→                  zoneId: currentZone.id,
   480→                  projectId: currentProject.id,
   481→                }),
   482→              });
   483→            }
   484→          } catch (notifError) {
   485→            console.error('Error notifying responsables of audit disfunciones:', notifError);
   486→          }
   487→        }
   488→
   489→        // ─── Upload audit photos to library with traceability ───
   490→        if (auditPhotos.length > 0) {
   491→          setIsUploadingPhotos(true);
   492→          for (let idx = 0; idx < auditPhotos.length; idx++) {
   493→            const photo = auditPhotos[idx];
   494→            try {
   495→              const formData = new FormData();
   496→              formData.append('file', photo.file);
   497→              formData.append('filename', `S${sStep}_auditoria_${currentZone?.name || 'zona'}_${idx + 1}_${Date.now()}.jpg`);
   498→
   499→              const uploadRes = await fetch('/api/upload', {
   500→                method: 'POST',
   501→                body: formData,
   502→              });
   503→              const uploadData = await uploadRes.json();
   504→
   505→              if (uploadData.success && uploadData.url) {
   506→                await fetch('/api/photo-library', {
   507→                  method: 'POST',
   508→                  headers: { 'Content-Type': 'application/json' },
   509→                  body: JSON.stringify({
   510→                    sStep,
   511→                    miniStep: 5,
   512→                    title: `Auditoría S${sStep} - Foto ${idx + 1}`,
   513→                    description: `${sStepData?.japaneseName || 'S' + sStep} - ${currentZone?.name || 'Zona'} - Paso 5 Auditoría - Auditor: ${auditorName}`,
   514→                    photoUrl: uploadData.url,
