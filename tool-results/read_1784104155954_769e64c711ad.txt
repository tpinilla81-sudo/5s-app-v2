   179→  }

   282→function ChecklistEditor({ content, onChange }: { content: string; onChange: (v: string) => void }) {
   283→  let parsed: { sections: ChecklistSection[] }
   284→  try {
   285→    parsed = JSON.parse(content)
   286→  } catch {
   287→    return (
   288→      <div className="p-4 bg-red-50 border border-red-200 rounded-lg text-sm text-red-600">
   289→        El JSON no es válido. Corrígelo en modo JSON o carga el contenido por defecto.
   290→      </div>
   291→    )
   292→  }
   293→
   294→  const sections = parsed.sections || []
   295→  const sPrefix = String(content.match(/"id"\s*:\s*"(\d+)\./)?.[1] || '1')
   296→
   297→  const update = (newSections: ChecklistSection[]) => {
   298→    onChange(JSON.stringify({ ...parsed, sections: newSections }, null, 2))
   299→  }
   300→
   301→  const addSection = () => {
   302→    const newId = `${sPrefix}.${sections.length + 1}`
   303→    update([...sections, { id: newId, title: 'Nueva Sección', items: [] }])
   304→  }
   305→
   306→  const removeSection = (idx: number) => {
   307→    update(sections.filter((_, i) => i !== idx))
   308→  }
   309→
   310→  const updateSection = (idx: number, field: 'id' | 'title', value: string) => {
   311→    const updated = [...sections]
   312→    updated[idx] = { ...updated[idx], [field]: value }
   313→    update(updated)
   314→  }
   315→
   316→  const moveSection = (idx: number, dir: -1 | 1) => {
   317→    const target = idx + dir
   318→    if (target < 0 || target >= sections.length) return
   319→    const updated = [...sections]
   320→    ;[updated[idx], updated[target]] = [updated[target], updated[idx]]
   321→    update(updated)
   322→  }
   323→
   324→  const addItem = (sectionIdx: number) => {
   325→    const sec = sections[sectionIdx]
   326→    const newItemId = `${sec.id}.${sec.items.length + 1}`
   327→    const updated = [...sections]
   328→    updated[sectionIdx] = {
   329→      ...updated[sectionIdx],
   330→      items: [...updated[sectionIdx].items, { id: newItemId, description: '', hasOther: false }]
   331→    }
   332→    update(updated)
   333→  }
   334→
   335→  const removeItem = (sectionIdx: number, itemIdx: number) => {
   336→    const updated = [...sections]
   337→    updated[sectionIdx] = {
   338→      ...updated[sectionIdx],
   339→      items: updated[sectionIdx].items.filter((_, i) => i !== itemIdx)
   340→    }
   341→    update(updated)
   342→  }
   343→
   344→  const updateItem = (sectionIdx: number, itemIdx: number, field: string, value: string | boolean) => {
   345→    const updated = [...sections]
   346→    updated[sectionIdx] = {
   347→      ...updated[sectionIdx],
   348→      items: updated[sectionIdx].items.map((item, i) =>
   349→        i === itemIdx ? { ...item, [field]: value } : item
   350→      )
   351→    }
   352→    update(updated)
   353→  }
   354→
   355→  const moveItem = (sectionIdx: number, itemIdx: number, dir: -1 | 1) => {
   356→    const items = sections[sectionIdx].items
   357→    const target = itemIdx + dir
   358→    if (target < 0 || target >= items.length) return
   359→    const updated = [...sections]
   360→    const newItems = [...updated[sectionIdx].items]
   361→    ;[newItems[itemIdx], newItems[target]] = [newItems[target], newItems[itemIdx]]
   362→    updated[sectionIdx] = { ...updated[sectionIdx], items: newItems }
   363→    update(updated)
   364→  }
   365→
   366→  return (
   367→    <div className="space-y-4">
   368→      {sections.map((section, sIdx) => (
   369→        <div key={sIdx} className="border-2 rounded-lg overflow-hidden bg-white shadow-sm">
   370→          {/* Section header */}
   371→          <div className="flex items-center gap-3 px-4 py-3 bg-gray-50 border-b">
   372→            <div className="flex flex-col gap-0.5">
   373→              <button onClick={() => moveSection(sIdx, -1)} className="text-gray-400 hover:text-gray-600 leading-none" title="Subir sección">
   374→                <ChevronUp className="h-4 w-4" />
   375→              </button>
   376→              <button onClick={() => moveSection(sIdx, 1)} className="text-gray-400 hover:text-gray-600 leading-none" title="Bajar sección">
   377→                <ChevronDown className="h-4 w-4" />
   378→              </button>
   379→            </div>
   380→            <Input
   381→              value={section.id}
   382→              onChange={(e) => updateSection(sIdx, 'id', e.target.value)}
   383→              className="w-20 h-9 text-sm font-mono"
   384→              placeholder="ID"
   385→            />
   386→            <Input
   387→              value={section.title}
   388→              onChange={(e) => updateSection(sIdx, 'title', e.target.value)}
   389→              className="flex-1 h-9 text-base font-semibold"
   390→              placeholder="Título de sección"
   391→            />
   392→            <Button variant="ghost" size="sm" onClick={() => addItem(sIdx)}
   393→              className="h-9 w-9 p-0 text-green-600 hover:text-green-700 hover:bg-green-50" title="Añadir item">
   394→              <Plus className="h-5 w-5" />
   395→            </Button>
   396→            <Button variant="ghost" size="sm" onClick={() => removeSection(sIdx)}
   397→              className="h-9 w-9 p-0 text-red-500 hover:text-red-600 hover:bg-red-50" title="Eliminar sección">
   398→              <Trash2 className="h-4 w-4" />
   399→            </Button>
   400→          </div>
   401→
   402→          {/* Items */}
   403→          <div className="p-3 space-y-2">
   404→            {section.items.length === 0 && (
   405→              <p className="text-sm text-muted-foreground italic px-2 py-2">Sin items. Pulsa + para añadir.</p>
   406→            )}
   407→            {section.items.map((item, iIdx) => (
   408→              <div key={iIdx} className="flex items-center gap-3 group">
   409→                <div className="flex flex-col gap-0.5">
   410→                  <button onClick={() => moveItem(sIdx, iIdx, -1)} className="text-gray-300 hover:text-gray-500 leading-none" title="Subir">
   411→                    <ChevronUp className="h-3.5 w-3.5" />
   412→                  </button>
   413→                  <button onClick={() => moveItem(sIdx, iIdx, 1)} className="text-gray-300 hover:text-gray-500 leading-none" title="Bajar">
   414→                    <ChevronDown className="h-3.5 w-3.5" />
   415→                  </button>
   416→                </div>
   417→                <Input
   418→                  value={item.id}
   419→                  onChange={(e) => updateItem(sIdx, iIdx, 'id', e.target.value)}
   420→                  className="w-20 h-8 text-xs font-mono"
   421→                  placeholder="ID"
   422→                />
   423→                <Input
   424→                  value={item.description}
   425→                  onChange={(e) => updateItem(sIdx, iIdx, 'description', e.target.value)}
   426→                  className="flex-1 h-8 text-sm"
   427→                  placeholder="Descripción del item"
   428→                />
   429→                <label className="flex items-center gap-1.5 text-xs text-muted-foreground whitespace-nowrap cursor-pointer shrink-0">
   430→                  <input
   431→                    type="checkbox"
   432→                    checked={item.hasOther}
   433→                    onChange={(e) => updateItem(sIdx, iIdx, 'hasOther', e.target.checked)}
   434→                    className="rounded border-gray-300 h-4 w-4"
   435→                  />
   436→                  Otros
   437→                </label>
   438→                <Button variant="ghost" size="sm"
   439→                  onClick={() => removeItem(sIdx, iIdx)}
   440→                  className="h-8 w-8 p-0 text-red-400 hover:text-red-500 opacity-0 group-hover:opacity-100 transition-opacity" title="Eliminar">
   441→                  <Trash2 className="h-4 w-4" />
   442→                </Button>
   443→              </div>
   444→            ))}
   445→          </div>
   446→        </div>
   447→      ))}
   448→
   449→      <Button variant="outline" onClick={addSection}
   450→        className="w-full border-dashed border-2 text-green-600 hover:bg-green-50 hover:border-green-400 gap-1 h-10">
   451→        <Plus className="h-5 w-5" />
   452→        Añadir sección
   453→      </Button>
   454→
   455→      <div className="text-sm text-muted-foreground text-center">
   456→        {sections.length} sección(es) · {sections.reduce((s, sec) => s + sec.items.length, 0)} item(s) en total
   457→      </div>
   458→    </div>
   459→  )
   460→}

   500→    update(updated)
