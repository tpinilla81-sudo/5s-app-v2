     1→'use client'
     2→
     3→import { useState, useEffect } from 'react'
     4→import { Button } from '../ui/button'
     5→import { Input } from '../ui/input'
     6→import { Label } from '../ui/label'
     7→import { Badge } from '../ui/badge'
     8→import {
     9→  Dialog,
    10→  DialogContent,
    11→  DialogHeader,
    12→  DialogTitle,
    13→} from '../ui/dialog'
    14→import { Loader2, Save, Building2, Users, Shield, Mail, Phone, Key, AlertTriangle, RefreshCw, Eye, EyeOff, Lock, CheckCircle2, RotateCcw } from 'lucide-react'
    15→import { toast } from 'sonner'
    16→
    17→interface FullEditCompanyModalProps {
    18→  open: boolean
    19→  onOpenChange: (open: boolean) => void
    20→  companyId: string | null
    21→}
    22→
    23→interface AdminInfo {
    24→  id: string
    25→  name: string
    26→  email: string
    27→  role: string
    28→  active: boolean
    29→  joinedAt: string
    30→  plainPassword?: string  // Contraseña en texto plano
    31→}
    32→
    33→export function FullEditCompanyModal({ open, onOpenChange, companyId }: FullEditCompanyModalProps) {
    34→  const [data, setData] = useState({
    35→    name: '',
    36→    description: '',
    37→    active: true,
    38→    nif: '',
    39→    sector: '',
    40→    address: '',
    41→    city: '',
    42→    province: '',
    43→    postalCode: '',
    44→    country: 'España',
    45→    phone: '',
    46→    website: '',
    47→    billingEmail: '',
    48→    billingName: '',
    49→    billingNif: '',
    50→    billingAddress: '',
    51→    billingCity: '',
    52→    billingPostalCode: '',
    53→    iban: '',
    54→    contactName: '',
    55→    contactEmail: '',
    56→    contactPhone: ''
    57→  })
    58→  
    59→  const [admin, setAdmin] = useState<AdminInfo | null>(null)
    60→  const [loading, setLoading] = useState(false)
    61→  const [saving, setSaving] = useState(false)
    62→  const [error, setError] = useState<string | null>(null)
    63→  const [errorDetails, setErrorDetails] = useState<string>('')
    64→  
    65→  // Estados para la contraseña del admin
    66→  const [showPassword, setShowPassword] = useState(false)
    67→  const [isEditingPassword, setIsEditingPassword] = useState(false)
    68→  const [newPassword, setNewPassword] = useState('')
    69→  const [resettingPassword, setResettingPassword] = useState(false)
    70→
    71→  // Reset state when modal closes
    72→  useEffect(() => {
    73→    if (!open) {
    74→      setError(null)
    75→      setErrorDetails('')
    76→      setLoading(false)
    77→      setShowPassword(false)
    78→      setIsEditingPassword(false)
    79→      setNewPassword('')
    80→      setResettingPassword(false)
    81→    }
    82→  }, [open])
    83→
    84→  // Cargar datos cuando se abre el modal
    85→  useEffect(() => {
    86→    if (open && companyId) {
    87→      loadCompanyData(companyId)
    88→    }
    89→  }, [open, companyId])
    90→
    91→  const loadCompanyData = async (id: string) => {
    92→    setLoading(true)
    93→    setError(null)
    94→    setErrorDetails('')
    95→    console.log('[FullEditModal] Cargando empresa:', id)
    96→    
    97→    try {
    98→      const response = await fetch(`/api/companies/${id}`, { 
    99→        credentials: 'include',
   100→        headers: {
   101→          'Cache-Control': 'no-cache',
   102→          'Pragma': 'no-cache'
   103→        }
   104→      })
   105→      
   106→      console.log('[FullEditModal] Response status:', response.status)
   107→      
   108→      const result = await response.json()
   109→      console.log('[FullEditModal] Respuesta API:', result)
   110→      
   111→      if (!response.ok) {
   112→        // Manejar diferentes tipos de error
   113→        let errorMsg = result.error || result.details || `Error ${response.status}`
   114→        let details = result.details || result.stack || ''
   115→        
   116→        if (response.status === 401) {
   117→          errorMsg = 'No autorizado - Sesión expirada'
   118→          details = 'Tu sesión ha expirado. Por favor, recarga la página e inicia sesión nuevamente.'
   119→        } else if (response.status === 403) {
   120→          errorMsg = 'Sin permisos'
   121→          details = 'No tienes permisos para ver esta empresa. Necesitas rol de Gestor o Admin.'
   122→        } else if (response.status === 404) {
   123→          errorMsg = 'Empresa no encontrada'
   124→          details = 'La empresa ya no existe o fue eliminada.'
   125→        }
   126→        
   127→        throw new Error(errorMsg + (details ? ` | ${details}` : ''))
   128→      }
   129→      
   130→      const company = result.company || result
   131→      if (company && company.name) {
   132→        console.log('[FullEditModal] Datos empresa cargados:', company.name)
   133→        setData({
   134→          name: company.name || '',
   135→          description: company.description || '',
   136→          active: company.active !== undefined ? company.active : true,
   137→          nif: company.nif || '',
   138→          sector: company.sector || '',
   139→          address: company.address || '',
   140→          city: company.city || '',
   141→          province: company.province || '',
   142→          postalCode: company.postalCode || '',
   143→          country: company.country || 'España',
   144→          phone: company.phone || '',
   145→          website: company.website || '',
   146→          billingEmail: company.billingEmail || '',
   147→          billingName: company.billingName || '',
   148→          billingNif: company.billingNif || '',
   149→          billingAddress: company.billingAddress || '',
   150→          billingCity: company.billingCity || '',
   151→          billingPostalCode: company.billingPostalCode || '',
   152→          iban: company.iban || '',
   153→          contactName: company.contactName || '',
   154→          contactEmail: company.contactEmail || '',
   155→          contactPhone: company.contactPhone || ''
   156→        })
   157→
   158→        // Buscar administrador entre los miembros
   159→        if (company.members && Array.isArray(company.members)) {
   160→          const adminMember = company.members.find((m: any) => 
   161→            m.role === 'admin' || m.user?.role === 'admin'
   162→          )
   163→          if (adminMember && adminMember.user) {
   164→            console.log('[FullEditModal] Admin encontrado:', adminMember.user.name)
   165→            const adminData = {
   166→              id: adminMember.user.id,
   167→              name: adminMember.user.name,
   168→              email: adminMember.user.email,
   169→              role: adminMember.user.role,
   170→              active: adminMember.user.active !== false, // default true
   171→              joinedAt: adminMember.joinedAt
   172→            }
   173→            setAdmin(adminData)
   174→            
   175→            // Cargar contraseña del admin desde API de usuarios
   176→            loadAdminPassword(adminMember.user.id)
   177→          } else {
   178→            console.log('[FullEditModal] No se encontró admin en miembros:', company.members.length, 'miembros')
   179→            setAdmin(null)
   180→          }
   181→        }
   182→      } else {
   183→        throw new Error('Datos de empresa inválidos o vacíos')
   184→      }
   185→    } catch (err: any) {
   186→      console.error('[FullEditModal] Error cargando empresa:', err)
   187→      const message = err.message || 'Error desconocido'
   188→      setError(message.includes('|') ? message.split('|')[0] : message)
   189→      setErrorDetails(message.includes('|') ? message.split('|').slice(1).join('|') : '')
   190→      toast.error('Error al cargar: ' + (message.includes('|') ? message.split('|')[0] : message))
   191→    } finally {
   192→      setLoading(false)
   193→    }
   194→  }
   195→
   196→  const handleSave = async () => {
   197→    if (!companyId || !data.name.trim()) {
   198→      toast.error('El nombre es obligatorio')
   199→      return
   200→    }
   201→    
   202→    setSaving(true)
   203→    try {
   204→      console.log('[FullEditModal] Guardando empresa:', data.name)
   205→      const res = await fetch(`/api/companies/${companyId}`, {
   206→        method: 'PUT',
   207→        credentials: 'include',
   208→        headers: { 'Content-Type': 'application/json' },
   209→        body: JSON.stringify(data)
   210→      })
   211→      
   212→      const result = await res.json()
   213→      console.log('[FullEditModal] Respuesta guardado:', result)
   214→      
   215→      if (res.ok) {
   216→        toast.success('✅ Empresa actualizada correctamente')
   217→        onOpenChange(false)
   218→        window.location.reload()
   219→      } else {
   220→        toast.error(result.error || 'Error al guardar')
   221→      }
   222→    } catch (error) {
   223→      console.error('[FullEditModal] Error saving:', error)
   224→      toast.error('Error de conexión')
   225→    }
   226→    
   227→    setSaving(false)
   228→  }
   229→
   230→  // Cargar contraseña del administrador
   231→  const loadAdminPassword = async (userId: string) => {
   232→    try {
   233→      console.log('[FullEditModal] Cargando contraseña del admin:', userId)
   234→      const res = await fetch(`/api/users?search=${admin?.email}`, { credentials: 'include' })
   235→      if (res.ok) {
   236→        const result = await res.json()
   237→        if (result.success && result.users && result.users.length > 0) {
   238→          const user = result.users.find((u: any) => u.id === userId)
   239→          if (user && user.plainPassword) {
   240→            console.log('[FullEditModal] Contraseña cargada')
   241→            setAdmin(prev => prev ? { ...prev, plainPassword: user.plainPassword } : null)
   242→          }
   243→        }
   244→      }
   245→    } catch (err) {
   246→      console.error('[FullEditModal] Error cargando contraseña:', err)
   247→    }
   248→  }
   249→
   250→  // Restablecer/cambiar contraseña del admin
   251→  const handleResetPassword = async () => {
   252→    if (!admin) return
   253→    
   254→    // Si no estamos en modo edición, entrar en modo edición
   255→    if (!isEditingPassword) {
   256→      setIsEditingPassword(true)
   257→      setNewPassword(admin.plainPassword || '')
   258→      return
   259→    }
   260→    
   261→    // Guardar nueva contraseña
   262→    if (!newPassword || newPassword.length < 6) {
   263→      toast.error('La contraseña debe tener al menos 6 caracteres')
   264→      return
   265→    }
   266→    
   267→    setResettingPassword(true)
   268→    try {
   269→      console.log('[FullEditModal] Reseteando contraseña para:', admin.id)
   270→      const res = await fetch('/api/users', {
   271→        method: 'PUT',
   272→        credentials: 'include',
   273→        headers: { 'Content-Type': 'application/json' },
   274→        body: JSON.stringify({
   275→          id: admin.id,
   276→          password: newPassword
   277→        })
   278→      })
   279→      
   280→      const result = await res.json()
   281→      if (res.ok) {
   282→        toast.success('✅ Contraseña actualizada correctamente')
   283→        setAdmin(prev => prev ? { ...prev, plainPassword: newPassword } : null)
   284→        setIsEditingPassword(false)
   285→        setNewPassword('')
   286→      } else {
   287→        toast.error(result.error || 'Error al actualizar contraseña')
   288→      }
   289→    } catch (error) {
   290→      console.error('[FullEditModal] Error reseteando contraseña:', error)
   291→      toast.error('Error de conexión')
   292→    }
   293→    
   294→    setResettingPassword(false)
   295→  }
   296→
   297→  // Generar contraseña aleatoria
   298→  const generateRandomPassword = () => {
   299→    const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZabcdefghjkmnpqrstuvwxyz23456789!@#$%'
   300→    let password = ''
   301→    for (let i = 0; i < 12; i++) {
   302→      password += chars.charAt(Math.floor(Math.random() * chars.length))
   303→    }
   304→    setNewPassword(password)
   305→    toast.info('Contraseña aleatoria generada')
   306→  }
   307→
   308→  const updateField = (field: string, value: string | boolean) => {
   309→    setData(prev => ({ ...prev, [field]: value }))
   310→  }
   311→
   312→  return (
   313→    <Dialog open={open} onOpenChange={onOpenChange}>
   314→      <DialogContent className="max-w-4xl max-h-[90vh] overflow-y-auto bg-white">
   315→        <DialogHeader className="bg-gradient-to-r from-blue-600 to-cyan-600 text-white p-6 rounded-t-lg -m-6 mb-4">
   316→          <DialogTitle className="text-xl font-bold flex items-center gap-3">
   317→            <Building2 className="h-6 w-6" />
   318→            ✏️ Editar Empresa: {data.name || 'Cargando...'}
   319→          </DialogTitle>
   320→          <p className="text-blue-100 text-sm mt-1">Modifica todos los datos de la empresa</p>
   321→        </DialogHeader>
   322→        
   323→        {error ? (
   324→          <div className="py-8 px-6">
   325→            <div className="bg-red-50 border border-red-200 rounded-lg p-6 text-center">
   326→              <AlertTriangle className="h-12 w-12 text-red-400 mx-auto mb-3" />
   327→              <h3 className="text-lg font-semibold text-red-700 mb-2">⚠️ {error}</h3>
   328→              {errorDetails && (
   329→                <div className="mt-3 p-3 bg-red-100 rounded-md text-left">
   330→                  <p className="text-sm text-red-600 font-mono break-all">{errorDetails}</p>
   331→                </div>
   332→              )}
   333→              <div className="flex justify-center gap-3 mt-5">
   334→                <Button 
   335→                  onClick={() => loadCompanyData(companyId!)} 
   336→                  variant="outline"
   337→                  className="border-blue-300 text-blue-600 hover:bg-blue-50"
   338→                >
   339→                  <RefreshCw className="h-4 w-4 mr-2" />
   340→                  Reintentar
   341→                </Button>
   342→                <Button 
   343→                  onClick={() => window.location.reload()} 
   344→                  variant="outline"
   345→                >
   346→                  Recargar página
   347→                </Button>
   348→              </div>
   349→            </div>
   350→          </div>
   351→        ) : loading ? (
   352→          <div className="flex flex-col items-center justify-center py-12">
   353→            <Loader2 className="h-10 w-10 text-blue-500 animate-spin mb-3" />
   354→            <p className="text-slate-600 font-medium">Cargando datos de la empresa...</p>
   355→            <p className="text-slate-400 text-sm mt-1">Por favor espera un momento</p>
   356→          </div>
   357→        ) : (
   358→          <div className="space-y-5 p-2">
   359→            
   360→            {/* 👤 SECCIÓN ADMINISTRADOR ASIGNADO */}
   361→            <section className="bg-gradient-to-r from-violet-50 to-purple-50 border border-violet-200 rounded-lg p-4">
   362→              <h3 className="font-bold text-violet-700 mb-3 flex items-center gap-2">
   363→                <Shield className="h-5 w-5" />
   364→                👤 Administrador Asignado
   365→              </h3>
   366→              {admin ? (
   367→                <div className="bg-white rounded-lg p-4 border border-violet-100 space-y-4">
   368→                  {/* Fila principal: Nombre y Email */}
   369→                  <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   370→                    <div className="flex items-center gap-3 bg-slate-50 p-3 rounded-lg">
   371→                      <div className="w-10 h-10 bg-violet-100 rounded-full flex items-center justify-center">
   372→                        <Users className="h-5 w-5 text-violet-600" />
   373→                      </div>
   374→                      <div className="min-w-0">
   375→                        <p className="text-[10px] text-slate-500 uppercase font-semibold">Nombre completo</p>
   376→                        <p className="font-bold text-slate-800 truncate">{admin.name}</p>
   377→                      </div>
   378→                    </div>
   379→                    <div className="flex items-center gap-3 bg-slate-50 p-3 rounded-lg">
   380→                      <div className="w-10 h-10 bg-blue-100 rounded-full flex items-center justify-center">
   381→                        <Mail className="h-5 w-5 text-blue-600" />
   382→                      </div>
   383→                      <div className="min-w-0">
   384→                        <p className="text-[10px] text-slate-500 uppercase font-semibold">Email de acceso</p>
   385→                        <p className="font-medium text-slate-800 truncate">{admin.email}</p>
   386→                      </div>
   387→                    </div>
   388→                  </div>
   389→
   390→                  {/* Fila de estado y rol */}
   391→                  <div className="flex flex-wrap items-center gap-3 pt-3 border-t border-violet-100">
   392→                    <Badge className={`${
   393→                      admin.active 
   394→                        ? 'bg-green-100 text-green-700 border-green-300' 
   395→                        : 'bg-red-100 text-red-700 border-red-300'
   396→                    } px-3 py-1`}>
   397→                      {admin.active ? (
   398→                        <><CheckCircle2 className="h-3 w-3 mr-1" /> Cuenta Activa</>
   399→                      ) : (
   400→                        <><EyeOff className="h-3 w-3 mr-1" /> Cuenta Inactiva</>
   401→                      )}
   402→                    </Badge>
   403→                    <Badge className="bg-violet-100 text-violet-700 border-violet-300 px-3 py-1">
   404→                      <Key className="h-3 w-3 mr-1" />
   405→                      Rol: {admin.role === 'admin' ? 'Administrador' : admin.role}
   406→                    </Badge>
   407→                  </div>
   408→
   409→                  {/* Sección de Credenciales */}
   410→                  <div className="mt-3 p-4 bg-gradient-to-r from-amber-50 to-orange-50 border border-amber-200 rounded-lg">
   411→                    <h4 className="font-semibold text-amber-800 mb-3 flex items-center gap-2 text-sm">
   412→                      <Lock className="h-4 w-4" />
   413→                      🔐 Credenciales de Acceso
   414→                    </h4>
   415→                    
   416→                    {/* Usuario/Email */}
   417→                    <div className="mb-3">
   418→                      <p className="text-[10px] text-amber-600 uppercase font-semibold mb-1">Usuario (Email)</p>
   419→                      <div className="bg-white p-2.5 rounded border border-amber-200 flex items-center gap-2">
   420→                        <Mail className="h-4 w-4 text-amber-500 shrink-0" />
   421→                        <p className="font-mono text-slate-700 font-medium text-sm">{admin.email}</p>
   422→                      </div>
   423→                    </div>
   424→
   425→                    {/* Contraseña */}
   426→                    <div className="mb-3">
   427→                      <p className="text-[10px] text-amber-600 uppercase font-semibold mb-1">Contraseña</p>
   428→                      
   429→                      {isEditingPassword ? (
   430→                        // Modo edición de contraseña
   431→                        <div className="space-y-2">
   432→                          <div className="flex gap-2">
   433→                            <Input
   434→                              type={showPassword ? "text" : "password"}
   435→                              value={newPassword}
   436→                              onChange={(e) => setNewPassword(e.target.value)}
   437→                              className="bg-white border-amber-300 font-mono text-sm"
   438→                              placeholder="Nueva contraseña"
   439→                            />
   440→                            <Button
   441→                              type="button"
   442→                              variant="outline"
   443→                              size="sm"
   444→                              onClick={() => setShowPassword(!showPassword)}
   445→                              className="shrink-0 border-amber-300 text-amber-600 hover:bg-amber-50"
   446→                            >
   447→                              {showPassword ? <EyeOff className="h-4 w-4" /> : <Eye className="h-4 w-4" />}
   448→                            </Button>
   449→                          </div>
   450→                          <div className="flex flex-wrap gap-2">
   451→                            <Button
   452→                              type="button"
   453→                              variant="outline"
   454→                              size="sm"
   455→                              onClick={generateRandomPassword}
   456→                              className="text-xs border-blue-300 text-blue-600 hover:bg-blue-50"
   457→                            >
   458→                              🎲 Generar aleatoria
   459→                            </Button>
   460→                            <Button
   461→                              type="button"
   462→                              variant="outline"
   463→                              size="sm"
   464→                              onClick={() => setIsEditingPassword(false)}
   465→                              className="text-xs border-slate-300 text-slate-600 hover:bg-slate-50"
   466→                            >
   467→                              Cancelar
   468→                            </Button>
   469→                            <Button
   470→                              type="button"
   471→                              size="sm"
   472→                              onClick={handleResetPassword}
   473→                              disabled={resettingPassword || !newPassword || newPassword.length < 6}
   474→                              className="text-xs bg-green-600 hover:bg-green-700 text-white"
   475→                            >
   476→                              {resettingPassword ? (
   477→                                <><Loader2 className="h-3 w-3 mr-1 animate-spin" /> Guardando...</>
   478→                              ) : (
   479→                                <><Save className="h-3 w-3 mr-1" /> Guardar</>
   480→                              )}
   481→                            </Button>
   482→                          </div>
   483→                        </div>
   484→                      ) : (
   485→                        // Modo visualización de contraseña
   486→                        <div className="flex gap-2">
   487→                          <div className="flex-1 bg-white p-2.5 rounded border border-amber-200 flex items-center gap-2">
   488→                            <Key className="h-4 w-4 text-amber-500 shrink-0" />
   489→                            <p className="font-mono text-slate-700 font-medium text-sm flex-1 break-all">
   490→                              {showPassword ? (admin.plainPassword || '••••••••') : '••••••••••••'}
   491→                            </p>
   492→                          </div>
   493→                          <Button
   494→                            type="button"
   495→                            variant="outline"
   496→                            size="sm"
   497→                            onClick={() => setShowPassword(!showPassword)}
   498→                            className="shrink-0 border-amber-300 text-amber-600 hover:bg-amber-50"
   499→                            title={showPassword ? "Ocultar contraseña" : "Mostrar contraseña"}
   500→                          >
   501→                            {showPassword ? <EyeOff className="h-4 w-4" /> : <Eye className="h-4 w-4" />}
   502→                          </Button>
   503→                          <Button
   504→                            type="button"
   505→                            variant="outline"
   506→                            size="sm"
   507→                            onClick={handleResetPassword}
   508→                            className="shrink-0 border-blue-300 text-blue-600 hover:bg-blue-50"
   509→                            title="Restablecer/cambiar contraseña"
   510→                          >
   511→                            <RotateCcw className="h-4 w-4" />
   512→                          </Button>
   513→                        </div>
   514→                      )}
   515→                    </div>
   516→
   517→                    {!isEditingPassword && (
   518→                      <p className="text-xs text-amber-600 mt-2 flex items-center gap-1">
   519→                        <AlertTriangle className="h-3 w-3" />
   520→                        Click en 👁️ para ver contraseña o 🔄 para cambiarla
   521→                      </p>
   522→                    )}
   523→                  </div>
   524→
   525→                  {/* Fecha de asignación */}
   526→                  <div className="text-xs text-slate-500 pt-2 border-t border-violet-100 flex items-center justify-between">
