    63→    63→    63→
    64→    64→    64→export default function FotosModal({ open, onClose, sStep, miniStep }: FotosModalProps) {
    65→    65→    65→  const { fetchProgress, currentUser, adminFreeNavigation, currentProject, currentZone, canPerform, canView, hasPermission } = use5SStore();
    66→    66→    66→  const sStepData = S_STEPS.find(s => s.id === sStep);
    67→    67→    67→  const miniStepData = MINI_STEPS.find(m => m.id === miniStep);
    68→    68→    68→
    69→    69→    69→  const fileInputRef = useRef<HTMLInputElement>(null);
    70→    70→    70→  const cameraInputRef = useRef<HTMLInputElement>(null);
    71→    71→    71→  const videoRef = useRef<HTMLVideoElement>(null);
    72→    72→    72→  const canvasRef = useRef<HTMLCanvasElement>(null);
    73→    73→    73→
    74→    74→    74→  const canSkipSteps = hasPermission('skip_steps');
    75→    75→    75→  const canPerformStep = canPerform(sStep, miniStep);
    76→    76→    76→  const canViewStep = canView(sStep, miniStep);
    77→    77→    77→  // Permission-driven: read-only if no execute perm OR if candado closed for skip_steps users
    78→    78→    78→  const isReadOnly = !canPerformStep || (canSkipSteps && !adminFreeNavigation);
    79→    79→    79→
    80→    80→    80→  const [isFullscreen, setIsFullscreen] = useState(true);
    81→    81→    81→  const [photos, setPhotos] = useState<PhotoItem[]>([]);
    82→    82→    82→  const [isSubmitting, setIsSubmitting] = useState(false);
    83→    83→    83→  const [isCompleted, setIsCompleted] = useState(false);
    84→    84→    84→  const [cameraMode, setCameraMode] = useState<'idle' | 'streaming' | 'capturing'>('idle');
    85→    85→    85→  const [stream, setStream] = useState<MediaStream | null>(null);
    86→    86→    86→  const [facingMode, setFacingMode] = useState<'environment' | 'user'>('environment');
    87→    87→    87→  const [cameraError, setCameraError] = useState<string | null>(null);
    88→    88→    88→  const [captureFlash, setCaptureFlash] = useState(false);
    89→    89→    89→  const [activeTab, setActiveTab] = useState<'camera' | 'upload'>('camera');
    90→    90→    90→  const [defaultPhotoType, setDefaultPhotoType] = useState<string>('antes');
    91→    91→    91→
    92→    92→    92→  const beforePrompt = BEFORE_PROMPT_BY_S[sStep] || 'Fotografía la zona para documentar su estado actual.';
    93→    93→    93→
    94→    94→    94→  // Generate auto title with traceability
    95→    95→    95→  const generatePhotoTitle = (index: number, photoType: string): string => {
    96→    96→    96→    const sName = sStepData?.japaneseName || `S${sStep}`
    97→    97→    97→    const zoneName = currentZone?.name || 'Zona'
    98→    98→    98→    const typeLabel = photoType === 'antes' ? 'ANTES' : photoType === 'despues' ? 'DESPUÉS' : photoType === 'referencia' ? 'Referencia' : 'Hallazgo'
    99→    99→    99→    const date = new Date().toLocaleDateString('es-ES')
   100→   100→   100→    return `S${sStep} ${sName} - ${zoneName} - ${typeLabel} ${index + 1} (${date})`
   101→   101→   101→  }
   102→   102→   102→
   103→   103→   103→  const uploadPhoto = async (base64Data: string, index: number): Promise<string | null> => {
   104→   104→   104→    try {
   105→   105→   105→      const projectId = currentProject?.id || 'unknown';
   106→   106→   106→      const filename = generatePhotoFilename(projectId, sStep, miniStep, index);
   107→   107→   107→      const file = base64toFile(base64Data, filename);
   108→   108→   108→
   109→   109→   109→      const formData = new FormData();
   110→   110→   110→      formData.append('file', file);
   111→   111→   111→      formData.append('filename', filename);
   112→   112→   112→
   113→   113→   113→      const res = await fetch('/api/upload', {
   114→   114→   114→        method: 'POST',
   115→   115→   115→        body: formData,
   116→   116→   116→      });
   117→   117→   117→
   118→   118→   118→      const data = await res.json();
   119→   119→   119→      if (data.success && data.url) return data.url;
   120→   120→   120→      return null;
   121→   121→   121→    } catch (error) {
   122→   122→   122→      console.error('Upload error:', error);
   123→   123→   123→      return null;
   124→   124→   124→    }
   125→   125→   125→  };
   126→   126→   126→
   127→   127→   127→  const addPhoto = useCallback(async (rawBase64: string) => {
   128→   128→   128→    try {
   129→   129→   129→      const compressed = await compressImage(rawBase64);
   130→   130→   130→      const estimatedSize = estimateBase64Size(compressed);
   131→   131→   131→      const photoType = defaultPhotoType;
   132→   132→   132→
   133→   133→   133→      const newPhoto: PhotoItem = {
   134→   134→   134→        preview: compressed,
   135→   135→   135→        serverUrl: '',
   136→   136→   136→        uploaded: false,
   137→   137→   137→        uploading: true,
   138→   138→   138→        estimatedSize,
   139→   139→   139→        title: generatePhotoTitle(photos.length, photoType),
   140→   140→   140→        photoType,
   141→   141→   141→        savedToLibrary: false,
   142→   142→   142→      };
   143→   143→   143→
   144→   144→   144→      setPhotos(prev => [...prev, newPhoto]);
   145→   145→   145→
   146→   146→   146→      const url = await uploadPhoto(compressed, photos.length);
   147→   147→   147→
   148→   148→   148→      setPhotos(prev =>
   149→   149→   149→        prev.map((p, i) =>
   150→   150→   150→          i === prev.length - 1 && p.uploading
   151→   151→   151→            ? { ...p, serverUrl: url || '', uploaded: !!url, uploading: false }
   152→   152→   152→            : p
   153→   153→   153→        )
   154→   154→   154→      );
   155→   155→   155→    } catch (error) {
   156→   156→   156→      console.error('Error processing photo:', error);
   157→   157→   157→    }
   158→   158→   158→  }, [sStep, miniStep, currentProject?.id, photos.length, defaultPhotoType]);
   159→   159→   159→
   160→   160→   160→  const stopStream = useCallback(() => {
   161→   161→   161→    if (stream) {
   162→   162→   162→      stream.getTracks().forEach(track => track.stop());
   163→   163→   163→      setStream(null);
   164→   164→   164→    }
   165→   165→   165→    setCameraMode('idle');
   166→   166→   166→  }, [stream]);
   167→   167→   167→
   168→   168→   168→  const startCamera = useCallback(async () => {
   169→   169→   169→    setCameraError(null);
   170→   170→   170→    try {
   171→   171→   171→      const mediaStream = await navigator.mediaDevices.getUserMedia({
   172→   172→   172→        video: { facingMode, width: { ideal: 1920 }, height: { ideal: 1080 } },
   173→   173→   173→        audio: false,
   174→   174→   174→      });
   175→   175→   175→      setStream(mediaStream);
   176→   176→   176→      setCameraMode('streaming');
   177→   177→   177→      if (videoRef.current) {
   178→   178→   178→        videoRef.current.srcObject = mediaStream;
   179→   179→   179→        await videoRef.current.play();
   180→   180→   180→      }
   181→   181→   181→    } catch (err: any) {
   182→   182→   182→      if (err.name === 'NotAllowedError') setCameraError('Permiso denegado. Permite el acceso a la cámara.');
   183→   183→   183→      else if (err.name === 'NotFoundError') setCameraError('No se encontró ninguna cámara.');
   184→   184→   184→      else setCameraError('Error al acceder a la cámara. Intenta subir una foto desde tu galería.');
   185→   185→   185→    }
   186→   186→   186→  }, [facingMode]);
   187→   187→   187→
   188→   188→   188→  const switchCamera = useCallback(async () => {
   189→   189→   189→    stopStream();
   190→   190→   190→    setFacingMode(prev => (prev === 'environment' ? 'user' : 'environment'));
   191→   191→   191→  }, [stopStream]);
   192→   192→   192→
   193→   193→   193→  const capturePhoto = useCallback(() => {
   194→   194→   194→    if (!videoRef.current || !canvasRef.current) return;
   195→   195→   195→    const video = videoRef.current;
   196→   196→   196→    const canvas = canvasRef.current;
   197→   197→   197→    canvas.width = video.videoWidth;
   198→   198→   198→    canvas.height = video.videoHeight;
   199→   199→   199→    const ctx = canvas.getContext('2d');
   200→   200→   200→    if (!ctx) return;
   201→   201→   201→    if (facingMode === 'user') { ctx.translate(canvas.width, 0); ctx.scale(-1, 1); }
   202→   202→   202→    ctx.drawImage(video, 0, 0);
   203→   203→   203→    const base64 = canvas.toDataURL('image/jpeg', 0.9);
   204→   204→   204→    setCaptureFlash(true);
   205→   205→   205→    setTimeout(() => setCaptureFlash(false), 300);
   206→   206→   206→    addPhoto(base64);
   207→   207→   207→  }, [facingMode, addPhoto]);
   208→   208→   208→
   209→   209→   209→  useEffect(() => {
   210→   210→   210→    if (!open) {
   211→   211→   211→      stopStream();
   212→   212→   212→      setPhotos([]);
   213→   213→   213→      setIsCompleted(false);
   214→   214→   214→      setCameraError(null);
   215→   215→   215→      setCameraMode('idle');
   216→   216→   216→      setActiveTab('camera');
   217→   217→   217→    }
   218→   218→   218→  }, [open, stopStream]);
   219→   219→   219→
   220→   220→   220→  const handleFileSelect = async (e: React.ChangeEvent<HTMLInputElement>) => {
   221→   221→   221→    const files = e.target.files;
   222→   222→   222→    if (!files) return;
   223→   223→   223→    for (const file of Array.from(files)) {
   224→   224→   224→      const reader = new FileReader();
   225→   225→   225→      reader.onloadend = async () => { await addPhoto(reader.result as string); };
   226→   226→   226→      reader.readAsDataURL(file);
   227→   227→   227→    }
   228→   228→   228→    if (fileInputRef.current) fileInputRef.current.value = '';
   229→   229→   229→  };
   230→   230→   230→
   231→   231→   231→  const handleCameraCapture = async (e: React.ChangeEvent<HTMLInputElement>) => {
   232→   232→   232→    const files = e.target.files;
   233→   233→   233→    if (!files) return;
   234→   234→   234→    for (const file of Array.from(files)) {
   235→   235→   235→      const reader = new FileReader();
   236→   236→   236→      reader.onloadend = async () => { await addPhoto(reader.result as string); };
   237→   237→   237→      reader.readAsDataURL(file);
   238→   238→   238→    }
   239→   239→   239→    if (cameraInputRef.current) cameraInputRef.current.value = '';
   240→   240→   240→  };
   241→   241→   241→
   242→   242→   242→  const removePhoto = (index: number) => setPhotos(prev => prev.filter((_, i) => i !== index));
   243→   243→   243→
   244→   244→   244→  const uploadingCount = photos.filter(p => p.uploading).length;
   245→   245→   245→  const canSubmit = photos.length >= MIN_PHOTOS && uploadingCount === 0;
   246→   246→   246→  const totalSize = photos.reduce((sum, p) => sum + p.estimatedSize, 0);
   247→   247→   247→
   248→   248→   248→  const handleSubmit = async () => {
   249→   249→   249→    if (!canSubmit) return;
   250→   250→   250→    setIsSubmitting(true);
   251→   251→   251→    try {
   252→   252→   252→      const urls = photos.map(p => p.serverUrl || p.preview).join(',');
   253→   253→   253→
   254→   254→   254→      // Save each photo to the PhotoLibrary with full traceability
   255→   255→   255→      const libraryPromises = photos
   256→   256→   256→        .filter(p => p.serverUrl && !p.savedToLibrary)
   257→   257→   257→        .map((p, idx) => {
   258→   258→   258→          const allUploadedBefore = photos.slice(0, idx).filter(pp => pp.serverUrl).length;
   259→   259→   259→          return fetch('/api/photo-library', {
   260→   260→   260→            method: 'POST',
   261→   261→   261→            headers: { 'Content-Type': 'application/json' },
   262→   262→   262→            body: JSON.stringify({
   263→   263→   263→              sStep,
   264→   264→   264→              miniStep: 2,
   265→   265→   265→              title: p.title || generatePhotoTitle(allUploadedBefore, p.photoType),
   266→   266→   266→              description: `${sStepData?.japaneseName || 'S' + sStep} - ${currentZone?.name || 'Zona'} - Paso 2 Fotos - Subida por ${currentUser?.name || 'Usuario'}`,
   267→   267→   267→              photoUrl: p.serverUrl,
   268→   268→   268→              photoType: p.photoType || 'antes',
   269→   269→   269→              category: `paso2_s${sStep}`,
   270→   270→   270→              tags: JSON.stringify([`S${sStep}`, sStepData?.japaneseName || '', currentZone?.name || '', `paso2`, p.photoType]),
   271→   271→   271→              projectId: currentProject?.id,
   272→   272→   272→              zoneId: currentZone?.id || null,
   273→   273→   273→              uploadedBy: currentUser?.id || null,
   274→   274→   274→            }),
   275→   275→   275→          });
   276→   276→   276→        });
   277→   277→   277→      
   278→   278→   278→      // Save to library in parallel (don't block if it fails)
   279→   279→   279→      Promise.allSettled(libraryPromises).then(results => {
   280→   280→   280→        const saved = results.filter(r => r.status === 'fulfilled').length;
   281→   281→   281→        const failed = results.filter(r => r.status === 'rejected').length;
   282→   282→   282→        if (failed > 0) console.warn(`[FotosModal] ${failed} photos failed to save to library`);
   283→   283→   283→        else if (saved > 0) console.log(`[FotosModal] ${saved} photos saved to library`);
   284→   284→   284→      });
   285→   285→   285→
   286→   286→   286→      // Save progress
   287→   287→   287→      const res = await fetch(`/api/progress/step?sStep=${sStep}&miniStep=${miniStep}`, {
   288→   288→   288→        method: 'PUT',
   289→   289→   289→        headers: { 'Content-Type': 'application/json' },
   290→   290→   290→        body: JSON.stringify({ completed: true, photoUrls: urls, score: 100, projectId: currentProject?.id, zoneId: currentZone?.id || null }),
   291→   291→   291→      });
   292→   292→   292→      const json = await res.json();
   293→   293→   293→      if (json.success) { setIsCompleted(true); stopStream(); await fetchProgress(); }
   294→   294→   294→    } catch (error) {
   295→   295→   295→      console.error('Error submitting photos:', error);
   296→   296→   296→    } finally {
   297→   297→   297→      setIsSubmitting(false);
   298→   298→   298→    }
   299→   299→   299→  };
   300→   300→   300→
   301→   301→   301→  const handleAdminSkip = async () => {
   302→   302→   302→    try {
   303→   303→   303→      const res = await fetch(`/api/progress/step?sStep=${sStep}&miniStep=${miniStep}`, {
   304→   304→   304→        method: 'PUT',
   305→   305→   305→        headers: { 'Content-Type': 'application/json' },
   306→   306→   306→        body: JSON.stringify({ completed: true, score: 100, notes: 'Completado por administrador (skip)', projectId: currentProject?.id, zoneId: currentZone?.id || null }),
   307→   307→   307→      });
   308→   308→   308→      const json = await res.json();
   309→   309→   309→      if (json.success) { await fetchProgress(); onClose(); }
   310→   310→   310→    } catch (error) { console.error('Error admin skip:', error); }
   311→   311→   311→  };
   312→   312→   312→
   313→   313→   313→  const isMobile = typeof navigator !== 'undefined' && /Android|webOS|iPhone|iPad|iPod|BlackBerry|IEMobile|Opera Mini/i.test(navigator.userAgent);
   314→   314→   314→
   315→   315→   315→  return (
   316→   316→   316→    <Dialog open={open} onOpenChange={() => { stopStream(); onClose(); }}>
   317→   317→   317→      <DialogContent size={isFullscreen ? "fullscreen" : "xl"} className="flex flex-col overflow-hidden p-0">
   318→   318→   318→        <DialogHeader className="px-6 pt-6 pb-2 flex-shrink-0">
   319→   319→   319→          <DialogTitle className="flex items-center gap-2">
   320→   320→   320→            <Camera className="h-5 w-5" style={{ color: sStepData?.color }} />
   321→   321→   321→            <span>Fotografías (Antes)</span>
   322→   322→   322→            <Badge variant="outline" style={{ borderColor: sStepData?.color, color: sStepData?.color }}>
   323→   323→   323→              {sStepData?.name}
   324→   324→   324→            </Badge>
   325→   325→   325→            <button
   326→   326→   326→              onClick={() => setIsFullscreen(!isFullscreen)}
   327→   327→   327→              className="ml-auto p-1 rounded hover:bg-muted transition-colors"
   328→   328→   328→              title={isFullscreen ? "Reducir ventana" : "Pantalla completa"}
   329→   329→   329→            >
   330→   330→   330→              {isFullscreen ? <Minimize2 className="h-4 w-4 text-muted-foreground" /> : <Maximize2 className="h-4 w-4 text-muted-foreground" />}
   331→   331→   331→            </button>
   332→   332→   332→          </DialogTitle>
   333→   333→   333→        </DialogHeader>
   334→   334→   334→
   335→   335→   335→        {canSkipSteps && !isCompleted && (
   336→   336→   336→          <div className="flex items-center gap-2 p-2 mx-6 flex-shrink-0 bg-amber-50 border border-amber-200 rounded-lg">
   337→   337→   337→            <span className="text-xs text-amber-700 font-medium">Modo Admin:</span>
   338→   338→   338→            <Button variant="outline" size="sm" className="text-xs border-amber-300 text-amber-700 hover:bg-amber-100" onClick={handleAdminSkip}>
   339→   339→   339→              Completar paso sin subir fotos
   340→   340→   340→            </Button>
   341→   341→   341→          </div>
   342→   342→   342→        )}
   343→   343→   343→
   344→   344→   344→        {isReadOnly && (
   345→   345→   345→          <div className="flex items-center gap-2 p-2 mx-6 flex-shrink-0 bg-blue-50 border border-blue-200 rounded-lg">
   346→   346→   346→            <span className="text-xs text-blue-700 font-medium">Solo lectura: {canSkipSteps ? 'Activa el candado para poder realizar pasos.' : 'Puedes ver pero no modificar.'}</span>
   347→   347→   347→          </div>
   348→   348→   348→        )}
   349→   349→   349→
   350→   350→   350→        <div className="flex-1 overflow-y-auto px-6 pb-6 min-h-0">
   351→   351→   351→        {isCompleted ? (
   352→   352→   352→          <div className="text-center py-8">
   353→   353→   353→            <CheckCircle className="h-16 w-16 text-green-500 mx-auto mb-3" />
   354→   354→   354→            <h3 className="text-xl font-bold mb-2">Fotografías del ANTES Guardadas</h3>
   355→   355→   355→            <p className="text-muted-foreground">Ha guardado {photos.length} fotos como evidencia del estado inicial.</p>
   356→   356→   356→            <p className="text-xs text-muted-foreground mt-2">Tamaño total optimizado: {formatBytes(totalSize)}</p>
   357→   357→   357→          </div>
   358→   358→   358→        ) : (
