   142→// ─── Field helper for Datos Empresa ─────────────────────────────────────────
   143→// Muestra el valor en modo lectura o un Input editable según `editing`.
   144→function Field({
   145→  label, k, value, edit, setEdit, editing,
   146→}: {
   147→  label: string
   148→  k: string
   149→  value: string | null | undefined
   150→  edit: Record<string, string>
   151→  setEdit: React.Dispatch<React.SetStateAction<Record<string, string>>>
   152→  editing: boolean
   153→}) {
   154→  return (
   155→    <div className="space-y-1">
   156→      <Label className="text-xs text-muted-foreground">{label}</Label>
   157→      {editing ? (
   158→        <Input
   159→          value={edit[k] ?? ''}
   160→          onChange={e => setEdit(prev => ({ ...prev, [k]: e.target.value }))}
   161→          className="h-8 text-sm"
   162→          placeholder={label}
   163→        />
   164→      ) : (
   165→        <div className={`h-8 px-2 flex items-center text-sm rounded-md border border-transparent bg-gray-50 ${
   166→          (value ?? '').trim() === '' ? 'text-muted-foreground italic' : 'text-gray-900'
   167→        }`}>
   168→          {(value ?? '').trim() === '' ? '— sin rellenar —' : value}
   169→        </div>
   170→      )}
   171→    </div>
   172→  )
   173→}

   429→    const email = newCompanyUserEmail.trim().toLowerCase()

   438→        body: JSON.stringify({ name, email, password, role: newCompanyUserRole, active: true }),
   439→      })
   440→      const createData = await createRes.json()
   441→      let userId: string
   442→      if (createRes.ok && createData.user) {
   443→        userId = createData.user.id
   444→      } else if (
   445→        createRes.status === 400 &&
   446→        typeof createData.error === 'string' &&
   447→        createData.error.toLowerCase().includes('email')
   448→      ) {
   449→        // Email ya existe — hacer lookup y reutilizar
   450→        const lookupRes = await fetch(`/api/users/lookup-by-email?email=${encodeURIComponent(email)}`)
   451→        const lookupData = await lookupRes.json()
   452→        if (!lookupData.found || !lookupData.user) {
   453→          alert('Ya existe un usuario con ese email pero no se pudo localizar. Usa otro email.')
   454→          return
   455→        }
   456→        userId = lookupData.user.id
   457→      } else {
   458→        alert(createData.error || 'Error al crear el usuario')
   459→        return
   460→      }
   461→
   462→      // 2. Ligar a la empresa (si no lo está ya)
   463→      await fetch(`/api/companies/${myCompany.id}/members`, {
   464→        method: 'POST',
   465→        headers: { 'Content-Type': 'application/json' },
   466→        body: JSON.stringify({ userId, role: newCompanyUserRole }),
   467→      })
