     5→// Helper para generar IDs únicos (el modelo Template requiere id manual)
     6→function generateTemplateId(type: string, sStep: number): string {
     7→  const timestamp = Date.now()
     8→  const random = Math.random().toString(36).substring(2, 8)
     9→  return `tpl_${type}_s${sStep}_${timestamp}_${random}`
    10→}
