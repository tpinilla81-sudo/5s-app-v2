    98→export default function StandardsLibrary({ open, onClose }: StandardsLibraryProps) {
    99→  const { currentProject, currentZone, currentUser } = use5SStore()
   100→  const [standards, setStandards] = useState<StandardItem[]>([])
   101→  const [isLoading, setIsLoading] = useState(false)
   102→  const [filterS, setFilterS] = useState<number | null>(null)
   103→  const [filterCategory, setFilterCategory] = useState<string>('all')
   104→  const [isCreating, setIsCreating] = useState(false)
   105→  const [editingId, setEditingId] = useState<string | null>(null)
   106→  const [showLayoutEditor, setShowLayoutEditor] = useState(false)
   107→  const [showColorCodeTable, setShowColorCodeTable] = useState(false)
   108→  const [viewingStandard, setViewingStandard] = useState<StandardItem | null>(null)
   109→
   110→  // Template-driven fields
   111→  const [templateFields, setTemplateFields] = useState<StdTemplateField[]>([])
   112→  const [templateLoaded, setTemplateLoaded] = useState(false)
   113→
   114→  // Form state
   115→  const [formSStep, setFormSStep] = useState(1)
   116→  const [formTitle, setFormTitle] = useState('')
   117→  const [formDescription, setFormDescription] = useState('')
   118→  const [formCategory, setFormCategory] = useState('formato_mejora')
   119→  const [formContent, setFormContent] = useState('')
   120→  const [formStatus, setFormStatus] = useState('activo')
   121→  const [formVersion, setFormVersion] = useState(1)
   122→  const [formBeforePhotoUrl, setFormBeforePhotoUrl] = useState<string | null>(null)
   123→  const [formAfterPhotoUrl, setFormAfterPhotoUrl] = useState<string | null>(null)
   124→  const [formResponsable, setFormResponsable] = useState('')
   125→  const [formContacto, setFormContacto] = useState('')
   126→  const [formMejoraTipo, setFormMejoraTipo] = useState<string>('')
   127→  // Extra fields for fotos_antes template
   128→  const [formZona, setFormZona] = useState('')
   129→  const [formFecha, setFormFecha] = useState('')
   130→  const [formObservaciones, setFormObservaciones] = useState('')
   131→  const [isSaving, setIsSaving] = useState(false)
   132→  const [isUploading, setIsUploading] = useState<'before' | 'after' | null>(null)
   133→
   134→  const beforeInputRef = useRef<HTMLInputElement>(null)
   135→  const afterInputRef = useRef<HTMLInputElement>(null)
   136→
   137→  const loadStandards = useCallback(async () => {
   138→    if (!currentProject) return
   139→    setIsLoading(true)
   140→    try {
   141→      const params = new URLSearchParams({ projectId: currentProject.id })
   142→      if (filterS) params.set('sStep', String(filterS))
   143→      if (filterCategory !== 'all') params.set('category', filterCategory)
   144→      if (currentZone?.id) params.set('zoneId', currentZone.id)
   145→
   146→      const res = await fetch(`/api/standards?${params}`)
   147→      const json = await res.json()
   148→      if (json.success) {
   149→        setStandards(json.data || [])
   150→      }
   151→    } catch (e) {
   152→      console.error('Error loading standards:', e)
   153→    } finally {
   154→      setIsLoading(false)
   155→    }
   156→  }, [currentProject, filterS, filterCategory, currentZone])
   157→
   158→  // Load the estandar template for the current S step to determine required fields
   159→  const loadStandardTemplate = useCallback(async (sStep: number) => {
   160→    try {
   161→      // If the zone has a board config, fetch from that config
   162→      if (currentZone?.boardConfigId) {
   163→        const slotsRes = await fetch(`/api/board-slots?boardConfigId=${currentZone.boardConfigId}&sStep=${sStep}&miniStep=4`)
   164→        const slotsJson = await slotsRes.json()
   165→        if (slotsJson.success && slotsJson.data.length > 0) {
   166→          const slot = slotsJson.data[0]
   167→          const estandarTemplates = (slot.templates || []).filter(
   168→            (t: any) => t.template?.type === 'estandar'
   169→          )
   170→          if (estandarTemplates.length > 0) {
   171→            const content = typeof estandarTemplates[0].template.content === 'string'
   172→              ? JSON.parse(estandarTemplates[0].template.content)
   173→              : estandarTemplates[0].template.content
   174→            if (content.fields && Array.isArray(content.fields)) {
   175→              setTemplateFields(content.fields)
   176→              setTemplateLoaded(true)
   177→              return
   178→            }
   179→          }
   180→        }
   181→      }
   182→
   183→      // Fallback: global template
   184→      const res = await fetch(`/api/templates?type=estandar&sStep=${sStep}`)
   185→      const json = await res.json()
   186→      if (json.success && json.data && json.data.length > 0) {
   187→        const content = typeof json.data[0].content === 'string'
   188→          ? JSON.parse(json.data[0].content)
   189→          : json.data[0].content
   190→        if (content.fields && Array.isArray(content.fields)) {
   191→          setTemplateFields(content.fields)
   192→          setTemplateLoaded(true)
   193→          return
   194→        }
   195→      }
   196→    } catch (e) {
   197→      console.error('Error loading standard template:', e)
   198→    }
   199→    // Default fields if no template found
   200→    setTemplateFields([
   201→      { key: 'beforePhotoUrl', label: 'Foto Antes', type: 'photo', required: true },
   202→      { key: 'afterPhotoUrl', label: 'Foto Después', type: 'photo', required: true },
   203→      { key: 'responsable', label: 'Quién lo ha hecho', type: 'text', required: true },
   204→      { key: 'contacto', label: 'Contacto', type: 'text', required: true },
   205→      { key: 'mejoraTipo', label: 'Tipo de Mejora', type: 'select', options: ['seguridad', 'calidad', 'proceso', 'logistica'], required: true },
   206→    ])
   207→    setTemplateLoaded(true)
   208→  }, [currentZone?.boardConfigId])
   209→
   210→  useEffect(() => {
   211→    if (open) loadStandards()
   212→  }, [open, loadStandards])
   213→
   214→  const resetForm = () => {
   215→    setFormSStep(1)
   216→    setFormTitle('')
   217→    setFormDescription('')
   218→    setFormCategory('formato_mejora')
   219→    setFormContent('')
   220→    setFormStatus('activo')
   221→    setFormVersion(1)
   222→    setFormBeforePhotoUrl(null)
   223→    setFormAfterPhotoUrl(null)
   224→    setFormResponsable('')
   225→    setFormContacto('')
   226→    setFormMejoraTipo('')
   227→    setFormZona('')
   228→    setFormFecha('')
   229→    setFormObservaciones('')
   230→    setIsCreating(false)
   231→    setEditingId(null)
   232→  }
   233→
   234→  const handleEdit = (std: StandardItem) => {
   235→    setEditingId(std.id)
   236→    setFormSStep(std.sStep)
   237→    setFormTitle(std.title)
   238→    setFormDescription(std.description || '')
   239→    setFormCategory(std.category)
   240→    setFormContent(std.content || '')
   241→    setFormStatus(std.status)
   242→    setFormVersion(std.version)
   243→    setFormBeforePhotoUrl(std.beforePhotoUrl)
   244→    setFormAfterPhotoUrl(std.afterPhotoUrl)
   245→    setFormResponsable(std.responsable || '')
   246→    setFormContacto(std.contacto || '')
   247→    setFormMejoraTipo(std.mejoraTipo || '')
   248→    // Load extra fields from content JSON for fotos_antes
   249→    if (std.category === 'fotos_antes' && std.content) {
   250→      try {
   251→        const extra = typeof std.content === 'string' ? JSON.parse(std.content) : std.content
   252→        setFormZona(extra.zona || '')
   253→        setFormFecha(extra.fecha || '')
   254→        setFormObservaciones(extra.observaciones || '')
   255→      } catch {
   256→        setFormZona('')
   257→        setFormFecha('')
   258→        setFormObservaciones('')
   259→      }
   260→    } else {
   261→      setFormZona('')
   262→      setFormFecha('')
   263→      setFormObservaciones('')
   264→    }
   265→    setIsCreating(true)
   266→  }
   267→
   268→  const handlePhotoUpload = async (file: File, type: 'before' | 'after') => {
   269→    if (!currentProject) return
   270→    setIsUploading(type)
   271→    try {
   272→      const formData = new FormData()
   273→      formData.append('file', file)
   274→      formData.append('projectId', currentProject.id)
   275→      formData.append('filename', `${currentProject.id}_std_${type}_${Date.now()}_${file.name}`)
   276→      const res = await fetch('/api/upload', { method: 'POST', body: formData })
   277→      const json = await res.json()
   278→      if (json.success && json.url) {
   279→        if (type === 'before') setFormBeforePhotoUrl(json.url)
   280→        else setFormAfterPhotoUrl(json.url)
   281→        toast.success(`Foto ${type === 'before' ? 'ANTES' : 'DESPUÉS'} subida correctamente`)
   282→      } else {
   283→        toast.error('Error al subir la foto')
   284→      }
   285→    } catch (e) {
   286→      console.error('Upload error:', e)
   287→      toast.error('Error al subir la foto')
   288→    } finally {
   289→      setIsUploading(null)
   290→    }
   291→  }
   292→
   293→  const handleSave = async () => {
   294→    if (!currentProject || !formTitle) return
   295→    // Validate required fields from template when creating a formato_mejora
   296→    if (isFormatoMejora) {
   297→      const validationError = validateRequiredFields()
   298→      if (validationError) {
   299→        toast.error(validationError)
   300→        return
   301→      }
   302→    }
   303→    // Validate required fields for fotos_antes
   304→    if (isFotosAntes) {
   305→      if (!formBeforePhotoUrl) { toast.error('La Foto del ANTES es obligatoria'); return }
   306→      if (!formResponsable.trim()) { toast.error('El nombre de quien hace la foto es obligatorio'); return }
   307→      if (!formZona.trim()) { toast.error('La zona es obligatoria'); return }
   308→    }
   309→    setIsSaving(true)
   310→    try {
   311→      const payload: any = {
   312→        sStep: formSStep,
   313→        title: formTitle,
   314→        description: formDescription || null,
   315→        category: formCategory,
   316→        content: formContent || null,
   317→        status: formStatus,
   318→        version: formVersion,
   319→        projectId: currentProject.id,
   320→        zoneId: currentZone?.id || null,
   321→        createdBy: currentUser?.id || null,
   322→      }
   323→
   324→      // Add mejora-specific fields
   325→      if (formCategory === 'formato_mejora' || formBeforePhotoUrl || formAfterPhotoUrl || formResponsable || formMejoraTipo) {
   326→        payload.beforePhotoUrl = formBeforePhotoUrl || null
   327→        payload.afterPhotoUrl = formAfterPhotoUrl || null
   328→        payload.responsable = formResponsable || null
   329→        payload.contacto = formContacto || null
   330→        payload.mejoraTipo = formMejoraTipo || null
   331→      }
   332→
   333→      // Add fotos_antes-specific fields
   334→      if (formCategory === 'fotos_antes') {
   335→        payload.beforePhotoUrl = formBeforePhotoUrl || null
   336→        payload.responsable = formResponsable || null
   337→        payload.content = JSON.stringify({
   338→          zona: formZona,
   339→          fecha: formFecha,
   340→          observaciones: formObservaciones,
   341→        })
   342→      }
   343→
   344→      if (editingId) {
   345→        await fetch('/api/standards', {
   346→          method: 'PUT',
   347→          headers: { 'Content-Type': 'application/json' },
   348→          body: JSON.stringify({ id: editingId, ...payload }),
   349→        })
   350→        toast.success('Estándar actualizado correctamente')
   351→      } else {
   352→        await fetch('/api/standards', {
   353→          method: 'POST',
   354→          headers: { 'Content-Type': 'application/json' },
   355→          body: JSON.stringify(payload),
   356→        })
   357→        toast.success('Estándar creado correctamente')
   358→      }
   359→
   360→      resetForm()
   361→      await loadStandards()
   362→    } catch (e) {
   363→      console.error('Error saving standard:', e)
   364→      alert('Error al guardar el estándar')
   365→    } finally {
   366→      setIsSaving(false)
   367→    }
   368→  }
   369→
   370→  const handleDelete = async (id: string) => {
   371→    if (!confirm('¿Eliminar este estándar?')) return
   372→    try {
   373→      await fetch(`/api/standards?id=${id}`, { method: 'DELETE' })
   374→      await loadStandards()
   375→      toast.success('Estándar eliminado')
   376→    } catch (e) {
   377→      console.error('Error deleting standard:', e)
   378→    }
   379→  }
   380→
   381→  const getCategoryInfo = (cat: string) => CATEGORIES.find(c => c.value === cat) || CATEGORIES[CATEGORIES.length - 1]
   382→  const getStatusInfo = (status: string) => STATUS_OPTIONS.find(s => s.value === status) || STATUS_OPTIONS[0]
   383→  const getMejoraTipoInfo = (tipo: string | null) => MEJORA_TIPOS.find(m => m.value === tipo)
   384→
   385→  // Group standards by S-step
   386→  const groupedByS = S_STEPS.map(s => ({
   387→    ...s,
   388→    items: standards.filter(std => std.sStep === s.id),
   389→  }))
   390→
   391→  // Check if form is in "formato mejora" mode
   392→  const isFormatoMejora = formCategory === 'formato_mejora'
   393→  const isFotosAntes = formCategory === 'fotos_antes'
   394→
   395→  // Load template when S step changes during creation
   396→  useEffect(() => {
   397→    if (isCreating && formSStep) {
   398→      loadStandardTemplate(formSStep)
   399→    }
   400→  }, [isCreating, formSStep, loadStandardTemplate])
   401→
   402→  // Check if a field is required based on the template
   403→  const isFieldRequired = (key: string): boolean => {
   404→    const field = templateFields.find(f => f.key === key)
   405→    return field?.required ?? false
   406→  }
   407→
   408→  // Validate required fields from template before saving
   409→  const validateRequiredFields = (): string | null => {
   410→    for (const field of templateFields) {
   411→      if (!field.required) continue
   412→      if (field.key === 'beforePhotoUrl' && !formBeforePhotoUrl) return `Foto ANTES es obligatoria`
   413→      if (field.key === 'afterPhotoUrl' && !formAfterPhotoUrl) return `Foto DESPUÉS es obligatoria`
   414→      if (field.key === 'responsable' && !formResponsable.trim()) return `Responsable es obligatorio`
   415→      if (field.key === 'contacto' && !formContacto.trim()) return `Contacto es obligatorio`
   416→      if (field.key === 'mejoraTipo' && !formMejoraTipo) return `Tipo de Mejora es obligatorio`
   417→    }
   418→    return null
   419→  }
   420→
   421→  return (
   422→    <Dialog open={open} onOpenChange={(o) => { if (!o) { onClose(); resetForm(); setViewingStandard(null) } }}>
   423→      <DialogContent className="max-w-5xl max-h-[95vh] overflow-hidden flex flex-col p-0">
   424→        <DialogHeader className="px-6 py-4 border-b shrink-0">
   425→          <DialogTitle className="flex items-center gap-2 text-lg">
   426→            <BookOpen className="h-5 w-5 text-teal-600" />
   427→            Biblioteca de Estándares
   428→          </DialogTitle>
   429→          <p className="text-xs text-muted-foreground mt-1">
   430→            Registra y consulta los estándares de cada S: Formato de Mejora (antes/después), Layout, Código de colores, Señalización visual, Procedimientos...
   431→          </p>
   432→        </DialogHeader>
   433→
   434→        <div className="flex-1 overflow-hidden flex flex-col">
   435→          {/* Filters + Add button */}
   436→          <div className="px-6 py-3 border-b bg-gray-50/50 flex items-center gap-3 flex-wrap shrink-0">
   437→            <Select value={filterS ? String(filterS) : 'all'} onValueChange={v => setFilterS(v === 'all' ? null : Number(v))}>
   438→              <SelectTrigger className="w-32 h-8 text-xs"><SelectValue placeholder="Todas las S" /></SelectTrigger>
   439→              <SelectContent>
   440→                <SelectItem value="all">Todas las S</SelectItem>
   441→                {S_STEPS.map(s => <SelectItem key={s.id} value={String(s.id)}>S{s.id} — {s.japaneseName}</SelectItem>)}
   442→              </SelectContent>
   443→            </Select>
   444→            <Select value={filterCategory} onValueChange={setFilterCategory}>
   445→              <SelectTrigger className="w-52 h-8 text-xs"><SelectValue placeholder="Categoría" /></SelectTrigger>
   446→              <SelectContent>
   447→                <SelectItem value="all">Todas las categorías</SelectItem>
   448→                {CATEGORIES.map(c => <SelectItem key={c.value} value={c.value}>{c.label}</SelectItem>)}
   449→              </SelectContent>
   450→            </Select>
   451→            <div className="flex-1" />
   452→            {!isCreating && !viewingStandard && (
   453→              <>
   454→                <Button size="sm" onClick={() => setShowColorCodeTable(true)}
   455→                  className="gap-1 bg-yellow-500 hover:bg-yellow-600 text-white text-xs h-8">
   456→                  <Paintbrush className="h-3 w-3" /> Cuadro Colores
   457→                </Button>
   458→                <Button size="sm" onClick={() => setShowLayoutEditor(true)}
   459→                  className="gap-1 bg-blue-600 hover:bg-blue-700 text-white text-xs h-8">
   460→                  <PenTool className="h-3 w-3" /> Dibujar Layout
   461→                </Button>
   462→                <Button size="sm" onClick={() => { resetForm(); setFormCategory('fotos_antes'); setIsCreating(true) }}
   463→                  className="gap-1 bg-red-600 hover:bg-red-700 text-white text-xs h-8">
   464→                  <Camera className="h-3 w-3" /> Fotos Antes
   465→                </Button>
   466→                <Button size="sm" onClick={() => { resetForm(); setFormCategory('formato_mejora'); setIsCreating(true) }}
   467→                  className="gap-1 bg-teal-600 hover:bg-teal-700 text-white text-xs h-8">
   468→                  <Plus className="h-3 w-3" /> Formato Mejora
   469→                </Button>
   470→                <Button size="sm" variant="outline" onClick={() => { resetForm(); setIsCreating(true) }}
   471→                  className="gap-1 text-xs h-8">
   472→                  <Plus className="h-3 w-3" /> Otro Estándar
   473→                </Button>
   474→              </>
   475→            )}
   476→          </div>
   477→
   478→          {/* ═══════════════ VIEW STANDARD DETAIL (Formato Mejora) ═══════════════ */}
