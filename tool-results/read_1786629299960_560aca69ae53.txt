   139→// ─── Field helper for Datos Empresa ─────────────────────────────────────────
   140→// Muestra el valor en modo lectura o un Input editable según `editing`.
   141→function Field({
   142→  label, k, value, edit, setEdit, editing,
   143→}: {
   144→  label: string
   145→  k: string
   146→  value: string | null | undefined
   147→  edit: Record<string, string>
   148→  setEdit: React.Dispatch<React.SetStateAction<Record<string, string>>>
   149→  editing: boolean
   150→}) {
   151→  return (
   152→    <div className="space-y-1">
   153→      <Label className="text-xs text-muted-foreground">{label}</Label>
   154→      {editing ? (
   155→        <Input
   156→          value={edit[k] ?? ''}
   157→          onChange={e => setEdit(prev => ({ ...prev, [k]: e.target.value }))}
   158→          className="h-8 text-sm"
   159→          placeholder={label}
   160→        />
   161→      ) : (
   162→        <div className={`h-8 px-2 flex items-center text-sm rounded-md border border-transparent bg-gray-50 ${
   163→          (value ?? '').trim() === '' ? 'text-muted-foreground italic' : 'text-gray-900'
   164→        }`}>
   165→          {(value ?? '').trim() === '' ? '— sin rellenar —' : value}
   166→        </div>
   167→      )}
   168→    </div>
   169→  )
   170→}
