   127→export default function GestorPanel() {

   129→  const [activeTab, setActiveTab] = useState<GestorTab>('dashboard')
   130→
   131→  // Platform stats
   132→  const [stats, setStats] = useState<PlatformStats | null>(null)
   133→  const [isLoadingStats, setIsLoadingStats] = useState(false)
   134→
   135→  // Companies management
   136→  const [showNewCompany, setShowNewCompany] = useState(false)
   137→  const [newCompanyName, setNewCompanyName] = useState('')
   138→  const [newCompanyDesc, setNewCompanyDesc] = useState('')
   139→  const [editingCompany, setEditingCompany] = useState<string | null>(null)
   140→  const [editCompanyData, setEditCompanyData] = useState({ name: '', description: '', active: true })
   141→
   142→  // Admin management
   143→  const [assigningAdminTo, setAssigningAdminTo] = useState<string | null>(null)
   144→  const [searchEmail, setSearchEmail] = useState('')
   145→  const [searchResults, setSearchResults] = useState<PlatformStats['users']>([])
   146→  const [isSearching, setIsSearching] = useState(false)
   147→  const [showCreateUser, setShowCreateUser] = useState(false)
   148→  const [newAdminData, setNewAdminData] = useState({ name: '', email: '', password: '' })
   149→  const [isAssigning, setIsAssigning] = useState(false)
   150→  const [sendingEmailFor, setSendingEmailFor] = useState<string | null>(null) // company id being sent email
   151→  const [emailSentFor, setEmailSentFor] = useState<Set<string>>(new Set()) // company ids where email was already sent
   152→  const [pendingCredentials, setPendingCredentials] = useState<Record<string, { name: string; email: string; password: string }>>({}) // companyId -> credentials
   153→  const [deleteCompanyDialog, setDeleteCompanyDialog] = useState<{ open: boolean; companyId: string; companyName: string; projectCount: number }>({ open: false, companyId: '', companyName: '', projectCount: 0 })

   155→  // ─── Data loading ────────────────────────────────────────────────────────
   156→  const loadStats = useCallback(async () => {
   157→    setIsLoadingStats(true)
   158→    try {
   159→      const res = await fetch('/api/platform-stats')
   160→      const data = await res.json()
   161→      if (data.success) {
   162→        setStats(data.data)
   163→      }
   164→    } catch (error) {
   165→      console.error('Error loading platform stats:', error)
   166→    } finally {
   167→      setIsLoadingStats(false)
   168→    }
   169→  }, [])
   170→
   171→  useEffect(() => {
   172→    loadStats()
   173→  }, [loadStats])
   174→
   175→  // ─── Company actions ────────────────────────────────────────────────────
   176→  const handleCreateCompany = async () => {
   177→    if (!newCompanyName.trim()) return
   178→    try {
   179→      const res = await fetch('/api/companies', {
   180→        method: 'POST',
   181→        headers: { 'Content-Type': 'application/json' },
   182→        body: JSON.stringify({ name: newCompanyName, description: newCompanyDesc || undefined }),
   183→      })
   184→      if (res.ok) {
   185→        setShowNewCompany(false)
   186→        setNewCompanyName('')
   187→        setNewCompanyDesc('')
   188→        await loadStats()
   189→        await fetchCompanies()
   190→      } else {
   191→        const data = await res.json()
   192→        alert(data.error || 'Error al crear empresa')
   193→      }
   194→    } catch (error) {
   195→      console.error('Error creating company:', error)
   196→    }
   197→  }
   198→
   199→  const handleUpdateCompany = async (companyId: string) => {
   200→    try {
   201→      const res = await fetch(`/api/companies/${companyId}`, {
   202→        method: 'PUT',
   203→        headers: { 'Content-Type': 'application/json' },
   204→        body: JSON.stringify(editCompanyData),
   205→      })
   206→      if (res.ok) {
   207→        setEditingCompany(null)
   208→        await loadStats()
   209→        await fetchCompanies()
   210→      } else {
   211→        const data = await res.json()
   212→        alert(data.error || 'Error al actualizar empresa')
   213→      }
   214→    } catch (error) {
   215→      console.error('Error updating company:', error)
   216→    }
   217→  }
   218→
   219→  const handleToggleCompanyActive = async (companyId: string, currentActive: boolean) => {
   220→    try {
   221→      const res = await fetch(`/api/companies/${companyId}`, {
   222→        method: 'PUT',
   223→        headers: { 'Content-Type': 'application/json' },
   224→        body: JSON.stringify({ active: !currentActive }),
   225→      })
   226→      if (res.ok) {
   227→        await loadStats()
   228→        await fetchCompanies()
   229→      }
   230→    } catch (error) {
   231→      console.error('Error toggling company active:', error)
   232→    }
   233→  }
   234→
   235→  const handleDeleteCompany = (companyId: string) => {
   236→    const company = stats?.companies.find(c => c.id === companyId)
   237→    if (!company) return

   239→      open: true,
   240→      companyId: company.id,
   241→      companyName: company.name,
   242→      projectCount: company.projectCount,
   243→    })
   244→  }
   245→
   246→  const confirmDeleteCompany = async () => {
   247→    const { companyId } = deleteCompanyDialog
   248→    setDeleteCompanyDialog(d => ({ ...d, open: false }))
   249→    try {
   250→      const res = await fetch(`/api/companies/${companyId}`, { method: 'DELETE' })
   251→      if (res.ok) {
   252→        await loadStats()
   253→        await fetchCompanies()
   254→      } else {
   255→        const data = await res.json()
   256→        alert(data.error || 'Error al eliminar empresa')
   257→      }
   258→    } catch (error) {
   259→      console.error('Error deleting company:', error)
   260→    }
   261→  }
   262→
   263→  // ─── Admin assignment actions ───────────────────────────────────────────
   264→  const handleSearchUser = async () => {
   265→    if (!searchEmail.trim() || searchEmail.length < 3) {
   266→      setSearchResults([])
   267→      return
   268→    }
   269→    setIsSearching(true)
   270→    try {
   271→      const res = await fetch(`/api/users?search=${encodeURIComponent(searchEmail)}`)
   272→      const data = await res.json()

   274→        setSearchResults(data.users || [])
   275→      }
   276→    } catch (error) {
   277→      console.error('Error searching users:', error)
   278→    } finally {
   279→      setIsSearching(false)
   280→    }
   281→  }
   282→
   283→  useEffect(() => {
   284→    if (searchEmail.length >= 3) {
   285→      const t = setTimeout(() => handleSearchUser(), 300)
   286→      return () => clearTimeout(t)
   287→    } else {
   288→      setSearchResults([])
   289→    }
   290→  }, [searchEmail])
   291→
   292→  const handleAssignExistingUser = async (userId: string) => {
   293→    if (!assigningAdminTo) return
   294→    setIsAssigning(true)
   295→    try {
   296→      // First, ensure the user has 'admin' role
   297→      await fetch('/api/users', {
   298→        method: 'PUT',
   299→        headers: { 'Content-Type': 'application/json' },
   300→        body: JSON.stringify({ id: userId, role: 'admin' }),
   301→      })
   302→
   303→      // Then assign them to the company as admin
   304→      const res = await fetch(`/api/companies/${assigningAdminTo}/members`, {
   305→        method: 'POST',
   306→        headers: { 'Content-Type': 'application/json' },
   307→        body: JSON.stringify({ userId, role: 'admin_empresa' }),
   308→      })
   309→
   310→      if (res.ok) {
   311→        setAssigningAdminTo(null)
   312→        setSearchEmail('')
   313→        setSearchResults([])
   314→        await loadStats()
   315→      } else {
   316→        const data = await res.json()
   317→        alert(data.error || 'Error al asignar administrador')
   318→      }
   319→    } catch (error) {
   320→      console.error('Error assigning admin:', error)
   321→    } finally {
   322→      setIsAssigning(false)
   323→    }
   324→  }
   325→
   326→  const handleCreateAndAssignAdmin = async () => {
   327→    if (!assigningAdminTo) return
   328→    if (!newAdminData.name.trim() || !newAdminData.email.trim() || newAdminData.password.length < 6) {
   329→      alert('Completa todos los campos. La contraseña debe tener al menos 6 caracteres.')
   330→      return
   331→    }
   332→    setIsAssigning(true)
   333→    try {
   334→      // 1. Create user with role 'admin'
   335→      const createRes = await fetch('/api/users', {
   336→        method: 'POST',
   337→        headers: { 'Content-Type': 'application/json' },
   338→        body: JSON.stringify({
   339→          name: newAdminData.name,
   340→          email: newAdminData.email,
   341→          password: newAdminData.password,
   342→          role: 'admin',
   343→          active: true,
   344→        }),
   345→      })
   346→
   347→      if (!createRes.ok) {
   348→        const data = await createRes.json()
   349→        alert(data.error || 'Error al crear usuario')
   350→        setIsAssigning(false)
   351→        return
   352→      }
   353→
   354→      const createData = await createRes.json()
   355→      const newUserId = createData.user?.id
   356→
   357→      if (!newUserId) {
   358→        alert('Usuario creado pero no se pudo obtener el ID. Asigna el admin manualmente.')
   359→        setIsAssigning(false)
   360→        return
   361→      }
   362→
   363→      // 2. Assign user to company
   364→      const assignRes = await fetch(`/api/companies/${assigningAdminTo}/members`, {
   365→        method: 'POST',
   366→        headers: { 'Content-Type': 'application/json' },
   367→        body: JSON.stringify({ userId: newUserId, role: 'admin_empresa' }),
   368→      })
   369→
   370→      if (assignRes.ok) {
   371→        // Store credentials temporarily so the gestor can send them via email later
   372→        setPendingCredentials(prev => ({
   373→          ...prev,
   374→          [assigningAdminTo]: {
   375→            name: newAdminData.name,
   376→            email: newAdminData.email,
   377→            password: newAdminData.password,
   378→          },
   379→        }))
   380→
   381→        setAssigningAdminTo(null)
   382→        setShowCreateUser(false)
   383→        setNewAdminData({ name: '', email: '', password: '' })
   384→        await loadStats()
   385→      } else {
   386→        const data = await assignRes.json()
   387→        alert(data.error || 'Usuario creado pero error al asignar a la empresa')
   388→      }
   389→    } catch (error) {
   390→      console.error('Error creating admin:', error)
   391→    } finally {
   392→      setIsAssigning(false)
   393→    }
   394→  }
   395→
   396→  const handleRemoveAdmin = async (companyId: string, userId: string) => {
   397→    if (!confirm('¿Quitar a este administrador de la empresa?')) return
   398→    try {
   399→      const res = await fetch(`/api/companies/${companyId}/members`, {
   400→        method: 'DELETE',
   401→        headers: { 'Content-Type': 'application/json' },
   402→        body: JSON.stringify({ memberUserId: userId }),
   403→      })
   404→      if (res.ok) {
   405→        await loadStats()
   406→      } else {
   407→        const data = await res.json()
   408→        alert(data.error || 'Error al quitar administrador')
   409→      }
   410→    } catch (error) {
   411→      console.error('Error removing admin:', error)
   412→    }
   413→  }
   414→  // ─── Render ──────────────────────────────────────────────────────────────
   415→  const tabs: { key: GestorTab; label: string; icon: React.ReactNode }[] = [
   416→    { key: 'dashboard', label: 'KPIs', icon: <Activity className="h-4 w-4" /> },
   417→    { key: 'empresas', label: 'Empresas', icon: <Building2 className="h-4 w-4" /> },
   418→    { key: 'admins', label: 'Administradores', icon: <Shield className="h-4 w-4" /> },
   419→    { key: 'recursos', label: 'Recursos', icon: <Database className="h-4 w-4" /> },
   420→  ]
   421→
   422→  // ─── Send invitation email manually ────────────────────────────────────
   423→  const handleSendInvitationEmail = async (companyId: string) => {
   424→    const company = stats?.companies?.find(c => c.id === companyId)
   425→    if (!company?.adminUser) {
   426→      alert('Esta empresa no tiene administrador asignado')
   427→      return
   428→    }
   429→
   430→    const adminName = company.adminUser.name
   431→    const adminEmail = company.adminUser.email
   432→    const companyName = company.name
   433→    const gestorEmail = currentUser?.email
   434→
   435→    // Check if we have the plain-text password from creation
   436→    const credentials = pendingCredentials[companyId]
   437→    const adminPassword = credentials?.password
   438→
   439→    const hasPassword = !!adminPassword
   440→    const confirmMsg = hasPassword
   441→      ? `¿Enviar email de inscripción a ${adminName} (${adminEmail}) con sus credenciales?\nSe enviará copia al gestor (${gestorEmail}).`
   442→      : `¿Enviar email de inscripción a ${adminName} (${adminEmail})?\n\nNota: No se dispone de la contraseña en texto plano. El email indicará que debe restablecerla.\nSe enviará copia al gestor (${gestorEmail}).`
   443→
   444→    if (!confirm(confirmMsg)) return
   445→
   446→    setSendingEmailFor(companyId)
   447→    try {
   448→      const res = await fetch('/api/email', {
   449→        method: 'POST',
   450→        headers: { 'Content-Type': 'application/json' },
   451→        body: JSON.stringify({
   452→          type: 'admin_welcome',
   453→          adminName,
   454→          adminEmail,
   455→          adminPassword: adminPassword || '',
   456→          companyName,
   457→          gestorEmail,
   458→          sendCopy: true,
   459→        }),
   460→      })
   461→      const data = await res.json()
   462→      if (data.success) {
   463→        setEmailSentFor(prev => new Set(prev).add(companyId))
   464→        // Clear stored credentials after sending
   465→        setPendingCredentials(prev => {
   466→          const next = { ...prev }
   467→          delete next[companyId]
   468→          return next
   469→        })
   470→      } else {
   471→        alert(data.error || 'Error al enviar el email')
   472→      }
   473→    } catch (error) {
   474→      console.error('Error sending invitation email:', error)
   475→      alert('Error al enviar el email')
   476→    } finally {
   477→      setSendingEmailFor(null)
   478→    }
   479→  }
   480→
   481→  return (
   482→    <div className="min-h-screen bg-gradient-to-b from-slate-950 via-violet-950/20 to-slate-950 text-white">
   483→      {/* Header */}
   484→      <header className="border-b border-violet-800/30 bg-slate-950/80 backdrop-blur-sm sticky top-0 z-10">
   485→        <div className="max-w-7xl mx-auto px-4 py-3 flex items-center justify-between">
   486→          <div className="flex items-center gap-3">
   487→            <div className="w-10 h-10 rounded-xl bg-gradient-to-br from-violet-500 to-purple-700 flex items-center justify-center shadow-lg shadow-violet-500/30">
   488→              <Crown className="h-5 w-5 text-yellow-300" />
   489→            </div>
   490→            <div>
   491→              <h1 className="text-lg font-bold bg-gradient-to-r from-violet-300 to-purple-200 bg-clip-text text-transparent">
   492→                Gestor Panel
   493→              </h1>
   494→              <p className="text-xs text-violet-400">Dueño de la Plataforma</p>
   495→            </div>
   496→          </div>
   497→          <div className="flex items-center gap-2">
   498→            <Button
   499→              variant="outline"
   500→              size="sm"
   501→              onClick={() => {
   502→                if (!currentProject && projects.length > 0) {
   503→                  setCurrentProject(projects[0])
   504→                }
   505→                setCurrentView('board')
   506→              }}
   507→              className="gap-1.5 border-violet-700/50 text-violet-300 hover:bg-violet-900/30 hover:text-violet-200"
   508→            >
   509→              ← Vista App
   510→            </Button>
   511→            <Badge className="bg-gradient-to-r from-violet-600 to-purple-600 text-white border-0 px-3 py-1">
   512→              <Crown className="h-3 w-3 mr-1" /> GESTOR
   513→            </Badge>
   514→          </div>
   515→        </div>
   516→      </header>
   517→
   518→      {/* Tabs */}
   519→      <div className="border-b border-violet-800/20 bg-slate-950/60 backdrop-blur-sm">
   520→        <div className="max-w-7xl mx-auto px-4 flex gap-0 overflow-x-auto">
   521→          {tabs.map(tab => (
   522→            <button
   523→              key={tab.key}
   524→              onClick={() => setActiveTab(tab.key)}
   525→              className={`flex items-center gap-2 px-4 py-3 text-sm font-medium border-b-2 transition-all whitespace-nowrap ${
   526→                activeTab === tab.key
   527→                  ? 'border-violet-400 text-violet-300 bg-violet-900/20'
   528→                  : 'border-transparent text-violet-500 hover:text-violet-300 hover:border-violet-700/50'
   529→              }`}
   530→            >
   531→              {tab.icon}
   532→              {tab.label}
   533→            </button>
   534→          ))}
   535→        </div>
   536→      </div>
   537→
   538→      {/* Content */}
   539→      <main className="max-w-7xl mx-auto px-4 py-6">
   540→        <AnimatePresence mode="wait">
   541→          {/* ═══ DASHBOARD TAB (KPIs) ═══ */}
   542→          {activeTab === 'dashboard' && (
   543→            <motion.div key="dashboard" initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0 }} className="space-y-6">
   544→              {isLoadingStats ? (
   545→                <div className="flex justify-center py-20"><Loader2 className="h-10 w-10 text-violet-400 animate-spin" /></div>
   546→              ) : stats ? (
   547→                <>
   548→                  {/* KPI Cards */}
   549→                  <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   550→                    <Card className="bg-gradient-to-br from-violet-900/40 to-purple-900/20 border-violet-700/30 text-white">
   551→                      <CardContent className="p-4">
   552→                        <div className="flex items-center justify-between mb-2">
   553→                          <Building2 className="h-5 w-5 text-violet-400" />
   554→                          <Badge className="bg-violet-800/50 text-violet-300 border-0 text-[10px]">
   555→                            {stats.totals.activeCompanies}/{stats.totals.companies} activas
   556→                          </Badge>
   557→                        </div>
   558→                        <p className="text-2xl font-bold">{stats.totals.companies}</p>
   559→                        <p className="text-xs text-violet-400">Empresas</p>
   560→                      </CardContent>
   561→                    </Card>
   562→                    <Card className="bg-gradient-to-br from-blue-900/40 to-indigo-900/20 border-blue-700/30 text-white">
   563→                      <CardContent className="p-4">
