   428→ const handleCreateCompanyFull = async () => {
   429→  if (!newCompanyName.trim()) return
   430→  setIsCreatingCompany(true)
   431→  try {
   432→   // Step 1: Create company
   433→   const companyRes = await fetch('/api/companies', {
   434→    method: 'POST',
   435→    headers: { 'Content-Type': 'application/json' },
   436→    body: JSON.stringify({ name: newCompanyName, description: newCompanyDesc || undefined }),
   437→   })
   438→   const companyData = await companyRes.json()
   439→   if (!companyRes.ok) {
   440→    alert(companyData.error || 'Error al crear empresa')
   441→    return
   442→   }
   443→   const companyId = companyData.company?.id || companyData.id
   444→
   445→   // Step 2: Create or assign admin
   446→   let adminUserId: string | null = null
   447→   if (adminMode === 'create') {
   448→    if (!newAdminName.trim() || !newAdminEmail.trim() || !newAdminPassword.trim()) {
   449→     alert('Complete los datos del administrador')
   450→     return
   451→    }
   452→    // Create user with role 'admin' via /api/users
   453→    const userRes = await fetch('/api/users', {
   454→     method: 'POST',
   455→     headers: { 'Content-Type': 'application/json' },
   456→     body: JSON.stringify({
   457→      name: newAdminName,
   458→      email: newAdminEmail,
   459→      password: newAdminPassword,
   460→      role: 'admin',
   461→      active: true,
   462→     }),
   463→    })
   464→    const userData = await userRes.json()
   465→    if (!userRes.ok) {
   466→     alert(userData.error || 'Error al crear usuario admin')
   467→     return
   468→    }
   469→    adminUserId = userData.user?.id
   470→   } else {
   471→    // Assign existing user
   472→    if (!assignUserId) {
   473→     alert('Seleccione un usuario para asignar como administrador')
   474→     return
   475→    }
   476→    adminUserId = assignUserId
   477→    // Update user role to 'admin' if not already
   478→    const existingUser = users.find(u => u.id === assignUserId)
   479→    if (existingUser && existingUser.role !== 'admin') {
   480→     await fetch('/api/users', {
   481→      method: 'PUT',
   482→      headers: { 'Content-Type': 'application/json' },
   483→      body: JSON.stringify({ id: assignUserId, role: 'admin' }),
   484→     })
   485→    }
   486→   }
   487→
   488→   // Step 3: Add admin to company as admin_empresa
   489→   if (adminUserId) {
   490→    await fetch(`/api/companies/${companyId}/members`, {
   491→     method: 'POST',
   492→     headers: { 'Content-Type': 'application/json' },
   493→     body: JSON.stringify({ userId: adminUserId, role: 'admin_empresa' }),
   494→    })
   495→   }
   496→
   497→   // Step 4: Create subscription
   498→   await fetch('/api/subscriptions', {
   499→    method: 'POST',
   500→    headers: { 'Content-Type': 'application/json' },
   501→    body: JSON.stringify({
   502→     companyId,
   503→     plan: newSubPlan,
   504→     status: newSubStatus,
   505→     maxUsers: newSubMaxUsers,
   506→     maxProjects: newSubMaxProjects,
   507→     price: newSubPrice,
   508→     endDate: newSubEndDate || undefined,
   509→     trialEndsAt: newSubTrialEndsAt || undefined,
   510→     notes: newSubNotes || undefined,
   511→    }),
   512→   })
   513→
   514→   // Step 5: Send welcome email to admin + notification to gestor
   515→   if (adminMode === 'create' && adminUserId) {
   516→    try {
   517→     await fetch('/api/email', {
   518→      method: 'POST',
   519→      headers: { 'Content-Type': 'application/json' },
   520→      body: JSON.stringify({
   521→       type: 'admin_welcome',
   522→       adminName: newAdminName,
   523→       adminEmail: newAdminEmail,
   524→       adminPassword: newAdminPassword,
   525→       companyName: newCompanyName,
   526→       gestorEmail: currentUser?.email || undefined,
   527→      }),
   528→     })
   529→    } catch (emailErr) {
   530→     console.error('Error sending welcome email (non-blocking):', emailErr)
   531→    }
   532→   }
   533→
   534→   setShowNewCompany(false)
   535→   resetNewCompanyForm()
   536→   await loadStats()
   537→   await fetchCompanies()
   538→  } catch (error) {
   539→   console.error('Error creating company with admin:', error)
   540→   alert('Error al crear la empresa. Revise la consola para detalles.')
   541→  } finally {
   542→   setIsCreatingCompany(false)
   543→  }
   544→ }
