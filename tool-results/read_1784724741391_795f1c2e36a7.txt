     1→'use client';
     2→
     3→import { useState, useEffect, useMemo, useRef } from 'react';
     4→import {
     5→  Dialog,
     6→  DialogContent,
     7→  DialogHeader,
     8→  DialogTitle,
     9→} from '@/components/ui/dialog';
    10→import { Button } from '@/components/ui/button';
    11→import { Card, CardContent } from '@/components/ui/card';
    12→import { Badge } from '@/components/ui/badge';
    13→import { Input } from '@/components/ui/input';
    14→import { Textarea } from '@/components/ui/textarea';
    15→import { Label } from '@/components/ui/label';
    16→import {
    17→  ShieldCheck,
    18→  CheckCircle,
    19→  XCircle,
    20→  Camera,
    21→  ChevronDown,
    22→  ChevronRight,
    23→  AlertCircle,
    24→  AlertTriangle,
    25→  Plus,
    26→  Trash2,
    27→  TrendingUp,
    28→  Maximize2,
    29→  Minimize2,
    30→  Upload,
    31→  X,
    32→  Image as ImageIcon,
    33→  Calendar,
    34→} from 'lucide-react';
    35→import { use5SStore } from '@/lib/store';
    36→import {
    37→  S_STEPS,
    38→  AUDIT_PASS_THRESHOLD,
    39→} from '@/lib/5s-constants';
    40→import type { AuditSection, AuditItemResult } from '@/lib/5s-constants';
    41→import { useChecklistTemplate } from '@/lib/checklist-templates';
    42→
    43→interface AuditoriaModalProps {
    44→  open: boolean;
    45→  onClose: () => void;
    46→  sStep: number;
    47→  miniStep: number;
    48→}
    49→
    50→
    51→
    52→export default function AuditoriaModal({ open, onClose, sStep, miniStep }: AuditoriaModalProps) {
    53→  const { fetchProgress, currentUser, adminFreeNavigation, currentProject, currentZone, canPerform, hasPermission } = use5SStore();
    54→  const sStepData = S_STEPS.find(s => s.id === sStep);
    55→  const canSkipSteps = hasPermission('skip_steps');
    56→  const canAudit = canPerform(sStep, 5);
    57→
    58→  const [auditorName, setAuditorName] = useState('');
    59→  const [results, setResults] = useState<Record<string, AuditItemResult>>({});
    60→  const [expandedSections, setExpandedSections] = useState<Record<string, boolean>>({});
    61→  const [observaciones, setObservaciones] = useState('');
    62→  const [isSubmitting, setIsSubmitting] = useState(false);
    63→  const [isCompleted, setIsCompleted] = useState(false);
    64→  const [finalScore, setFinalScore] = useState(0);
    65→  const [notaMinima, setNotaMinima] = useState(75);
    66→  const [fechaAuditoria, setFechaAuditoria] = useState('');
    67→  const [horaAuditoria, setHoraAuditoria] = useState('');
    68→  const [auditPhotos, setAuditPhotos] = useState<{ file: File; preview: string; uploading?: boolean; serverUrl?: string }[]>([]);
    69→  const [isUploadingPhotos, setIsUploadingPhotos] = useState(false);
    70→  const photoInputRef = useRef<HTMLInputElement>(null);
    71→
    72→  // Scheduling for audit
    73→  const [fechaProgramada, setFechaProgramada] = useState('');
    74→  const [horaProgramada, setHoraProgramada] = useState('');
    75→
    76→  // Load template from API (uses board config if zone has one)
    77→  const { sections, isLoading: isLoadingTemplate, notaMinima: templateNotaMinima } = useChecklistTemplate('auditoria', sStep, open, currentZone?.boardConfigId);
    78→
    79→  // Apply template notaMinima when loaded
    80→  useEffect(() => {
    81→    if (templateNotaMinima !== null) setNotaMinima(templateNotaMinima);
    82→  }, [templateNotaMinima]);
    83→
    84→  // Mejoras realizadas
    85→  const [isFullscreen, setIsFullscreen] = useState(true);
    86→  const [haMejoras, setHaMejoras] = useState<boolean | null>(null);
    87→  const [mejoras, setMejoras] = useState<Array<{ id: string; descripcion: string; responsable: string; fecha: string }>>([]);
    88→
    89→
    90→
    91→  // Fetch dynamic threshold
    92→  useEffect(() => {
    93→    if (open && currentProject?.id) {
    94→      const fetchThreshold = async () => {
    95→        try {
    96→          const params = new URLSearchParams({ projectId: currentProject.id });
    97→          if (currentZone?.id) params.set('zoneId', currentZone.id);
    98→          const res = await fetch(`/api/audit-targets?${params}`);
    99→          const json = await res.json();
   100→          if (json.success && json.data) {
   101→            const zoneTarget = json.data.find((t: any) => t.sStep === sStep && t.miniStep === 5 && t.zoneId === currentZone?.id);
   102→            const projectTarget = json.data.find((t: any) => t.sStep === sStep && t.miniStep === 5 && t.zoneId === null);
   103→            const target = zoneTarget || projectTarget;
   104→            if (target?.notaMinima) setNotaMinima(target.notaMinima);
   105→          }
   106→        } catch (e) { console.error('Error fetching threshold:', e); }
   107→      };
   108→      fetchThreshold();
   109→    }
   110→  }, [open, sStep, currentProject?.id, currentZone?.id]);
   111→
   112→  useEffect(() => {
   113→    if (open && sections.length > 0) {
   114→      const expanded: Record<string, boolean> = {};
   115→      sections.forEach(s => { expanded[s.id] = true; });
   116→      setExpandedSections(expanded);
   117→      setResults({});
   118→      setAuditorName(currentUser?.name || '');
   119→      setObservaciones('');
   120→      setIsCompleted(false);
   121→      setFinalScore(0);
   122→      setHaMejoras(null);
   123→      setAuditPhotos([]);
   124→      setMejoras([]);
   125→      // Auto-fill date and time of the audit
   126→      const now = new Date();
   127→      setFechaAuditoria(now.toISOString().slice(0, 10));
   128→      setHoraAuditoria(now.toTimeString().slice(0, 5));
   129→      // Load scheduled date if available
   130→      loadScheduledDate();
   131→    }
   132→  }, [open, sStep, sections]);
   133→
   134→  // Load scheduled date/time for this auditoría
   135→  const loadScheduledDate = async () => {
   136→    if (!currentProject?.id || !currentZone?.id) return;
   137→    try {
   138→      const res = await fetch(`/api/evaluation-schedule?sStep=${sStep}&miniStep=5&projectId=${currentProject.id}&zoneId=${currentZone.id}`);
   139→      const json = await res.json();
   140→      if (json.success && json.data) {
   141→        setFechaProgramada(json.data.fechaProgramada || '');
   142→        setHoraProgramada(json.data.horaProgramada || '');
   143→      }
   144→    } catch (e) {
   145→      console.error('Error loading scheduled date:', e);
   146→    }
   147→  };
   148→
   149→  // Save scheduled date/time
   150→  const handleSaveSchedule = async () => {
   151→    if (!currentProject?.id || !currentZone?.id) return;
   152→    try {
   153→      await fetch('/api/evaluation-schedule', {
   154→        method: 'POST',
   155→        headers: { 'Content-Type': 'application/json' },
   156→        body: JSON.stringify({
   157→          sStep,
   158→          miniStep: 5,
   159→          projectId: currentProject.id,
   160→          zoneId: currentZone.id,
   161→          fechaProgramada,
   162→          horaProgramada,
   163→        }),
   164→      });
   165→      toast.success('Fecha programada guardada');
   166→    } catch (e) {
   167→      console.error('Error saving schedule:', e);
   168→      toast.error('Error al guardar la fecha programada');
   169→    }
   170→  };
   171→
   172→  const totalItems = sections.reduce((sum, s) => sum + s.items.length, 0) || 26;
   173→
   174→  const scoring = useMemo(() => {
   175→    const allResults = Object.values(results);
   176→    const okCount = allResults.filter(r => r.status === 'ok').length;
   177→    const nokCount = allResults.filter(r => r.status === 'nok').length;
   178→    const answeredCount = okCount + nokCount;
   179→    // Checklist maxes at 90%
   180→    const checklistScore = totalItems > 0 ? Math.round((okCount / totalItems) * 90) : 0;
   181→    // Each mejora with description adds 5%, max 2 mejoras = +10%
   182→    const validMejorasCount = haMejoras ? mejoras.filter(m => m.descripcion.trim()).length : 0;
   183→    const mejorasScore = Math.min(validMejorasCount, 2) * 5;
   184→    const scorePercent = Math.min(checklistScore + mejorasScore, 100); // HARD CAP: never exceeds 100%
   185→    return { okCount, nokCount, answeredCount, checklistScore, mejorasScore, validMejorasCount, scorePercent };
   186→  }, [results, totalItems, haMejoras, mejoras]);
   187→
   188→  const canSubmit = canAudit && auditorName.trim() !== '' && scoring.answeredCount > 0;
   189→
   190→  // Check that all NOK items have hallazgo and mejora filled
   191→  const nokItems = Object.values(results).filter(r => r.status === 'nok');
   192→  const allNokCompleted = nokItems.length === 0 || nokItems.every(r => (r.hallazgo || '').trim() !== '' && (r.mejora || '').trim() !== '');
   193→  const canSubmitFinal = canSubmit && allNokCompleted;
   194→
   195→  const toggleSection = (sectionId: string) => {
   196→    setExpandedSections(prev => ({ ...prev, [sectionId]: !prev[sectionId] }));
   197→  };
   198→
   199→  const setItemStatus = (itemId: string, status: 'ok' | 'nok' | 'na') => {
   200→    setResults(prev => ({
   201→      ...prev,
   202→      [itemId]: { ...prev[itemId], itemId, status },
   203→    }));
   204→  };
   205→
   206→  const setItemField = (itemId: string, field: 'hallazgo' | 'mejora' | 'otherText', value: string) => {
   207→    setResults(prev => ({
   208→      ...prev,
   209→      [itemId]: { ...prev[itemId], itemId, [field]: value },
   210→    }));
   211→  };
   212→
   213→  const handleAuditPhotoSelect = async (e: React.ChangeEvent<HTMLInputElement>) => {
   214→    const files = e.target.files;
   215→    if (!files) return;
   216→    for (const file of Array.from(files)) {
   217→      const preview = URL.createObjectURL(file);
   218→      setAuditPhotos(prev => [...prev, { file, preview }]);
   219→    }
   220→    if (photoInputRef.current) photoInputRef.current.value = '';
   221→  };
   222→
   223→  const removeAuditPhoto = (index: number) => {
   224→    setAuditPhotos(prev => {
   225→      const photo = prev[index];
   226→      if (photo.preview) URL.revokeObjectURL(photo.preview);
   227→      return prev.filter((_, i) => i !== index);
   228→    });
   229→  };
   230→
   231→  const handleSubmit = async () => {
   232→    if (!canSubmit) return;
   233→    setIsSubmitting(true);
   234→    const isApto = scoring.scorePercent >= notaMinima;
   235→    try {
   236→      // Save audit result (always save, even if not apto)
   237→      const auditRes = await fetch('/api/audit', {
   238→        method: 'POST',
   239→        headers: { 'Content-Type': 'application/json' },
   240→        body: JSON.stringify({
   241→          sStep,
   242→          auditorName,
   243→          result: isApto ? 'apto' : 'no_apto',
   244→          score: Math.min(scoring.scorePercent, 100),
   245→          observations: observaciones || null,
   246→          fechaAuditoria,
   247→          horaAuditoria,
   248→          projectId: currentProject?.id,
   249→          zoneId: currentZone?.id || null,
   250→          checklistData: JSON.stringify(Object.values(results)),
   251→          mejorasData: haMejoras ? JSON.stringify(mejoras.filter(m => m.descripcion.trim())) : null,
   252→        }),
   253→      });
   254→
   255→      const auditJson = await auditRes.json();
   256→
   257→      // Handle server-side role rejection
   258→      if (auditRes.status === 403) {
   259→        setIsSubmitting(false);
   260→        alert(auditJson.error || 'No tienes permisos para realizar auditorías externas');
   261→        return;
   262→      }
   263→
   264→      if (!auditJson.success) {
   265→        setIsSubmitting(false);
   266→        alert(auditJson.error || 'Error al guardar la auditoría');
   267→        return;
   268→      }
   269→      if (auditJson.success) {
   270→        // Create action items for each NOK to transmit disfunciones to the operator
   271→        const nokResults = Object.values(results).filter(r => r.status === 'nok');
   272→        for (const nok of nokResults) {
   273→          await fetch('/api/actions', {
   274→            method: 'POST',
   275→            headers: { 'Content-Type': 'application/json' },
   276→            body: JSON.stringify({
   277→              sStep,
   278→              miniStep: 5,
   279→              itemId: nok.itemId,
   280→              itemDescription: `Disfunción detectada en auditoría: ${nok.itemId}`,
   281→              hallazgo: nok.hallazgo || nok.itemId,
   282→              mejora: nok.mejora || '',
   283→              responsable: null,
   284→              prioridad: 'alta',
   285→              estado: 'abierta',
   286→              source: 'auditoria',
   287→              auditor: auditorName,
   288→              projectId: currentProject?.id,
   289→              zoneId: currentZone?.id || null,
   290→            }),
   291→          });
   292→        }
   293→
   294→        // ─── Notify responsables of NOK disfunciones from audit ───
   295→        if (nokResults.length > 0 && currentProject?.id && currentZone?.id) {
   296→          try {
   297→            const membersRes = await fetch(`/api/projects/${currentProject.id}/members`);
   298→            const membersData = await membersRes.json();
   299→            const allMembers = membersData?.members || [];
   300→
   301→            const sStepData = S_STEPS.find(s => s.id === sStep);
   302→            const disfuncionMessage = `El auditor ha detectado ${nokResults.length} disfunción(es) en la auditoría de S${sStep} (${sStepData?.japaneseName || ''}) en la zona "${currentZone.name}". Revisa el Plan de Acción.`;
   303→
   304→            // Notify zone responsable
   305→            const responsableIds = new Set<string>();
   306→            if (currentZone.responsableId) responsableIds.add(currentZone.responsableId);
   307→            const responsables = allMembers.filter((m: any) => m.role === 'responsable');
   308→            for (const resp of responsables) responsableIds.add(resp.userId);
   309→
   310→            for (const respId of responsableIds) {
   311→              await fetch('/api/notifications', {
   312→                method: 'POST',
   313→                headers: { 'Content-Type': 'application/json' },
   314→                body: JSON.stringify({
   315→                  userId: respId,
   316→                  type: 'disfuncion',
   317→                  title: `Disfunciones en auditoría: S${sStep} — ${sStepData?.japaneseName || ''}`,
   318→                  message: disfuncionMessage,
   319→                  sStep,
   320→                  zoneId: currentZone.id,
   321→                  projectId: currentProject.id,
   322→                }),
   323→              });
   324→            }
   325→          } catch (notifError) {
   326→            console.error('Error notifying responsables of audit disfunciones:', notifError);
   327→          }
   328→        }
   329→
   330→        // ─── Upload audit photos to library with traceability ───
   331→        if (auditPhotos.length > 0) {
   332→          setIsUploadingPhotos(true);
   333→          for (let idx = 0; idx < auditPhotos.length; idx++) {
   334→            const photo = auditPhotos[idx];
   335→            try {
   336→              const formData = new FormData();
   337→              formData.append('file', photo.file);
   338→              formData.append('filename', `S${sStep}_auditoria_${currentZone?.name || 'zona'}_${idx + 1}_${Date.now()}.jpg`);
   339→
   340→              const uploadRes = await fetch('/api/upload', {
   341→                method: 'POST',
   342→                body: formData,
   343→              });
   344→              const uploadData = await uploadRes.json();
   345→
   346→              if (uploadData.success && uploadData.url) {
   347→                await fetch('/api/photo-library', {
   348→                  method: 'POST',
   349→                  headers: { 'Content-Type': 'application/json' },
   350→                  body: JSON.stringify({
   351→                    sStep,
   352→                    miniStep: 5,
   353→                    title: `Auditoría S${sStep} - Foto ${idx + 1}`,
   354→                    description: `${sStepData?.japaneseName || 'S' + sStep} - ${currentZone?.name || 'Zona'} - Paso 5 Auditoría - Auditor: ${auditorName}`,
   355→                    photoUrl: uploadData.url,
   356→                    photoType: 'hallazgo',
   357→                    category: `paso5_s${sStep}`,
   358→                    tags: JSON.stringify([`S${sStep}`, sStepData?.japaneseName || '', currentZone?.name || '', 'paso5', 'auditoria', 'hallazgo']),
   359→                    projectId: currentProject?.id,
   360→                    zoneId: currentZone?.id || null,
   361→                    uploadedBy: currentUser?.id || null,
   362→                  }),
   363→                });
   364→              }
   365→            } catch (photoErr) {
   366→              console.error('Error uploading audit photo:', photoErr);
   367→            }
   368→          }
   369→          setIsUploadingPhotos(false);
   370→        }
   371→
   372→        // Mark the mini-step as completed
   373→        const progressRes = await fetch(`/api/progress/step?sStep=${sStep}&miniStep=${miniStep}`, {
   374→          method: 'PUT',
   375→          headers: { 'Content-Type': 'application/json' },
   376→          body: JSON.stringify({
   377→            completed: isApto,
   378→            score: Math.min(scoring.scorePercent, 100),
   379→            notes: JSON.stringify({
   380→              type: 'auditoria_externa',
   381→              auditor: auditorName,
   382→              result: isApto ? 'apto' : 'no_apto',
   383→              fecha: fechaAuditoria,
   384→              hora: horaAuditoria,
   385→              results: Object.values(results),
   386→              observaciones,
   387→              mejorasRealizadas: haMejoras,
   388→              mejoras: haMejoras ? mejoras.filter(m => m.descripcion.trim()) : [],
   389→              disfuncionesCount: nokResults.length,
   390→            }),
   391→            projectId: currentProject?.id,
   392→            zoneId: currentZone?.id || null,
   393→          }),
   394→        });
   395→
   396→        const progressJson = await progressRes.json();
   397→
   398→        // Handle progress update failure
   399→        if (!progressJson.success) {
   400→          console.error('Error updating progress after audit:', progressJson.error);
   401→          if (progressRes.status === 403) {
   402→            alert(`La auditoría se ha guardado, pero no se pudo marcar el paso como completado: ${progressJson.error}. Verifica los permisos del auditor.`);
   403→          } else {
   404→            alert(`La auditoría se ha guardado, pero hubo un error al actualizar el progreso: ${progressJson.error || 'Error desconocido'}`);
   405→          }
   406→        }
   407→
   408→        // Always mark as completed in UI if audit was saved successfully
   409→        // (even if progress update failed, the audit result is saved)
   410→        setIsCompleted(true);
   411→        setFinalScore(scoring.scorePercent);
   412→
   413→        // Refresh progress (this also refreshes employeeProgress now)
   414→        await fetchProgress();
   415→      }
   416→    } catch (error) {
   417→      console.error('Error submitting audit:', error);
   418→    } finally {
   419→      setIsSubmitting(false);
   420→    }
   421→  };
   422→
   423→  const handleAdminSkip = async () => {
   424→    try {
   425→      // Save audit result
   426→      await fetch('/api/audit', {
   427→        method: 'POST',
   428→        headers: { 'Content-Type': 'application/json' },
   429→        body: JSON.stringify({
   430→          sStep,
   431→          auditorName: 'Admin (skip)',
   432→          result: 'apto',
   433→          score: 100,
   434→          observations: 'Completado por administrador (skip)',
   435→          projectId: currentProject?.id,
   436→          zoneId: currentZone?.id || null,
   437→        }),
   438→      });
   439→      // Mark progress as completed
   440→      const res = await fetch(`/api/progress/step?sStep=${sStep}&miniStep=${miniStep}`, {
   441→        method: 'PUT',
   442→        headers: { 'Content-Type': 'application/json' },
   443→        body: JSON.stringify({ completed: true, score: 100, notes: 'Completado por administrador (skip)', skipMissingTemplate: true, projectId: currentProject?.id, zoneId: currentZone?.id || null }),
   444→      });
   445→      const json = await res.json();
   446→      if (json.success) {
   447→        await fetchProgress();
   448→        onClose();
   449→      }
   450→    } catch (error) {
   451→      console.error('Error admin skip:', error);
   452→    }
   453→  };
   454→
   455→  return (
   456→    <Dialog open={open} onOpenChange={() => onClose()}>
   457→      <DialogContent size={isFullscreen ? "fullscreen" : "xl"} className="flex flex-col overflow-hidden p-0">
   458→        <DialogHeader className="px-6 pt-6 pb-2 flex-shrink-0">
   459→          <DialogTitle className="flex items-center gap-2">
   460→            <ShieldCheck className="h-5 w-5" style={{ color: sStepData?.color }} />
   461→            <span>Auditoría Externa</span>
   462→            <Badge variant="outline" style={{ borderColor: sStepData?.color, color: sStepData?.color }}>
   463→              {sStepData?.japaneseName} — {sStepData?.spanishName}
   464→            </Badge>
   465→            <button
   466→              onClick={() => setIsFullscreen(!isFullscreen)}
   467→              className="ml-auto p-1 rounded hover:bg-muted transition-colors"
   468→              title={isFullscreen ? "Reducir ventana" : "Pantalla completa"}
   469→            >
   470→              {isFullscreen ? <Minimize2 className="h-4 w-4 text-muted-foreground" /> : <Maximize2 className="h-4 w-4 text-muted-foreground" />}
   471→            </button>
   472→          </DialogTitle>
   473→        </DialogHeader>
   474→
   475→        {canSkipSteps && !isCompleted && (
   476→          <div className="flex items-center gap-2 p-2 mx-6 flex-shrink-0 bg-amber-50 border border-amber-200 rounded-lg">
   477→            <span className="text-xs text-amber-700 font-medium">Modo Admin:</span>
   478→            <Button
   479→              variant="outline"
   480→              size="sm"
   481→              className="text-xs border-amber-300 text-amber-700 hover:bg-amber-100"
   482→              onClick={handleAdminSkip}
   483→            >
   484→              Completar paso sin auditoría
   485→            </Button>
   486→          </div>
   487→        )}
   488→
   489→        <div className="flex-1 overflow-y-auto px-6 pb-6 min-h-0">
   490→        {/* Permission check */}
   491→        {!canAudit && !isCompleted && (
   492→          <div className="flex items-center gap-2 p-3 bg-red-50 border border-red-200 rounded-lg">
   493→            <AlertCircle className="h-4 w-4 text-red-600" />
   494→            <span className="text-sm text-red-700 font-medium">No tienes permiso para realizar auditorías en este paso. Tu rol: {currentUser?.role || 'sin rol'}</span>
   495→          </div>
   496→        )}
   497→
   498→        {!canAudit && !isCompleted ? null : isLoadingTemplate ? (
   499→          <div className="flex items-center justify-center py-12">
   500→            <div className="h-8 w-8 border-4 border-green-500 border-t-transparent rounded-full animate-spin" />
   501→          </div>
   502→        ) : sections.length === 0 ? (
   503→          <div className="text-center py-12">
   504→            <AlertCircle className="h-12 w-12 text-amber-500 mx-auto mb-3" />
   505→            <h3 className="text-lg font-bold mb-2">No hay checklist configurado</h3>
   506→            <p className="text-sm text-muted-foreground max-w-md mx-auto">
   507→              No se encontró una plantilla de auditoría para S{sStep}. 
   508→              Crea una plantilla en <strong>Administración → Plantillas → Auditoría Externa</strong> 
   509→              o pulsa el botón &quot;Crear plantillas por defecto&quot;.
