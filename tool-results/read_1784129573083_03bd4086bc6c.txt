   123→// ─── Component ───────────────────────────────────────────────────────────────
   124→
   125→export default function GestorPanel() {
   126→  const { setCurrentView, fetchProjects, fetchCompanies, projects, setCurrentProject, currentProject } = use5SStore()
   127→  const [activeTab, setActiveTab] = useState<GestorTab>('dashboard')
   128→
   129→  // Platform stats
   130→  const [stats, setStats] = useState<PlatformStats | null>(null)
   131→  const [isLoadingStats, setIsLoadingStats] = useState(false)
   132→
   133→  // Companies management
   134→  const [showNewCompany, setShowNewCompany] = useState(false)
   135→  const [newCompanyName, setNewCompanyName] = useState('')
   136→  const [newCompanyDesc, setNewCompanyDesc] = useState('')
   137→  const [editingCompany, setEditingCompany] = useState<string | null>(null)
   138→  const [editCompanyData, setEditCompanyData] = useState({ name: '', description: '', active: true })
   139→
   140→  // Admin management
   141→  const [assigningAdminTo, setAssigningAdminTo] = useState<string | null>(null)
   142→  const [searchEmail, setSearchEmail] = useState('')
   143→  const [searchResults, setSearchResults] = useState<PlatformStats['users']>([])
   144→  const [isSearching, setIsSearching] = useState(false)
   145→  const [showCreateUser, setShowCreateUser] = useState(false)
   146→  const [newAdminData, setNewAdminData] = useState({ name: '', email: '', password: '' })
   147→  const [isAssigning, setIsAssigning] = useState(false)
   148→
   149→  // ─── Data loading ────────────────────────────────────────────────────────
   150→  const loadStats = useCallback(async () => {
   151→    setIsLoadingStats(true)
   152→    try {
   153→      const res = await fetch('/api/platform-stats')
   154→      const data = await res.json()
   155→      if (data.success) {
   156→        setStats(data.data)
   157→      }
   158→    } catch (error) {
   159→      console.error('Error loading platform stats:', error)
   160→    } finally {
   161→      setIsLoadingStats(false)
   162→    }
   163→  }, [])
   164→
   165→  useEffect(() => {
   166→    loadStats()
   167→  }, [loadStats])
   168→
   169→  // ─── Company actions ────────────────────────────────────────────────────
   170→  const handleCreateCompany = async () => {
   171→    if (!newCompanyName.trim()) return
   172→    try {
   173→      const res = await fetch('/api/companies', {
   174→        method: 'POST',
   175→        headers: { 'Content-Type': 'application/json' },
   176→        body: JSON.stringify({ name: newCompanyName, description: newCompanyDesc || undefined }),
   177→      })
   178→      if (res.ok) {
   179→        setShowNewCompany(false)
   180→        setNewCompanyName('')
   181→        setNewCompanyDesc('')
   182→        await loadStats()
   183→        await fetchCompanies()
   184→      } else {
   185→        const data = await res.json()
   186→        alert(data.error || 'Error al crear empresa')
   187→      }
   188→    } catch (error) {
   189→      console.error('Error creating company:', error)
   190→    }
   191→  }
   192→
   193→  const handleUpdateCompany = async (companyId: string) => {
   194→    try {
   195→      const res = await fetch(`/api/companies/${companyId}`, {
   196→        method: 'PUT',
   197→        headers: { 'Content-Type': 'application/json' },
   198→        body: JSON.stringify(editCompanyData),
   199→      })
   200→      if (res.ok) {
   201→        setEditingCompany(null)
   202→        await loadStats()
   203→        await fetchCompanies()
   204→      } else {
   205→        const data = await res.json()
   206→        alert(data.error || 'Error al actualizar empresa')
   207→      }
   208→    } catch (error) {
   209→      console.error('Error updating company:', error)
   210→    }
   211→  }
   212→
   213→  const handleToggleCompanyActive = async (companyId: string, currentActive: boolean) => {
   214→    try {
   215→      const res = await fetch(`/api/companies/${companyId}`, {
   216→        method: 'PUT',
   217→        headers: { 'Content-Type': 'application/json' },
   218→        body: JSON.stringify({ active: !currentActive }),
   219→      })
   220→      if (res.ok) {
   221→        await loadStats()
   222→        await fetchCompanies()
   223→      }
   224→    } catch (error) {
   225→      console.error('Error toggling company active:', error)
   226→    }
   227→  }
   228→
   229→  // ─── Admin assignment actions ───────────────────────────────────────────
   230→  const handleSearchUser = async () => {
   231→    if (!searchEmail.trim() || searchEmail.length < 3) {
   232→      setSearchResults([])
   233→      return
   234→    }
   235→    setIsSearching(true)
   236→    try {
   237→      const res = await fetch(`/api/users?search=${encodeURIComponent(searchEmail)}`)
   238→      const data = await res.json()
   239→      if (data.success) {
   240→        setSearchResults(data.users || [])
   241→      }
   242→    } catch (error) {
   243→      console.error('Error searching users:', error)
   244→    } finally {
   245→      setIsSearching(false)
   246→    }
   247→  }
   248→
   249→  useEffect(() => {
   250→    if (searchEmail.length >= 3) {
   251→      const t = setTimeout(() => handleSearchUser(), 300)
   252→      return () => clearTimeout(t)
   253→    } else {
   254→      setSearchResults([])
   255→    }
   256→  }, [searchEmail])
   257→
   258→  const handleAssignExistingUser = async (userId: string) => {
   259→    if (!assigningAdminTo) return
   260→    setIsAssigning(true)
   261→    try {
   262→      // First, ensure the user has 'admin' role
   263→      await fetch('/api/users', {
   264→        method: 'PUT',
   265→        headers: { 'Content-Type': 'application/json' },
   266→        body: JSON.stringify({ id: userId, role: 'admin' }),
   267→      })
   268→
   269→      // Then assign them to the company as admin
   270→      const res = await fetch(`/api/companies/${assigningAdminTo}/members`, {
   271→        method: 'POST',
   272→        headers: { 'Content-Type': 'application/json' },
   273→        body: JSON.stringify({ userId, role: 'admin_empresa' }),
   274→      })
   275→
   276→      if (res.ok) {
   277→        setAssigningAdminTo(null)
   278→        setSearchEmail('')
   279→        setSearchResults([])
   280→        await loadStats()
   281→      } else {
   282→        const data = await res.json()
   283→        alert(data.error || 'Error al asignar administrador')
   284→      }
   285→    } catch (error) {
   286→      console.error('Error assigning admin:', error)
   287→    } finally {
   288→      setIsAssigning(false)
   289→    }
   290→  }
   291→
   292→  const handleCreateAndAssignAdmin = async () => {
   293→    if (!assigningAdminTo) return
   294→    if (!newAdminData.name.trim() || !newAdminData.email.trim() || newAdminData.password.length < 6) {
   295→      alert('Completa todos los campos. La contraseña debe tener al menos 6 caracteres.')
   296→      return
   297→    }
   298→    setIsAssigning(true)
   299→    try {
   300→      // 1. Create user with role 'admin'
   301→      const createRes = await fetch('/api/users', {
   302→        method: 'POST',
   303→        headers: { 'Content-Type': 'application/json' },
   304→        body: JSON.stringify({
   305→          name: newAdminData.name,
   306→          email: newAdminData.email,
   307→          password: newAdminData.password,
   308→          role: 'admin',
   309→          active: true,
   310→        }),
   311→      })
   312→
   313→      if (!createRes.ok) {
   314→        const data = await createRes.json()
   315→        alert(data.error || 'Error al crear usuario')
   316→        setIsAssigning(false)
   317→        return
   318→      }
   319→
   320→      const createData = await createRes.json()
   321→      const newUserId = createData.user?.id
   322→
   323→      if (!newUserId) {
   324→        alert('Usuario creado pero no se pudo obtener el ID. Asigna el admin manualmente.')
   325→        setIsAssigning(false)
   326→        return
   327→      }
   328→
   329→      // 2. Assign user to company
   330→      const assignRes = await fetch(`/api/companies/${assigningAdminTo}/members`, {
   331→        method: 'POST',
   332→        headers: { 'Content-Type': 'application/json' },
   333→        body: JSON.stringify({ userId: newUserId, role: 'admin_empresa' }),
   334→      })
   335→
   336→      if (assignRes.ok) {
   337→        // 3. Send welcome email to the new admin (non-blocking)
   338→        const company = stats?.companies?.find(c => c.id === assigningAdminTo)
   339→        const companyName = company?.name || 'la empresa'
   340→        fetch('/api/email', {
   341→          method: 'POST',
   342→          headers: { 'Content-Type': 'application/json' },
   343→          body: JSON.stringify({
   344→            type: 'admin_welcome',
   345→            adminName: newAdminData.name,
   346→            adminEmail: newAdminData.email,
   347→            adminPassword: newAdminData.password,
   348→            companyName,
   349→            gestorEmail: undefined, // could add gestor email here later
   350→          }),
   351→        }).catch(e => console.error('Email send error (non-critical):', e))
   352→
   353→        setAssigningAdminTo(null)
   354→        setShowCreateUser(false)
   355→        setNewAdminData({ name: '', email: '', password: '' })
   356→        await loadStats()
   357→      } else {
   358→        const data = await assignRes.json()
   359→        alert(data.error || 'Usuario creado pero error al asignar a la empresa')
   360→      }
   361→    } catch (error) {
   362→      console.error('Error creating admin:', error)
   363→    } finally {
   364→      setIsAssigning(false)
   365→    }
   366→  }
   367→
   368→  const handleRemoveAdmin = async (companyId: string, userId: string) => {
   369→    if (!confirm('¿Quitar a este administrador de la empresa?')) return
   370→    try {
   371→      const res = await fetch(`/api/companies/${companyId}/members`, {
   372→        method: 'DELETE',
   373→        headers: { 'Content-Type': 'application/json' },
   374→        body: JSON.stringify({ memberUserId: userId }),
   375→      })
   376→      if (res.ok) {
   377→        await loadStats()
   378→      } else {
   379→        const data = await res.json()
   380→        alert(data.error || 'Error al quitar administrador')
   381→      }
   382→    } catch (error) {
   383→      console.error('Error removing admin:', error)
   384→    }
   385→  }
   386→  // ─── Render ──────────────────────────────────────────────────────────────
   387→  const tabs: { key: GestorTab; label: string; icon: React.ReactNode }[] = [
   388→    { key: 'dashboard', label: 'KPIs', icon: <Activity className="h-4 w-4" /> },
   389→    { key: 'empresas', label: 'Empresas', icon: <Building2 className="h-4 w-4" /> },
   390→    { key: 'admins', label: 'Administradores', icon: <Shield className="h-4 w-4" /> },
   391→  ]
   392→
   393→  return (
   394→    <div className="min-h-screen bg-gradient-to-b from-slate-950 via-violet-950/20 to-slate-950 text-white">
   395→      {/* Header */}
   396→      <header className="border-b border-violet-800/30 bg-slate-950/80 backdrop-blur-sm sticky top-0 z-10">
   397→        <div className="max-w-7xl mx-auto px-4 py-3 flex items-center justify-between">
   398→          <div className="flex items-center gap-3">
   399→            <div className="w-10 h-10 rounded-xl bg-gradient-to-br from-violet-500 to-purple-700 flex items-center justify-center shadow-lg shadow-violet-500/30">
   400→              <Crown className="h-5 w-5 text-yellow-300" />
   401→            </div>
   402→            <div>
   403→              <h1 className="text-lg font-bold bg-gradient-to-r from-violet-300 to-purple-200 bg-clip-text text-transparent">
   404→                Gestor Panel
   405→              </h1>
   406→              <p className="text-xs text-violet-400">Dueño de la Plataforma</p>
   407→            </div>
   408→          </div>
   409→          <div className="flex items-center gap-2">
   410→            <Button
   411→              variant="outline"
   412→              size="sm"
   413→              onClick={() => {
   414→                if (!currentProject && projects.length > 0) {
   415→                  setCurrentProject(projects[0])
   416→                }
   417→                setCurrentView('board')
   418→              }}
   419→              className="gap-1.5 border-violet-700/50 text-violet-300 hover:bg-violet-900/30 hover:text-violet-200"
   420→            >
   421→              ← Vista App
   422→            </Button>
   423→            <Badge className="bg-gradient-to-r from-violet-600 to-purple-600 text-white border-0 px-3 py-1">
   424→              <Crown className="h-3 w-3 mr-1" /> GESTOR
   425→            </Badge>
   426→          </div>
   427→        </div>
   428→      </header>
   429→
   430→      {/* Tabs */}
   431→      <div className="border-b border-violet-800/20 bg-slate-950/60 backdrop-blur-sm">
   432→        <div className="max-w-7xl mx-auto px-4 flex gap-0 overflow-x-auto">
   433→          {tabs.map(tab => (
   434→            <button
   435→              key={tab.key}
   436→              onClick={() => setActiveTab(tab.key)}
   437→              className={`flex items-center gap-2 px-4 py-3 text-sm font-medium border-b-2 transition-all whitespace-nowrap ${
   438→                activeTab === tab.key
   439→                  ? 'border-violet-400 text-violet-300 bg-violet-900/20'
   440→                  : 'border-transparent text-violet-500 hover:text-violet-300 hover:border-violet-700/50'
   441→              }`}
   442→            >
   443→              {tab.icon}
   444→              {tab.label}
   445→            </button>
   446→          ))}
   447→        </div>
   448→      </div>
   449→
   450→      {/* Content */}
   451→      <main className="max-w-7xl mx-auto px-4 py-6">
   452→        <AnimatePresence mode="wait">
   453→          {/* ═══ DASHBOARD TAB (KPIs) ═══ */}
   454→          {activeTab === 'dashboard' && (
   455→            <motion.div key="dashboard" initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0 }} className="space-y-6">
   456→              {isLoadingStats ? (
   457→                <div className="flex justify-center py-20"><Loader2 className="h-10 w-10 text-violet-400 animate-spin" /></div>
   458→              ) : stats ? (
   459→                <>
   460→                  {/* KPI Cards */}
   461→                  <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   462→                    <Card className="bg-gradient-to-br from-violet-900/40 to-purple-900/20 border-violet-700/30 text-white">
   463→                      <CardContent className="p-4">
   464→                        <div className="flex items-center justify-between mb-2">
   465→                          <Building2 className="h-5 w-5 text-violet-400" />
   466→                          <Badge className="bg-violet-800/50 text-violet-300 border-0 text-[10px]">
   467→                            {stats.totals.activeCompanies}/{stats.totals.companies} activas
   468→                          </Badge>
   469→                        </div>
   470→                        <p className="text-2xl font-bold">{stats.totals.companies}</p>
   471→                        <p className="text-xs text-violet-400">Empresas</p>
   472→                      </CardContent>
   473→                    </Card>
   474→                    <Card className="bg-gradient-to-br from-blue-900/40 to-indigo-900/20 border-blue-700/30 text-white">
   475→                      <CardContent className="p-4">
   476→                        <div className="flex items-center justify-between mb-2">
   477→                          <Users className="h-5 w-5 text-blue-400" />
   478→                          <Badge className="bg-blue-800/50 text-blue-300 border-0 text-[10px]">
   479→                            {stats.totals.activeUsers}/{stats.totals.users} activos
   480→                          </Badge>
   481→                        </div>
   482→                        <p className="text-2xl font-bold">{stats.totals.users}</p>
   483→                        <p className="text-xs text-blue-400">Usuarios</p>
   484→                      </CardContent>
   485→                    </Card>
   486→                    <Card className="bg-gradient-to-br from-emerald-900/40 to-green-900/20 border-emerald-700/30 text-white">
   487→                      <CardContent className="p-4">
   488→                        <div className="flex items-center justify-between mb-2">
   489→                          <TrendingUp className="h-5 w-5 text-emerald-400" />
   490→                          <Badge className="bg-emerald-800/50 text-emerald-300 border-0 text-[10px]">
   491→                            {stats.totals.activeProjects}/{stats.totals.projects} activos
   492→                          </Badge>
   493→                        </div>
   494→                        <p className="text-2xl font-bold">{stats.totals.projects}</p>
   495→                        <p className="text-xs text-emerald-400">Proyectos</p>
   496→                      </CardContent>
   497→                    </Card>
   498→                    <Card className="bg-gradient-to-br from-orange-900/40 to-amber-900/20 border-orange-700/30 text-white">
   499→                      <CardContent className="p-4">
   500→                        <div className="flex items-center justify-between mb-2">
   501→                          <AlertTriangle className="h-5 w-5 text-orange-400" />
   502→                          <Badge className="bg-orange-800/50 text-orange-300 border-0 text-[10px]">
   503→                            de {stats.totals.actions} total
   504→                          </Badge>
   505→                        </div>
   506→                        <p className="text-2xl font-bold">{stats.totals.openActions}</p>
   507→                        <p className="text-xs text-orange-400">Acciones Abiertas</p>
   508→                      </CardContent>
   509→                    </Card>
   510→                  </div>
   511→
   512→                  {/* Role distribution + Platform health */}
   513→                  <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   514→                    <Card className="bg-slate-900/60 border-violet-700/20">
   515→                      <CardHeader className="pb-3">
   516→                        <CardTitle className="text-sm text-violet-300 flex items-center gap-2">
   517→                          <Users className="h-4 w-4" /> Distribución de Roles
   518→                        </CardTitle>
   519→                      </CardHeader>
   520→                      <CardContent>
   521→                        <div className="space-y-2">
   522→                          {Object.entries(stats.roleDistribution).map(([role, count]) => (
   523→                            <div key={role} className="flex items-center justify-between">
   524→                              <div className="flex items-center gap-2">
   525→                                <Badge className={`${ROLE_COLORS[role] || 'bg-gray-100 text-gray-700 border-gray-200'} border text-xs`}>
   526→                                  {ROLE_LABELS[role] || role}
   527→                                </Badge>
   528→                              </div>
   529→                              <div className="flex items-center gap-2">
   530→                                <div className="w-24 h-2 bg-slate-800 rounded-full overflow-hidden">
   531→                                  <div
   532→                                    className="h-full bg-gradient-to-r from-violet-500 to-purple-500 rounded-full"
   533→                                    style={{ width: `${Math.min((count / stats.totals.users) * 100, 100)}%` }}
   534→                                  />
   535→                                </div>
   536→                                <span className="text-sm font-semibold text-white w-8 text-right">{count}</span>
   537→                              </div>
   538→                            </div>
   539→                          ))}
   540→                        </div>
   541→                      </CardContent>
   542→                    </Card>
   543→
   544→                    <Card className="bg-slate-900/60 border-violet-700/20">
