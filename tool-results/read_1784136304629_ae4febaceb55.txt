   446→ const handleCreateCompanyFull = async () => {
   447→  if (!newCompanyName.trim()) return
   448→  setIsCreatingCompany(true)
   449→  let companyId: string | null = null
   450→  try {
   451→   // Step 1: Create company
   452→   const companyRes = await fetch('/api/companies', {
   453→    method: 'POST',
   454→    headers: { 'Content-Type': 'application/json' },
   455→    body: JSON.stringify({ name: newCompanyName, description: newCompanyDesc || undefined, nif: newCompanyNif || undefined, sector: newCompanySector || undefined, city: newCompanyCity || undefined, phone: newCompanyPhone || undefined }),
   456→   })
   457→   const companyData = await companyRes.json()
   458→   if (!companyRes.ok) {
   459→    alert(companyData.error || 'Error al crear empresa')
   460→    return
   461→   }
   462→   companyId = companyData.company?.id || companyData.id
   463→
   464→   // Step 2: Create or assign admin
   465→   let adminUserId: string | null = null
   466→   if (adminMode === 'create') {
   467→    if (!newAdminName.trim() || !newAdminEmail.trim() || !newAdminPassword.trim()) {
   468→     alert('Complete los datos del administrador')
   469→     // Company was created but admin data is missing — still close dialog and refresh
   470→     setShowNewCompany(false)
   471→     resetNewCompanyForm()
   472→     await loadStats()
   473→     await fetchCompanies()
   474→     return
   475→    }
   476→    // Create user with role 'admin' via /api/users
   477→    const userRes = await fetch('/api/users', {
   478→     method: 'POST',
   479→     headers: { 'Content-Type': 'application/json' },
   480→     body: JSON.stringify({
   481→      name: newAdminName,
   482→      email: newAdminEmail,
   483→      password: newAdminPassword,
   484→      role: 'admin',
   485→      active: true,
   486→     }),
   487→    })
   488→    const userData = await userRes.json()
   489→    if (!userRes.ok) {
   490→     alert(userData.error || 'Error al crear usuario admin')
   491→     // Company was created but admin creation failed — still close dialog and refresh
   492→     setShowNewCompany(false)
   493→     resetNewCompanyForm()
   494→     await loadStats()
   495→     await fetchCompanies()
   496→     return
   497→    }
   498→    adminUserId = userData.user?.id
   499→   } else {
   500→    // Assign existing user
   501→    if (!assignUserId) {
   502→     alert('Seleccione un usuario para asignar como administrador')
   503→     setShowNewCompany(false)
   504→     resetNewCompanyForm()
   505→     await loadStats()
   506→     await fetchCompanies()
   507→     return
   508→    }
   509→    adminUserId = assignUserId
   510→    // Update user role to 'admin' if not already
   511→    const existingUser = users.find(u => u.id === assignUserId)
   512→    if (existingUser && existingUser.role !== 'admin') {
   513→     await fetch('/api/users', {
   514→      method: 'PUT',
   515→      headers: { 'Content-Type': 'application/json' },
   516→      body: JSON.stringify({ id: assignUserId, role: 'admin' }),
   517→     })
   518→    }
   519→   }
   520→
   521→   // Step 3: Add admin to company as admin_empresa
   522→   if (adminUserId && companyId) {
   523→    const memberRes = await fetch(`/api/companies/${companyId}/members`, {
   524→     method: 'POST',
   525→     headers: { 'Content-Type': 'application/json' },
   526→     body: JSON.stringify({ userId: adminUserId, role: 'admin_empresa' }),
   527→    })
   528→    if (!memberRes.ok) {
   529→     const memberData = await memberRes.json()
   530→     console.error('Error adding admin to company:', memberData.error)
   531→    }
   532→   }
   533→
   534→   // Step 4: Create subscription
   535→   if (companyId) {
   536→    const subRes = await fetch('/api/subscriptions', {
   537→     method: 'POST',
   538→     headers: { 'Content-Type': 'application/json' },
   539→     body: JSON.stringify({
   540→      companyId,
   541→      plan: newSubPlan,
   542→      status: newSubStatus,
   543→      maxUsers: newSubMaxUsers,
   544→      maxProjects: newSubMaxProjects,
   545→      price: newSubPrice,
   546→      endDate: newSubEndDate || undefined,
   547→      trialEndsAt: newSubTrialEndsAt || undefined,
   548→      notes: newSubNotes || undefined,
   549→     }),
   550→    })
   551→    if (!subRes.ok) {
   552→     console.error('Error creating subscription')
   553→    }
   554→   }
   555→
   556→   // Note: Welcome email is NOT sent automatically.
   557→   // The gestor can send it manually using the "Enviar Email" button on the company card.
   558→
   559→   setShowNewCompany(false)
   560→   resetNewCompanyForm()
   561→   await loadStats()
   562→   await fetchCompanies()
   563→  } catch (error) {
   564→   console.error('Error creating company with admin:', error)
   565→   alert('Error al crear la empresa. Revise la consola para detalles.')
   566→   // Still refresh data even on error — the company might have been created
   567→   await loadStats()
   568→   await fetchCompanies()
   569→  } finally {
   570→   setIsCreatingCompany(false)
   571→  }
   572→ }
