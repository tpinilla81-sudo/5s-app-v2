   235→  // Al guardar, los ActionItems verificados como resueltos se cierran con verificadoPorId=auditor.

   260→          auditorOrig: a.auditor,
   261→          decisionRevision: 'pendiente' as const,
   262→          nuevaPrioridad: a.prioridad,
   263→          notasAuditor: '',
   264→        }));
   265→        setHeredados(pendientes);
   266→      }
   267→    } catch (e) {
   268→      console.error('[auditoria] Error cargando heredados:', e);
   269→    }

   285→  };
   286→
   287→  // Load previously saved audit results from the backend

   329→  // v2.100: Track también el estado del schedule, el id y las propuestas (si las hay)

   331→  const loadScheduledDate = async () => {

   375→        setCurrentPropuestas([]);
   376→        // Default: primer auditor del proyecto
   377→        const auditors = projectMembers.filter(m => m.role === 'auditor');
   378→        if (auditors.length > 0) {
   379→          setSelectedAuditorId(auditors[0].userId);
   380→        }
   381→      }
   382→    } catch (e) {
   383→      console.error('Error loading scheduled date:', e);

   386→
   387→  // Save scheduled date/time
   388→  // v2.75: enviamos responsableId (ejecutor=auditor), empleadoId
   389→  // (asistente=responsable de zona), createdBy y notifyUser:true
   390→  // para que el backend dispare la notif 'evaluation_scheduled' al
   391→  // asistente (responsable de zona).
   392→  // En auditoría (miniStep=5):
   393→  //   - ejecutor = primer auditor del proyecto
   394→  //   - asistente = responsable de la zona (currentZone.responsableId)
   395→  //     fallback al primer responsable del proyecto
   396→  //   - createdBy = currentUser.id (auditor que programa)
   397→  // v2.75.2: tras guardar, refresca el store (schedules + notifs) para

   399→  // y reabrir el modal.
   400→  const handleSaveSchedule = async () => {
   401→    if (!currentProject?.id || !currentZone?.id) return;

   403→      // Cargar miembros si no están cargados aún (race condition)
   404→      let members = projectMembers;

   406→        try {
   407→          const r = await fetch(`/api/projects/${currentProject.id}/members`);
   408→          const d = await r.json();
   409→          members = d?.members || [];
   410→          setProjectMembers(members);
   411→        } catch (e) { /* ignore */ }
   412→      }
   413→      const auditorMember = members.find(m => m.role === 'auditor');
   414→      const responsableMember = members.find(m => m.role === 'responsable');
   415→      const responsableId = auditorMember?.userId || currentUser?.id || null;
   416→      const empleadoId = currentZone.responsableId || responsableMember?.userId || null;
   417→
   418→      if (!responsableId) {
   419→        toast.error('No se puede programar: no hay auditor en el proyecto');
   420→        return;
   421→      }
   422→      if (!empleadoId) {
   423→        toast.error('No se puede programar: no hay responsable de zona');
   424→        return;
   425→      }
   426→
   427→      const res = await fetch('/api/evaluation-schedule', {
   428→        method: 'POST',
   429→        headers: { 'Content-Type': 'application/json' },
   430→        body: JSON.stringify({
   431→          sStep,
   432→          miniStep: 5,
   433→          projectId: currentProject.id,
   434→          zoneId: currentZone.id,
   435→          fechaProgramada,
   436→          horaProgramada,
   437→          // v2.75: roles para que se disparen los avisos
   438→          responsableId,   // EJECUTOR (auditor)
   439→          empleadoId,      // ASISTENTE (responsable de zona)
   440→          createdBy: currentUser?.id || null,
   441→          rolEjecutor: 'auditor',
   442→          notifyUser: true,
   443→        }),
   444→      });
   445→      const json = await res.json();
   446→      if (!json.success) {
   447→        // v2.77: si el backend bloquea por hallazgos pendientes (409),
   448→        // mostrar un diálogo con el listado para que el auditor sepa
   449→        // exactamente qué debe cerrar el responsable antes de programar.
   450→        if (res.status === 409 && json.code === 'PENDING_HALLAZGOS') {
   451→          const pendingList = Array.isArray(json.pending) ? json.pending : [];
   452→          const detail = pendingList.length > 0
   453→            ? pendingList.slice(0, 5).map((p: any, i: number) =>
   454→                `${i + 1}. ${p.hallazgo || p.itemId || '—'} (${p.source || '?'}, ${p.estado})`
   455→              ).join('\n')
   456→            : '';
   457→          const extra = pendingList.length > 5 ? `\n…y ${pendingList.length - 5} más.` : '';
   458→          alert(
   459→            `${json.error}\n\nHallazgos pendientes:\n${detail}${extra}\n\n` +
   460→            `El responsable de zona debe cerrar estos hallazgos en el Plan de Acción antes de programar la auditoría.`
   461→          );
   462→        } else {
   463→          toast.error(`Error al programar: ${json.error || 'desconocido'}`);
   464→        }
   465→        return;
   466→      }
   467→      toast.success('Fecha programada guardada — responsable avisado');
   468→      // v2.75.2: refrescar store para que el badge "Programada" aparezca
   469→      // inmediatamente sobre el globo 5 y el responsable lo vea sin tener
   470→      // que refrescar la página.
   471→      try {
   472→        await use5SStore.getState().fetchEvaluationSchedules();
   473→        await use5SStore.getState().fetchNotifications(true);
   474→      } catch (e) { /* ignore */ }
   475→    } catch (e) {
   476→      console.error('Error saving schedule:', e);
   477→      toast.error('Error al guardar la fecha programada');
