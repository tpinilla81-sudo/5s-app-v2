   438→ const handleCreateCompanyFull = async () => {
   439→  if (!newCompanyName.trim()) return
   440→  setIsCreatingCompany(true)
   441→  let companyId: string | null = null
   442→  try {
   443→   // Step 1: Create company
   444→   const companyRes = await fetch('/api/companies', {
   445→    method: 'POST',
   446→    headers: { 'Content-Type': 'application/json' },
   447→    body: JSON.stringify({ name: newCompanyName, description: newCompanyDesc || undefined }),
   448→   })
   449→   const companyData = await companyRes.json()
   450→   if (!companyRes.ok) {
   451→    alert(companyData.error || 'Error al crear empresa')
   452→    return
   453→   }
   454→   companyId = companyData.company?.id || companyData.id
   455→
   456→   // Step 2: Create or assign admin
   457→   let adminUserId: string | null = null
   458→   if (adminMode === 'create') {
   459→    if (!newAdminName.trim() || !newAdminEmail.trim() || !newAdminPassword.trim()) {
   460→     alert('Complete los datos del administrador')
   461→     // Company was created but admin data is missing — still close dialog and refresh
   462→     setShowNewCompany(false)
   463→     resetNewCompanyForm()
   464→     await loadStats()
   465→     await fetchCompanies()
   466→     return
   467→    }
   468→    // Create user with role 'admin' via /api/users
   469→    const userRes = await fetch('/api/users', {
   470→     method: 'POST',
   471→     headers: { 'Content-Type': 'application/json' },
   472→     body: JSON.stringify({
   473→      name: newAdminName,
   474→      email: newAdminEmail,
   475→      password: newAdminPassword,
   476→      role: 'admin',
   477→      active: true,
   478→     }),
   479→    })
   480→    const userData = await userRes.json()
   481→    if (!userRes.ok) {
   482→     alert(userData.error || 'Error al crear usuario admin')
   483→     // Company was created but admin creation failed — still close dialog and refresh
   484→     setShowNewCompany(false)
   485→     resetNewCompanyForm()
   486→     await loadStats()
   487→     await fetchCompanies()
   488→     return
   489→    }
   490→    adminUserId = userData.user?.id
   491→   } else {
   492→    // Assign existing user
   493→    if (!assignUserId) {
   494→     alert('Seleccione un usuario para asignar como administrador')
   495→     setShowNewCompany(false)
   496→     resetNewCompanyForm()
   497→     await loadStats()
   498→     await fetchCompanies()
   499→     return
   500→    }
   501→    adminUserId = assignUserId
   502→    // Update user role to 'admin' if not already
   503→    const existingUser = users.find(u => u.id === assignUserId)
   504→    if (existingUser && existingUser.role !== 'admin') {
   505→     await fetch('/api/users', {
   506→      method: 'PUT',
   507→      headers: { 'Content-Type': 'application/json' },
   508→      body: JSON.stringify({ id: assignUserId, role: 'admin' }),
   509→     })
   510→    }
   511→   }
   512→
   513→   // Step 3: Add admin to company as admin_empresa
   514→   if (adminUserId && companyId) {
   515→    const memberRes = await fetch(`/api/companies/${companyId}/members`, {
   516→     method: 'POST',
   517→     headers: { 'Content-Type': 'application/json' },
   518→     body: JSON.stringify({ userId: adminUserId, role: 'admin_empresa' }),
   519→    })
   520→    if (!memberRes.ok) {
   521→     const memberData = await memberRes.json()
   522→     console.error('Error adding admin to company:', memberData.error)
   523→    }
   524→   }
   525→
   526→   // Step 4: Create subscription
   527→   if (companyId) {
   528→    const subRes = await fetch('/api/subscriptions', {
   529→     method: 'POST',
   530→     headers: { 'Content-Type': 'application/json' },
   531→     body: JSON.stringify({
   532→      companyId,
   533→      plan: newSubPlan,
   534→      status: newSubStatus,
   535→      maxUsers: newSubMaxUsers,
   536→      maxProjects: newSubMaxProjects,
   537→      price: newSubPrice,
   538→      endDate: newSubEndDate || undefined,
   539→      trialEndsAt: newSubTrialEndsAt || undefined,
   540→      notes: newSubNotes || undefined,
   541→     }),
   542→    })
   543→    if (!subRes.ok) {
   544→     console.error('Error creating subscription')
   545→    }
   546→   }
   547→
   548→   // Note: Welcome email is NOT sent automatically.
   549→   // The gestor can send it manually using the "Enviar Email" button on the company card.
   550→
   551→   setShowNewCompany(false)
   552→   resetNewCompanyForm()
   553→   await loadStats()
   554→   await fetchCompanies()
   555→  } catch (error) {
   556→   console.error('Error creating company with admin:', error)
   557→   alert('Error al crear la empresa. Revise la consola para detalles.')
   558→   // Still refresh data even on error — the company might have been created
   559→   await loadStats()
   560→   await fetchCompanies()
   561→  } finally {
   562→   setIsCreatingCompany(false)
   563→  }
   564→ }
