     1→'use client';
     2→
     3→import { useState, useEffect } from 'react';
     4→import { motion, AnimatePresence } from 'framer-motion';
     5→import { use5SStore } from '@/lib/store';
     6→import { S_STEPS } from '@/lib/5s-constants';
     7→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
     8→import { Badge } from '@/components/ui/badge';
     9→import { Button } from '@/components/ui/button';
    10→import { Input } from '@/components/ui/input';
    11→import { Textarea } from '@/components/ui/textarea';
    12→import { Label } from '@/components/ui/label';
    13→import {
    14→  Select,
    15→  SelectContent,
    16→  SelectItem,
    17→  SelectTrigger,
    18→  SelectValue,
    19→} from '@/components/ui/select';
    20→import {
    21→  Sparkles, Plan, CheckCircle2, ArrowRight, Plus, Trash2, Edit3,
    22→  Save, Loader2, BarChart3, Target, Clock, AlertTriangle, TrendingUp,
    23→  RotateCcw, ChevronDown, ChevronRight, FileText, Users, Calendar,
    24→} from 'lucide-react';
    25→
    26→// ═══════════════════════════════════════════════════════
    27→// PDCA Cycle Types
    28→// ═══════════════════════════════════════════════════════
    29→
    30→interface PDCAItem {
    31→  id: string;
    32→  title: string;
    33→  description: string;
    34→  phase: 'plan' | 'do' | 'check' | 'act';
    35→  sStep: number;
    36→  responsable: string | null;
    37→  prioridad: 'baja' | 'media' | 'alta' | 'critica';
    38→  estado: 'pendiente' | 'en_progreso' | 'completada' | 'cancelada';
    39→  fechaInicio: string;
    40→  fechaLimite: string | null;
    41→  resultado: string | null;
    42→  projectId: string;
    43→  zoneId: string | null;
    44→  createdAt: string;
    45→  updatedAt: string;
    46→}
    47→
    48→interface PDCAStat {
    49→  plan: number;
    50→  do: number;
    51→  check: number;
    52→  act: number;
    53→  completadas: number;
    54→  pendientes: number;
    55→  en_progreso: number;
    56→}
    57→
    58→const PHASE_CONFIG = {
    59→  plan: { label: 'Plan', color: 'bg-blue-100 text-blue-800 border-blue-300', icon: Target, description: 'Identificar oportunidades de mejora y planificar acciones' },
    60→  do: { label: 'Do', color: 'bg-green-100 text-green-800 border-green-300', icon: ArrowRight, description: 'Ejecutar las acciones planificadas' },
    61→  check: { label: 'Check', color: 'bg-amber-100 text-amber-800 border-amber-300', icon: BarChart3, description: 'Verificar resultados y comparar con objetivos' },
    62→  act: { label: 'Act', color: 'bg-purple-100 text-purple-800 border-purple-300', icon: TrendingUp, description: 'Estandarizar o ajustar según resultados' },
    63→};
    64→
    65→const PRIORIDAD_CONFIG = {
    66→  baja: { label: 'Baja', color: 'bg-gray-100 text-gray-700' },
    67→  media: { label: 'Media', color: 'bg-blue-100 text-blue-700' },
    68→  alta: { label: 'Alta', color: 'bg-orange-100 text-orange-700' },
    69→  critica: { label: 'Crítica', color: 'bg-red-100 text-red-700' },
    70→};
    71→
    72→const ESTADO_CONFIG = {
    73→  pendiente: { label: 'Pendiente', color: 'bg-gray-100 text-gray-700' },
    74→  en_progreso: { label: 'En Progreso', color: 'bg-blue-100 text-blue-700' },
    75→  completada: { label: 'Completada', color: 'bg-green-100 text-green-700' },
    76→  cancelada: { label: 'Cancelada', color: 'bg-red-100 text-red-700' },
    77→};
    78→
    79→export default function MejoraContinuaAdmin() {
    80→  const { currentProject, currentZone } = use5SStore();
    81→  const [activeView, setActiveView] = useState<'overview' | 'pdca' | 'actions' | 'indicators'>('overview');
    82→  const [items, setItems] = useState<PDCAItem[]>([]);
    83→  const [isLoading, setIsLoading] = useState(true);
    84→  const [showForm, setShowForm] = useState(false);
    85→  const [editingId, setEditingId] = useState<string | null>(null);
    86→  const [expandedId, setExpandedId] = useState<string | null>(null);
    87→
    88→  // Form state
    89→  const [formData, setFormData] = useState({
    90→    title: '',
    91→    description: '',
    92→    phase: 'plan' as 'plan' | 'do' | 'check' | 'act',
    93→    sStep: 1,
    94→    responsable: '',
    95→    prioridad: 'media' as 'baja' | 'media' | 'alta' | 'critica',
    96→    fechaLimite: '',
    97→  });
    98→
    99→  useEffect(() => {
   100→    loadItems();
   101→  }, [currentProject, currentZone]);
   102→
   103→  const loadItems = async () => {
   104→    if (!currentProject?.id) return;
   105→    setIsLoading(true);
   106→    try {
   107→      const params = new URLSearchParams({ projectId: currentProject.id });
   108→      if (currentZone?.id) params.set('zoneId', currentZone.id);
   109→      const res = await fetch(`/api/pdca?${params}`);
   110→      const data = await res.json();
   111→      if (data.success) {
   112→        setItems(data.data || []);
   113→      }
   114→    } catch (e) {
   115→      console.error('Error loading PDCA items:', e);
   116→    } finally {
   117→      setIsLoading(false);
   118→    }
   119→  };
   120→
   121→  const handleSave = async () => {
   122→    if (!currentProject?.id || !formData.title.trim()) return;
   123→    try {
   124→      const body: any = {
   125→        ...formData,
   126→        projectId: currentProject.id,
   127→        zoneId: currentZone?.id || null,
   128→        responsable: formData.responsable || null,
   129→        fechaLimite: formData.fechaLimite || null,
   130→      };
   131→
   132→      if (editingId) {
   133→        body.id = editingId;
   134→        await fetch('/api/pdca', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) });
   135→      } else {
   136→        await fetch('/api/pdca', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) });
   137→      }
   138→      await loadItems();
   139→      resetForm();
   140→    } catch (e) {
   141→      console.error('Error saving PDCA item:', e);
   142→    }
   143→  };
   144→
   145→  const handleDelete = async (id: string) => {
   146→    if (!confirm('¿Eliminar este elemento?')) return;
   147→    try {
   148→      await fetch(`/api/pdca?id=${id}`, { method: 'DELETE' });
   149→      await loadItems();
   150→    } catch (e) {
   151→      console.error('Error deleting PDCA item:', e);
   152→    }
   153→  };
   154→
   155→  const handleStatusChange = async (id: string, estado: string) => {
   156→    try {
   157→      await fetch('/api/pdca', {
   158→        method: 'PUT',
   159→        headers: { 'Content-Type': 'application/json' },
   160→        body: JSON.stringify({ id, estado }),
   161→      });
   162→      await loadItems();
   163→    } catch (e) {
   164→      console.error('Error updating status:', e);
   165→    }
   166→  };
   167→
   168→  const resetForm = () => {
   169→    setShowForm(false);
   170→    setEditingId(null);
   171→    setFormData({ title: '', description: '', phase: 'plan', sStep: 1, responsable: '', prioridad: 'media', fechaLimite: '' });
   172→  };
   173→
   174→  const startEdit = (item: PDCAItem) => {
   175→    setEditingId(item.id);
   176→    setFormData({
   177→      title: item.title,
   178→      description: item.description || '',
   179→      phase: item.phase,
   180→      sStep: item.sStep,
   181→      responsable: item.responsable || '',
   182→      prioridad: item.prioridad,
   183→      fechaLimite: item.fechaLimite || '',
   184→    });
   185→    setShowForm(true);
   186→  };
   187→
   188→  // Stats
   189→  const stats: PDCAStat = {
   190→    plan: items.filter(i => i.phase === 'plan').length,
   191→    do: items.filter(i => i.phase === 'do').length,
   192→    check: items.filter(i => i.phase === 'check').length,
   193→    act: items.filter(i => i.phase === 'act').length,
   194→    completadas: items.filter(i => i.estado === 'completada').length,
   195→    pendientes: items.filter(i => i.estado === 'pendiente').length,
   196→    en_progreso: items.filter(i => i.estado === 'en_progreso').length,
   197→  };
   198→
   199→  return (
   200→    <div className="space-y-6">
   201→      {/* Header */}
   202→      <div className="flex items-center justify-between">
   203→        <div>
   204→          <h2 className="text-xl font-bold text-gray-900 flex items-center gap-2">
   205→            <Sparkles className="h-5 w-5 text-green-500" />
   206→            Mejora Continua — Ciclo PDCA
   207→          </h2>
   208→          <p className="text-sm text-muted-foreground mt-1">
   209→            Gestión del ciclo Plan-Do-Check-Act para la mejora continua tras completar las 5S
   210→          </p>
   211→        </div>
   212→        <Button onClick={() => { resetForm(); setShowForm(true); }} className="gap-2 bg-green-600 hover:bg-green-700 text-white">
   213→          <Plus className="h-4 w-4" /> Nueva Acción
   214→        </Button>
   215→      </div>
   216→
   217→      {/* Sub-tabs */}
   218→      <div className="flex gap-1 border-b">
   219→        {[
   220→          { key: 'overview', label: 'Resumen PDCA', icon: BarChart3 },
   221→          { key: 'pdca', label: 'Ciclo PDCA', icon: RotateCcw },
   222→          { key: 'actions', label: 'Acciones', icon: FileText },
   223→          { key: 'indicators', label: 'Indicadores', icon: TrendingUp },
   224→        ].map(tab => (
   225→          <button
   226→            key={tab.key}
   227→            onClick={() => setActiveView(tab.key as any)}
   228→            className={`flex items-center gap-2 px-4 py-2.5 text-sm font-medium border-b-2 transition-colors ${
   229→              activeView === tab.key
   230→                ? 'border-green-500 text-green-700'
   231→                : 'border-transparent text-muted-foreground hover:text-foreground'
   232→            }`}
   233→          >
   234→            <tab.icon className="h-4 w-4" />
   235→            {tab.label}
   236→          </button>
   237→        ))}
   238→      </div>
   239→
   240→      <AnimatePresence mode="wait">
   241→        {/* ═══ OVERVIEW ═══ */}
   242→        {activeView === 'overview' && (
   243→          <motion.div key="overview" initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0 }} className="space-y-6">
   244→            {/* PDCA Stats Cards */}
   245→            <div className="grid grid-cols-4 gap-4">
   246→              {Object.entries(PHASE_CONFIG).map(([key, cfg]) => (
   247→                <Card key={key} className="border-l-4" style={{ borderLeftColor: key === 'plan' ? '#3B82F6' : key === 'do' ? '#22C55E' : key === 'check' ? '#F59E0B' : '#8B5CF6' }}>
   248→                  <CardContent className="p-4">
   249→                    <div className="flex items-center justify-between mb-2">
   250→                      <cfg.icon className="h-5 w-5" style={{ color: key === 'plan' ? '#3B82F6' : key === 'do' ? '#22C55E' : key === 'check' ? '#F59E0B' : '#8B5CF6' }} />
   251→                      <span className="text-2xl font-bold">{stats[key as keyof PDCAStat]}</span>
   252→                    </div>
   253→                    <p className="text-sm font-semibold">{cfg.label}</p>
   254→                    <p className="text-xs text-muted-foreground mt-1">{cfg.description}</p>
   255→                  </CardContent>
   256→                </Card>
   257→              ))}
   258→            </div>
   259→
   260→            {/* Summary */}
   261→            <div className="grid grid-cols-3 gap-4">
   262→              <Card>
   263→                <CardContent className="p-4 text-center">
   264→                  <Clock className="h-6 w-6 text-gray-500 mx-auto mb-2" />
   265→                  <p className="text-2xl font-bold">{stats.pendientes}</p>
   266→                  <p className="text-sm text-muted-foreground">Pendientes</p>
   267→                </CardContent>
   268→              </Card>
   269→              <Card>
   270→                <CardContent className="p-4 text-center">
   271→                  <ArrowRight className="h-6 w-6 text-blue-500 mx-auto mb-2" />
   272→                  <p className="text-2xl font-bold">{stats.en_progreso}</p>
   273→                  <p className="text-sm text-muted-foreground">En Progreso</p>
   274→                </CardContent>
   275→              </Card>
   276→              <Card>
   277→                <CardContent className="p-4 text-center">
   278→                  <CheckCircle2 className="h-6 w-6 text-green-500 mx-auto mb-2" />
   279→                  <p className="text-2xl font-bold">{stats.completadas}</p>
   280→                  <p className="text-sm text-muted-foreground">Completadas</p>
   281→                </CardContent>
   282→              </Card>
   283→            </div>
   284→
   285→            {/* Recent items */}
   286→            <Card>
   287→              <CardHeader>
   288→                <CardTitle className="text-base">Últimas Acciones</CardTitle>
   289→              </CardHeader>
   290→              <CardContent>
   291→                {items.length === 0 ? (
   292→                  <p className="text-sm text-muted-foreground text-center py-8">No hay acciones de mejora registradas. Crea la primera acción para comenzar el ciclo PDCA.</p>
   293→                ) : (
   294→                  <div className="space-y-2">
   295→                    {items.slice(0, 5).map(item => (
   296→                      <div key={item.id} className="flex items-center gap-3 p-3 rounded-lg border hover:bg-gray-50">
   297→                        <Badge className={PHASE_CONFIG[item.phase].color}>{PHASE_CONFIG[item.phase].label}</Badge>
   298→                        <span className="text-sm font-medium flex-1">{item.title}</span>
   299→                        <Badge variant="outline" className={ESTADO_CONFIG[item.estado].color}>{ESTADO_CONFIG[item.estado].label}</Badge>
   300→                        <Badge variant="outline" className={PRIORIDAD_CONFIG[item.prioridad].color}>{PRIORIDAD_CONFIG[item.prioridad].label}</Badge>
   301→                      </div>
   302→                    ))}
   303→                  </div>
   304→                )}
   305→              </CardContent>
   306→            </Card>
   307→          </motion.div>
   308→        )}
   309→
   310→        {/* ═══ PDCA CYCLE VIEW ═══ */}
   311→        {activeView === 'pdca' && (
   312→          <motion.div key="pdca" initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0 }} className="space-y-4">
   313→            {/* PDCA Visual Cycle */}
   314→            <div className="grid grid-cols-2 gap-4">
   315→              {Object.entries(PHASE_CONFIG).map(([key, cfg]) => {
   316→                const phaseItems = items.filter(i => i.phase === key);
   317→                return (
   318→                  <Card key={key} className="border-t-4" style={{ borderTopColor: key === 'plan' ? '#3B82F6' : key === 'do' ? '#22C55E' : key === 'check' ? '#F59E0B' : '#8B5CF6' }}>
   319→                    <CardHeader className="pb-2">
   320→                      <CardTitle className="text-base flex items-center gap-2">
   321→                        <cfg.icon className="h-4 w-4" />
   322→                        {cfg.label} — {cfg.description.split(' ').slice(0, 3).join(' ')}
   323→                      </CardTitle>
   324→                    </CardHeader>
   325→                    <CardContent>
   326→                      {phaseItems.length === 0 ? (
   327→                        <p className="text-xs text-muted-foreground text-center py-4">Sin acciones en esta fase</p>
   328→                      ) : (
   329→                        <div className="space-y-2 max-h-64 overflow-y-auto">
   330→                          {phaseItems.map(item => (
   331→                            <div key={item.id} className="p-2 rounded border text-sm hover:bg-gray-50 cursor-pointer" onClick={() => setExpandedId(expandedId === item.id ? null : item.id)}>
   332→                              <div className="flex items-center gap-2">
   333→                                <Badge variant="outline" className={ESTADO_CONFIG[item.estado].color}>{ESTADO_CONFIG[item.estado].label}</Badge>
   334→                                <span className="font-medium truncate">{item.title}</span>
   335→                              </div>
   336→                              {expandedId === item.id && (
   337→                                <div className="mt-2 space-y-1 text-xs text-muted-foreground">
   338→                                  {item.description && <p>{item.description}</p>}
   339→                                  {item.responsable && <p>Responsable: {item.responsable}</p>}
   340→                                  {item.fechaLimite && <p>Fecha límite: {new Date(item.fechaLimite).toLocaleDateString('es-ES')}</p>}
   341→                                  <div className="flex gap-1 mt-2">
   342→                                    {item.estado !== 'completada' && (
   343→                                      <Button size="sm" variant="outline" className="h-6 text-[10px]" onClick={(e) => { e.stopPropagation(); handleStatusChange(item.id, 'completada'); }}>Completar</Button>
   344→                                    )}
   345→                                    <Button size="sm" variant="outline" className="h-6 text-[10px]" onClick={(e) => { e.stopPropagation(); startEdit(item); }}>Editar</Button>
   346→                                    <Button size="sm" variant="outline" className="h-6 text-[10px] text-red-600" onClick={(e) => { e.stopPropagation(); handleDelete(item.id); }}>Eliminar</Button>
   347→                                  </div>
   348→                                </div>
   349→                              )}
   350→                            </div>
   351→                          ))}
   352→                        </div>
   353→                      )}
   354→                    </CardContent>
   355→                  </Card>
   356→                );
   357→              })}
   358→            </div>
   359→          </motion.div>
   360→        )}
   361→
   362→        {/* ═══ ACTIONS LIST ═══ */}
   363→        {activeView === 'actions' && (
   364→          <motion.div key="actions" initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0 }} className="space-y-4">
   365→            {/* Filters */}
   366→            <div className="flex gap-3 items-center flex-wrap">
   367→              <Select value={formData.phase} onValueChange={(v) => setFormData(prev => ({ ...prev, phase: v as any }))}>
   368→                <SelectTrigger className="w-40"><SelectValue placeholder="Fase" /></SelectTrigger>
   369→                <SelectContent>
   370→                  <SelectItem value="plan">Plan</SelectItem>
   371→                  <SelectItem value="do">Do</SelectItem>
   372→                  <SelectItem value="check">Check</SelectItem>
   373→                  <SelectItem value="act">Act</SelectItem>
   374→                </SelectContent>
   375→              </Select>
   376→            </div>
   377→
   378→            {/* Items table */}
   379→            {isLoading ? (
   380→              <div className="flex justify-center py-12"><Loader2 className="h-6 w-6 animate-spin text-green-500" /></div>
   381→            ) : items.length === 0 ? (
   382→              <div className="text-center py-12">
   383→                <Sparkles className="h-10 w-10 text-green-300 mx-auto mb-3" />
   384→                <p className="text-sm text-muted-foreground">No hay acciones de mejora. Crea la primera para iniciar el ciclo PDCA.</p>
   385→              </div>
   386→            ) : (
   387→              <div className="border rounded-lg overflow-hidden">
   388→                <table className="w-full text-sm">
   389→                  <thead className="bg-gray-50 border-b">
   390→                    <tr>
   391→                      <th className="px-4 py-3 text-left font-medium">Fase</th>
   392→                      <th className="px-4 py-3 text-left font-medium">Título</th>
   393→                      <th className="px-4 py-3 text-left font-medium">S-Step</th>
   394→                      <th className="px-4 py-3 text-left font-medium">Prioridad</th>
   395→                      <th className="px-4 py-3 text-left font-medium">Estado</th>
   396→                      <th className="px-4 py-3 text-left font-medium">Responsable</th>
   397→                      <th className="px-4 py-3 text-left font-medium">Acciones</th>
   398→                    </tr>
   399→                  </thead>
   400→                  <tbody className="divide-y">
   401→                    {items.map(item => (
   402→                      <tr key={item.id} className="hover:bg-gray-50">
   403→                        <td className="px-4 py-3"><Badge className={PHASE_CONFIG[item.phase].color}>{PHASE_CONFIG[item.phase].label}</Badge></td>
   404→                        <td className="px-4 py-3 font-medium">{item.title}</td>
   405→                        <td className="px-4 py-3"><Badge variant="outline" style={{ color: S_STEPS.find(s => s.id === item.sStep)?.color }}>S{item.sStep}</Badge></td>
   406→                        <td className="px-4 py-3"><Badge variant="outline" className={PRIORIDAD_CONFIG[item.prioridad].color}>{PRIORIDAD_CONFIG[item.prioridad].label}</Badge></td>
   407→                        <td className="px-4 py-3">
   408→                          <Select value={item.estado} onValueChange={(v) => handleStatusChange(item.id, v)}>
   409→                            <SelectTrigger className="h-7 w-32 text-xs"><SelectValue /></SelectTrigger>
   410→                            <SelectContent>
   411→                              <SelectItem value="pendiente">Pendiente</SelectItem>
   412→                              <SelectItem value="en_progreso">En Progreso</SelectItem>
   413→                              <SelectItem value="completada">Completada</SelectItem>
   414→                              <SelectItem value="cancelada">Cancelada</SelectItem>
   415→                            </SelectContent>
   416→                          </Select>
   417→                        </td>
   418→                        <td className="px-4 py-3 text-muted-foreground">{item.responsable || '—'}</td>
   419→                        <td className="px-4 py-3">
   420→                          <div className="flex gap-1">
   421→                            <Button size="sm" variant="ghost" className="h-7 w-7 p-0" onClick={() => startEdit(item)}><Edit3 className="h-3.5 w-3.5" /></Button>
   422→                            <Button size="sm" variant="ghost" className="h-7 w-7 p-0 text-red-500" onClick={() => handleDelete(item.id)}><Trash2 className="h-3.5 w-3.5" /></Button>
   423→                          </div>
   424→                        </td>
   425→                      </tr>
   426→                    ))}
   427→                  </tbody>
   428→                </table>
   429→              </div>
   430→            )}
   431→          </motion.div>
   432→        )}
   433→
   434→        {/* ═══ INDICATORS ═══ */}
   435→        {activeView === 'indicators' && (
   436→          <motion.div key="indicators" initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0 }} className="space-y-4">
   437→            <Card>
   438→              <CardHeader>
   439→                <CardTitle className="text-base flex items-center gap-2"><TrendingUp className="h-4 w-4" /> Indicadores de Mejora Continua</CardTitle>
   440→              </CardHeader>
   441→              <CardContent>
   442→                <div className="grid grid-cols-2 gap-6">
   443→                  {/* Completion rate */}
   444→                  <div>
   445→                    <p className="text-sm font-medium mb-2">Tasa de Completación</p>
   446→                    <div className="flex items-center gap-3">
   447→                      <div className="flex-1 bg-gray-200 rounded-full h-4">
   448→                        <div className="bg-green-500 h-4 rounded-full transition-all" style={{ width: `${items.length > 0 ? Math.round((stats.completadas / items.length) * 100) : 0}%` }} />
