     1→'use client';
     2→
     3→import { useState, useEffect, useCallback } from 'react';
     4→import { Button } from '@/components/ui/button';
     5→import { Badge } from '@/components/ui/badge';
     6→import { Input } from '@/components/ui/input';
     7→import {
     8→  Select,
     9→  SelectContent,
    10→  SelectItem,

    12→  SelectValue,
    13→} from '@/components/ui/select';
    14→import {

    18→import { toast } from 'sonner';

    20→import { INVENTORY_CONFIGS } from '@/lib/5s-constants';
    21→import type { InventoryConfig } from '@/lib/5s-constants';
    22→import TagPrinter from '@/components/5s/TagPrinter';
    23→

    41→  price: number | null;
    42→  photoUrl: string | null;
    43→  photoUrls: string | null;

    58→const defaultConfig = INVENTORY_CONFIGS[1];
    59→

    61→  { value: 'en_jaula', label: 'En Jaula', color: 'bg-red-100 text-red-800' },
    62→  { value: 'reclamado', label: 'Reclamado', color: 'bg-amber-100 text-amber-800' },
    63→  { value: 'transferido', label: 'Transferido', color: 'bg-green-100 text-green-800' },
    64→];

    66→// ═══════════════════════════════════════════════════════
    67→// Component
    68→// ═══════════════════════════════════════════════════════
    69→export default function JaulaView() {
    70→  const { currentProject, currentZone } = use5SStore();
    71→  const [jaulaItems, setJaulaItems] = useState<JaulaItem[]>([]);
    72→  const [jaulaFilter, setJaulaFilter] = useState<string>('all');
    73→  const [isLoading, setIsLoading] = useState(true);
    74→  const [lightboxPhoto, setLightboxPhoto] = useState<string | null>(null);
    75→  const [customConfig, setCustomConfig] = useState<InventoryConfig | null>(null);
    76→  const [hasTemplate, setHasTemplate] = useState<boolean | null>(null);
    77→
    78→  const config: InventoryConfig = customConfig || INVENTORY_CONFIGS[1];
    79→
    80→  const loadCustomInventoryConfig = async () => {
    81→    try {
    82→      // If the zone has a board config, fetch inventory template from that config
    83→      if (currentZone?.boardConfigId) {
    84→        const slotsRes = await fetch(`/api/board-slots?boardConfigId=${currentZone.boardConfigId}&sStep=1&miniStep=3`);
    85→        const slotsJson = await slotsRes.json();
    86→        if (slotsJson.success && slotsJson.data.length > 0) {
    87→          const slot = slotsJson.data[0];
    88→          const inventarioTemplates = (slot.templates || []).filter(
    89→            (t: any) => t.template?.type === 'inventario'
    90→          );
    91→          if (inventarioTemplates.length > 0) {
    92→            const content = JSON.parse(inventarioTemplates[0].template.content);
    93→            if (content.categories && content.extraFields) {
    94→              setCustomConfig({
    95→                title: content.title || defaultConfig.title,
    96→                subtitle: content.subtitle || defaultConfig.subtitle,
    97→                templateName: content.templateName || defaultConfig.templateName,
    98→                categories: content.categories,
    99→                extraFields: content.extraFields,
   100→                ...(content.desplegables_jerarquicos ? { desplegables_jerarquicos: content.desplegables_jerarquicos } : {}),
   101→              });
   102→              setHasTemplate(true);
   103→            } else {
   104→              // Unknown format — use default config as fallback
   105→              setCustomConfig(null);
   106→              setHasTemplate(true);
   107→            }
   108→          } else {
   109→            // No inventario template assigned in this board slot — use default config
   110→            setCustomConfig(null);
   111→            setHasTemplate(true);
   112→          }
   113→        } else {
   114→          // No slot configured for this step — use default config
   115→          setCustomConfig(null);
   116→          setHasTemplate(true);
   117→        }
   118→      } else {
   119→        // Fallback: load global template
   120→        const res = await fetch(`/api/templates?type=inventario&sStep=1&miniStep=3`);
   121→        const json = await res.json();
   122→        if (json.success && json.data && json.data.length > 0) {
   123→          const content = JSON.parse(json.data[0].content);
   124→          if (content.categories && content.extraFields) {
   125→            setCustomConfig({
   126→              title: content.title || defaultConfig.title,
   127→              subtitle: content.subtitle || defaultConfig.subtitle,
   128→              templateName: content.templateName || defaultConfig.templateName,
   129→              categories: content.categories,
   130→              extraFields: content.extraFields,
   131→              ...(content.desplegables_jerarquicos ? { desplegables_jerarquicos: content.desplegables_jerarquicos } : {}),
   132→            });
   133→            setHasTemplate(true);
   134→          } else {
   135→            // Unknown format — use default config as fallback
   136→            setCustomConfig(null);
   137→            setHasTemplate(true);
   138→          }
   139→        } else {
   140→          // No global template — use default config (INVENTORY_CONFIGS has entries for all 5 S steps)
   141→          setCustomConfig(null);
   142→          setHasTemplate(true);
   143→        }
   144→      }
   145→    } catch (e) {
   146→      console.error('Error loading custom inventory config:', e);
   147→      // On error, use default config so the view still works
   148→      setCustomConfig(null);
   149→      setHasTemplate(true);
   150→    }
   151→  };
   152→
   153→  // Load data on mount
   154→  useEffect(() => {
   155→    loadJaulaItems();
   156→    loadCustomInventoryConfig();
   157→  }, [currentProject, currentZone]);
   158→
   159→  const loadJaulaItems = async () => {
   160→    setIsLoading(true);
   161→    try {
   162→      let url = '/api/inventory?jaulaOnly=true';
   163→      if (currentProject?.id) url += `&projectId=${currentProject.id}`;
   164→      if (currentZone?.id) url += `&zoneId=${currentZone.id}`;
   165→      const res = await fetch(url);
   166→      const json = await res.json();
   167→      if (json.success) {
   168→        setJaulaItems(json.data || []);
   169→      }
   170→    } catch (error) {
   171→      console.error('Error loading jaula items:', error);
   172→      toast.error('Error al cargar la jaula');
   173→    } finally {
   174→      setIsLoading(false);
   175→    }
   176→  };
   177→
   178→  const handleAddItem = async () => {
   179→    if (!currentProject?.id) {
   180→      toast.error('No hay proyecto seleccionado');
   181→      return;
   182→    }
   183→    try {
   184→      const res = await fetch('/api/inventory', {
   185→        method: 'POST',
   186→        headers: { 'Content-Type': 'application/json' },
   187→        body: JSON.stringify({
   188→          sStep: 1,
   189→          name: '',
   190→          location: currentZone?.name || '',
   191→          category: 'innecesario',
   192→          quantity: 1,
   193→          quantityUnneeded: 1,
   194→          price: null,
   195→          action: 'jaula',
   196→          projectId: currentProject.id,
   197→          zoneId: currentZone?.id || null,
   198→          jaulaStatus: 'en_jaula',
   199→          jaulaFechaEntrada: new Date().toISOString(),
   200→          extra: JSON.stringify({
   201→            estado: '',
   202→            frecuenciaUso: '',
   203→            decision: 'Retirar', // v2.50: was 'Jaula'
   204→            diasCuarentena: '40',
   205→          }),
   206→        }),
   207→      });
   208→      const json = await res.json();
   209→      if (json.success) {
   210→        toast.success('Nueva entrada agregada a la Jaula');
   211→        await loadJaulaItems();
   212→      } else {
   213→        toast.error(`Error: ${json.error || 'Error desconocido'}`);
   214→      }
   215→    } catch (e) {
   216→      console.error('Error adding item:', e);
   217→      toast.error('Error de conexión');
   218→    }
   219→  };
   220→
   221→  const handleUpdateField = async (itemId: string, field: string, value: any) => {
   222→    setJaulaItems(prev => prev.map(item => {
   223→      if (item.id !== itemId) return item;
   224→      if (field.startsWith('extra.')) {
   225→        const extraKey = field.replace('extra.', '');
   226→        return { ...item, extra: { ...item.extra, [extraKey]: value } };
   227→      }
   228→      return { ...item, [field]: value };
   229→    }));
   230→
   231→    try {
   232→      let body: any;
   233→      if (field.startsWith('extra.')) {
   234→        const extraKey = field.replace('extra.', '');
   235→        const item = jaulaItems.find(i => i.id === itemId);
   236→        const updatedExtra = { ...(item?.extra || {}), [extraKey]: value };
   237→        body = { extra: JSON.stringify(updatedExtra) };
   238→      } else {
   239→        body = { [field]: value };
   240→      }
   241→      const res = await fetch(`/api/inventory?id=${itemId}`, {
   242→        method: 'PUT',
   243→        headers: { 'Content-Type': 'application/json' },
   244→        body: JSON.stringify(body),
   245→      });
   246→      const json = await res.json();
   247→      if (!json.success) {
   248→        toast.error(`Error al actualizar: ${json.error || 'Error'}`);
   249→        await loadJaulaItems();
   250→      }
   251→    } catch (e) {
   252→      console.error('Error updating:', e);
   253→      await loadJaulaItems();
   254→    }
   255→  };
   256→
   257→  const filteredJaulaItems = jaulaFilter === 'all'
   258→    ? jaulaItems
   259→    : jaulaItems.filter(i => i.jaulaStatus === jaulaFilter);
   260→
   261→  const totalJaulaValue = filteredJaulaItems.reduce(
   262→    (sum, i) => sum + (i.price || 0) * (i.quantityUnneeded || i.quantity), 0
   263→  );
   264→
   265→  // Stats
   266→  const enJaula = jaulaItems.filter(i => i.jaulaStatus === 'en_jaula').length;
   267→  const reclamados = jaulaItems.filter(i => i.jaulaStatus === 'reclamado').length;
   268→  const transferidos = jaulaItems.filter(i => i.jaulaStatus === 'transferido').length;
   269→
   270→  const getStatusBadge = (status: string) => {
   271→    const opt = JAULA_STATUS.find(s => s.value === status);
   272→    if (!opt) return <Badge className="bg-gray-100 text-gray-400 text-[10px]">—</Badge>;
   273→    return <Badge className={`text-[10px] px-1.5 py-0 ${opt.color}`}>{opt.label}</Badge>;
   274→  };
   275→
   276→  // v2.50: helper para normalizar decision (legacy 'Jaula' → 'Retirar', 'Tirar' → 'Eliminar')
   277→  const displayDecision = (d?: string | null): string => {
   278→    if (!d || d === 'Retirar' || d === 'Jaula') return 'Retirar';
   279→    if (d === 'Tirar') return 'Eliminar';
   280→    return d;
   281→  };
   282→  const isJaulaDecision = (d?: string | null) => !d || d === 'Retirar' || d === 'Jaula';
   283→
   284→  // Tag data for printing
   285→  const tagItemsAndIds = filteredJaulaItems
   286→    .filter(i => isJaulaDecision(i.extra?.decision))
   287→    .map(i => {
   288→      const dias = Number(i.extra?.diasCuarentena ?? 40);
   289→      let fechaRevision: string | null = null;
   290→      if (i.jaulaFechaEntrada) {
   291→        try {
   292→          const d = new Date(i.jaulaFechaEntrada);
   293→          d.setDate(d.getDate() + dias);
   294→          fechaRevision = d.toISOString();
   295→        } catch {}
   296→      }
   297→      return {
   298→        itemId: i.id, // v2.48: para que TagPrinter pueda persistir el snapshot
   299→        tag: {
   300→          nombre: i.name,
   301→          ubicacion: i.location,
   302→          cantidad: i.quantityUnneeded || i.quantity,
   303→          estado: String(i.extra?.estado ?? ''),
   304→          frecuenciaUso: String(i.extra?.frecuenciaUso ?? ''),
   305→          decision: 'Retirar' as string,
   306→          fechaEntrada: i.jaulaFechaEntrada,
   307→          fechaRevision,
   308→          diasCuarentena: dias,
   309→          zonaOrigen: i.zonaOrigen || i.jaulaOrigen,
   310→        },
   311→      };
   312→    });
   313→  const tagItems = tagItemsAndIds.map(x => x.tag);
   314→  const tagItemIds = tagItemsAndIds.map(x => x.itemId);
   315→
   316→  return (
   317→    <div className="flex flex-col h-full">
   318→      {/* Header */}
   319→      <div className="px-3 sm:px-4 py-2.5 border-b bg-white shrink-0">
   320→        <div className="flex items-center justify-between gap-2">
   321→          <div className="flex items-center gap-2 min-w-0">
   322→            <Package className="h-5 w-5 text-red-600 shrink-0" />
   323→            <h2 className="text-base sm:text-lg font-bold truncate">Jaula de Excedentes</h2>
   324→            {currentZone && (
   325→              <Badge variant="outline" className="border-red-300 text-red-700 text-[10px] shrink-0 hidden sm:inline-flex">
   326→                {currentZone.name}
   327→              </Badge>
   328→            )}
   329→            <Badge className="bg-red-100 text-red-800 text-[10px] shrink-0">
   330→              {jaulaItems.length} elemento{jaulaItems.length !== 1 ? 's' : ''}
   331→            </Badge>
   332→          </div>
   333→          <Button size="sm" className="gap-1 text-xs h-8 shrink-0 bg-red-600 hover:bg-red-700" onClick={handleAddItem}>
   334→            <Plus className="h-3.5 w-3.5" />
   335→            <span className="hidden sm:inline">Nueva Entrada</span>
   336→            <span className="sm:hidden">Nueva</span>
   337→          </Button>
   338→        </div>
   339→
   340→        {/* Info panel */}
   341→        <div className="p-2.5 rounded-lg border-l-4 border-red-500 bg-red-50/30 mt-2">
   342→          <div className="flex items-center gap-2 mb-0.5">
   343→            <div className="w-3 h-3 rounded bg-red-500" />
   344→            <span className="text-xs font-medium text-red-800">Etiqueta ROJA — Innecesario</span>
   345→          </div>
   346→          <p className="text-[11px] text-muted-foreground">
   347→            Elementos innecesarios clasificados en S1. Se envían a la JAULA o se ELIMINAN.
   348→          </p>
   349→        </div>
   350→
   351→        {/* Filter pills */}
   352→        <div className="flex items-center gap-1.5 mt-2 overflow-x-auto pb-1">
   353→          {tagItems.length > 0 && <TagPrinter items={tagItems} itemIds={tagItemIds} onAfterPrint={loadJaulaItems} />}
   354→          {tagItems.length > 0 && <div className="h-4 w-px bg-gray-200 mx-1 shrink-0" />}
   355→          {/* Status filters */}
   356→          {[
   357→            { key: 'all', label: 'Todos', count: jaulaItems.length, active: jaulaFilter === 'all' },
   358→            { key: 'en_jaula', label: 'En Jaula', count: enJaula, active: jaulaFilter === 'en_jaula' },
   359→            { key: 'reclamado', label: 'Reclamado', count: reclamados, active: jaulaFilter === 'reclamado' },
   360→            { key: 'transferido', label: 'Transferido', count: transferidos, active: jaulaFilter === 'transferido' },
   361→          ].map(f => (
   362→            <button
   363→              key={f.key}
   364→              onClick={() => setJaulaFilter(f.key)}
   365→              className={`px-2 py-1 rounded-full text-[10px] font-medium border transition-colors whitespace-nowrap ${
   366→                f.active ? 'bg-gray-900 text-white border-gray-900' : 'bg-white text-gray-600 border-gray-200 hover:bg-gray-50'
   367→              }`}
   368→            >
   369→              {f.label} {f.count}
   370→            </button>
   371→          ))}
   372→          {jaulaFilter !== 'all' && (
   373→            <button onClick={() => setJaulaFilter('all')}
   374→              className="flex items-center gap-0.5 px-2 py-1 rounded-full text-[10px] text-gray-500 hover:bg-gray-100">
   375→              <X className="h-3 w-3" /> Limpiar
   376→            </button>
   377→          )}
   378→        </div>
   379→      </div>
   380→
   381→      {/* Content */}
   382→      <div className="flex-1 overflow-hidden flex flex-col px-3 sm:px-4 pb-4 min-h-0">
   383→        {isLoading ? (
   384→          <div className="flex-1 flex items-center justify-center">
   385→            <Loader2 className="h-8 w-8 text-red-500 animate-spin" />
   386→          </div>
   387→        ) : filteredJaulaItems.length === 0 ? (
   388→          <div className="flex-1 flex flex-col items-center justify-center text-center">
   389→            <Package className="h-10 w-10 text-gray-300 mb-2" />
   390→            <p className="text-sm text-muted-foreground">
   391→              {jaulaItems.length === 0 ? 'No hay elementos en la jaula' : 'No hay elementos con este filtro'}
   392→            </p>
   393→            {jaulaItems.length === 0 && (
   394→              <Button size="sm" className="mt-3 gap-1 text-xs bg-red-600" onClick={handleAddItem}>
   395→                <Plus className="h-3.5 w-3.5" /> Registrar primer elemento
   396→              </Button>
   397→            )}
   398→          </div>
   399→        ) : (
   400→          <div className="flex-1 overflow-auto">
   401→            {/* Desktop table */}
   402→            <div className="overflow-x-auto border rounded-lg hidden md:block">
   403→              <table className="w-full text-xs border-collapse min-w-[1000px]">
   404→                <thead className="sticky top-0 z-10">
   405→                  <tr>
   406→                    <th className="bg-red-500 text-white px-2 py-1.5 text-center font-bold border border-red-600">ELEMENTO</th>
   407→                    <th className="bg-amber-400 text-white px-2 py-1.5 text-center font-bold border border-amber-500" colSpan={4}>CLASIFICACIÓN</th>
   408→                    <th className="bg-sky-400 text-white px-2 py-1.5 text-center font-bold border border-sky-500" colSpan={3}>JAULA</th>
   409→                    <th className="bg-green-500 text-white px-2 py-1.5 text-center font-bold border border-green-600" colSpan={2}>SEGUIMIENTO</th>
   410→                    <th className="bg-gray-400 text-white px-1 py-1.5 text-center font-bold border border-gray-500 w-8">🗑</th>
   411→                  </tr>
   412→                  <tr>
   413→                    <th className="bg-red-400 text-white px-1 py-1 font-semibold border border-red-300 whitespace-nowrap">Nombre</th>
   414→                    <th className="bg-amber-300 text-white px-1 py-1 font-semibold border border-amber-300 whitespace-nowrap">Ubicación</th>
   415→                    <th className="bg-amber-300 text-white px-1 py-1 font-semibold border border-amber-300 whitespace-nowrap">Cant.</th>
   416→                    <th className="bg-amber-300 text-white px-1 py-1 font-semibold border border-amber-300 whitespace-nowrap">Estado</th>
   417→                    <th className="bg-amber-300 text-white px-1 py-1 font-semibold border border-amber-300 whitespace-nowrap">Decisión</th>
   418→                    <th className="bg-sky-300 text-white px-1 py-1 font-semibold border border-sky-300 whitespace-nowrap">F. Entrada</th>
   419→                    <th className="bg-sky-300 text-white px-1 py-1 font-semibold border border-sky-300 whitespace-nowrap">Días cuar.</th>
   420→                    <th className="bg-sky-300 text-white px-1 py-1 font-semibold border border-sky-300 whitespace-nowrap">F. Límite</th>
   421→                    <th className="bg-green-400 text-white px-1 py-1 font-semibold border border-green-400 whitespace-nowrap">Estado Jaula</th>
   422→                    <th className="bg-green-400 text-white px-1 py-1 font-semibold border border-green-400 whitespace-nowrap">F. Salida / Destino</th>
   423→                  </tr>
   424→                </thead>
   425→                <tbody>
   426→                  {filteredJaulaItems.map(item => {
   427→                    const diasCuarentena = Number(item.extra?.diasCuarentena ?? 40);
   428→                    let fechaLimite: string | null = item.jaulaFechaLimite || null;
   429→                    if (!fechaLimite && item.jaulaFechaEntrada) {
   430→                      try {
   431→                        const d = new Date(item.jaulaFechaEntrada);
   432→                        d.setDate(d.getDate() + diasCuarentena);
   433→                        fechaLimite = d.toISOString();
   434→                      } catch {}
   435→                    }
   436→                    const isExpired = fechaLimite && new Date(fechaLimite) < new Date();
   437→
   438→                    return (
   439→                      <tr key={item.id} className={`border-b hover:bg-gray-50 ${
   440→                        item.jaulaStatus === 'transferido' ? 'bg-green-50/30' : ''
   441→                      }`}>
   442→                        {/* Elemento */}
   443→                        <td className="px-1 py-1 border bg-red-50 font-medium">
   444→                          <Input value={item.name} onChange={e => handleUpdateField(item.id, 'name', e.target.value)}
   445→                            className="h-6 text-[10px] p-0 px-1 border-0 bg-transparent font-medium" placeholder="Elemento" />
   446→                        </td>
   447→                        {/* Clasificación */}
   448→                        <td className="px-1 py-1 border bg-amber-50">
   449→                          <Input value={item.location} onChange={e => handleUpdateField(item.id, 'location', e.target.value)}
   450→                            className="h-6 text-[10px] p-0 px-1 border-0 bg-transparent" placeholder="Ubicación" />
   451→                        </td>
   452→                        <td className="px-1 py-1 border bg-amber-50 text-center">
   453→                          <Input type="number" min={1} value={item.quantityUnneeded || item.quantity}
   454→                            onChange={e => handleUpdateField(item.id, 'quantityUnneeded', Number(e.target.value))}
   455→                            className="h-6 text-[10px] p-0 px-1 border-0 bg-transparent w-10 text-center" />
   456→                        </td>
   457→                        <td className="px-1 py-1 border bg-amber-50 text-center">
   458→                          <Select value={String(item.extra?.estado || '')} onValueChange={v => handleUpdateField(item.id, 'extra.estado', v)}>
   459→                            <SelectTrigger className="h-6 text-[10px] p-0 px-1 border-0 bg-transparent w-16">
   460→                              <SelectValue />
   461→                            </SelectTrigger>
   462→                            <SelectContent>
   463→                              {(config.extraFields.find(f => f.key === 'estado')?.options || ['Bueno', 'Regular', 'Malo']).map(o => (
   464→                                <SelectItem key={o} value={o} className="text-xs">{o}</SelectItem>

   466→                            </SelectContent>
   467→                          </Select>

   469→                        <td className="px-1 py-1 border bg-amber-50 text-center">

   473→                            </SelectTrigger>

   484→                            onChange={e => handleUpdateField(item.id, 'jaulaFechaEntrada', e.target.value ? new Date(e.target.value).toISOString() : null)}

   489→                            <SelectTrigger className="h-6 text-[10px] p-0 px-1 border-0 bg-transparent w-12">
   490→                              <SelectValue />
   491→                            </SelectTrigger>
   492→                            <SelectContent>
   493→                              {(config.extraFields.find(f => f.key === 'diasCuarentena')?.options || ['7', '15', '20', '30', '40', '60', '90']).map(o => (
   494→                                <SelectItem key={o} value={o} className="text-xs">{o}d</SelectItem>
   495→                              ))}
   496→                            </SelectContent>
   497→                          </Select>
   498→                        </td>
   499→                        <td className="px-1 py-1 border bg-sky-50 text-center">
   500→                          {fechaLimite ? (

   511→                              <SelectValue />
   512→                            </SelectTrigger>
   513→                            <SelectContent>
   514→                              {JAULA_STATUS.map(opt => (
   515→                                <SelectItem key={opt.value} value={opt.value} className="text-xs">{opt.label}</SelectItem>
   516→                              ))}
   517→                            </SelectContent>

   537→                          }} className="text-red-400 hover:text-red-600">
   538→                            <Trash2 className="h-3.5 w-3.5" />
   539→                          </button>
   540→                        </td>
   541→                      </tr>
   542→                    );
   543→                  })}
   544→                </tbody>
   545→              </table>
   546→            </div>
   547→
   548→            {/* Mobile cards */}
   549→            <div className="space-y-2 md:hidden">
   550→              {filteredJaulaItems.map(item => {
   551→                const diasCuarentena = Number(item.extra?.diasCuarentena ?? 40);
   552→                let fechaLimite: string | null = item.jaulaFechaLimite || null;
   553→                if (!fechaLimite && item.jaulaFechaEntrada) {
   554→                  try {
   555→                    const d = new Date(item.jaulaFechaEntrada);
   556→                    d.setDate(d.getDate() + diasCuarentena);
   557→                    fechaLimite = d.toISOString();
   558→                  } catch {}
   559→                }

   561→
   562→                return (
   563→                  <div key={item.id} className="rounded-xl border bg-white shadow-sm overflow-hidden">
   564→                    <div className="flex items-center gap-2 px-3 py-2.5">

   567→                      <Badge className={`text-[10px] px-1.5 py-0 ${
   568→                        isJaulaDecision(item.extra?.decision) ? 'bg-orange-100 text-orange-800'
   569→                        : 'bg-red-100 text-red-800'
   570→                      }`}>
   571→                        {displayDecision(item.extra?.decision)}
   572→                      </Badge>
   573→                    </div>
   574→                    <div className="px-3 pb-2 text-[11px] text-muted-foreground flex items-center gap-3">
   575→                      <span>📦 {item.quantityUnneeded || item.quantity} uds</span>
   576→                      {item.price && <span>💰 {(item.price * (item.quantityUnneeded || item.quantity)).toFixed(2)}€</span>}
   577→                      {fechaLimite && (
   578→                        <span className={isExpired ? 'text-red-600 font-bold' : 'text-amber-600'}>
   579→                          📅 {new Date(fechaLimite).toLocaleDateString('es-ES')}{isExpired && ' ⚠'}
   580→                        </span>
   581→                      )}
   582→                    </div>
   583→                  </div>
   584→                );
   585→              })}
   586→            </div>
   587→
   588→            {/* Totals */}
   589→            <div className="flex gap-6 text-xs mt-3 pt-3 border-t">
   590→              <span className="text-gray-700">Total: <strong>{filteredJaulaItems.length}</strong> elementos</span>
   591→              <span className="text-gray-700">Valor: <strong className="text-red-600">{totalJaulaValue.toFixed(2)} €</strong></span>
   592→            </div>
   593→          </div>
   594→        )}
   595→      </div>
   596→
   597→      {/* Photo lightbox */}
   598→      {lightboxPhoto && (
   599→        <div className="fixed inset-0 z-50 bg-black/80 flex items-center justify-center" onClick={() => setLightboxPhoto(null)}>
   600→          <button className="absolute top-4 right-4 text-white" onClick={() => setLightboxPhoto(null)}>
   601→            <X className="h-6 w-6" />
   602→          </button>
   603→          <img src={lightboxPhoto} alt="Foto" className="max-w-[90vw] max-h-[85vh] object-contain rounded-lg" />
   604→        </div>
   605→      )}
   606→    </div>
   607→  );
