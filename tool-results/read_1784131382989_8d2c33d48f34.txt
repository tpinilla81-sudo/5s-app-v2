   431→ const handleCreateCompanyFull = async () => {
   432→  if (!newCompanyName.trim()) return
   433→  setIsCreatingCompany(true)
   434→  try {
   435→   // Step 1: Create company
   436→   const companyRes = await fetch('/api/companies', {
   437→    method: 'POST',
   438→    headers: { 'Content-Type': 'application/json' },
   439→    body: JSON.stringify({ name: newCompanyName, description: newCompanyDesc || undefined }),
   440→   })
   441→   const companyData = await companyRes.json()
   442→   if (!companyRes.ok) {
   443→    alert(companyData.error || 'Error al crear empresa')
   444→    return
   445→   }
   446→   const companyId = companyData.company?.id || companyData.id
   447→
   448→   // Step 2: Create or assign admin
   449→   let adminUserId: string | null = null
   450→   if (adminMode === 'create') {
   451→    if (!newAdminName.trim() || !newAdminEmail.trim() || !newAdminPassword.trim()) {
   452→     alert('Complete los datos del administrador')
   453→     return
   454→    }
   455→    // Create user with role 'admin' via /api/users
   456→    const userRes = await fetch('/api/users', {
   457→     method: 'POST',
   458→     headers: { 'Content-Type': 'application/json' },
   459→     body: JSON.stringify({
   460→      name: newAdminName,
   461→      email: newAdminEmail,
   462→      password: newAdminPassword,
   463→      role: 'admin',
   464→      active: true,
   465→     }),
   466→    })
   467→    const userData = await userRes.json()
   468→    if (!userRes.ok) {
   469→     alert(userData.error || 'Error al crear usuario admin')
   470→     return
   471→    }
   472→    adminUserId = userData.user?.id
   473→   } else {
   474→    // Assign existing user
   475→    if (!assignUserId) {
   476→     alert('Seleccione un usuario para asignar como administrador')
   477→     return
   478→    }
   479→    adminUserId = assignUserId
   480→    // Update user role to 'admin' if not already
   481→    const existingUser = users.find(u => u.id === assignUserId)
   482→    if (existingUser && existingUser.role !== 'admin') {
   483→     await fetch('/api/users', {
   484→      method: 'PUT',
   485→      headers: { 'Content-Type': 'application/json' },
   486→      body: JSON.stringify({ id: assignUserId, role: 'admin' }),
   487→     })
   488→    }
   489→   }
   490→
   491→   // Step 3: Add admin to company as admin_empresa
   492→   if (adminUserId) {
   493→    await fetch(`/api/companies/${companyId}/members`, {
   494→     method: 'POST',
   495→     headers: { 'Content-Type': 'application/json' },
   496→     body: JSON.stringify({ userId: adminUserId, role: 'admin_empresa' }),
   497→    })
   498→   }
   499→
   500→   // Step 4: Create subscription
   501→   await fetch('/api/subscriptions', {
   502→    method: 'POST',
   503→    headers: { 'Content-Type': 'application/json' },
   504→    body: JSON.stringify({
   505→     companyId,
   506→     plan: newSubPlan,
   507→     status: newSubStatus,
   508→     maxUsers: newSubMaxUsers,
   509→     maxProjects: newSubMaxProjects,
   510→     price: newSubPrice,
   511→     endDate: newSubEndDate || undefined,
   512→     trialEndsAt: newSubTrialEndsAt || undefined,
   513→     notes: newSubNotes || undefined,
   514→    }),
   515→   })
   516→
   517→   // Step 5: Send welcome email to admin + notification to gestor
   518→   if (adminMode === 'create' && adminUserId) {
   519→    try {
   520→     await fetch('/api/email', {
   521→      method: 'POST',
   522→      headers: { 'Content-Type': 'application/json' },
   523→      body: JSON.stringify({
   524→       type: 'admin_welcome',
   525→       adminName: newAdminName,
   526→       adminEmail: newAdminEmail,
   527→       adminPassword: newAdminPassword,
   528→       companyName: newCompanyName,
   529→       gestorEmail: currentUser?.email || undefined,
   530→      }),
   531→     })
   532→    } catch (emailErr) {
   533→     console.error('Error sending welcome email (non-blocking):', emailErr)
   534→    }
   535→   }
   536→
   537→   setShowNewCompany(false)
   538→   resetNewCompanyForm()
   539→   await loadStats()
   540→   await fetchCompanies()
   541→  } catch (error) {
   542→   console.error('Error creating company with admin:', error)
   543→   alert('Error al crear la empresa. Revise la consola para detalles.')
   544→  } finally {
   545→   setIsCreatingCompany(false)
   546→  }
   547→ }
