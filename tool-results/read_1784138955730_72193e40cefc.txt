    78→  const { setCurrentView, currentProject, currentZone, fetchProgress } = use5SStore();
    79→  const [showQuarterlyAudit, setShowQuarterlyAudit] = useState(false);
    80→  const [showWeeklyAudit, setShowWeeklyAudit] = useState(false);
    81→  const [showMonthlyAudit, setShowMonthlyAudit] = useState(false);
    82→
    83→  // Dynamic checklist data loaded from API templates
    84→  const [weeklySections, setWeeklySections] = useState<AuditSection[]>([]);
    85→  const [monthlySections, setMonthlySections] = useState<AuditSection[]>([]);
    86→  const [weeklyTotalItems, setWeeklyTotalItems] = useState(0);
    87→  const [monthlyTotalItems, setMonthlyTotalItems] = useState(0);
    88→
    89→  const [auditHistory, setAuditHistory] = useState<AuditRecord[]>([]);
    90→  const [actionItems, setActionItems] = useState<ActionItemData[]>([]);
    91→  const [actionStats, setActionStats] = useState({ abierta: 0, en_proceso: 0, resuelta: 0, cerrada: 0 });
    92→  const [stats, setStats] = useState<any>(null);
    93→  const [isLoadingHistory, setIsLoadingHistory] = useState(true);
    94→  const [activeTab, setActiveTab] = useState<'overview' | 'weekly' | 'monthly' | 'quarterly' | 'actions' | 'counters'>('overview');
    95→
    96→  // Action plan form
    97→  const [showNewAction, setShowNewAction] = useState(false);
    98→  const [newAction, setNewAction] = useState({ descripcion: '', responsable: '', prioridad: 'media', fechaLimite: '' });
    99→
   100→  useEffect(() => {
   101→    loadData();
   102→  }, [currentProject]);
   103→
   104→  // Load checklist templates from API and compute weekly/monthly variants
   105→  useEffect(() => {
   106→    const loadChecklists = async () => {
   107→      const templatesMap = await fetchAllChecklistTemplates('auditoria');
   108→      // Weekly: only S3 (Seiso/cleaning) sections
   109→      const s3Sections = templatesMap[3] || [];
   110→      setWeeklySections(s3Sections);
   111→      setWeeklyTotalItems(s3Sections.reduce((sum, s) => sum + s.items.length, 0));
   112→      // Monthly: first 2 items from each section across all S
   113→      const monthlyList: AuditSection[] = [];
   114→      for (let s = 1; s <= 5; s++) {
   115→        const sSections = templatesMap[s] || [];
   116→        for (const section of sSections) {
   117→          monthlyList.push({
   118→            ...section,
   119→            items: section.items.slice(0, 2),
   120→          });
   121→        }
   122→      }
   123→      setMonthlySections(monthlyList);
   124→      setMonthlyTotalItems(monthlyList.reduce((sum, s) => sum + s.items.length, 0));
   125→    };
   126→    loadChecklists();
   127→  }, []);
   128→
   129→  const loadData = async () => {
   130→    setIsLoadingHistory(true);
   131→    await Promise.all([loadAuditHistory(), loadActionItems(), loadStats()]);
   132→    setIsLoadingHistory(false);
   133→  };
   134→
   135→  const loadAuditHistory = async () => {
   136→    try {
   137→      const params = currentProject ? `?projectId=${currentProject.id}` : '';
   138→      const res = await fetch(`/api/audit${params}`);
   139→      const data = await res.json();
   140→      if (data.success && data.audits) {
   141→        const sorted = data.audits.sort((a: any, b: any) => new Date(b.auditDate).getTime() - new Date(a.auditDate).getTime());
   142→        setAuditHistory(sorted);
   143→      }
   144→    } catch (error) {
   145→      console.error('Error loading audit history:', error);
   146→    }
   147→  };
   148→
   149→  const loadActionItems = async () => {
   150→    try {
   151→      const params = currentProject ? `?projectId=${currentProject.id}` : '';
   152→      const res = await fetch(`/api/actions${params}`);
   153→      const data = await res.json();
   154→      if (data.success && data.data) {
   155→        setActionItems(data.data);
   156→        const counts = { abierta: 0, en_proceso: 0, resuelta: 0, cerrada: 0 };
   157→        data.data.forEach((a: ActionItemData) => {
   158→          if (counts[a.estado as keyof typeof counts] !== undefined) counts[a.estado as keyof typeof counts]++;
   159→        });
   160→        setActionStats(counts);
   161→      }
   162→    } catch (error) {
   163→      console.error('Error loading action items:', error);
   164→    }
   165→  };
   166→
   167→  const loadStats = async () => {
   168→    try {
   169→      const params = currentProject ? `?projectId=${currentProject.id}` : '';
   170→      const res = await fetch(`/api/stats${params}`);
   171→      const data = await res.json();
   172→      if (data.success) setStats(data.data);
   173→    } catch (error) {
   174→      console.error('Error loading stats:', error);
   175→    }
   176→  };
   177→
   178→  const handleAddAction = async () => {
   179→    if (!newAction.descripcion.trim()) return;
   180→    try {
   181→      await fetch('/api/actions', {
   182→        method: 'POST',
   183→        headers: { 'Content-Type': 'application/json' },
   184→        body: JSON.stringify({
   185→          sStep: 0,
   186→          miniStep: 0,
   187→          itemId: `MC-${Date.now()}`,
   188→          itemDescription: newAction.descripcion,
   189→          hallazgo: newAction.descripcion,
   190→          responsable: newAction.responsable || null,
   191→          prioridad: newAction.prioridad,
   192→          estado: 'abierta',
   193→          fechaLimite: newAction.fechaLimite || null,
   194→          source: 'mejora_continua',
   195→          projectId: currentProject?.id,
   196→          zoneId: currentZone?.id || null,
   197→        }),
   198→      });
   199→      setNewAction({ descripcion: '', responsable: '', prioridad: 'media', fechaLimite: '' });
   200→      setShowNewAction(false);
   201→      await loadActionItems();
   202→    } catch (error) {
   203→      console.error('Error adding action:', error);
   204→    }
   205→  };
   206→
   207→  const handleUpdateActionEstado = async (id: string, estado: string) => {
   208→    try {
   209→      await fetch(`/api/actions?id=${id}`, {
   210→        method: 'PUT',
   211→        headers: { 'Content-Type': 'application/json' },
   212→        body: JSON.stringify({ estado }),
   213→      });
   214→      await loadActionItems();
   215→    } catch (error) {
   216→      console.error('Error updating action:', error);
   217→    }
   218→  };
   219→
   220→  const handleDeleteAction = async (id: string) => {
   221→    try {
   222→      await fetch(`/api/actions?id=${id}`, { method: 'DELETE' });
   223→      await loadActionItems();
   224→    } catch (error) {
   225→      console.error('Error deleting action:', error);
   226→    }
   227→  };
   228→
   229→  const totalActions = actionStats.abierta + actionStats.en_proceso + actionStats.resuelta + actionStats.cerrada;
   230→
   231→  const weeklyAudits = auditHistory.filter(a => a.auditType === 'weekly');
   232→  const monthlyAudits = auditHistory.filter(a => a.auditType === 'monthly');
   233→  const quarterlyAudits = auditHistory.filter(a => a.auditType === 'quarterly' || (!a.auditType && a.score !== undefined && a.auditorName));
   234→
   235→  // Calculate next audit dates
   236→  const getNextAuditDate = (audits: AuditRecord[], months: number) => {
   237→    if (audits.length === 0) return 'Pendiente';

   239→    const nextDate = new Date(lastAudit);
   240→    nextDate.setMonth(nextDate.getMonth() + months);
   241→    const today = new Date();
   242→    const diffDays = Math.ceil((nextDate.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
   243→    if (diffDays < 0) return `¡Vencida hace ${Math.abs(diffDays)} días!`;
   244→    if (diffDays === 0) return '¡Hoy!';
   245→    return `En ${diffDays} días (${nextDate.toLocaleDateString('es-ES')})`;
   246→  };
   247→
   248→  const getNextWeeklyDate = () => {
   249→    if (weeklyAudits.length === 0) return 'Pendiente';
   250→    const lastAudit = new Date(weeklyAudits[0].auditDate);
   251→    const nextDate = new Date(lastAudit);
   252→    nextDate.setDate(nextDate.getDate() + 7);
   253→    const today = new Date();
   254→    const diffDays = Math.ceil((nextDate.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
   255→    if (diffDays < 0) return `¡Vencida hace ${Math.abs(diffDays)} días!`;
   256→    if (diffDays === 0) return '¡Hoy!';
   257→    return `En ${diffDays} días`;
   258→  };
   259→
   260→  const isWeeklyOverdue = () => {
   261→    if (weeklyAudits.length === 0) return true;
   262→    const lastAudit = new Date(weeklyAudits[0].auditDate);
   263→    const nextDate = new Date(lastAudit);
   264→    nextDate.setDate(nextDate.getDate() + 7);
   265→    return nextDate < new Date();
   266→  };
   267→
   268→  const isMonthlyOverdue = () => getNextAuditDate(monthlyAudits, 1).includes('Vencida');
   269→  const isQuarterlyOverdue = () => getNextAuditDate(quarterlyAudits, 3).includes('Vencida');

   280→    alta: 'bg-red-100 text-red-800',

   287→    en_proceso: 'bg-blue-100 text-blue-800',
   288→    resuelta: 'bg-green-100 text-green-800',
   289→    cerrada: 'bg-gray-100 text-gray-600',
   290→  };
   291→
   292→  const ESTADO_LABELS: Record<string, string> = {
   293→    abierta: 'Abierta',
   294→    en_proceso: 'En proceso',
   295→    resuelta: 'Resuelta',
   296→    cerrada: 'Cerrada',
   297→  };
   298→
   299→  const tabs = [
   300→    { key: 'overview', label: 'Resumen', icon: BarChart3 },
   301→    { key: 'weekly', label: 'Semanal', icon: SprayCan },
   302→    { key: 'monthly', label: 'Mensual', icon: Calendar },
   303→    { key: 'quarterly', label: 'Trimestral', icon: ShieldCheck },

   305→    { key: 'counters', label: 'Contadores', icon: BarChart3 },
   306→  ];
   307→
   308→  return (
   309→    <div className="space-y-4">
   310→      {/* Header - only in standalone mode */}
   311→      {!embedded && (
   312→        <motion.div initial={{ opacity: 0, y: -10 }} animate={{ opacity: 1, y: 0 }}>
   313→          <div className="flex items-center justify-between mb-4">
   314→            <Button variant="ghost" size="sm" onClick={() => setCurrentView('board')} className="gap-1.5">
   315→              <ArrowLeft className="h-4 w-4" />
   316→              Volver al Tablero 5S
   317→            </Button>
   318→          </div>
   319→        </motion.div>
   320→      )}
   321→
   322→      <Card className="border-green-200 bg-gradient-to-r from-green-50 to-emerald-50">
   323→        <CardContent className="p-6">
   324→          <div className="flex items-center gap-4">
   325→            <div className="w-16 h-16 rounded-full bg-gradient-to-br from-green-500 to-emerald-600 flex items-center justify-center text-white shadow-lg">
   326→              <Trophy className="h-8 w-8" />
   327→            </div>
   328→            <div className="flex-1">
   329→              <h2 className="text-xl font-bold text-green-900">Fase de Mejora Continua</h2>
   330→              <p className="text-sm text-green-700 mt-1">
   331→                Taller de implementación completado. La zona está en <strong>mejora continua permanente</strong> con auditorías periódicas, plan de acción y contadores.
   332→              </p>
   333→            </div>
   334→            <div className="hidden sm:flex items-center gap-1">
   335→              {S_STEPS.map(s => (
   336→                <div key={s.id} className="w-8 h-8 rounded-full flex items-center justify-center text-white text-xs font-bold" style={{ backgroundColor: s.color }}>
   337→                  S{s.id}
   338→                </div>
   339→              ))}
   340→            </div>
   341→          </div>
   342→        </CardContent>
   343→      </Card>
   344→
   345→      {/* Tab navigation */}
   346→      <div className="flex gap-1 overflow-x-auto pb-1">
   347→        {tabs.map(tab => (

   350→            onClick={() => setActiveTab(tab.key as any)}
   351→            className={`flex items-center gap-1.5 px-3 py-2 rounded-lg text-xs font-medium whitespace-nowrap transition-colors ${
   352→              activeTab === tab.key
   353→                ? 'bg-green-100 text-green-800 border border-green-200'
   354→                : 'text-muted-foreground hover:bg-muted'
   355→            }`}
   356→          >
   357→            <tab.icon className="h-3.5 w-3.5" />
   358→            {tab.label}
   359→          </button>
   360→        ))}
   361→      </div>
   362→
   363→      {/* OVERVIEW TAB */}
   364→      {activeTab === 'overview' && (
   365→        <div className="space-y-4">
   366→          {/* Key metrics */}
   367→          <div className="grid grid-cols-2 sm:grid-cols-3 gap-3">
   368→            <Card className={isWeeklyOverdue() ? 'border-red-200 bg-red-50' : 'border-blue-100'}>
   369→              <CardContent className="p-3 text-center">
   370→                <SprayCan className={`h-5 w-5 mx-auto mb-1 ${isWeeklyOverdue() ? 'text-red-500' : 'text-blue-500'}`} />
   371→                <p className="text-2xl font-bold text-blue-700">{weeklyAudits.length}</p>
   372→                <p className="text-xs text-muted-foreground">Auditorías Semanales</p>

   374→                  {getNextWeeklyDate()}
   375→                </p>
   376→              </CardContent>
   377→            </Card>
   378→
   379→            <Card className={isMonthlyOverdue() ? 'border-red-200 bg-red-50' : 'border-purple-100'}>
   380→              <CardContent className="p-3 text-center">
   381→                <Calendar className={`h-5 w-5 mx-auto mb-1 ${isMonthlyOverdue() ? 'text-red-500' : 'text-purple-500'}`} />
   382→                <p className="text-2xl font-bold text-purple-700">{monthlyAudits.length}</p>
   383→                <p className="text-xs text-muted-foreground">Auditorías Mensuales</p>

   385→                  {getNextAuditDate(monthlyAudits, 1)}
   386→                </p>
   387→              </CardContent>
   388→            </Card>
   389→
   390→            <Card className={isQuarterlyOverdue() ? 'border-red-200 bg-red-50' : 'border-orange-100'}>
   391→              <CardContent className="p-3 text-center">
   392→                <ShieldCheck className={`h-5 w-5 mx-auto mb-1 ${isQuarterlyOverdue() ? 'text-red-500' : 'text-orange-500'}`} />
   393→                <p className="text-2xl font-bold text-orange-700">{quarterlyAudits.length}</p>

   395→                <p className={`text-[10px] mt-1 ${isQuarterlyOverdue() ? 'text-red-600 font-medium' : 'text-muted-foreground'}`}>
   396→                  {getNextAuditDate(quarterlyAudits, 3)}
   397→                </p>
   398→              </CardContent>
   399→            </Card>
   400→          </div>
   401→
   402→          {/* Action plan summary */}
   403→          <Card className="border-orange-100">
   404→            <CardHeader className="pb-2">
   405→              <CardTitle className="text-sm font-medium flex items-center gap-2">
   406→                <Zap className="h-4 w-4 text-orange-500" />
   407→                Plan de Acción — Resumen
   408→              </CardTitle>
   409→            </CardHeader>
   410→            <CardContent>
   411→              {totalActions === 0 ? (
   412→                <p className="text-xs text-muted-foreground text-center py-4">No hay acciones de mejora registradas</p>
   413→              ) : (
   414→                <div className="grid grid-cols-4 gap-3 text-center">
   415→                  <div className="p-2 rounded-lg bg-red-50">
   416→                    <p className="text-lg font-bold text-red-600">{actionStats.abierta}</p>
   417→                    <p className="text-[10px] text-red-700 font-medium">Abiertas</p>
   418→                  </div>
   419→                  <div className="p-2 rounded-lg bg-amber-50">
   420→                    <p className="text-lg font-bold text-amber-600">{actionStats.en_proceso}</p>
   421→                    <p className="text-[10px] text-amber-700 font-medium">En Proceso</p>
   422→                  </div>
   423→                  <div className="p-2 rounded-lg bg-green-50">
   424→                    <p className="text-lg font-bold text-green-600">{actionStats.resuelta}</p>

   426→                  </div>
   427→                  <div className="p-2 rounded-lg bg-gray-50">
   428→                    <p className="text-lg font-bold text-gray-600">{actionStats.cerrada}</p>
   429→                    <p className="text-[10px] text-gray-700 font-medium">Cerradas</p>
   430→                  </div>
   431→                </div>
   432→              )}
   433→            </CardContent>
   434→          </Card>
   435→
   436→          {/* Scoring explanation */}
   437→          <Card>
   438→            <CardContent className="p-4">
   439→              <h4 className="text-sm font-semibold mb-2">Sistema de Puntuación — Todas las Auditorías</h4>
   440→              <div className="grid grid-cols-3 gap-3 text-center">
   441→                <div className="p-2 rounded-lg bg-blue-50 border border-blue-100">
   442→                  <p className="text-lg font-bold text-blue-700">90%</p>
   443→                  <p className="text-[10px] text-blue-600">Checklist (máx.)</p>

   445→                <div className="p-2 rounded-lg bg-green-50 border border-green-100">

   450→                  <p className="text-lg font-bold text-orange-700">75%</p>

   452→                </div>
   453→              </div>

   456→              </p>

   458→          </Card>
   459→        </div>

   461→
   462→      {/* WEEKLY TAB */}
   463→      {activeTab === 'weekly' && (
   464→        <div className="space-y-4">
   465→          {/* Weekly audit info */}
   466→          <Card className="border-blue-100">
   467→            <CardContent className="p-4">
   468→              <div className="flex items-center gap-3 mb-3">
   469→                <div className="w-10 h-10 rounded-full bg-blue-100 flex items-center justify-center">

   477→                </div>
   478→              </div>

   480→                <div className="text-xs text-muted-foreground">
   481→                  {weeklyTotalItems} puntos a evaluar · Frecuencia: semanal

   483→                <Button
   484→                  onClick={() => setShowWeeklyAudit(true)}
   485→                  className={`gap-1.5 ${isWeeklyOverdue() ? 'bg-red-500 hover:bg-red-600 text-white' : 'bg-blue-500 hover:bg-blue-600 text-white'}`}
