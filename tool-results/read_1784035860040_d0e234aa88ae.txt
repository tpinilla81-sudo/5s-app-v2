   361→   361→  // getMiniStepStatus — Permission-Driven with Business Rules
   362→   362→  // Permissions are the source of truth for WHO can do WHAT.
   363→   363→  // Business rules define WHEN things are accessible:
   364→   364→  //   - INTER-S progression: S2 requires S1 completed, S3 requires S2, etc.
   365→   365→  //   - INTRA-S progression: step N requires step N-1 completed
   366→   366→  //   - Step 5 (audit): requires steps 1-4 ALL completed within the same S
   367→   367→  //   - a1 (perform) → can enter and act (if previous step done)
   368→   368→  //   - a0 (view) only → can see the step exists, but CANNOT enter it
   369→   369→  //   - No permission → locked (not even visible)
   370→   370→  //   - IMPORTANT: Previous step completion is checked at ZONE level,
   371→   371→  //     so an auditor can access step 5 if employees completed 1-4
   372→   372→  //   - For step 1 (individual): also checks EmployeeProgress for the
   373→   373→  //     current user, so the employee's own completion unlocks step 2
   374→   374→  // ═══════════════════════════════════════════════════════
   375→   375→  getMiniStepStatus: (sStep, miniStep) => {
   376→   376→    const { progress, employeeProgress, currentUser, adminFreeNavigation, currentZone } = get()
   377→   377→    if (!currentUser) return 'locked'
   378→   378→
   379→   379→    const canViewStep = get().canView(sStep, miniStep)
   380→   380→    const canPerformStep = get().canPerform(sStep, miniStep)
   381→   381→    // Permission-driven: skip_steps from DB replaces hardcoded isAdmin check
   382→   382→    const canSkipSteps = get().hasPermission('skip_steps')
   383→   383→    const skipLocks = canSkipSteps && adminFreeNavigation
   384→   384→
   385→   385→    // Check if a specific mini-step of a specific S is completed at zone level
   386→   386→    const isStepCompletedAt = (s: number, ms: number): boolean => {
   387→   387→      if (currentZone) {
   388→   388→        const zoneStep = progress.find(p =>
   389→   389→          p.sStep === s &&
   390→   390→          p.miniStep === ms &&
   391→   391→          (p.zoneId === currentZone.id || p.zoneId === null) &&
   392→   392→          p.completed
   393→   393→        )
   394→   394→        return !!zoneStep
   395→   395→      }
   396→   396→      const anyStep = progress.find(p =>
   397→   397→        p.sStep === s &&
   398→   398→        p.miniStep === ms &&
   399→   399→        p.completed
   400→   400→      )
   401→   401→      return !!anyStep
   402→   402→    }
   403→   403→
   404→   404→    // Check if a specific S-step (all 5 mini-steps) is fully completed (quesito earned)
   405→   405→    // Also checks employeeProgress for individual steps, consistent with areSteps1to4Completed
   406→   406→    const isSCompleted = (s: number): boolean => {
   407→   407→      for (let ms = 1; ms <= 5; ms++) {
   408→   408→        // Check zone-level progress first
   409→   409→        if (isStepCompletedAt(s, ms)) continue
   410→   410→        // Also check employeeProgress for this step (e.g., individual formación step 1)
   411→   411→        const anyEmpCompleted = currentZone
   412→   412→          ? employeeProgress.some(ep =>
   413→   413→              ep.sStep === s &&
   414→   414→              ep.miniStep === ms &&
   415→   415→              ep.zoneId === currentZone.id &&
   416→   416→              ep.completed
   417→   417→            )
   418→   418→          : employeeProgress.some(ep =>
   419→   419→              ep.sStep === s &&
   420→   420→              ep.miniStep === ms &&
   421→   421→              ep.completed
   422→   422→            )
   423→   423→        if (!anyEmpCompleted) return false
   424→   424→      }
   425→   425→      return true
   426→   426→    }
   427→   427→
   428→   428→    // Check if the CURRENT USER has completed step 1 individually (EmployeeProgress)
   429→   429→    // This is crucial for unlocking step 2 for an employee who passed the exam
   430→   430→    // even if not ALL employees in the zone have completed it yet
   431→   431→    const hasUserCompletedStep1 = (): boolean => {
   432→   432→      if (!currentZone) return false
   433→   433→      const userStep = employeeProgress.find(ep =>
   434→   434→        ep.sStep === sStep &&
   435→   435→        ep.miniStep === 1 &&
   436→   436→        ep.zoneId === currentZone.id &&
   437→   437→        ep.userId === currentUser.id &&
   438→   438→        ep.completed
   439→   439→      )
   440→   440→      return !!userStep
   441→   441→    }
   442→   442→
   443→   443→    // Check if already completed at zone level (or project level if no zone)
   444→   444→    const isStepCompleted = (): boolean => isStepCompletedAt(sStep, miniStep)
   445→   445→
   446→   446→    // Check if the CURRENT USER has completed this individual step (EmployeeProgress)
   447→   447→    // For step 1 (formación) and step 4 (autoevaluación), completion is tracked per employee
   448→   448→    const hasUserCompletedIndividualStep = (): boolean => {
   449→   449→      if (!currentZone) return false
   450→   450→      if (miniStep !== 1 && miniStep !== 4) return false // Only individual steps
   451→   451→      const userStep = employeeProgress.find(ep =>
   452→   452→        ep.sStep === sStep &&
   453→   453→        ep.miniStep === miniStep &&
   454→   454→        ep.zoneId === currentZone.id &&
   455→   455→        ep.userId === currentUser.id &&
   456→   456→        ep.completed
   457→   457→      )
   458→   458→      return !!userStep
   459→   459→    }
   460→   460→
   461→   461→    // Combined: step is "done" if either zone-level completed OR user completed individually
   462→   462→    const isStepDoneForUser = (): boolean => isStepCompleted() || hasUserCompletedIndividualStep()
   463→   463→
   464→   464→    // Check if step is completed by ANYONE (zone or any employee)
   465→   465→    // This is crucial: ALL users must see completed steps in green,
   466→   466→    // even if they don't have permission to enter them (e.g., auditors viewing steps 1-4)
   467→   467→    const isStepCompletedByAnyone = (): boolean => {
   468→   468→      if (isStepCompleted()) return true
   469→   469→      if (!currentZone) {
   470→   470→        return employeeProgress.some(ep =>
   471→   471→          ep.sStep === sStep &&
   472→   472→          ep.miniStep === miniStep &&
   473→   473→          ep.completed
   474→   474→        )
   475→   475→      }
   476→   476→      return employeeProgress.some(ep =>
   477→   477→        ep.sStep === sStep &&
   478→   478→        ep.miniStep === miniStep &&
   479→   479→        ep.zoneId === currentZone.id &&
   480→   480→        ep.completed
   481→   481→      )
   482→   482→    }
   483→   483→
   484→   484→    // Helper: check if steps 1-4 are all completed for this S-step
   485→   485→    // Uses BOTH zone-level progress AND any employee progress,
   486→   486→    // so auditors can access step 5 when employees have completed 1-4
   487→   487→    const areSteps1to4Completed = (): boolean => {
   488→   488→      for (let ms = 1; ms <= 4; ms++) {
   489→   489→        const zoneCompleted = isStepCompletedAt(sStep, ms)
   490→   490→        if (zoneCompleted) continue
   491→   491→        // Check if ANY employee has completed this step individually
   492→   492→        const anyEmpCompleted = currentZone
   493→   493→          ? employeeProgress.some(ep =>
   494→   494→              ep.sStep === sStep &&
   495→   495→              ep.miniStep === ms &&
   496→   496→              ep.zoneId === currentZone.id &&
   497→   497→              ep.completed
   498→   498→            )
   499→   499→          : employeeProgress.some(ep =>
   500→   500→              ep.sStep === sStep &&
   501→   501→              ep.miniStep === ms &&
   502→   502→              ep.completed
   503→   503→            )
   504→   504→        if (!anyEmpCompleted) return false
   505→   505→      }
   506→   506→      return true
   507→   507→    }
   508→   508→
   509→   509→    // ═══ INTER-S PROGRESSION CHECK ═══
   510→   510→    // S1 is always available. S2 requires S1 completed. S3 requires S2. Etc.
   511→   511→    // This enforces the 5S methodology sequence: Clasificar → Ordenar → Limpiar → Estandarizar → Mantener
   512→   512→    const isPreviousSCompleted = (): boolean => {
   513→   513→      if (sStep === 1) return true // S1 has no prerequisite
   514→   514→      return isSCompleted(sStep - 1)
   515→   515→    }
   516→   516→
   517→   517→    // ═══ INTRA-S PROGRESSION CHECK ═══
   518→   518→    // Step 1: always available (if you have permission AND previous S is done)
   519→   519→    // Steps 2-4: need the immediately previous step completed
   520→   520→    // Step 5: needs ALL steps 1-4 completed
   521→   521→    // IMPORTANT: Completion checks include BOTH zone-level AND any employee progress,
   522→   522→    // so auditors/responsables can see steps as available when employees completed them
   523→   523→    const isPreviousStepCompleted = (): boolean => {
   524→   524→      if (miniStep === 1) return true
   525→   525→      if (miniStep === 5) return areSteps1to4Completed()
   526→   526→      // Steps 2, 3, 4: need previous step completed at zone level
   527→   527→      if (isStepCompletedAt(sStep, miniStep - 1)) return true
   528→   528→      // For step 2: current user completed step 1 individually (exam passed)
   529→   529→      if (miniStep === 2 && hasUserCompletedStep1()) return true
   530→   530→      // Also check: ANY employee completed previous step (for auditors/responsables)
   531→   531→      const anyEmpCompletedPrev = currentZone
   532→   532→        ? employeeProgress.some(ep =>
   533→   533→            ep.sStep === sStep &&
   534→   534→            ep.miniStep === miniStep - 1 &&
   535→   535→            ep.zoneId === currentZone.id &&
   536→   536→            ep.completed
   537→   537→          )
   538→   538→        : employeeProgress.some(ep =>
   539→   539→            ep.sStep === sStep &&
   540→   540→            ep.miniStep === miniStep - 1 &&
   541→   541→            ep.completed
   542→   542→          )
   543→   543→      if (anyEmpCompletedPrev) return true
   544→   544→      return false
   545→   545→    }
   546→   546→
   547→   547→    // ── If already completed (zone-level OR user's individual), VERIFY COHERENCY ──
   548→   548→    if (isStepDoneForUser()) {
   549→   549→      const isChainCoherent = (): boolean => {
   550→   550→        // First check inter-S: previous S must be completed
   551→   551→        if (!isPreviousSCompleted()) return false
   552→   552→        // Then check intra-S chain — reuse same logic as isPreviousStepCompleted
   553→   553→        if (miniStep === 1) return true // Step 1 is always coherent if previous S is done
   554→   554→        if (miniStep === 5) return areSteps1to4Completed()
   555→   555→        if (isStepCompletedAt(sStep, miniStep - 1)) return true
   556→   556→        if (miniStep === 2 && hasUserCompletedStep1()) return true
   557→   557→        // Also check any employee completed previous step
   558→   558→        const anyEmpPrev = currentZone
   559→   559→          ? employeeProgress.some(ep =>
   560→   560→              ep.sStep === sStep &&
   561→   561→              ep.miniStep === miniStep - 1 &&
   562→   562→              ep.zoneId === currentZone.id &&
   563→   563→              ep.completed
   564→   564→            )
   565→   565→          : employeeProgress.some(ep =>
   566→   566→              ep.sStep === sStep &&
   567→   567→              ep.miniStep === miniStep - 1 &&
   568→   568→              ep.completed
   569→   569→            )
   570→   570→        if (anyEmpPrev) return true
   571→   571→        return false
   572→   572→      }
   573→   573→
   574→   574→      if (isChainCoherent()) {
   575→   575→        if (canPerformStep || canSkipSteps) return 'completed'
   576→   576→        if (canViewStep) return 'completed_viewonly'
   577→   577→        return 'locked'
   578→   578→      }
   579→   579→      // Chain is incoherent: treat as NOT completed — fall through
   580→   580→    }
   581→   581→
   582→   582→    // ── Step completed by ANYONE (zone or employee) → show as green for ALL users ──
   583→   583→    // Even users without permission must see completed steps in green
   584→   584→    if (isStepCompletedByAnyone()) {
   585→   585→      if (canPerformStep || canSkipSteps) return 'completed'
   586→   586→      // Everyone else sees it as completed_viewonly (green, read-only)
   587→   587→      return 'completed_viewonly'
   588→   588→    }
   589→   589→
   590→   590→    // ── Admin with lock open: skip all checks ──
   591→   591→    if (skipLocks) return 'available'
   592→   592→
   593→   593→    // ── No permission at all = locked ──
   594→   594→    if (!canViewStep && !canPerformStep) return 'locked'
   595→   595→
   596→   596→    // ── Has perform permission (a1) → check progression ──
   597→   597→    if (canPerformStep) {
   598→   598→      // INTER-S: can't start this S until previous S is completed
   599→   599→      if (!isPreviousSCompleted()) return 'locked'
   600→   600→      // INTRA-S: can't enter step until previous step is completed
   601→   601→      if (!isPreviousStepCompleted()) return 'locked'
   602→   602→      return 'available'
   603→   603→    }
   604→   604→
   605→   605→    // ── Has view permission only (a0) → can see but CANNOT enter ──
   606→   606→    if (canViewStep) return 'locked'
   607→   607→
   608→   608→    return 'locked'
   609→   609→  },
   610→   610→
   611→   611→  isZoneMiniStepComplete: (sStep, miniStep, zoneId) => {
   612→   612→    const { progress, employeeProgress, currentProject } = get()
   613→   613→    if (!currentProject) return false
   614→   614→
   615→   615→    if (ZONE_MINI_STEPS.includes(miniStep)) {

   636→   636→
   637→   637→    return false
   638→   638→  },
   639→   639→

   643→   643→
