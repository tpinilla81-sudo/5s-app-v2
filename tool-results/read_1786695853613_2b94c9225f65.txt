    93→    93→  // v2.35: límite dinámico de fotos configurable por plantilla / override por zona.

    98→    98→  const beforePrompt = BEFORE_PROMPT_BY_S[sStep] || 'Fotografía la zona para documentar su estado actual.';

   100→   100→  // Generate auto title with traceability
   101→   101→  const generatePhotoTitle = (index: number, photoType: string): string => {

   103→   103→    const zoneName = currentZone?.name || 'Zona'
   104→   104→    const typeLabel = photoType === 'antes' ? 'ANTES' : photoType === 'despues' ? 'DESPUÉS' : photoType === 'referencia' ? 'Referencia' : 'Hallazgo'

   107→   107→  }
   108→   108→
   109→   109→  // Refs para cola de subida secuencial — evita saturar CPU/memoria en móvil
   110→   110→  // cuando se seleccionan varias fotos a la vez (era la causa del bloqueo).
   111→   111→  const uploadQueueRef = useRef<{ rawBase64: string; photoType: string }[]>([]);
   112→   112→  const isProcessingRef = useRef(false);
   113→   113→  const photoCounterRef = useRef(0);
   114→   114→  const [queueLength, setQueueLength] = useState(0);
   115→   115→
   116→   116→  const uploadPhoto = async (base64Data: string, index: number): Promise<string | null> => {
   117→   117→    try {
   118→   118→      const projectId = currentProject?.id || 'unknown';
   119→   119→      const filename = generatePhotoFilename(projectId, sStep, miniStep, index);
   120→   120→      const file = base64toFile(base64Data, filename);
   121→   121→
   122→   122→      const formData = new FormData();
   123→   123→      formData.append('file', file);
   124→   124→      formData.append('filename', filename);
   125→   125→
   126→   126→      const res = await fetch('/api/upload', {
   127→   127→        method: 'POST',
   128→   128→        body: formData,
   129→   129→      });
   130→   130→
   131→   131→      const data = await res.json();
   132→   132→      if (data.success && data.url) return data.url;
   133→   133→      return null;
   134→   134→    } catch (error) {
   135→   135→      console.error('Upload error:', error);
   136→   136→      return null;
   137→   137→    }
   138→   138→  };
   139→   139→
   140→   140→  // Procesa la cola de fotos UNA A UNA. Así el canvas + toDataURL + fetch
   141→   141→  // no compiten por CPU/memoria, y la UI del móvil no se congela.
   142→   142→  const processQueue = useCallback(async () => {
   143→   143→    if (isProcessingRef.current) return;
   144→   144→    if (uploadQueueRef.current.length === 0) return;
   145→   145→    isProcessingRef.current = true;
   146→   146→    console.log(`[FotosModal] processQueue: iniciando procesamiento de ${uploadQueueRef.current.length} foto(s)`);
   147→   147→
   148→   148→    try {
   149→   149→      while (uploadQueueRef.current.length > 0) {
   150→   150→        const item = uploadQueueRef.current.shift()!;
   151→   151→        setQueueLength(uploadQueueRef.current.length);
   152→   152→
   153→   153→        const { rawBase64, photoType } = item;
   154→   154→        const index = photoCounterRef.current++;
   155→   155→        const photoId = `photo_${Date.now()}_${index}_${Math.random().toString(36).slice(2, 8)}`;
   156→   156→        console.log(`[FotosModal] processQueue: procesando foto #${index + 1} (id=${photoId}), quedan ${uploadQueueRef.current.length} en cola`);
   157→   157→
   158→   158→        try {
   159→   159→          const compressed = await compressImage(rawBase64);
   160→   160→          const estimatedSize = estimateBase64Size(compressed);
   161→   161→
   162→   162→          const newPhoto: PhotoItem = {
   163→   163→            id: photoId,
   164→   164→            preview: compressed,
   165→   165→            serverUrl: '',
   166→   166→            uploaded: false,
   167→   167→            uploading: true,
   168→   168→            estimatedSize,
   169→   169→            title: generatePhotoTitle(index, photoType),
   170→   170→            photoType,
   171→   171→            savedToLibrary: false,
   172→   172→          };
   173→   173→
   174→   174→          setPhotos(prev => [...prev, newPhoto]);
   175→   175→
   176→   176→          const url = await uploadPhoto(compressed, index);
   177→   177→
   178→   178→          // Actualiza por ID — robusto frente a reordenamientos/eliminaciones
   179→   179→          setPhotos(prev =>
   180→   180→            prev.map(p =>
   181→   181→              p.id === photoId
   182→   182→                ? { ...p, serverUrl: url || '', uploaded: !!url, uploading: false }
   183→   183→                : p
   184→   184→            )
   185→   185→          );
   186→   186→        } catch (err) {
   187→   187→          console.error('Error processing queued photo:', err);
   188→   188→          // Marcamos la foto como fallida (uploaded=false, uploading=false) para que el usuario la vea
   189→   189→          setPhotos(prev =>
   190→   190→            prev.map(p =>
   191→   191→              p.id === photoId
   192→   192→                ? { ...p, uploading: false, uploaded: false }
   193→   193→                : p
   194→   194→            )
   195→   195→          );
   196→   196→        }
   197→   197→
   198→   198→        // Cede al event loop para que el navegador pinte (evita congelar la UI)
   199→   199→        await new Promise(r => setTimeout(r, 30));
   200→   200→      }
   201→   201→    } finally {
   202→   202→      isProcessingRef.current = false;
   203→   203→      setQueueLength(0);
   204→   204→    }
   205→   205→  }, [sStep, miniStep, currentProject?.id]);
   206→   206→
   207→   207→  // Encolar (no procesa directamente). Múltiples llamadas paralelas solo llenan la cola,
   208→   208→  // luego processQueue las consume de una en una.
   209→   209→  const addPhoto = useCallback((rawBase64: string) => {
   210→   210→    uploadQueueRef.current.push({ rawBase64, photoType: defaultPhotoType });
   211→   211→    setQueueLength(uploadQueueRef.current.length);
   212→   212→    processQueue();
   213→   213→  }, [processQueue, defaultPhotoType]);
   214→   214→
   215→   215→  const stopStream = useCallback(() => {
   216→   216→    if (stream) {
   217→   217→      stream.getTracks().forEach(track => track.stop());
   218→   218→      setStream(null);
   219→   219→    }
   220→   220→    setCameraMode('idle');
   221→   221→  }, [stream]);
   222→   222→
   223→   223→  const startCamera = useCallback(async () => {
   224→   224→    setCameraError(null);
   225→   225→    try {
   226→   226→      const mediaStream = await navigator.mediaDevices.getUserMedia({
   227→   227→        video: { facingMode, width: { ideal: 1920 }, height: { ideal: 1080 } },
   228→   228→        audio: false,
   229→   229→      });
   230→   230→      setStream(mediaStream);
   231→   231→      setCameraMode('streaming');
   232→   232→      if (videoRef.current) {
   233→   233→        videoRef.current.srcObject = mediaStream;
   234→   234→        await videoRef.current.play();
   235→   235→      }
   236→   236→    } catch (err: any) {
   237→   237→      if (err.name === 'NotAllowedError') setCameraError('Permiso denegado. Permite el acceso a la cámara.');
   238→   238→      else if (err.name === 'NotFoundError') setCameraError('No se encontró ninguna cámara.');
   239→   239→      else setCameraError('Error al acceder a la cámara. Intenta subir una foto desde tu galería.');
   240→   240→    }
   241→   241→  }, [facingMode]);
   242→   242→
   243→   243→  const switchCamera = useCallback(async () => {
   244→   244→    stopStream();
   245→   245→    setFacingMode(prev => (prev === 'environment' ? 'user' : 'environment'));
   246→   246→  }, [stopStream]);
   247→   247→
   248→   248→  const capturePhoto = useCallback(() => {
   249→   249→    if (!videoRef.current || !canvasRef.current) return;
   250→   250→    const video = videoRef.current;
   251→   251→    const canvas = canvasRef.current;
   252→   252→    canvas.width = video.videoWidth;
   253→   253→    canvas.height = video.videoHeight;
   254→   254→    const ctx = canvas.getContext('2d');
   255→   255→    if (!ctx) return;
   256→   256→    if (facingMode === 'user') { ctx.translate(canvas.width, 0); ctx.scale(-1, 1); }
   257→   257→    ctx.drawImage(video, 0, 0);
   258→   258→    const base64 = canvas.toDataURL('image/jpeg', 0.9);
   259→   259→    setCaptureFlash(true);
   260→   260→    setTimeout(() => setCaptureFlash(false), 300);
   261→   261→    addPhoto(base64);
   262→   262→  }, [facingMode, addPhoto]);
   263→   263→
   264→   264→  // Load existing photos from DB when modal opens
   265→   265→  const [loadingPhotos, setLoadingPhotos] = useState(false);
   266→   266→
   267→   267→  // v2.35: cargar límite dinámico de fotos al abrir el modal.
   268→   268→  // Llama a /api/photo-limits que resuelve override > template > 10.
   269→   269→  useEffect(() => {
   270→   270→    if (!open || !currentProject?.id || !currentZone?.id) return;
   271→   271→    const url = `/api/photo-limits?projectId=${currentProject.id}&zoneId=${currentZone.id}&sStep=${sStep}&miniStep=${miniStep}`;
   272→   272→    fetch(url)
   273→   273→      .then(r => r.json())
   274→   274→      .then(data => {
   275→   275→        if (data.success && data.data && typeof data.data.minPhotos === 'number') {
   276→   276→          setMinPhotos(data.data.minPhotos);
   277→   277→          setPhotoLimitSource(data.data.source);
   278→   278→          console.log(`[FotosModal] minPhotos resuelto: ${data.data.minPhotos} (origen: ${data.data.source})`);
   279→   279→        }
   280→   280→      })
   281→   281→      .catch(err => console.error('[FotosModal] Error al cargar photo-limits:', err));
   282→   282→  }, [open, currentProject?.id, currentZone?.id, sStep, miniStep]);
   283→   283→
   284→   284→  useEffect(() => {
   285→   285→    if (open && currentProject?.id) {
   286→   286→      setLoadingPhotos(true);
   287→   287→      const zoneIdParam = currentZone?.id ? `&zoneId=${currentZone.id}` : ''
   288→   288→      fetch(`/api/photo-library?projectId=${currentProject.id}&sStep=${sStep}&miniStep=2${zoneIdParam}`)
   289→   289→        .then(res => res.json())
   290→   290→        .then(data => {
   291→   291→          if (data.success && data.data && data.data.length > 0) {
   292→   292→            const existingPhotos: PhotoItem[] = data.data.map((p: any, idx: number) => ({
   293→   293→              id: `existing_${p.id || idx}_${Math.random().toString(36).slice(2, 8)}`,
   294→   294→              preview: p.photoUrl, // Use the stored URL as preview
   295→   295→              serverUrl: p.photoUrl,
   296→   296→              uploaded: true,
   297→   297→              uploading: false,
   298→   298→              estimatedSize: 0,
   299→   299→              title: p.title || '',
   300→   300→              photoType: p.photoType || 'antes',
   301→   301→              savedToLibrary: true, // Already saved in DB
   302→   302→            }));
   303→   303→            photoCounterRef.current = existingPhotos.length;
   304→   304→            setPhotos(existingPhotos);
   305→   305→            // If step was already completed, mark it
   306→   306→            const antesCount = existingPhotos.filter(p => p.photoType === 'antes').length;
   307→   307→            const despuesCount = existingPhotos.filter(p => p.photoType === 'despues').length;
   308→   308→            if (antesCount >= minPhotos) setIsCompleted(true);
   309→   309→          }
   310→   310→          setLoadingPhotos(false);
   311→   311→        })
   312→   312→        .catch(err => {
   313→   313→          console.error('Error loading existing photos:', err);
   314→   314→          setLoadingPhotos(false);
   315→   315→        });
   316→   316→    }
   317→   317→  }, [open, currentProject?.id, sStep, currentZone?.id]);
   318→   318→
   319→   319→  useEffect(() => {
   320→   320→    if (!open) {
   321→   321→      stopStream();
   322→   322→      // Don't clear photos on close — they should persist
   323→   323→      // Only clear transient UI state
   324→   324→      setCameraError(null);
   325→   325→      setCameraMode('idle');
   326→   326→      setActiveTab('camera');
   327→   327→    }
   328→   328→  }, [open, stopStream]);
   329→   329→
   330→   330→  // Lee todos los archivos y los encola. NO procesa aquí (eso saturationa el hilo UI).
   331→   331→  // El procesamiento pesado (compresión + upload) lo hace processQueue secuencialmente.
   332→   332→  const readFileAsDataURL = (file: File): Promise<string> =>
   333→   333→    new Promise((resolve, reject) => {
   334→   334→      const reader = new FileReader();
   335→   335→      reader.onloadend = () => resolve(reader.result as string);
   336→   336→      reader.onerror = () => reject(reader.error);
   337→   337→      reader.readAsDataURL(file);
   338→   338→    });
   339→   339→
   340→   340→  const handleFileSelect = async (e: React.ChangeEvent<HTMLInputElement>) => {
   341→   341→    const files = e.target.files;
   342→   342→    if (!files || files.length === 0) return;
   343→   343→    // Captura el snapshot del FileList ANTES de limpiar el input.
   344→   344→    // Si limpiamos antes, el navegador vacía el FileList y Array.from(files) da [].
   345→   345→    const fileArr = Array.from(files);
   346→   346→    if (fileInputRef.current) fileInputRef.current.value = '';
   347→   347→    console.log(`[FotosModal] handleFileSelect: ${fileArr.length} archivo(s) seleccionado(s)`);
   348→   348→    await Promise.all(fileArr.map(f => readFileAsDataURL(f).then(addPhoto).catch(err => console.error('Error leyendo archivo:', err))));
   349→   349→  };
   350→   350→
   351→   351→  const handleCameraCapture = async (e: React.ChangeEvent<HTMLInputElement>) => {
   352→   352→    const files = e.target.files;
   353→   353→    if (!files || files.length === 0) return;
   354→   354→    const fileArr = Array.from(files);
   355→   355→    if (cameraInputRef.current) cameraInputRef.current.value = '';
   356→   356→    console.log(`[FotosModal] handleCameraCapture: ${fileArr.length} archivo(s) capturado(s)`);
   357→   357→    await Promise.all(fileArr.map(f => readFileAsDataURL(f).then(addPhoto).catch(err => console.error('Error leyendo archivo de cámara:', err))));
   358→   358→  };
   359→   359→
   360→   360→  const removePhoto = (index: number) => setPhotos(prev => prev.filter((_, i) => i !== index));
   361→   361→
   362→   362→  const uploadingCount = photos.filter(p => p.uploading).length;
   363→   363→  const isQueueBusy = queueLength > 0 || uploadingCount > 0;
   364→   364→  const canSubmit = photos.length >= minPhotos && uploadingCount === 0 && queueLength === 0;
   365→   365→  const totalSize = photos.reduce((sum, p) => sum + p.estimatedSize, 0);
   366→   366→
   367→   367→  const handleSubmit = async () => {
   368→   368→    if (!canSubmit) return;
   369→   369→    setIsSubmitting(true);
   370→   370→    try {
   371→   371→      const urls = photos.map(p => p.serverUrl || p.preview).join(',');
   372→   372→
   373→   373→      // Save each photo to the PhotoLibrary with full traceability
   374→   374→      // Use serverUrl if available, otherwise use preview (base64) as the photoUrl
   375→   375→      const libraryPromises = photos
   376→   376→        .filter(p => !p.savedToLibrary)
   377→   377→        .map((p, idx) => {
   378→   378→          const allUploadedBefore = photos.slice(0, idx).filter(pp => pp.serverUrl || pp.preview).length;
   379→   379→          const photoUrlForDb = p.serverUrl || p.preview; // Use base64 as fallback if upload failed
   380→   380→          return fetch('/api/photo-library', {
   381→   381→            method: 'POST',
   382→   382→            headers: { 'Content-Type': 'application/json' },
   383→   383→            body: JSON.stringify({
   384→   384→              sStep,
   385→   385→              miniStep: 2,
   386→   386→              title: p.title || generatePhotoTitle(allUploadedBefore, p.photoType),
   387→   387→              description: `${sStepData?.japaneseName || 'S' + sStep} - ${currentZone?.name || 'Zona'} - Paso 2 Fotos - Subida por ${currentUser?.name || 'Usuario'}`,
   388→   388→              photoUrl: photoUrlForDb,
   389→   389→              photoType: p.photoType || 'antes',
   390→   390→              category: `paso2_s${sStep}`,
   391→   391→              tags: JSON.stringify([`S${sStep}`, sStepData?.japaneseName || '', currentZone?.name || '', `paso2`, p.photoType]),
   392→   392→              projectId: currentProject?.id,
   393→   393→              zoneId: currentZone?.id || null,
   394→   394→              uploadedBy: currentUser?.id || null,
   395→   395→            }),
   396→   396→          });
   397→   397→        });
   398→   398→      
   399→   399→      // Wait for ALL photo library saves to complete before proceeding
   400→   400→      const libraryResults = await Promise.allSettled(libraryPromises);
   401→   401→      const savedCount = libraryResults.filter(r => r.status === 'fulfilled').length;
   402→   402→      const failedCount = libraryResults.filter(r => r.status === 'rejected').length;
   403→   403→      if (failedCount > 0) console.warn(`[FotosModal] ${failedCount} photos failed to save to library`);
   404→   404→      else if (savedCount > 0) console.log(`[FotosModal] ${savedCount} photos saved to library`);
   405→   405→      
   406→   406→      // Mark all photos as saved in local state
   407→   407→      setPhotos(prev => prev.map(p => ({ ...p, savedToLibrary: true })));
