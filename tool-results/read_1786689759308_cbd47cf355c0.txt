   365→// ═══════════════════════════════════════════════════════
   366→// VISUAL EDITOR: ChecklistEditor (autoevaluacion / auditoria)
   367→// ═══════════════════════════════════════════════════════
   368→interface ChecklistItem {
   369→  id: string
   370→  description: string
   371→  hasOther: boolean
   372→  active?: boolean // true (default) → aparece en el tablero (apartados 4/5); false → oculta
   373→}
   374→
   375→interface ChecklistSection {
   376→  id: string
   377→  title: string
   378→  items: ChecklistItem[]
   379→}
   380→
   381→function ChecklistEditor({ content, onChange }: { content: string; onChange: (v: string) => void }) {
   382→  let parsed: { sections: ChecklistSection[] }
   383→  try {
   384→    parsed = JSON.parse(content)
   385→  } catch {
   386→    return (
   387→      <div className="p-4 bg-red-50 border border-red-200 rounded-lg text-sm text-red-600">
   388→        El JSON no es válido. Corrígelo en modo JSON o carga el contenido por defecto.
   389→      </div>
   390→    )
   391→  }
   392→
   393→  const sections = parsed.sections || []
   394→  const sPrefix = String(content.match(/"id"\s*:\s*"(\d+)\./)?.[1] || '1')
   395→
   396→  const update = (newSections: ChecklistSection[]) => {
   397→    onChange(JSON.stringify({ ...parsed, sections: newSections }, null, 2))
   398→  }
   399→
   400→  const addSection = () => {
   401→    const newId = `${sPrefix}.${sections.length + 1}`
   402→    update([...sections, { id: newId, title: 'Nueva Sección', items: [] }])
   403→  }
   404→
   405→  const removeSection = (idx: number) => {
   406→    update(sections.filter((_, i) => i !== idx))
   407→  }
   408→
   409→  const updateSection = (idx: number, field: 'id' | 'title', value: string) => {
   410→    const updated = [...sections]
   411→    updated[idx] = { ...updated[idx], [field]: value }
   412→    update(updated)
   413→  }
   414→
   415→  const moveSection = (idx: number, dir: -1 | 1) => {
   416→    const target = idx + dir
   417→    if (target < 0 || target >= sections.length) return
   418→    const updated = [...sections]
   419→    ;[updated[idx], updated[target]] = [updated[target], updated[idx]]
   420→    update(updated)
   421→  }
   422→
   423→  const addItem = (sectionIdx: number) => {
   424→    const sec = sections[sectionIdx]
   425→    const newItemId = `${sec.id}.${sec.items.length + 1}`
   426→    const updated = [...sections]
   427→    updated[sectionIdx] = {
   428→      ...updated[sectionIdx],
   429→      items: [...updated[sectionIdx].items, { id: newItemId, description: '', hasOther: false, active: true }]
   430→    }
   431→    update(updated)
   432→  }
   433→
   434→  const removeItem = (sectionIdx: number, itemIdx: number) => {
   435→    const updated = [...sections]
   436→    updated[sectionIdx] = {
   437→      ...updated[sectionIdx],
   438→      items: updated[sectionIdx].items.filter((_, i) => i !== itemIdx)
   439→    }
   440→    update(updated)
   441→  }
   442→
   443→  const updateItem = (sectionIdx: number, itemIdx: number, field: string, value: string | boolean) => {
   444→    const updated = [...sections]
   445→    updated[sectionIdx] = {
   446→      ...updated[sectionIdx],
   447→      items: updated[sectionIdx].items.map((item, i) =>
   448→        i === itemIdx ? { ...item, [field]: value } : item
   449→      )
   450→    }
   451→    update(updated)
   452→  }
   453→
   454→  const moveItem = (sectionIdx: number, itemIdx: number, dir: -1 | 1) => {
   455→    const items = sections[sectionIdx].items
   456→    const target = itemIdx + dir
   457→    if (target < 0 || target >= items.length) return
   458→    const updated = [...sections]
   459→    const newItems = [...updated[sectionIdx].items]
   460→    ;[newItems[itemIdx], newItems[target]] = [newItems[target], newItems[itemIdx]]
   461→    updated[sectionIdx] = { ...updated[sectionIdx], items: newItems }
   462→    update(updated)
   463→  }
   464→
   465→  return (
   466→    <div className="space-y-4">
   467→      {/* Leyenda rápida */}
   468→      <div className="flex items-center gap-3 px-3 py-2 bg-indigo-50 border border-indigo-200 rounded-md text-xs text-indigo-800">
   469→        <Eye className="h-3.5 w-3.5 shrink-0" />
   470→        <span><b>En tablero</b> = la pregunta aparece en los apartados 4/5 de las S del tablero.</span>
   471→        <span className="text-indigo-400">·</span>
   472→        <EyeOff className="h-3.5 w-3.5 shrink-0" />
   473→        <span><b>Oculta</b> = no aparece en el tablero. Pulsa el badge para alternar.</span>
   474→      </div>
   475→      {sections.map((section, sIdx) => {
   476→        const activeCount = section.items.filter(it => it.active !== false).length
   477→        const totalCount = section.items.length
   478→        return (
   479→        <div key={sIdx} className="border-2 rounded-lg overflow-hidden bg-white shadow-sm">
   480→          {/* Section header */}
   481→          <div className="flex items-center gap-3 px-4 py-3 bg-gray-50 border-b">
   482→            <div className="flex flex-col gap-0.5">
   483→              <button onClick={() => moveSection(sIdx, -1)} className="text-gray-400 hover:text-gray-600 leading-none" title="Subir sección">
   484→                <ChevronUp className="h-4 w-4" />
   485→              </button>
   486→              <button onClick={() => moveSection(sIdx, 1)} className="text-gray-400 hover:text-gray-600 leading-none" title="Bajar sección">
   487→                <ChevronDown className="h-4 w-4" />
   488→              </button>
   489→            </div>
   490→            <Input
   491→              value={section.id}
   492→              onChange={(e) => updateSection(sIdx, 'id', e.target.value)}
   493→              className="w-20 h-9 text-sm font-mono"
   494→              placeholder="ID"
   495→            />
   496→            <Input
   497→              value={section.title}
   498→              onChange={(e) => updateSection(sIdx, 'title', e.target.value)}
   499→              className="flex-1 h-9 text-base font-semibold"
   500→              placeholder="Título de sección"
   501→            />
   502→            <span
   503→              className={`text-xs font-medium px-2 py-1 rounded-md whitespace-nowrap shrink-0 ${
   504→                activeCount === totalCount
   505→                  ? 'bg-green-100 text-green-700'
   506→                  : activeCount === 0
   507→                  ? 'bg-gray-200 text-gray-600'
   508→                  : 'bg-amber-100 text-amber-700'
   509→              }`}
   510→              title={`${activeCount} de ${totalCount} preguntas aparecen en el tablero`}
   511→            >
   512→              {activeCount}/{totalCount} en tablero
   513→            </span>
   514→            <Button variant="ghost" size="sm" onClick={() => addItem(sIdx)}
   515→              className="h-9 w-9 p-0 text-green-600 hover:text-green-700 hover:bg-green-50" title="Añadir item">
   516→              <Plus className="h-5 w-5" />
   517→            </Button>
   518→            <Button variant="ghost" size="sm" onClick={() => removeSection(sIdx)}
   519→              className="h-9 w-9 p-0 text-red-500 hover:text-red-600 hover:bg-red-50" title="Eliminar sección">
   520→              <Trash2 className="h-4 w-4" />
   521→            </Button>
   522→          </div>
   523→
   524→          {/* Items */}
   525→          <div className="p-3 space-y-2">
   526→            {section.items.length === 0 && (
   527→              <p className="text-sm text-muted-foreground italic px-2 py-2">Sin items. Pulsa + para añadir.</p>
   528→            )}
   529→            {section.items.map((item, iIdx) => {
   530→              // Un item se considera "en tablero" a menos que explícitamente esté active:false.
   531→              // Esto mantiene compatibilidad con plantillas antiguas que no tienen el campo `active`.
   532→              const isOnBoard = item.active !== false
   533→              return (
   534→                <div
   535→                  key={iIdx}
   536→                  className={`flex items-center gap-3 group rounded-md px-1 py-0.5 transition-colors ${
   537→                    isOnBoard ? '' : 'bg-gray-50'
   538→                  }`}
   539→                >
   540→                  <div className="flex flex-col gap-0.5">
   541→                    <button onClick={() => moveItem(sIdx, iIdx, -1)} className="text-gray-300 hover:text-gray-500 leading-none" title="Subir">
   542→                      <ChevronUp className="h-3.5 w-3.5" />
   543→                    </button>
   544→                    <button onClick={() => moveItem(sIdx, iIdx, 1)} className="text-gray-300 hover:text-gray-500 leading-none" title="Bajar">
   545→                      <ChevronDown className="h-3.5 w-3.5" />
   546→                    </button>
   547→                  </div>
   548→                  <Input
   549→                    value={item.id}
   550→                    onChange={(e) => updateItem(sIdx, iIdx, 'id', e.target.value)}
   551→                    className="w-20 h-8 text-xs font-mono"
   552→                    placeholder="ID"
   553→                  />
   554→                  <Input
   555→                    value={item.description}
   556→                    onChange={(e) => updateItem(sIdx, iIdx, 'description', e.target.value)}
   557→                    className={`flex-1 h-8 text-sm ${
   558→                      isOnBoard ? '' : 'text-gray-400 line-through decoration-gray-300'
   559→                    }`}
   560→                    placeholder="Descripción del item"
   561→                  />
   562→                  {/* Toggle "En tablero" — Controla si esta pregunta aparece en los apartados 4/5 de las S del tablero */}
   563→                  <button
   564→                    type="button"
   565→                    onClick={() => updateItem(sIdx, iIdx, 'active', !isOnBoard)}
   566→                    className={`flex items-center gap-1 px-2 h-8 rounded-md text-xs font-medium border transition-colors whitespace-nowrap shrink-0 ${
   567→                      isOnBoard
   568→                        ? 'bg-green-50 text-green-700 border-green-200 hover:bg-green-100'
   569→                        : 'bg-gray-100 text-gray-500 border-gray-200 hover:bg-gray-200'
   570→                    }`}
   571→                    title={
   572→                      isOnBoard
