     4→import {
     5→  Dialog,

     8→  DialogTitle,
     9→} from '@/components/ui/dialog';
    10→import { Button } from '@/components/ui/button';

    19→  S_STEPS,

    24→
    25→interface AutoevaluacionModalProps {
    26→  open: boolean;
    27→  onClose: () => void;
    28→  sStep: number;

    31→
    32→
    33→
    34→export default function AutoevaluacionModal({ open, onClose, sStep, miniStep }: AutoevaluacionModalProps) {
    35→  const { fetchProgress, currentUser, adminFreeNavigation, currentProject, currentZone, canPerform, canView, hasPermission, openModal } = use5SStore();
    36→  const sStepData = S_STEPS.find(s => s.id === sStep);
    37→  const canSkipSteps = hasPermission('skip_steps');
    38→  const canPerformStep = canPerform(sStep, miniStep);
    39→  const canViewStep = canView(sStep, miniStep);
    40→  // Permission-driven: read-only if no execute perm OR if candado closed for skip_steps users
    41→  const isReadOnly = !canPerformStep || (canSkipSteps && !adminFreeNavigation);
    42→  const canPerformAutoeval = canPerformStep;
    43→
    44→  const [isFullscreen, setIsFullscreen] = useState(true);
    45→  const [results, setResults] = useState<Record<string, AuditItemResult>>({});
    46→  const [expandedSections, setExpandedSections] = useState<Record<string, boolean>>({});
    47→  const [observaciones, setObservaciones] = useState('');
    48→  const [isSubmitting, setIsSubmitting] = useState(false);
    49→  const [isCompleted, setIsCompleted] = useState(false);
    50→  const [finalScore, setFinalScore] = useState(0);
    51→  const [notaMinima, setNotaMinima] = useState(70);
    52→  const [autoevalPhotos, setAutoevalPhotos] = useState<{ file: File; preview: string; uploading?: boolean; serverUrl?: string }[]>([]);
    53→  const [isUploadingPhotos, setIsUploadingPhotos] = useState(false);
    54→  const photoInputRef = useRef<HTMLInputElement>(null);
    55→
    56→  // Date/Time for scheduling and recording the evaluation
    57→  const [fechaAutoevaluacion, setFechaAutoevaluacion] = useState('');
    58→  const [horaAutoevaluacion, setHoraAutoevaluacion] = useState('');
    59→  const [fechaProgramada, setFechaProgramada] = useState('');
    60→  const [horaProgramada, setHoraProgramada] = useState('');
    61→  // Track auditoría schedule (miniStep 5) — employee sees this as read-only (auditor sets it)
    62→  const [auditFechaProgramada, setAuditFechaProgramada] = useState('');
    63→  const [auditHoraProgramada, setAuditHoraProgramada] = useState('');
    64→
    65→  // Load template from API (uses board config if zone has one)
    66→  const { sections, isLoading: isLoadingTemplate, notaMinima: templateNotaMinima } = useChecklistTemplate('autoevaluacion', sStep, open, currentZone?.boardConfigId);
    67→
    68→  // Apply template notaMinima when loaded
    69→  useEffect(() => {
    70→    if (templateNotaMinima !== null) setNotaMinima(templateNotaMinima);
    71→  }, [templateNotaMinima]);
    72→
    73→
    74→
    75→  // Fetch dynamic threshold (overrides template if present)
    76→  useEffect(() => {
    77→    if (open && currentProject?.id) {
    78→      const fetchThreshold = async () => {
    79→        try {
    80→          const params = new URLSearchParams({ projectId: currentProject.id });
    81→          if (currentZone?.id) params.set('zoneId', currentZone.id);
    82→          const res = await fetch(`/api/audit-targets?${params}`);
    83→          const json = await res.json();
    84→          if (json.success && json.data) {
    85→            const zoneTarget = json.data.find((t: any) => t.sStep === sStep && t.miniStep === 4 && t.zoneId === currentZone?.id);
    86→            const projectTarget = json.data.find((t: any) => t.sStep === sStep && t.miniStep === 4 && t.zoneId === null);
    87→            const target = zoneTarget || projectTarget;
    88→            if (target?.notaMinima) setNotaMinima(target.notaMinima);
    89→          }
    90→        } catch (e) { console.error('Error fetching threshold:', e); }
    91→      };
    92→      fetchThreshold();
    93→    }
    94→  }, [open, sStep, currentProject?.id, currentZone?.id]);
    95→
    96→  // Initialize expanded sections
    97→  useEffect(() => {
    98→    if (open && sections.length > 0) {
    99→      const expanded: Record<string, boolean> = {};
   100→      sections.forEach(s => { expanded[s.id] = true; });
   101→      setExpandedSections(expanded);
   102→      setResults({});
   103→      setObservaciones('');
   104→      setIsCompleted(false);
   105→      setFinalScore(0);
   106→      setAutoevalPhotos([]);
   107→      // Auto-fill current date/time for the evaluation
   108→      const now = new Date();
   109→      setFechaAutoevaluacion(now.toISOString().split('T')[0]);
   110→      setHoraAutoevaluacion(now.toTimeString().slice(0, 5));
   111→      // Load scheduled date if available
   112→      loadScheduledDate();
   113→    }
   114→  }, [open, sStep, sections]);
   115→
   116→  // Load scheduled date/time for this autoevaluación AND the auditoría schedule
   117→  const loadScheduledDate = async () => {
   118→    if (!currentProject?.id || !currentZone?.id) return;
   119→    try {
   120→      // Load autoevaluación schedule (miniStep 4)
   121→      const res = await fetch(`/api/evaluation-schedule?sStep=${sStep}&miniStep=4&projectId=${currentProject.id}&zoneId=${currentZone.id}`);
   122→      const json = await res.json();
   123→      if (json.success && json.data) {
   124→        setFechaProgramada(json.data.fechaProgramada || '');
   125→        setHoraProgramada(json.data.horaProgramada || '');
   126→      }
   127→      // Also load auditoría schedule (miniStep 5) — employee sees if auditor has set a date
   128→      const auditRes = await fetch(`/api/evaluation-schedule?sStep=${sStep}&miniStep=5&projectId=${currentProject.id}&zoneId=${currentZone.id}`);
   129→      const auditJson = await auditRes.json();
   130→      if (auditJson.success && auditJson.data) {
   131→        setAuditFechaProgramada(auditJson.data.fechaProgramada || '');
   132→        setAuditHoraProgramada(auditJson.data.horaProgramada || '');
   133→      }
   134→    } catch (e) {
   135→      console.error('Error loading scheduled date:', e);
   136→    }
   137→  };
   138→
   139→  // Save scheduled date/time
   140→  const handleSaveSchedule = async () => {
   141→    if (!currentProject?.id || !currentZone?.id) return;
   142→    try {
   143→      await fetch('/api/evaluation-schedule', {
   144→        method: 'POST',
   145→        headers: { 'Content-Type': 'application/json' },
   146→        body: JSON.stringify({
   147→          sStep,
   148→          miniStep: 4,
   149→          projectId: currentProject.id,
   150→          zoneId: currentZone.id,
   151→          fechaProgramada,
   152→          horaProgramada,
   153→        }),
   154→      });
   155→      toast.success('Fecha programada guardada');
   156→    } catch (e) {
   157→      console.error('Error saving schedule:', e);
   158→      toast.error('Error al guardar la fecha programada');
   159→    }
   160→  };
   161→
   162→  const totalItems = sections.reduce((sum, s) => sum + s.items.length, 0) || 26;
   163→
   164→  const scoring = useMemo(() => {
   165→    const allResults = Object.values(results);
   166→    const okCount = allResults.filter(r => r.status === 'ok').length;
   167→    const nokCount = allResults.filter(r => r.status === 'nok').length;
   168→    const answeredCount = okCount + nokCount;
   169→    const scorePercent = totalItems > 0 ? Math.min(Math.round((okCount / totalItems) * 100), 100) : 0;
   170→    return { okCount, nokCount, answeredCount, scorePercent };
   171→  }, [results, totalItems]);
   172→
   173→  // Check that all NOK items have hallazgo and mejora filled
   174→  const nokItems = Object.values(results).filter(r => r.status === 'nok');
   175→  const allNokCompleted = nokItems.length === 0 || nokItems.every(r => (r.hallazgo || '').trim() !== '' && (r.mejora || '').trim() !== '');
   176→
   177→  const passed = scoring.scorePercent >= notaMinima;
   178→  const canSubmit = canPerformAutoeval && scoring.answeredCount > 0 && allNokCompleted;
   179→
   180→  const handlePhotoSelect = async (e: React.ChangeEvent<HTMLInputElement>) => {
   181→    const files = e.target.files;
   182→    if (!files) return;
   183→    for (const file of Array.from(files)) {
   184→      const preview = URL.createObjectURL(file);
   185→      setAutoevalPhotos(prev => [...prev, { file, preview }]);
   186→    }
   187→    // Reset input
   188→    if (photoInputRef.current) photoInputRef.current.value = '';
   189→  };
   190→
   191→  const removeAutoevalPhoto = (index: number) => {
   192→    setAutoevalPhotos(prev => {
   193→      const photo = prev[index];
   194→      if (photo.preview) URL.revokeObjectURL(photo.preview);
   195→      return prev.filter((_, i) => i !== index);
   196→    });
   197→  };
   198→
   199→  const toggleSection = (sectionId: string) => {
   200→    setExpandedSections(prev => ({ ...prev, [sectionId]: !prev[sectionId] }));
   201→  };
   202→
   203→  const setItemStatus = (itemId: string, status: 'ok' | 'nok' | 'na') => {
   204→    setResults(prev => ({
   205→      ...prev,
   206→      [itemId]: { ...prev[itemId], itemId, status },
   207→    }));
   208→  };
   209→
   210→  const setItemField = (itemId: string, field: 'hallazgo' | 'mejora' | 'otherText', value: string) => {
   211→    setResults(prev => ({
   212→      ...prev,
   213→      [itemId]: { ...prev[itemId], itemId, [field]: value },
   214→    }));
   215→  };
   216→
   217→  const handleSubmit = async () => {
   218→    if (!canSubmit) return;
   219→    if (!canPerformAutoeval) return; // Only responsable/admin for S4, or any employee for S1/S2/S3/S5
   220→    setIsSubmitting(true);
   221→    try {
   222→      // Only mark as completed if score meets notaMinima threshold
   223→      // If not passed, still save results but step stays available for retry
   224→      const res = await fetch(`/api/progress/step?sStep=${sStep}&miniStep=${miniStep}`, {
   225→        method: 'PUT',
   226→        headers: { 'Content-Type': 'application/json' },
   227→        body: JSON.stringify({
   228→          completed: passed,
   229→          score: scoring.scorePercent,
   230→          notes: JSON.stringify({
   231→            type: 'autoevaluacion',
   232→            passed,
   233→            notaMinima,
   234→            results: Object.values(results),
   235→            observaciones,
   236→            fechaAutoevaluacion,
   237→            horaAutoevaluacion,
   238→            fechaProgramada: fechaProgramada || null,
   239→            horaProgramada: horaProgramada || null,
   240→          }),
   241→          projectId: currentProject?.id,
   242→          zoneId: currentZone?.id || null,
   243→        }),
   244→      });
   245→      const json = await res.json();
   246→      if (json.success) {
   247→        setIsCompleted(true);
   248→        setFinalScore(scoring.scorePercent);
   249→        await fetchProgress();
   250→
   251→        // ─── Create EmployeeProgress record for individual step 4 (autoevaluación) ───
   252→        // Step 4 is individual for S1/S2/S3/S5 (done by employees) and for S4 (done by responsable)
   253→        // We need to track individual completion so the gating system unlocks step 5
   254→        if (passed && currentUser?.id && currentProject?.id && currentZone?.id) {
   255→          try {
   256→            await fetch('/api/employee-progress', {
   257→              method: 'POST',
   258→              headers: { 'Content-Type': 'application/json' },
   259→              body: JSON.stringify({
   260→                sStep,
   261→                miniStep: 4,
   262→                completed: true,
   263→                score: scoring.scorePercent,
   264→                projectId: currentProject.id,
   265→                zoneId: currentZone.id,
   266→                userId: currentUser.id,
   267→              }),
   268→            });
   269→            // Also refresh employee progress so step 5 unlocks immediately
   270→            const { fetchEmployeeProgress } = use5SStore.getState();
   271→            await fetchEmployeeProgress(currentProject.id, currentZone.id);
   272→          } catch (empErr) {
   273→            console.error('Error creating employee progress for autoeval:', empErr);
   274→          }
   275→        }
   276→
   277→        // ─── Upload photos to library with traceability ───
   278→        if (autoevalPhotos.length > 0) {
   279→          setIsUploadingPhotos(true);
   280→          for (let idx = 0; idx < autoevalPhotos.length; idx++) {
   281→            const photo = autoevalPhotos[idx];
   282→            try {
   283→              // Step 1: Upload the file to get a server URL
   284→              const formData = new FormData();
   285→              formData.append('file', photo.file);
   286→              formData.append('filename', `S${sStep}_autoeval_${currentZone?.name || 'zona'}_${idx + 1}_${Date.now()}.jpg`);
   287→
   288→              const uploadRes = await fetch('/api/upload', {
   289→                method: 'POST',
   290→                body: formData,
   291→              });
   292→              const uploadData = await uploadRes.json();
   293→
   294→              if (uploadData.success && uploadData.url) {
   295→                // Step 2: Save to photo library with full traceability
   296→                await fetch('/api/photo-library', {
   297→                  method: 'POST',
   298→                  headers: { 'Content-Type': 'application/json' },
   299→                  body: JSON.stringify({
   300→                    sStep,
   301→                    miniStep: 4,
   302→                    title: `Autoeval S${sStep} - Foto ${idx + 1}`,
   303→                    description: `${sStepData?.japaneseName || 'S' + sStep} - ${currentZone?.name || 'Zona'} - Paso 4 Autoevaluación - Subida por ${currentUser?.name || 'Usuario'}`,
   304→                    photoUrl: uploadData.url,
   305→                    photoType: 'hallazgo',
   306→                    category: `paso4_s${sStep}`,
   307→                    tags: JSON.stringify([`S${sStep}`, sStepData?.japaneseName || '', currentZone?.name || '', 'paso4', 'autoevaluacion', 'hallazgo']),
   308→                    projectId: currentProject?.id,
   309→                    zoneId: currentZone?.id || null,
   310→                    uploadedBy: currentUser?.id || null,
   311→                  }),
   312→                });
   313→              }
   314→            } catch (photoErr) {
   315→              console.error('Error uploading autoeval photo:', photoErr);
   316→            }
   317→          }
   318→          setIsUploadingPhotos(false);
   319→        }
   320→
   321→        // ─── Create Action Items for NOK (disfunciones) ───
   322→        const nokResults = Object.values(results).filter((r: any) => r.status === 'nok');
   323→        for (const nok of nokResults) {
   324→          if (!nok.hallazgo && !nok.mejora) continue; // Skip items without description
   325→          try {
   326→            await fetch('/api/actions', {
   327→              method: 'POST',
   328→              headers: { 'Content-Type': 'application/json' },
   329→              body: JSON.stringify({
   330→                sStep,
   331→                miniStep: 4,
   332→                itemId: nok.itemId,
   333→                itemDescription: `Disfunción detectada en autoevaluación: ${nok.itemId}`,
   334→                hallazgo: nok.hallazgo || nok.itemId,
   335→                mejora: nok.mejora || '',
   336→                responsable: null,
   337→                prioridad: 'media',
   338→                estado: 'abierta',
   339→                source: 'autoevaluacion',
   340→                auditor: null,
   341→                projectId: currentProject?.id,
   342→                zoneId: currentZone?.id || null,
   343→              }),
   344→            });
   345→          } catch (actionError) {
   346→            console.error('Error creating action item from autoevaluación:', actionError);
   347→          }
   348→        }
   349→
   350→        // ─── Notify responsables of NOK disfunciones ───
   351→        if (nokResults.length > 0 && currentProject?.id && currentZone?.id) {
   352→          try {
   353→            const membersRes = await fetch(`/api/projects/${currentProject.id}/members`);
   354→            const membersData = await membersRes.json();
   355→            const allMembers = membersData?.members || [];
   356→
   357→            const sStepData = S_STEPS.find(s => s.id === sStep);
   358→            const disfuncionMessage = `Se han detectado ${nokResults.length} disfunción(es) en la autoevaluación de S${sStep} (${sStepData?.japaneseName || ''}) en la zona "${currentZone.name}". Revisa el Plan de Acción.`;
   359→
   360→            // Notify zone responsable
   361→            const responsableIds = new Set<string>();
   362→            if (currentZone.responsableId) responsableIds.add(currentZone.responsableId);
   363→            const responsables = allMembers.filter((m: any) => m.role === 'responsable');
   364→            for (const resp of responsables) responsableIds.add(resp.userId);
   365→
   366→            for (const respId of responsableIds) {
   367→              await fetch('/api/notifications', {
   368→                method: 'POST',
   369→                headers: { 'Content-Type': 'application/json' },
   370→                body: JSON.stringify({
   371→                  userId: respId,
   372→                  type: 'disfuncion',
   373→                  title: `Disfunciones detectadas: S${sStep} — ${sStepData?.japaneseName || ''}`,
   374→                  message: disfuncionMessage,
   375→                  sStep,
   376→                  zoneId: currentZone.id,
   377→                  projectId: currentProject.id,
   378→                }),
   379→              });
   380→            }
   381→          } catch (notifError) {
   382→            console.error('Error notifying responsables of disfunciones:', notifError);
   383→          }
   384→        }
   385→
   386→        // Check if Steps 1-4 are now all completed for this S-step in the zone
   387→        // If so, notify auditor(s) that they can perform Step 5
   388→        if (currentProject?.id && currentZone?.id) {
   389→          try {
   390→            // Fetch all progress for this zone and S-step
   391→            const progRes = await fetch(`/api/progress?projectId=${currentProject.id}`);
   392→            const progData = await progRes.json();
   393→            const allProgress = progData?.data || [];
   394→
   395→            // Also fetch employee progress for individual steps (formación = step 1)
   396→            const empProgRes = await fetch(`/api/employee-progress?projectId=${currentProject.id}&zoneId=${currentZone.id}`);
   397→            const empProgData = await empProgRes.json();
   398→            const allEmpProgress = empProgData?.data || [];
   399→
   400→            // Check steps 1-4 completed (both zone-level progress AND employee progress)
   401→            let allStepsCompleted = true;
   402→            for (let ms = 1; ms <= 4; ms++) {
   403→              const zoneStep = allProgress.find((p: any) =>
   404→                p.sStep === sStep &&
   405→                p.miniStep === ms &&
   406→                (p.zoneId === currentZone.id || p.zoneId === null) &&
   407→                p.completed
   408→              );
   409→              if (zoneStep) continue; // Zone-level completed
   410→
   411→              // Also check employee progress for individual steps
   412→              const empStep = allEmpProgress.some((ep: any) =>
   413→                ep.sStep === sStep &&
   414→                ep.miniStep === ms &&
   415→                ep.zoneId === currentZone.id &&
   416→                ep.completed
   417→              );
   418→              if (empStep) continue; // Some employee completed this step
   419→
   420→              allStepsCompleted = false;
   421→              break;
   422→            }
   423→
   424→            if (allStepsCompleted) {
   425→              // Find auditor and responsable users for this project and notify them
   426→              const membersRes = await fetch(`/api/projects/${currentProject.id}/members`);
   427→              const membersData = await membersRes.json();
   428→              const allMembers = membersData?.members || [];
   429→
   430→              const sStepData = S_STEPS.find(s => s.id === sStep);
   431→              const notifMessage = `Los pasos 1-4 de S${sStep} (${sStepData?.japaneseName || ''}) en la zona "${currentZone.name}" han sido completados. La auditoría (Paso 5) está lista para realizarse.`;
   432→
   433→              // Notify auditors
   434→              const auditors = allMembers.filter((m: any) => m.role === 'auditor');
   435→              for (const auditor of auditors) {
   436→                await fetch('/api/notifications', {
   437→                  method: 'POST',
   438→                  headers: { 'Content-Type': 'application/json' },
   439→                  body: JSON.stringify({
   440→                    userId: auditor.userId,
   441→                    type: 'audit_ready',
   442→                    title: `Auditoría lista: S${sStep} — ${sStepData?.japaneseName || ''}`,
   443→                    message: notifMessage,
   444→                    sStep,
   445→                    zoneId: currentZone.id,
   446→                    projectId: currentProject.id,
   447→                  }),
   448→                });
   449→              }
   450→
   451→              // Notify responsable of the zone (from zone.responsableId OR from project members)
   452→              const responsableIds = new Set<string>();
   453→              if (currentZone.responsableId) responsableIds.add(currentZone.responsableId);
   454→              const responsables = allMembers.filter((m: any) => m.role === 'responsable');
   455→              for (const resp of responsables) responsableIds.add(resp.userId);
   456→              for (const respId of responsableIds) {
   457→                await fetch('/api/notifications', {
   458→                  method: 'POST',
   459→                  headers: { 'Content-Type': 'application/json' },
   460→                  body: JSON.stringify({
   461→                    userId: respId,
   462→                    type: 'audit_ready',
   463→                    title: `Auditoría lista: S${sStep} — ${sStepData?.japaneseName || ''}`,
   464→                    message: notifMessage,
   465→                    sStep,
   466→                    zoneId: currentZone.id,
   467→                    projectId: currentProject.id,
   468→                  }),
   469→                });
   470→              }
   471→            }
   472→          } catch (notifError) {
   473→            console.error('Error sending notification to auditor:', notifError);
   474→          }
   475→        }
   476→      }
   477→    } catch (error) {
   478→      console.error('Error submitting self-evaluation:', error);
   479→    } finally {
   480→      setIsSubmitting(false);
   481→    }
   482→  };
   483→
   484→  const handleAdminSkip = async () => {
   485→    try {
   486→      const res = await fetch(`/api/progress/step?sStep=${sStep}&miniStep=${miniStep}`, {
   487→        method: 'PUT',
   488→        headers: { 'Content-Type': 'application/json' },
   489→        body: JSON.stringify({ completed: true, score: 100, notes: 'Completado por administrador (skip)', skipMissingTemplate: true, projectId: currentProject?.id, zoneId: currentZone?.id || null }),
   490→      });
   491→      const json = await res.json();
   492→      if (json.success) {
   493→        await fetchProgress();
   494→        onClose();
   495→      }
   496→    } catch (error) {
   497→      console.error('Error admin skip:', error);
   498→    }
   499→  };
   500→
   501→  return (
   502→    <Dialog open={open} onOpenChange={() => onClose()}>
   503→      <DialogContent size={isFullscreen ? "fullscreen" : "xl"} className="flex flex-col overflow-hidden p-0">
   504→        <DialogHeader className="px-6 pt-6 pb-2 flex-shrink-0">
